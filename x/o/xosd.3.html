XOSD(3) Manual de funciones de biblioteca XOSD(3)

NOMBRE

xosd - biblioteca de visualización en pantalla de X

SINOPSIS

#include <xosd.h>

xosd* xosd_create(int numero_lineas);

int xosd_uninit (xosd* osd);

int xosd_display (xosd* osd, int linea,
                  xosd_command comando, ...);

int xosd_is_onscreen(xosd* osd);

int xosd_wait_until_no_display(xosd* osd);

int xosd_hide (xosd* osd);

int xosd_show (xosd* osd);

int xosd_set_pos (xosd* osd, xosd_pos pos);

int xosd_set_vertical_offset (xosd* osd, int offset);

int xosd_set_horizontal_offset (xosd* osd, int offset);

int xosd_set_align (xosd* osd, xosd_align align)

int xosd_set_shadow_offset (xosd* osd, int offset_sombra);

int xosd_set_timeout (xosd* osd, int tiempo_espera);

int xosd_set_font (xosd* osd, char* fuente);

int xosd_set_colour (xosd* osd, char* color);

int xosd_get_colour (xosd* osd,
                     int* rojo, int* verde, int* azul);

int xosd_scroll (xosd* osd, int lineas);

int xosd_get_number_lines ( xosd* osd);

DESCRIPCIÓN

xosd es una biblioteca para mostrar una visualización en pantalla (como la que se ve en muchos televisores) en tu pantalla de X. Cada función debería tener su propia página de manual.

La última versión se puede encontrar en http://www.ignavus.net/software.html

Se distribuye bajo la Licencia Pública General de GNU.

EJEMPLO

#include <xosd.h>
int main (int argc, char *argv[])
{
	xosd *osd;
	osd = xosd_create (1);
	xosd_set_font(osd, "fixed");
	xosd_set_colour(osd, "LawnGreen");
	xosd_set_timeout(osd, 3);
	xosd_set_shadow_offset(osd, 1);

	xosd_display (osd, 0, XOSD_string, "Salida de ejemplo de XOSD");
	xosd_wait_until_no_display(osd);
	xosd_uninit (osd);
	return EXIT_SUCCESS;
}
