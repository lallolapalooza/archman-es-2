Documentación de función/subrutina

subroutine sqrt15 (integer scale, integer rksel, integer m, integer n, integer nrhs, real, dimension( lda, * ) a, integer lda, real, dimension( ldb, * ) b, integer ldb, real, dimension( * ) s, integer rank, real norma, real normb, integer, dimension( 4 ) iseed, real, dimension( lwork ) work, integer lwork)

SQRT15

Propósito:

!>
!> SQRT15 genera una matriz con rango completo o deficiente y de varias
!> normas.
!>

Parámetros

SCALE 
!>          SCALE es INTEGER
!>          SCALE = 1: matriz normalmente escalada
!>          SCALE = 2: matriz escalada hacia arriba
!>          SCALE = 3: matriz escalada hacia abajo
!>

RKSEL

!>          RKSEL es INTEGER
!>          RKSEL = 1: matriz de rango completo
!>          RKSEL = 2: matriz de rango deficiente
!>

M

!>          M es INTEGER
!>          El número de filas de la matriz A.
!>

N

!>          N es INTEGER
!>          El número de columnas de A.
!>

NRHS

!>          NRHS es INTEGER
!>          El número de columnas de B.
!>

A

!>          A es un array REAL, dimensión (LDA,N)
!>          La matriz M-por-N A.
!>

LDA

!>          LDA es INTEGER
!>          La dimensión principal del array A.
!>

B

!>          B es un array REAL, dimensión (LDB, NRHS)
!>          Una matriz que está en el espacio de rango de la matriz A.
!>

LDB

!>          LDB es INTEGER
!>          La dimensión principal del array B.
!>

S

!>          S es un array REAL, dimensión MIN(M,N)
!>          Valores singulares de A.
!>

RANK

!>          RANK es INTEGER
!>          número de valores singulares distintos de cero de A.
!>

NORMA

!>          NORMA es REAL
!>          norma uno de A.
!>

NORMB

!>          NORMB es REAL
!>          norma uno de B.
!>

ISEED

!>          ISEED es un array integer, dimensión (4)
!>          semilla para el generador de números aleatorios.
!>

WORK

!>          WORK es un array REAL, dimensión (LWORK)
!>

LWORK

!>          LWORK es INTEGER
!>          longitud del espacio de trabajo requerido.
!>          LWORK >= MAX(M+MIN(M,N),NRHS*MIN(M,N),2*N+M)
!>
