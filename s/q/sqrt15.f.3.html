DOCUMENTACIÓN DE FUNCIÓN/SUBRUTINA

subrutina sqrt15 (entero scale, entero rksel, entero m, entero n, entero nrhs, real, dimensión( lda, * ) a, entero lda, real, dimensión( ldb, * ) b, entero ldb, real, dimensión( * ) s, entero rank, real norma, real normb, entero, dimensión( 4 ) iseed, real, dimensión( lwork ) work, entero lwork)

SQRT15

Propósito:

!>
!> SQRT15 genera una matriz con rango completo o deficiente y de varias
!> normas.
!>

Parámetros

SCALE 
!>          SCALE es ENTERO
!>          SCALE = 1: matriz normalmente escalada
!>          SCALE = 2: matriz escalada hacia arriba
!>          SCALE = 3: matriz escalada hacia abajo
!>

RKSEL

!>          RKSEL es ENTERO
!>          RKSEL = 1: matriz de rango completo
!>          RKSEL = 2: matriz de rango deficiente
!>

M

!>          M es ENTERO
!>          El número de filas de la matriz A.
!>

N

!>          N es ENTERO
!>          El número de columnas de A.
!>

NRHS

!>          NRHS es ENTERO
!>          El número de columnas de B.
!>

A

!>          A es arreglo REAL, dimensión (LDA,N)
!>          La matriz M por N de A.
!>

LDA

!>          LDA es ENTERO
!>          La dimensión principal del arreglo A.
!>

B

!>          B es arreglo REAL, dimensión (LDB, NRHS)
!>          Una matriz que está en el espacio de rango de la matriz A.
!>

LDB

!>          LDB es ENTERO
!>          La dimensión principal del arreglo B.
!>

S

!>          S es arreglo REAL, dimensión MIN(M,N)
!>          Valores singulares de A.
!>

RANK

!>          RANK es ENTERO
!>          número de valores singulares distintos de cero de A.
!>

NORMA

!>          NORMA es REAL
!>          norma uno de A.
!>

NORMB

!>          NORMB es REAL
!>          norma uno de B.
!>

ISEED

!>          ISEED es arreglo entero, dimensión (4)
!>          semilla para el generador de números aleatorios.
!>

WORK

!>          WORK es arreglo REAL, dimensión (LWORK)
!>

LWORK

!>          LWORK es ENTERO
!>          longitud del espacio de trabajo requerido.
!>          LWORK >= MAX(M+MIN(M,N),NRHS*MIN(M,N),2*N+M)
!>
