DOCUMENTACIÓN DE FUNCIÓN/SUBRUTINA

subroutine sqrt01p (integer m, integer n, real, dimension( lda, * ) a, real, dimension( lda, * ) af, real, dimension( lda, * ) q, real, dimension( lda, * ) r, integer lda, real, dimension( * ) tau, real, dimension( lwork ) work, integer lwork, real, dimension( * ) rwork, real, dimension( * ) result)

SQRT01P

Propósito:

!>
!> SQRT01P prueba SGEQRFP, que calcula la factorización QR de una matriz m-por-n
!> A, y prueba parcialmente SORGQR que forma la matriz ortogonal m-por-m
!> Q.
!>
!> SQRT01P compara R con Q'*A y comprueba que Q es ortogonal.
!>

Parámetros

M 
!>          M es INTEGER
!>          El número de filas de la matriz A.  M >= 0.
!>

N

!>          N es INTEGER
!>          El número de columnas de la matriz A.  N >= 0.
!>

A

!>          A es array REAL, dimensión (LDA,N)
!>          La matriz A de m por n.
!>

AF

!>          AF es array REAL, dimensión (LDA,N)
!>          Detalles de la factorización QR de A, tal como devuelve SGEQRFP.
!>          Ver SGEQRFP para más detalles.
!>

Q

!>          Q es array REAL, dimensión (LDA,M)
!>          La matriz ortogonal Q de m por m.
!>

R

!>          R es array REAL, dimensión (LDA,max(M,N))
!>

LDA

!>          LDA es INTEGER
!>          La dimensión principal de los arrays A, AF, Q y R.
!>          LDA >= max(M,N).
!>

TAU

!>          TAU es array REAL, dimensión (min(M,N))
!>          Los factores escalares de los reflectores elementales, tal como
!>          devuelve SGEQRFP.
!>

WORK

!>          WORK es array REAL, dimensión (LWORK)
!>

LWORK

!>          LWORK es INTEGER
!>          La dimensión del array WORK.
!>

RWORK

!>          RWORK es array REAL, dimensión (M)
!>

RESULT

!>          RESULT es array REAL, dimensión (2)
!>          Las relaciones de prueba:
!>          RESULT(1) = norm( R - Q'*A ) / ( M * norm(A) * EPS )
!>          RESULT(2) = norm( I - Q'*Q ) / ( M * EPS )
!>
