función real sqrt17 (carácter trans, entero iresid, entero m, entero n, entero nrhs, real, dimensión( lda, * ) a, entero lda, real, dimensión( ldx, * ) x, entero ldx, real, dimensión( ldb, * ) b, entero ldb, real, dimensión( ldb, * ) c, real, dimensión( lwork ) work, entero lwork)

SQRT17

Propósito:

!>
!> SQRT17 calcula la razón
!>
!>    norm(R**T * op(A)) / ( norm(A) * alpha * max(M,N,NRHS) * EPS ),
!>
!> donde R = B - op(A)*X, op(A) es A o A**T, dependiendo de TRANS, EPS
!> es la epsilon de la máquina, y
!>
!>    alpha = norm(B) si IRESID = 1 (problema de residuo cero)
!>    alpha = norm(R) si IRESID = 2 (en caso contrario).
!>
!> La norma utilizada es la norma 1.
!>

Parámetros

TRANS 
!>          TRANS es CARACTER*1
!>          Especifica si se utiliza o no la transpuesta de A.
!>          = 'N':  Sin transpuesta, op(A) = A.
!>          = 'T':  Transpuesta, op(A) = A**T.
!>

IRESID

!>          IRESID es ENTERO
!>          IRESID = 1 indica problema de residuo cero.
!>          IRESID = 2 indica residuo no cero.
!>

M

!>          M es ENTERO
!>          El número de filas de la matriz A.
!>          Si TRANS = 'N', el número de filas de la matriz B.
!>          Si TRANS = 'T', el número de filas de la matriz X.
!>

N

!>          N es ENTERO
!>          El número de columnas de la matriz A.
!>          Si TRANS = 'N', el número de filas de la matriz X.
!>          Si TRANS = 'T', el número de filas de la matriz B.
!>

NRHS

!>          NRHS es ENTERO
!>          El número de columnas de las matrices X y B.
!>

A

!>          A es un array REAL, dimensión (LDA,N)
!>          La matriz A de m x n.
!>

LDA

!>          LDA es ENTERO
!>          La dimensión principal del array A. LDA >= M.
!>

X

!>          X es un array REAL, dimensión (LDX,NRHS)
!>          Si TRANS = 'N', la matriz X de n x nrhs.
!>          Si TRANS = 'T', la matriz X de m x nrhs.
!>

LDX

!>          LDX es ENTERO
!>          La dimensión principal del array X.
!>          Si TRANS = 'N', LDX >= N.
!>          Si TRANS = 'T', LDX >= M.
!>

B

!>          B es un array REAL, dimensión (LDB,NRHS)
!>          Si TRANS = 'N', la matriz B de m x nrhs.
!>          Si TRANS = 'T', la matriz B de n x nrhs.
!>

LDB

!>          LDB es ENTERO
!>          La dimensión principal del array B.
!>          Si TRANS = 'N', LDB >= M.
!>          Si TRANS = 'T', LDB >= N.
!>

C

!>          C es un array REAL, dimensión (LDB,NRHS)
!>

WORK

!>          WORK es un array REAL, dimensión (LWORK)
!>

LWORK

!>          LWORK es ENTERO
!>          La longitud del array WORK. LWORK >= NRHS*(M+N).
!>
