subroutine sppt03 (character uplo, integer n, real, dimension( * ) a, real, dimension( * ) ainv, real, dimension( ldwork, * ) work, integer ldwork, real, dimension( * ) rwork, real rcond, real resid)

SPPT03

Propósito:

!>
!> SPPT03 calcula el residuo para una matriz simétrica empaquetada multiplicada por su
!> inversa:
!>    norm( I - A*AINV ) / ( N * norm(A) * norm(AINV) * EPS ),
!> donde EPS es la epsilon de la máquina.
!>

Parámetros

UPLO 
!>          UPLO es CHARACTER*1
!>          Especifica si la parte triangular superior o inferior de la
!>          matriz simétrica A está almacenada:
!>          = 'U':  Triangular superior
!>          = 'L':  Triangular inferior
!>

N

!>          N es INTEGER
!>          El número de filas y columnas de la matriz A.  N >= 0.
!>

A

!>          A es un array REAL, de dimensión (N*(N+1)/2)
!>          La matriz simétrica original A, almacenada como una matriz
!>          triangular empaquetada.
!>

AINV

!>          AINV es un array REAL, de dimensión (N*(N+1)/2)
!>          La inversa (simétrica) de la matriz A, almacenada como una matriz
!>          triangular empaquetada.
!>

WORK

!>          WORK es un array REAL, de dimensión (LDWORK,N)
!>

LDWORK

!>          LDWORK es INTEGER
!>          La dimensión principal del array WORK.  LDWORK >= max(1,N).
!>

RWORK

!>          RWORK es un array REAL, de dimensión (N)
!>

RCOND

!>          RCOND es REAL
!>          El recíproco del número de condición de A, calculado como
!>          ( 1/norm(A) ) / norm(AINV).
!>

RESID

!>          RESID es REAL
!>          norm(I - A*AINV) / ( N * norm(A) * norm(AINV) * EPS )
!>
