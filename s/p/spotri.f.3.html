subroutine spotri (character uplo, integer n, real, dimension( lda, * ) a, integer lda, integer info)

SPOTRI

Propósito:

!>
!> SPOTRI calcula la inversa de una matriz real simétrica definida positiva
!> A utilizando la factorización de Cholesky A = U**T*U o A = L*L**T
!> calculada por SPOTRF.
!>

Parámetros

UPLO 
!>          UPLO es CHARACTER*1
!>          = 'U':  El triángulo superior de A está almacenado;
!>          = 'L':  El triángulo inferior de A está almacenado.
!>

N

!>          N es INTEGER
!>          El orden de la matriz A.  N >= 0.
!>

A

!>          A es un array REAL, de dimensión (LDA,N)
!>          En la entrada, el factor triangular U o L de la factorización de
!>          Cholesky A = U**T*U o A = L*L**T, tal como fue calculada por
!>          SPOTRF.
!>          En la salida, el triángulo superior o inferior de la inversa
!>          (simétrica) de A, sobrescribiendo el factor de entrada U o L.
!>

LDA

!>          LDA es INTEGER
!>          La dimensión principal del array A.  LDA >= max(1,N).
!>

INFO

!>          INFO es INTEGER
!>          = 0:  salida exitosa
!>          < 0:  si INFO = -i, el i-ésimo argumento tenía un valor ilegal
!>          > 0:  si INFO = i, el elemento (i,i) del factor U o L es cero,
!>                y la inversa no pudo ser calculada.
!>
