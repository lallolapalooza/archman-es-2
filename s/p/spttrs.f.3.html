subroutine spttrs (integer n, integer nrhs, real, dimension( * ) d, real, dimension( * ) e, real, dimension( ldb, * ) b, integer ldb, integer info)

SPTTRS

Propósito:

!>
!> SPTTRS resuelve un sistema tridiagonal de la forma
!>    A * X = B
!> utilizando la factorización L*D*L**T de A calculada por SPTTRF. D es una
!> matriz diagonal especificada en el vector D, L es una matriz bidiagonal
!> unitaria cuya subdiagonal está especificada en el vector E, y X y B
!> son matrices N por NRHS.
!>

Parámetros

N 
!>          N es INTEGER
!>          El orden de la matriz tridiagonal A.  N >= 0.
!>

NRHS

!>          NRHS es INTEGER
!>          El número de lados derechos, es decir, el número de columnas
!>          de la matriz B.  NRHS >= 0.
!>

D

!>          D es array REAL, dimension (N)
!>          Los n elementos diagonales de la matriz diagonal D de la
!>          factorización L*D*L**T de A.
!>

E

!>          E es array REAL, dimension (N-1)
!>          Los n-1 elementos subdiagonales del factor unitario bidiagonal
!>          L de la factorización L*D*L**T de A. E también puede considerarse
!>          como la superdiagonal del factor unitario bidiagonal U de la
!>          factorización A = U**T*D*U.
!>

B

!>          B es array REAL, dimension (LDB,NRHS)
!>          En la entrada, los vectores del lado derecho B para el sistema de
!>          ecuaciones lineales.
!>          En la salida, los vectores solución, X.
!>

LDB

!>          LDB es INTEGER
!>          La dimensión principal del array B.  LDB >= max(1,N).
!>

INFO

!>          INFO es INTEGER
!>          = 0: salida exitosa
!>          < 0: si INFO = -k, el k-ésimo argumento tenía un valor ilegal
!>
