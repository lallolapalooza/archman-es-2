subroutine sposv (character uplo, integer n, integer nrhs, real, dimension( lda, * ) a, integer lda, real, dimension( ldb, * ) b, integer ldb, integer info)

SPOSV calcula la solución de un sistema de ecuaciones lineales reales A * X = B,

Propósito:

!>
!> SPOSV calcula la solución de un sistema de ecuaciones lineales reales
!>    A * X = B,
!> donde A es una matriz N-por-N simétrica positiva definida y X y B
!> son matrices N-por-NRHS.
!>
!> La descomposición de Cholesky se utiliza para factorizar A como
!>    A = U**T* U,  si UPLO = 'U', o
!>    A = L * L**T,  si UPLO = 'L',
!> donde U es una matriz triangular superior y L es una matriz triangular
!> inferior. La forma factorizada de A se utiliza luego para resolver el sistema de
!> ecuaciones A * X = B.
!>

Parámetros

UPLO 
!>          UPLO es CHARACTER*1
!>          = 'U':  El triángulo superior de A se almacena;
!>          = 'L':  El triángulo inferior de A se almacena.
!>

N

!>          N es INTEGER
!>          El número de ecuaciones lineales, es decir, el orden de la
!>          matriz A.  N >= 0.
!>

NRHS

!>          NRHS es INTEGER
!>          El número de lados derechos, es decir, el número de columnas
!>          de la matriz B.  NRHS >= 0.
!>

A

!>          A es un array REAL, dimensionado (LDA,N)
!>          En la entrada, la matriz simétrica A.  Si UPLO = 'U', la parte
!>          superior triangular N-por-N líder de A contiene la parte
!>          triangular superior de la matriz A, y la parte estrictamente
!>          inferior triangular de A no se referencia. Si UPLO = 'L', la parte
!>          inferior triangular N-por-N líder de A contiene la parte
!>          triangular inferior de la matriz A, y la parte estrictamente
!>          superior triangular de A no se referencia.
!>
!>          En la salida, si INFO = 0, el factor U o L de la descomposición
!>          de Cholesky A = U**T*U o A = L*L**T.
!>

LDA

!>          LDA es INTEGER
!>          La dimensión líder del array A.  LDA >= max(1,N).
!>

B

!>          B es un array REAL, dimensionado (LDB,NRHS)
!>          En la entrada, la matriz B del lado derecho N-por-NRHS.
!>          En la salida, si INFO = 0, la matriz solución X de N-por-NRHS.
!>

LDB

!>          LDB es INTEGER
!>          La dimensión líder del array B.  LDB >= max(1,N).
!>

INFO

!>          INFO es INTEGER
!>          = 0:  salida exitosa
!>          < 0:  si INFO = -i, el i-ésimo argumento tenía un valor ilegal
!>          > 0:  si INFO = i, el menor principal líder de orden i
!>                de A no es positivo, por lo que la factorización no pudo
!>                ser completada, y la solución no ha sido calculada.
!>
