subrutina spbtrs (character uplo, integer n, integer kd, integer nrhs, real, dimension( ldab, * ) ab, integer ldab, real, dimension( ldb, * ) b, integer ldb, integer info)

SPBTRS

Propósito

!>
!> SPBTRS resuelve un sistema de ecuaciones lineales A*X = B con una matriz de banda
!> simétrica positiva definida A utilizando la factorización de Cholesky
!> A = U**T*U o A = L*L**T calculada por SPBTRF.
!>

Parámetros

UPLO 
!>          UPLO es CHARACTER*1
!>          = 'U':  Factor triangular superior almacenado en AB;
!>          = 'L':  Factor triangular inferior almacenado en AB.
!>

N

!>          N es INTEGER
!>          El orden de la matriz A.  N >= 0.
!>

KD

!>          KD es INTEGER
!>          El número de superdiagonales de la matriz A si UPLO = 'U',
!>          o el número de subdiagonales si UPLO = 'L'.  KD >= 0.
!>

NRHS

!>          NRHS es INTEGER
!>          El número de lados derechos, es decir, el número de columnas
!>          de la matriz B.  NRHS >= 0.
!>

AB

!>          AB es array REAL, dimension (LDAB,N)
!>          El factor triangular U o L de la factorización de Cholesky
!>          A = U**T*U o A = L*L**T de la matriz de banda A, almacenado en las
!>          primeras KD+1 filas del array. La j-ésima columna de U o L se
!>          almacena en la j-ésima columna del array AB de la siguiente manera:
!>          si UPLO ='U', AB(kd+1+i-j,j) = U(i,j) para max(1,j-kd)<=i<=j;
!>          si UPLO ='L', AB(1+i-j,j)    = L(i,j) para j<=i<=min(n,j+kd).
!>

LDAB

!>          LDAB es INTEGER
!>          La dimensión principal del array AB.  LDAB >= KD+1.
!>

B

!>          B es array REAL, dimension (LDB,NRHS)
!>          En la entrada, la matriz de lados derechos B.
!>          En la salida, la matriz solución X.
!>

LDB

!>          LDB es INTEGER
!>          La dimensión principal del array B.  LDB >= max(1,N).
!>

INFO

!>          INFO es INTEGER
!>          = 0:  salida exitosa
!>          < 0:  si INFO = -i, el i-ésimo argumento tenía un valor ilegal
!>
