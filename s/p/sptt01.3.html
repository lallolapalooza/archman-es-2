subroutine sptt01 (integer n, real, dimension( * ) d, real, dimension( * ) e, real, dimension( * ) df, real, dimension( * ) ef, real, dimension( * ) work, real resid)

SPTT01

Propósito:

!>
!> SPTT01 reconstruye una matriz tridiagonal A a partir de su
!> factorización L*D*L' y calcula el residuo
!>    norm(L*D*L' - A) / ( n * norm(A) * EPS ),
!> donde EPS es el epsilon de la máquina.
!>

Parámetros

N 
!>          N es ENTERO
!>          El orden de la matriz A.
!>

D

!>          D es un array REAL, dimension (N)
!>          Los n elementos diagonales de la matriz tridiagonal A.
!>

E

!>          E es un array REAL, dimension (N-1)
!>          Los n-1 elementos subdiagonales de la matriz tridiagonal A.
!>

DF

!>          DF es un array REAL, dimension (N)
!>          Los n elementos diagonales del factor L de la factorización L*D*L'
!>          de A.
!>

EF

!>          EF es un array REAL, dimension (N-1)
!>          Los n-1 elementos subdiagonales del factor L de la
!>          factorización L*D*L' de A.
!>

WORK

!>          WORK es un array REAL, dimension (2*N)
!>

RESID

!>          RESID es REAL
!>          norm(L*D*L' - A) / (n * norm(A) * EPS)
!>
