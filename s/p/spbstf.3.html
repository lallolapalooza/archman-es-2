subroutine spbstf (character uplo, integer n, integer kd, real, dimension( ldab, * ) ab, integer ldab, integer info)

SPBSTF

Propósito:

!>
!> SPBSTF calcula una factorización de Cholesky dividida de una matriz de banda
!> real simétrica positiva definida A.
!>
!> Esta rutina está diseñada para ser utilizada junto con SSBGST.
!>
!> La factorización tiene la forma  A = S**T*S  donde S es una matriz de banda
!> del mismo ancho de banda que A y la siguiente estructura:
!>
!>   S = ( U    )
!>       ( M  L )
!>
!> donde U es triangular superior de orden m = (n+kd)/2, y L es triangular inferior
!> de orden n-m.
!>

Parámetros

UPLO 
!>          UPLO es CHARACTER*1
!>          = 'U': El triángulo superior de A está almacenado;
!>          = 'L': El triángulo inferior de A está almacenado.
!>

N

!>          N es INTEGER
!>          El orden de la matriz A.  N >= 0.
!>

KD

!>          KD es INTEGER
!>          El número de superdiagonales de la matriz A si UPLO = 'U',
!>          o el número de subdiagonales si UPLO = 'L'.  KD >= 0.
!>

AB

!>          AB es array REAL, dimensión (LDAB,N)
!>          En la entrada, el triángulo superior o inferior de la matriz de banda
!>          simétrica A, almacenado en las primeras kd+1 filas del array. La
!>          columna j de A se almacena en la columna j del array AB
!>          como sigue:
!>          si UPLO = 'U', AB(kd+1+i-j,j) = A(i,j) para max(1,j-kd)<=i<=j;
!>          si UPLO = 'L', AB(1+i-j,j)    = A(i,j) para j<=i<=min(n,j+kd).
!>
!>          En la salida, si INFO = 0, el factor S de la factorización de Cholesky
!>          dividida A = S**T*S. Ver Detalles Adicionales.
!>

LDAB

!>          LDAB es INTEGER
!>          La dimensión líder del array AB.  LDAB >= KD+1.
!>

INFO

!>          INFO es INTEGER
!>          = 0: salida exitosa
!>          < 0: si INFO = -i, el i-ésimo argumento tuvo un valor ilegal
!>          > 0: si INFO = i, la factorización no pudo completarse,
!>               porque el elemento actualizado a(i,i) fue negativo; la
!>               matriz A no es positiva definida.
!>

Detalles Adicionales:

!>
!>  El esquema de almacenamiento de banda se ilustra con el siguiente ejemplo,
!>  cuando N = 7, KD = 2:
!>
!>  S = ( s11  s12  s13                     )
!>      (      s22  s23  s24                )
!>      (           s33  s34                )
!>      (                s44                )
!>      (           s53  s54  s55           )
!>      (                s64  s65  s66      )
!>      (                     s75  s76  s77 )
!>
!>  Si UPLO = 'U', el array AB contiene:
!>
!>  en la entrada:                    en la salida:
!>
!>   *    *   a13  a24  a35  a46  a57   *    *   s13  s24  s53  s64  s75
!>   *   a12  a23  a34  a45  a56  a67   *   s12  s23  s34  s54  s65  s76
!>  a11  a22  a33  a44  a55  a66  a77  s11  s22  s33  s44  s55  s66  s77
!>
!>  Si UPLO = 'L', el array AB contiene:
!>
!>  en la entrada:                    en la salida:
!>
!>  a11  a22  a33  a44  a55  a66  a77  s11  s22  s33  s44  s55  s66  s77
!>  a21  a32  a43  a54  a65  a76   *   s12  s23  s34  s54  s65  s76   *
!>  a31  a42  a53  a64  a64   *    *   s13  s24  s53  s64  s75   *    *
!>
!>  Los elementos del array marcados con * no son utilizados por la rutina.
!>
