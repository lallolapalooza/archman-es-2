subroutine spbcon (character uplo, integer n, integer kd, real, dimension( ldab, * ) ab, integer ldab, real anorm, real rcond, real, dimension( * ) work, integer, dimension( * ) iwork, integer info)

SPBCON

Propósito:

!>
!> SPBCON estima la inversa del número de condición (en la
!> norma 1) de una matriz de banda simétrica real y definida positiva
!> utilizando la factorización de Cholesky A = U**T*U o A = L*L**T calculada por SPBTRF.
!>
!> Se obtiene una estimación para norm(inv(A)), y la inversa del número de condición se
!> calcula como RCOND = 1 / (ANORM * norm(inv(A))).
!>

Parámetros

UPLO 
!>          UPLO es CHARACTER*1
!>          = 'U':  Factor triangular superior almacenado en AB;
!>          = 'L':  Factor triangular inferior almacenado en AB.
!>

N

!>          N es INTEGER
!>          El orden de la matriz A.  N >= 0.
!>

KD

!>          KD es INTEGER
!>          El número de superdiagonales de la matriz A si UPLO = 'U',
!>          o el número de subdiagonales si UPLO = 'L'.  KD >= 0.
!>

AB

!>          AB es array REAL, dimensión (LDAB,N)
!>          El factor triangular U o L de la factorización de Cholesky
!>          A = U**T*U o A = L*L**T de la matriz de banda A, almacenado en las
!>          primeras KD+1 filas del array.  La j-ésima columna de U o L se
!>          almacena en la j-ésima columna del array AB de la siguiente manera:
!>          si UPLO ='U', AB(kd+1+i-j,j) = U(i,j) para max(1,j-kd)<=i<=j;
!>          si UPLO ='L', AB(1+i-j,j)    = L(i,j) para j<=i<=min(n,j+kd).
!>

LDAB

!>          LDAB es INTEGER
!>          La dimensión principal del array AB.  LDAB >= KD+1.
!>

ANORM

!>          ANORM es REAL
!>          La norma 1 (o norma infinito) de la matriz de banda simétrica A.
!>

RCOND

!>          RCOND es REAL
!>          La inversa del número de condición de la matriz A,
!>          calculada como RCOND = 1/(ANORM * AINVNM), donde AINVNM es una
!>          estimación de la norma 1 de inv(A) calculada en esta rutina.
!>

WORK

!>          WORK es array REAL, dimensión (3*N)
!>

IWORK

!>          IWORK es array INTEGER, dimensión (N)
!>

INFO

!>          INFO es INTEGER
!>          = 0:  salida exitosa
!>          < 0:  si INFO = -i, el i-ésimo argumento tenía un valor ilegal
!>
