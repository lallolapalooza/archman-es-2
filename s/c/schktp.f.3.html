subrutina schktp (logical, dimension( * ) dotype, integer nn, integer, dimension( * ) nval, integer nns, integer, dimension( * ) nsval, real thresh, logical tsterr, integer nmax, real, dimension( * ) ap, real, dimension( * ) ainvp, real, dimension( * ) b, real, dimension( * ) x, real, dimension( * ) xact, real, dimension( * ) work, real, dimension( * ) rwork, integer, dimension( * ) iwork, integer nout)

SCHKTP

Propósito:

!>
!> SCHKTP prueba STPTRI, -TRS, -RFS, y -CON, y SLATPS
!>

Parámetros

DOTYPE 
!>          DOTYPE es un array LOGICAL, de dimensión (NTYPES)
!>          Los tipos de matriz a utilizar para las pruebas. Las matrices de tipo j
!>          (para 1 <= j <= NTYPES) se utilizan para las pruebas si DOTYPE(j) =
!>          .TRUE.; si DOTYPE(j) = .FALSE., entonces el tipo j no se utiliza.
!>

NN

!>          NN es INTEGER
!>          El número de valores de N contenidos en el vector NVAL.
!>

NVAL

!>          NVAL es un array INTEGER, de dimensión (NN)
!>          Los valores de la dimensión de columna de la matriz N.
!>

NNS

!>          NNS es INTEGER
!>          El número de valores de NRHS contenidos en el vector NSVAL.
!>

NSVAL

!>          NSVAL es un array INTEGER, de dimensión (NNS)
!>          Los valores del número de lados derechos NRHS.
!>

THRESH

!>          THRESH es REAL
!>          El valor umbral para las relaciones de prueba. Un resultado se
!>          incluye en el archivo de salida si RESULT >= THRESH. Para que
!>          cada relación de prueba se imprima, use THRESH = 0.
!>

TSTERR

!>          TSTERR es LOGICAL
!>          Indicador que señala si las salidas de error deben ser probadas.
!>

NMAX

!>          NMAX es INTEGER
!>          La dimensión principal de los arrays de trabajo. NMAX >= el
!>          valor máximo de N en NVAL.
!>

AP

!>          AP es un array REAL, de dimensión
!>                      (NMAX*(NMAX+1)/2)
!>

AINVP

!>          AINVP es un array REAL, de dimensión
!>                      (NMAX*(NMAX+1)/2)
!>

B

!>          B es un array REAL, de dimensión (NMAX*NSMAX)
!>          donde NSMAX es la entrada más grande en NSVAL.
!>

X

!>          X es un array REAL, de dimensión (NMAX*NSMAX)
!>

XACT

!>          XACT es un array REAL, de dimensión (NMAX*NSMAX)
!>

WORK

!>          WORK es un array REAL, de dimensión
!>                      (NMAX*max(3,NSMAX))
!>

IWORK

!>          IWORK es un array INTEGER, de dimensión (NMAX)
!>

RWORK

!>          RWORK es un array REAL, de dimensión
!>                      (max(NMAX,2*NSMAX))
!>

NOUT

!>          NOUT es INTEGER
!>          El número de unidad para la salida.
!>
