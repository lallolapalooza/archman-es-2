subroutine schkps (logical, dimension( * ) dotype, integer nn, integer, dimension( * ) nval, integer nnb, integer, dimension( * ) nbval, integer nrank, integer, dimension( * ) rankval, real thresh, logical tsterr, integer nmax, real, dimension( * ) a, real, dimension( * ) afac, real, dimension( * ) perm, integer, dimension( * ) piv, real, dimension( * ) work, real, dimension( * ) rwork, integer nout)

SCHKPS

Propósito:

!>
!> SCHKPS prueba SPSTRF.
!>

Parámetros

DOTYPE 
!>          DOTYPE es un array LOGICAL, de dimensión (NTYPES)
!>          Los tipos de matriz que se utilizarán para las pruebas. Las matrices de tipo j
!>          (para 1 <= j <= NTYPES) se utilizan para las pruebas si DOTYPE(j) =
!>          .TRUE.; si DOTYPE(j) = .FALSE., entonces el tipo j no se utiliza.
!>

NN

!>          NN es INTEGER
!>          El número de valores de N contenidos en el vector NVAL.
!>

NVAL

!>          NVAL es un array INTEGER, de dimensión (NN)
!>          Los valores de la dimensión de la matriz N.
!>

NNB

!>          NNB es INTEGER
!>          El número de valores de NB contenidos en el vector NBVAL.
!>

NBVAL

!>          NBVAL es un array INTEGER, de dimensión (NNB)
!>          Los valores del tamaño del bloque NB.
!>

NRANK

!>          NRANK es INTEGER
!>          El número de valores de RANK contenidos en el vector RANKVAL.
!>

RANKVAL

!>          RANKVAL es un array INTEGER, de dimensión (NBVAL)
!>          Los valores del tamaño del bloque NB.
!>

THRESH

!>          THRESH es REAL
!>          El valor umbral para las relaciones de prueba. Un resultado se
!>          incluye en el archivo de salida si RESULT >= THRESH. Para que
!>          cada relación de prueba se imprima, use THRESH = 0.
!>

TSTERR

!>          TSTERR es LOGICAL
!>          Indicador que señala si se deben probar las salidas de error.
!>

NMAX

!>          NMAX es INTEGER
!>          El valor máximo permitido para N, utilizado en la dimensionamiento de los
!>          arrays de trabajo.
!>

A

!>          A es un array REAL, de dimensión (NMAX*NMAX)
!>

AFAC

!>          AFAC es un array REAL, de dimensión (NMAX*NMAX)
!>

PERM

!>          PERM es un array REAL, de dimensión (NMAX*NMAX)
!>

PIV

!>          PIV es un array INTEGER, de dimensión (NMAX)
!>

WORK

!>          WORK es un array REAL, de dimensión (NMAX*3)
!>

RWORK

!>          RWORK es un array REAL, de dimensión (NMAX)
!>

NOUT

!>          NOUT es INTEGER
!>          El número de unidad para la salida.
!>
