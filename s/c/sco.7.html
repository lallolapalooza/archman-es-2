sco(7) Administración del sistema Linux sco(7)

Nombre

sco - transporte SCO

SINOPSIS

#include <sys/socket.h>
#include <bluetooth/bluetooth.h>
#include <bluetooth/sco.h>
sco_socket = socket(PF_BLUETOOTH, SOCK_SEQPACKET, BTPROTO_SCO);

DESCRIPCIÓN

El transporte lógico SCO es un transporte simétrico punto a punto entre el Central y un Periférico específico. El transporte lógico SCO reserva espacios (slots) y, por lo tanto, puede considerarse una conexión conmutada de circuitos entre el Central y el Periférico.

Además de los espacios reservados, cuando se soporta eSCO, le sigue inmediatamente una ventana de retransmisión. Juntos, los espacios reservados y la ventana de retransmisión forman la ventana eSCO completa.

DIRECCIÓN DE SOCKET

struct sockaddr_sco {
    sa_family_t     sco_family;
    bdaddr_t        sco_bdaddr;
};

Ejemplo:

struct sockaddr_sco addr;
memset(&addr, 0, sizeof(addr));
addr.sco_family = AF_BLUETOOTH;
bacpy(&addr.sco_bdaddr, bdaddr);

OPCIONES DE SOCKET (SOL_BLUETOOTH)

Las opciones de socket que se enumeran a continuación se pueden establecer usando setsockopt(2) y leer con getsockopt(2) con el nivel de socket establecido en SOL_BLUETOOTH.

BT_SECURITY (desde Linux 2.6.30)

Nivel de seguridad del canal, valores posibles:

Valor Nivel de Seguridad Tipo de Clave de Enlace Cifrado
BT_SECURITY_SDP 0 (Solo SDP) Ninguno No requerido
BT_SECURITY_LOW 1 (Bajo) No autenticado No requerido
BT_SECURITY_MEDIUM 2 (Medio - por defecto) No autenticado Deseado
BT_SECURITY_HIGH 3 (Alto) Autenticado Requerido
BT_SECURITY_FIPS (desde Linux 3.15) 4 (Solo Seguro) Autenticado (Emparejamiento Simple Seguro basado en P-256 y Autenticación Segura) Requerido

Ejemplo:

int level = BT_SECURITY_HIGH;
int err = setsockopt(sco_socket, SOL_BLUETOOTH, BT_SECURITY, &level,
                     sizeof(level));
if (err == -1) {
    perror("setsockopt");
    return 1;
}

BT_DEFER_SETUP (desde Linux 2.6.30)

Configuración diferida de la conexión del canal, esto controla si el procedimiento de conexión necesita ser autorizado por el espacio de usuario antes de responder, lo que permite la autorización a nivel de perfil, valores posibles:

Valor Descripción Autorización
0 Desactivar (por defecto) No requerida
1 Activar Requerida

Ejemplo:

int defer_setup = 1;
int err = setsockopt(sco_socket, SOL_BLUETOOTH, BT_DEFER_SETUP,
                     &defer_setup, sizeof(defer_setup));
if (err == -1) {
    perror("setsockopt");
    return err;
}
err = listen(sco_socket, 5);
if (err) {
    perror("listen");
    return err;
}
struct sockaddr_sco remote_addr = {0};
socklen_t addr_len = sizeof(remote_addr);
int new_socket = accept(sco_socket, (struct sockaddr*)&remote_addr,
                        &addr_len);
if (new_socket < 0) {
    perror("accept");
    return new_socket;
}
/* Para completar la configuración de la conexión de new_socket, lea 1 byte */
char c;
struct pollfd pfd;
memset(&pfd, 0, sizeof(pfd));
pfd.fd = new_socket;
pfd.events = POLLOUT;
err = poll(&pfd, 1, 0);
if (err) {
    perror("poll");
    return err;
}
if (!(pfd.revents & POLLOUT)) {
    err = read(sk, &c, 1);
    if (err < 0) {
        perror("read");
        return err;
    }
}

BT_VOICE (desde Linux 3.11)

Configuración de voz del transporte, valores posibles:

struct bt_voice {
    uint16_t setting;
};
Definir Valor Descripción
BT_VOICE_TRANSPARENT 0x0003 Salida transparente
BT_VOICE_CVSD_16BIT 0x0060 Entrada PCM de 16 bits de salida C-VSD
BT_VOICE_TRANSPARENT_16BIT 0x0063 Entrada PCM de 16 bits de salida transparente

Ejemplo:

struct bt_voice voice;
memset(&voice, 0, sizeof(voice));
voice.setting = BT_VOICE_TRANSPARENT;
int err = setsockopt(sco_socket, SOL_BLUETOOTH, BT_VOICE, &voice,
                     sizeof(voice));
if (err == -1) {
    perror("setsockopt");
    return 1;
}

BT_PKT_STATUS (desde Linux 5.9)

Habilitar la notificación del estado del paquete a través de BT_SCM_PKT_STATUS CMSG en los paquetes recibidos. Valores posibles:

Valor Descripción
0 Desactivar (por defecto)
1 Activar
BT_SCM_PKT_STATUS
Nivel SOL_BLUETOOTH CMSG con datos:
uint8_t pkt_status;

Los valores son iguales a "Packet_Status_Flag" definido en la Especificación Central v6.0 Sec. 5.4.3 pág. 1877:

pkt_status Descripción
0x0 Datos recibidos correctamente
0x1 Datos posiblemente inválidos
0x2 No se recibieron datos
0x3 Datos parcialmente perdidos

BT_PHY (desde Linux 5.10)

Transporte de PHY(s) compatibles, solo lectura (sin soporte para setsockopt). Posibles valores:

Define Valor Descripción
BT_PHY_BR_1M_1SLOT BIT 0 BR 1Mbps 1SLOT
BT_PHY_BR_1M_3SLOT BIT 1 BR 1Mbps 3SLOT
BT_PHY_BR_2M_1SLOT BIT 3 EDR 2Mbps 1SLOT
BT_PHY_BR_2M_3SLOT BIT 4 EDR 2Mbps 3SLOT
BT_PHY_BR_3M_1SLOT BIT 6 EDR 3Mbps 1SLOT
BT_PHY_BR_3M_3SLOT BIT 7 EDR 3Mbps 3SLOT

BT_CODEC (desde Linux 5.14)

Descarga de códec de transporte, posibles valores:

struct bt_codec {
    uint8_t id;
    uint16_t cid;
    uint16_t vid;
    uint8_t data_path_id;
    uint8_t num_caps;
    struct codec_caps {
        uint8_t len;
        uint8_t data[];
    } caps[];
} __attribute__((packed));
struct bt_codecs {
    uint8_t num_codecs;
    struct bt_codec codecs[];
} __attribute__((packed));

Ejemplo:

char buffer[sizeof(struct bt_codecs) + sizeof(struct bt_codec)];
struct bt_codec *codecs = (void *)buffer;
memset(codecs, 0, sizeof(codecs));
codec->num_codecs = 1;
codecs->codecs[0].id = 0x05;
codecs->codecs[0].data_path_id = 1;
int err = setsockopt(sco_socket, SOL_BLUETOOTH, BT_CODEC, codecs,
                     sizeof(buffer));
if (err == -1) {
    perror("setsockopt");
    return 1;
}

OPCIONES DE SOCKET (SOL_SOCKET)

Las opciones de socket de nivel SOL_SOCKET que modifican las características genéricas del socket (SO_SNDBUF, SO_RCVBUF, etc.) tienen su significado habitual, véase socket(7).

Las opciones de socket SCO de nivel SOL_SOCKET que tienen manejo específico de Bluetooth en el kernel se enumeran a continuación.

SO_TIMESTAMPING, SO_TIMESTAMP, SO_TIMESTAMPNS

Ver https://docs.kernel.org/networking/timestamping.html

Para sockets SCO, se admiten marcas de tiempo de recepción (RX) por software. Las marcas de tiempo de transmisión (TX) (SOF_TIMESTAMPING_TX_SOFTWARE) se admiten desde Linux 6.15.

La marca de tiempo de recepción (RX) por software es el momento en que el kernel recibió el paquete del controlador.

La marca de tiempo SCM_TSTAMP_SND se emite cuando el paquete se envía al controlador.

La marca de tiempo SCM_TSTAMP_COMPLETION se emite cuando el controlador informa de que el paquete se completó. Las marcas de tiempo de finalización solo se admiten en controladores que tienen control de flujo SCO. Otros tipos de marcas de tiempo de transmisión no son compatibles.

Puede usar SIOCETHTOOL para consultar las banderas admitidas.

Las marcas de tiempo están en el tiempo CLOCK_REALTIME.

Ejemplo (Habilitar marcas de tiempo de recepción (RX)):

int flags = SOF_TIMESTAMPING_SOFTWARE |
    SOF_TIMESTAMPING_RX_SOFTWARE;
setsockopt(fd, SOL_SOCKET, SO_TIMESTAMPING, &flags, sizeof(flags));

Ejemplo (Leer paquete y su marca de tiempo de recepción (RX)):

char data_buf[256];
union {
    char buf[CMSG_SPACE(sizeof(struct scm_timestamping))];
    struct cmsghdr align;
} control;
struct iovec data = {
    .iov_base = data_buf,
    .iov_len = sizeof(data_buf),
};
struct msghdr msg = {
    .msg_iov = &data,
    .msg_iovlen = 1,
    .msg_control = control.buf,
    .msg_controllen = sizeof(control.buf),
};
struct scm_timestamping tss;
res = recvmsg(fd, &msg, MSG_ERRQUEUE | MSG_DONTWAIT);
if (res < 0)
    goto error;
for (cmsg = CMSG_FIRSTHDR(&msg); cmsg; cmsg = CMSG_NXTHDR(&msg, cmsg)) {
    if (cmsg->cmsg_level == SOL_SOCKET && cmsg->cmsg_type == SCM_TIMESTAMPING)
        memcpy(&tss, CMSG_DATA(cmsg), sizeof(tss));
}
tstamp_clock_realtime = tss.ts[0];

Ejemplo (Habilitar marcas de tiempo de transmisión (TX)):

int flags = SOF_TIMESTAMPING_SOFTWARE |
    SOF_TIMESTAMPING_TX_SOFTWARE |
    SOF_TIMESTAMPING_OPT_ID;
setsockopt(fd, SOL_SOCKET, SO_TIMESTAMPING, &flags, sizeof(flags));

Ejemplo (Leer marcas de tiempo de transmisión (TX)):

union {
    char buf[CMSG_SPACE(sizeof(struct scm_timestamping))];
    struct cmsghdr align;
} control;
struct iovec data = {
    .iov_base = NULL,
    .iov_len = 0
};
struct msghdr msg = {
    .msg_iov = &data,
    .msg_iovlen = 1,
    .msg_control = control.buf,
    .msg_controllen = sizeof(control.buf),
};
struct cmsghdr *cmsg;
struct scm_timestamping tss;
struct sock_extended_err serr;
int res;
res = recvmsg(fd, &msg, MSG_ERRQUEUE | MSG_DONTWAIT);
if (res < 0)
    goto error;
for (cmsg = CMSG_FIRSTHDR(&msg); cmsg; cmsg = CMSG_NXTHDR(&msg, cmsg)) {
    if (cmsg->cmsg_level == SOL_SOCKET && cmsg->cmsg_type == SCM_TIMESTAMPING)
        memcpy(&tss, CMSG_DATA(cmsg), sizeof(tss));
    else if (cmsg->cmsg_level == SOL_BLUETOOTH && cmsg->cmsg_type == BT_SCM_ERROR)
        memcpy(&serr, CMSG_DATA(cmsg), sizeof(serr));
}
tstamp_clock_realtime = tss.ts[0];
tstamp_type = serr->ee_info;      /* SCM_TSTAMP_SND o SCM_TSTAMP_COMPLETION */
tstamp_seqnum = serr->ee_data;

IOCTLS

Los siguientes ioctls con operación específica para sockets SCO están disponibles.

SIOCETHTOOL (desde Linux 6.16-rc1)

Solo admite el comando ETHTOOL_GET_TS_INFO, que puede usarse para consultar las banderas SOF_TIMESTAMPING_* admitidas. Las banderas SOF_TIMESTAMPING_OPT_* siempre están disponibles según corresponda.

Ejemplo:

#include <linux/ethtool.h>
#include <linux/sockios.h>
#include <net/if.h>
#include <sys/socket.h>
#include <sys/ioctl.h>
...
struct ifreq ifr = {};
struct ethtool_ts_info cmd = {};
int sk;
snprintf(ifr.ifr_name, sizeof(ifr.ifr_name), "hci0");
ifr.ifr_data = (void *)&cmd;
cmd.cmd = ETHTOOL_GET_TS_INFO;
sk = socket(PF_BLUETOOTH, SOCK_SEQPACKET, BTPROTO_SCO);
if (sk < 0)
    goto error;
if (ioctl(sk, SIOCETHTOOL, &ifr))
    goto error;
sof_available = cmd.so_timestamping;
