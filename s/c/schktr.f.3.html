subrutina schktr (lógico, dimensión( * ) dotype, entero nn, entero, dimensión( * ) nval, entero nnb, entero, dimensión( * ) nbval, entero nns, entero, dimensión( * ) nsval, real thresh, lógico tsterr, entero nmax, real, dimensión( * ) a, real, dimensión( * ) ainv, real, dimensión( * ) b, real, dimensión( * ) x, real, dimensión( * ) xact, real, dimensión( * ) work, real, dimensión( * ) rwork, entero, dimensión( * ) iwork, entero nout)

SCHKTR

Propósito:

!>
!> SCHKTR prueba STRTRI, -TRS, -RFS, y -CON, y SLATRS(3)
!>

Parámetros

DOTYPE 
!>          DOTYPE es un array LOGICAL, dimensión (NTYPES)
!>          Los tipos de matrices a utilizar para la prueba. Las matrices de tipo j
!>          (para 1 <= j <= NTYPES) se utilizan para la prueba si DOTYPE(j) =
!>          .TRUE.; si DOTYPE(j) = .FALSE., entonces el tipo j no se utiliza.
!>

NN

!>          NN es INTEGER
!>          El número de valores de N contenidos en el vector NVAL.
!>

NVAL

!>          NVAL es un array INTEGER, dimensión (NN)
!>          Los valores de la dimensión de columna de la matriz N.
!>

NNB

!>          NNB es INTEGER
!>          El número de valores de NB contenidos en el vector NBVAL.
!>

NBVAL

!>          NBVAL es un array INTEGER, dimensión (NNB)
!>          Los valores del tamaño del bloque NB.
!>

NNS

!>          NNS es INTEGER
!>          El número de valores de NRHS contenidos en el vector NSVAL.
!>

NSVAL

!>          NSVAL es un array INTEGER, dimensión (NNS)
!>          Los valores del número de lados derechos NRHS.
!>

THRESH

!>          THRESH es REAL
!>          El valor umbral para las relaciones de prueba. Un resultado se
!>          incluye en el archivo de salida si RESULT >= THRESH. Para que
!>          cada relación de prueba se imprima, use THRESH = 0.
!>

TSTERR

!>          TSTERR es LOGICAL
!>          Indicador que señala si se deben probar las salidas de error.
!>

NMAX

!>          NMAX es INTEGER
!>          La dimensión principal de los arrays de trabajo.
!>          NMAX >= el valor máximo de N en NVAL.
!>

A

!>          A es un array REAL, dimensión (NMAX*NMAX)
!>

AINV

!>          AINV es un array REAL, dimensión (NMAX*NMAX)
!>

B

!>          B es un array REAL, dimensión (NMAX*NSMAX)
!>          donde NSMAX es la entrada más grande en NSVAL.
!>

X

!>          X es un array REAL, dimensión (NMAX*NSMAX)
!>

XACT

!>          XACT es un array REAL, dimensión (NMAX*NSMAX)
!>

WORK

!>          WORK es un array REAL, dimensión
!>                      (NMAX*max(3,NSMAX))
!>

RWORK

!>          RWORK es un array REAL, dimensión
!>                      (max(NMAX,2*NSMAX))
!>

IWORK

!>          IWORK es un array INTEGER, dimensión (NMAX)
!>

NOUT

!>          NOUT es INTEGER
!>          La unidad de salida.
!>
