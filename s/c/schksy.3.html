subroutine schksy (logical, dimension( * ) dotype, integer nn, integer, dimension( * ) nval, integer nnb, integer, dimension( * ) nbval, integer nns, integer, dimension( * ) nsval, real thresh, logical tsterr, integer nmax, real, dimension( * ) a, real, dimension( * ) afac, real, dimension( * ) ainv, real, dimension( * ) b, real, dimension( * ) x, real, dimension( * ) xact, real, dimension( * ) work, real, dimension( * ) rwork, integer, dimension( * ) iwork, integer nout)

SCHKSY

Propósito:

!>
!> SCHKSY prueba SSYTRF, -TRI2, -TRS, -TRS2, -RFS y -CON.
!>

Parámetros

DOTYPE 
!>          DOTYPE es un array LOGICAL, dimension (NTYPES)
!>          Los tipos de matriz a utilizar para las pruebas. Las matrices de tipo j
!>          (para 1 <= j <= NTYPES) se utilizan para las pruebas si DOTYPE(j) =
!>          .TRUE.; si DOTYPE(j) = .FALSE., entonces el tipo j no se utiliza.
!>

NN

!>          NN es un INTEGER
!>          El número de valores de N contenidos en el vector NVAL.
!>

NVAL

!>          NVAL es un array INTEGER, dimension (NN)
!>          Los valores de la dimensión de la matriz N.
!>

NNB

!>          NNB es un INTEGER
!>          El número de valores de NB contenidos en el vector NBVAL.
!>

NBVAL

!>          NBVAL es un array INTEGER, dimension (NNB)
!>          Los valores del tamaño del bloque NB.
!>

NNS

!>          NNS es un INTEGER
!>          El número de valores de NRHS contenidos en el vector NSVAL.
!>

NSVAL

!>          NSVAL es un array INTEGER, dimension (NNS)
!>          Los valores del número de lados derechos NRHS.
!>

THRESH

!>          THRESH es un REAL
!>          El valor umbral para las ratios de prueba. Un resultado se
!>          incluye en el archivo de salida si RESULT >= THRESH. Para que
!>          se imprima cada ratio de prueba, use THRESH = 0.
!>

TSTERR

!>          TSTERR es LOGICAL
!>          Indicador que señala si se deben probar las salidas de error.
!>

NMAX

!>          NMAX es un INTEGER
!>          El valor máximo permitido para N, utilizado en la dimensionación de los
!>          arrays de trabajo.
!>

A

!>          A es un array REAL, dimension (NMAX*NMAX)
!>

AFAC

!>          AFAC es un array REAL, dimension (NMAX*NMAX)
!>

AINV

!>          AINV es un array REAL, dimension (NMAX*NMAX)
!>

B

!>          B es un array REAL, dimension (NMAX*NSMAX)
!>          donde NSMAX es la entrada más grande en NSVAL.
!>

X

!>          X es un array REAL, dimension (NMAX*NSMAX)
!>

XACT

!>          XACT es un array REAL, dimension (NMAX*NSMAX)
!>

WORK

!>          WORK es un array REAL, dimension (NMAX*max(3,NSMAX))
!>

RWORK

!>          RWORK es un array REAL, dimension (max(NMAX,2*NSMAX))
!>

IWORK

!>          IWORK es un array INTEGER, dimension (2*NMAX)
!>

NOUT

!>          NOUT es un INTEGER
!>          El número de unidad para la salida.
!>
