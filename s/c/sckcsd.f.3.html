subroutine sckcsd (nm entero, entero, dimension( * ) mval, entero, dimension( * ) pval, entero, dimension( * ) qval, nmats entero, iseed entero, dimension( 4 ) thresh real, mmax entero, x real, dimension( * ) xf real, dimension( * ) u1 real, dimension( * ) u2 real, dimension( * ) v1t real, dimension( * ) v2t real, dimension( * ) theta real, dimension( * ) iwork entero, dimension( * ) work real, dimension( * ) rwork real, nin entero, nout entero, info entero)

SCKCSD

Propósito:

!>
!> SCKCSD prueba SORCSD:
!>        el CSD para una matriz ortogonal M-por-M X particionada como
!>        [ X11 X12; X21 X22 ]. X11 es P-por-Q.
!>

Parámetros

NM 
!>          NM es ENTERO
!>          El número de valores de M contenidos en el vector MVAL.
!>

MVAL

!>          MVAL es un array de ENTEROS, dimensionado (NM)
!>          Los valores de la dimensión de fila de la matriz M.
!>

PVAL

!>          PVAL es un array de ENTEROS, dimensionado (NM)
!>          Los valores de la dimensión de fila de la matriz P.
!>

QVAL

!>          QVAL es un array de ENTEROS, dimensionado (NM)
!>          Los valores de la dimensión de columna de la matriz Q.
!>

NMATS

!>          NMATS es ENTERO
!>          El número de tipos de matriz a probar para cada combinación
!>          de dimensiones de matriz. Si NMATS >= NTYPES (el número máximo
!>          de tipos de matriz), entonces se generan todos los tipos diferentes
!>          para la prueba. Si NMATS < NTYPES, se lee otra línea de entrada
!>          para obtener los números de los tipos de matriz a utilizar.
!>

ISEED

!>          ISEED es un array de ENTEROS, dimensionado (4)
!>          En la entrada, la semilla del generador de números aleatorios. Los
!>          elementos del array deben estar entre 0 y 4095, de lo contrario
!>          serán reducidos módulo 4096, y ISEED(4) debe ser impar.
!>          En la salida, la siguiente semilla en la secuencia de números
!>          aleatorios después de que todas las matrices de prueba han sido generadas.
!>

THRESH

!>          THRESH es REAL
!>          El valor umbral para las relaciones de prueba. Un resultado se
!>          incluye en el archivo de salida si RESULT >= THRESH. Para que
!>          se imprima cada relación de prueba, use THRESH = 0.
!>

MMAX

!>          MMAX es ENTERO
!>          El valor máximo permitido para M, utilizado en la dimensionamiento
!>          de los arrays de trabajo.
!>

X

!>          X es un array REAL, dimensionado (MMAX*MMAX)
!>

XF

!>          XF es un array REAL, dimensionado (MMAX*MMAX)
!>

U1

!>          U1 es un array REAL, dimensionado (MMAX*MMAX)
!>

U2

!>          U2 es un array REAL, dimensionado (MMAX*MMAX)
!>

V1T

!>          V1T es un array REAL, dimensionado (MMAX*MMAX)
!>

V2T

!>          V2T es un array REAL, dimensionado (MMAX*MMAX)
!>

THETA

!>          THETA es un array REAL, dimensionado (MMAX)
!>

IWORK

!>          IWORK es un array de ENTEROS, dimensionado (MMAX)
!>

WORK

!>          WORK es un array REAL
!>

RWORK

!>          RWORK es un array REAL
!>

NIN

!>          NIN es ENTERO
!>          El número de unidad para la entrada.
!>

NOUT

!>          NOUT es ENTERO
!>          El número de unidad para la salida.
!>

INFO

!>          INFO es ENTERO
!>          = 0 :  Salida exitosa
!>          > 0 :  Si SLAROR devuelve un código de error, se devuelve su
!>                 valor absoluto.
!>
