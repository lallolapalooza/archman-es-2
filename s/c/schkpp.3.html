subroutine schkpp (logical, dimension( * ) dotype, integer nn, integer, dimension( * ) nval, integer nns, integer, dimension( * ) nsval, real thresh, logical tsterr, integer nmax, real, dimension( * ) a, real, dimension( * ) afac, real, dimension( * ) ainv, real, dimension( * ) b, real, dimension( * ) x, real, dimension( * ) xact, real, dimension( * ) work, real, dimension( * ) rwork, integer, dimension( * ) iwork, integer nout)

SCHKPP

Propósito:

!>
!> SCHKPP prueba SPPTRF, -TRI, -TRS, -RFS y -CON
!>

Parámetros

DOTYPE 
!>          DOTYPE es un array LOGICAL, dimension (NTYPES)
!>          Los tipos de matriz a utilizar para la prueba. Las matrices de tipo j
!>          (para 1 <= j <= NTYPES) se utilizan para la prueba si DOTYPE(j) =
!>          .TRUE.; si DOTYPE(j) = .FALSE., entonces el tipo j no se utiliza.
!>

NN

!>          NN es un ENTERO
!>          El número de valores de N contenidos en el vector NVAL.
!>

NVAL

!>          NVAL es un array de ENTEROS, dimension (NN)
!>          Los valores de la dimensión de la matriz N.
!>

NNS

!>          NNS es un ENTERO
!>          El número de valores de NRHS contenidos en el vector NSVAL.
!>

NSVAL

!>          NSVAL es un array de ENTEROS, dimension (NNS)
!>          Los valores del número de lados derechos NRHS.
!>

THRESH

!>          THRESH es REAL
!>          El valor umbral para las relaciones de prueba. Un resultado se
!>          incluye en el archivo de salida si RESULT >= THRESH. Para tener
!>          cada relación de prueba impresa, use THRESH = 0.
!>

TSTERR

!>          TSTERR es LOGICAL
!>          Indicador que señala si se van a probar las salidas de error.
!>

NMAX

!>          NMAX es un ENTERO
!>          El valor máximo permitido para N, utilizado en la dimensionamiento de los
!>          arrays de trabajo.
!>

A

!>          A es un array REAL, dimension
!>                      (NMAX*(NMAX+1)/2)
!>

AFAC

!>          AFAC es un array REAL, dimension
!>                      (NMAX*(NMAX+1)/2)
!>

AINV

!>          AINV es un array REAL, dimension
!>                      (NMAX*(NMAX+1)/2)
!>

B

!>          B es un array REAL, dimension (NMAX*NSMAX)
!>          donde NSMAX es la entrada más grande en NSVAL.
!>

X

!>          X es un array REAL, dimension (NMAX*NSMAX)
!>

XACT

!>          XACT es un array REAL, dimension (NMAX*NSMAX)
!>

WORK

!>          WORK es un array REAL, dimension
!>                      (NMAX*max(3,NSMAX))
!>

RWORK

!>          RWORK es un array REAL, dimension
!>                      (max(NMAX,2*NSMAX))
!>

IWORK

!>          IWORK es un array de ENTEROS, dimension (NMAX)
!>

NOUT

!>          NOUT es un ENTERO
!>          El número de unidad para la salida.
!>
