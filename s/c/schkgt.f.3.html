subroutine schkgt (logical, dimension( * ) dotype, integer nn, integer, dimension( * ) nval, integer nns, integer, dimension( * ) nsval, real thresh, logical tsterr, real, dimension( * ) a, real, dimension( * ) af, real, dimension( * ) b, real, dimension( * ) x, real, dimension( * ) xact, real, dimension( * ) work, real, dimension( * ) rwork, integer, dimension( * ) iwork, integer nout)

SCHKGT

Propósito:

!>
!> SCHKGT prueba SGTTRF, -TRS, -RFS y -CON
!>

Parámetros

DOTYPE 
!>          DOTYPE es un array LOGICAL, de dimensión (NTYPES)
!>          Los tipos de matriz a utilizar para las pruebas. Las matrices de tipo j
!>          (para 1 <= j <= NTYPES) se utilizan para las pruebas si DOTYPE(j) =
!>          .TRUE.; si DOTYPE(j) = .FALSE., entonces el tipo j no se utiliza.
!>

NN

!>          NN es INTEGER
!>          El número de valores de N contenidos en el vector NVAL.
!>

NVAL

!>          NVAL es un array INTEGER, de dimensión (NN)
!>          Los valores de la dimensión de la matriz N.
!>

NNS

!>          NNS es INTEGER
!>          El número de valores de NRHS contenidos en el vector NSVAL.
!>

NSVAL

!>          NSVAL es un array INTEGER, de dimensión (NNS)
!>          Los valores del número de segundos miembros NRHS.
!>

THRESH

!>          THRESH es REAL
!>          El valor umbral para las relaciones de prueba. Un resultado se
!>          incluye en el archivo de salida si RESULT >= THRESH. Para que
!>          se imprima cada relación de prueba, utilice THRESH = 0.
!>

TSTERR

!>          TSTERR es LOGICAL
!>          Indicador que señala si se deben probar las salidas de error.
!>

A

!>          A es un array REAL, de dimensión (NMAX*4)
!>

AF

!>          AF es un array REAL, de dimensión (NMAX*4)
!>

B

!>          B es un array REAL, de dimensión (NMAX*NSMAX)
!>          donde NSMAX es la entrada más grande en NSVAL.
!>

X

!>          X es un array REAL, de dimensión (NMAX*NSMAX)
!>

XACT

!>          XACT es un array REAL, de dimensión (NMAX*NSMAX)
!>

WORK

!>          WORK es un array REAL, de dimensión
!>                      (NMAX*max(3,NSMAX))
!>

RWORK

!>          RWORK es un array REAL, de dimensión
!>                      (max(NMAX,2*NSMAX))
!>

IWORK

!>          IWORK es un array INTEGER, de dimensión (2*NMAX)
!>

NOUT

!>          NOUT es INTEGER
!>          El número de unidad para la salida.
!>
