subrutina scsdts (integer m, integer p, integer q, real, dimension( ldx, * ) x, real, dimension( ldx, * ) xf, integer ldx, real, dimension( ldu1, * ) u1, integer ldu1, real, dimension( ldu2, * ) u2, integer ldu2, real, dimension( ldv1t, * ) v1t, integer ldv1t, real, dimension( ldv2t, * ) v2t, integer ldv2t, real, dimension( * ) theta, integer, dimension( * ) iwork, real, dimension( lwork ) work, integer lwork, real, dimension( * ) rwork, real, dimension( 15 ) result)

SCSDTS

Propósito:

!>
!> SCSDTS prueba SORCSD, que, dada una matriz ortogonal particionada M-por-M
!> X,
!>              Q  M-Q
!>       X = [ X11 X12 ] P   ,
!>           [ X21 X22 ] M-P
!>
!> calcula la CSD (Decomposición de Matrices Singulares)
!>
!>       [ U1    ]**T * [ X11 X12 ] * [ V1    ]
!>       [    U2 ]      [ X21 X22 ]   [    V2 ]
!>
!>                             [  I  0  0 |  0  0  0 ]
!>                             [  0  C  0 |  0 -S  0 ]
!>                             [  0  0  0 |  0  0 -I ]
!>                           = [---------------------] = [ D11 D12 ] .
!>                             [  0  0  0 |  I  0  0 ]   [ D21 D22 ]
!>                             [  0  S  0 |  0  C  0 ]
!>                             [  0  0  I |  0  0  0 ]
!>
!> y también SORCSD2BY1, que, dada
!>          Q
!>       [ X11 ] P   ,
!>       [ X21 ] M-P
!>
!> calcula la CSD 2x1
!>
!>                                     [  I  0  0 ]
!>                                     [  0  C  0 ]
!>                                     [  0  0  0 ]
!>       [ U1    ]**T * [ X11 ] * V1 = [----------] = [ D11 ] ,
!>       [    U2 ]      [ X21 ]        [  0  0  0 ]   [ D21 ]
!>                                     [  0  S  0 ]
!>                                     [  0  0  I ]
!>

Parámetros

M 
!>          M es INTEGER
!>          El número de filas de la matriz X.  M >= 0.
!>

P

!>          P es INTEGER
!>          El número de filas de la matriz X11.  P >= 0.
!>

Q

!>          Q es INTEGER
!>          El número de columnas de la matriz X11.  Q >= 0.
!>

X

!>          X es un array REAL, dimension (LDX,M)
!>          La matriz M-por-M X.
!>

XF

!>          XF es un array REAL, dimension (LDX,M)
!>          Detalles de la CSD de X, como devuelve SORCSD;
!>          ver SORCSD para más detalles.
!>

LDX

!>          LDX es INTEGER
!>          La dimensión principal de los arrays X y XF.
!>          LDX >= max( 1,M ).
!>

U1

!>          U1 es un array REAL, dimension(LDU1,P)
!>          La matriz ortogonal P-por-P U1.
!>

LDU1

!>          LDU1 es INTEGER
!>          La dimensión principal del array U1. LDU >= max(1,P).
!>

U2

!>          U2 es un array REAL, dimension(LDU2,M-P)
!>          La matriz ortogonal (M-P)-por-(M-P) U2.
!>

LDU2

!>          LDU2 es INTEGER
!>          La dimensión principal del array U2. LDU >= max(1,M-P).
!>

V1T

!>          V1T es un array REAL, dimension(LDV1T,Q)
!>          La matriz ortogonal Q-por-Q V1T.
!>

LDV1T

!>          LDV1T es INTEGER
!>          La dimensión principal del array V1T. LDV1T >=
!>          max(1,Q).
!>

V2T

!>          V2T es un array REAL, dimension(LDV2T,M-Q)
!>          La matriz ortogonal (M-Q)-por-(M-Q) V2T.
!>

LDV2T

!>          LDV2T es INTEGER
!>          La dimensión principal del array V2T. LDV2T >=
!>          max(1,M-Q).
!>

THETA

!>          THETA es un array REAL, dimension MIN(P,M-P,Q,M-Q)
!>          Los valores CS de X; las matrices esencialmente diagonales C y
!>          S se construyen a partir de THETA; ver la subrutina SORCSD para
!>          detalles.
!>

IWORK

!>          IWORK es un array INTEGER, dimension (M)
!>

WORK

!>          WORK es un array REAL, dimension (LWORK)
!>

LWORK

!>          LWORK es INTEGER
!>          La dimensión del array WORK
!>

RWORK

!>          RWORK es un array REAL
!>

RESULT

!>          RESULT es un array REAL, dimension (15)
!>          Las razones de prueba:
!>          Primero, la CSD 2x2:
!>          RESULT(1) = norm( U1'*X11*V1 - D11 ) / ( MAX(1,P,Q)*EPS2 )
!>          RESULT(2) = norm( U1'*X12*V2 - D12 ) / ( MAX(1,P,M-Q)*EPS2 )
!>          RESULT(3) = norm( U2'*X21*V1 - D21 ) / ( MAX(1,M-P,Q)*EPS2 )
!>          RESULT(4) = norm( U2'*X22*V2 - D22 ) / ( MAX(1,M-P,M-Q)*EPS2 )
!>          RESULT(5) = norm( I - U1'*U1 ) / ( MAX(1,P)*ULP )
!>          RESULT(6) = norm( I - U2'*U2 ) / ( MAX(1,M-P)*ULP )
!>          RESULT(7) = norm( I - V1T'*V1T ) / ( MAX(1,Q)*ULP )
!>          RESULT(8) = norm( I - V2T'*V2T ) / ( MAX(1,M-Q)*ULP )
!>          RESULT(9) = 0        si THETA está en orden creciente y
!>                               todos los ángulos están en [0,pi/2]
!>                    = ULPINV   en caso contrario.
!>          Luego, la CSD 2x1:
!>          RESULT(10) = norm( U1'*X11*V1 - D11 ) / ( MAX(1,P,Q)*EPS2 )
!>          RESULT(11) = norm( U2'*X21*V1 - D21 ) / ( MAX(1,M-P,Q)*EPS2 )
!>          RESULT(12) = norm( I - U1'*U1 ) / ( MAX(1,P)*ULP )
!>          RESULT(13) = norm( I - U2'*U2 ) / ( MAX(1,M-P)*ULP )
!>          RESULT(14) = norm( I - V1T'*V1T ) / ( MAX(1,Q)*ULP )
!>          RESULT(15) = 0        si THETA está en orden creciente y
!>                               todos los ángulos están en [0,pi/2]
!>                     = ULPINV   en caso contrario.
!>          ( EPS2 = MAX( norm( I - X'*X ) / M, ULP ). )
!>
