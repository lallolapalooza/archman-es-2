sd_bus_set_watch_bind(3)

Paquete: systemd-libs

Licencias: LGPL-2.1-or-later, CC0-1.0, GPL-2.0-or-later WITH Linux-syscall-note

SD_BUS_SET_WATCH_BIND(3) sd_bus_set_watch_bind SD_BUS_SET_WATCH_BIND(3)

NOMBRE

sd_bus_set_watch_bind, sd_bus_get_watch_bind - Controlar la vigilancia de enlace de socket en conexiones de bus

SINOPSIS

#include <systemd/sd-bus.h>

int sd_bus_set_watch_bind(sd_bus *bus, int b);

int sd_bus_get_watch_bind(sd_bus *bus);

DESCRIPCIÓN

sd_bus_set_watch_bind() puede usarse para habilitar o deshabilitar la vigilancia del lado del cliente del enlace de socket del servidor para un objeto de conexión de bus. Si b es verdadero, la función se habilita; de lo contrario, se deshabilita (que es el comportamiento predeterminado). Cuando está habilitada, y la dirección del bus seleccionada se refiere a un socket AF_UNIX en el sistema de archivos que no existe mientras se intenta la conexión, se instala una vigilancia inotify(7) sobre él, esperando a que aparezca el socket. Tan pronto como aparece el socket, se realiza la conexión. Esta funcionalidad es útil en particular en programas de arranque temprano que necesitan ejecutarse antes de que el bus del sistema esté disponible, pero desean conectarse a él en el instante en que sea posible.

sd_bus_get_watch_bind() puede usarse para consultar la configuración actual de esta función. Devuelve cero cuando la función está deshabilitada y un valor positivo si está habilitada.

Tenga en cuenta que no se aplica ningún tiempo de espera mientras esperamos que aparezca el socket. Esto significa que cualquier operación remota síncrona (como sd_bus_call(3), sd_bus_add_match(3) o sd_bus_request_name(3)), que se utiliza en una conexión con esta función habilitada que aún no se ha establecido, podría bloquearse indefinidamente si el socket nunca se crea. Sin embargo, las operaciones remotas asíncronas (como sd_bus_send(3), sd_bus_call_async(3), sd_bus_add_match_async(3)) no se bloquean en este caso, y ponen en cola de forma segura las operaciones solicitadas para que se despachen en el instante en que se establezca la conexión.

Utilice sd_bus_is_ready(3) para determinar si la conexión está completamente establecida, es decir, si el socket del par ha sido enlazado, conectado y autenticado. Utilice sd_bus_set_connected_signal(3) para ser notificado cuando la conexión esté completamente establecida.

VALOR DE RETORNO

En caso de éxito, estas funciones devuelven 0 o un entero positivo. En caso de error, devuelven un código de error negativo al estilo errno.

Errores

Los errores devueltos pueden indicar los siguientes problemas:

-ECHILD

La conexión del bus se ha creado en un proceso, biblioteca o instancia de módulo diferente.

NOTAS

Las funciones descritas aquí están disponibles como una biblioteca compartida, que se puede compilar y enlazar con el archivo libsystemd pkg-config(1).

El código descrito aquí utiliza getenv(3), que se declara como no seguro para subprocesos. Esto significa que el código que llama a las funciones descritas aquí no debe llamar a setenv(3) desde un hilo paralelo. Se recomienda realizar llamadas a setenv() solo desde una fase temprana del programa cuando no se hayan iniciado otros hilos.

EJEMPLO

Ejemplo 1. Crear un servicio simple del sistema que publica una propiedad en el bus del sistema y puede reconectarse cuando D-Bus se desconecta y reconecta

/* SPDX-License-Identifier: MIT-0 */
/* Un servicio D-Bus que se reconecta automáticamente cuando se reinicia el bus del sistema.
 *
 * Compilar con 'cc sd_bus_service_reconnect.c $(pkg-config --libs --cflags libsystemd)'
 *
 * Para permitir que el programa tome posesión del nombre 'org.freedesktop.ReconnectExample',
 * agregue lo siguiente como /etc/dbus-1/system.d/org.freedesktop.ReconnectExample.conf
 * y luego recargue el broker con 'systemctl reload dbus':
<?xml version="1.0"?> <!--*-nxml-*-->
<!DOCTYPE busconfig PUBLIC "-//freedesktop//DTD D-BUS Bus Configuration 1.0//EN"
  "http://www.freedesktop.org/standards/dbus/1.0/busconfig.dtd">
<busconfig>
  <policy user="root">
    <allow own="org.freedesktop.ReconnectExample"/>
    <allow send_destination="org.freedesktop.ReconnectExample"/>
    <allow receive_sender="org.freedesktop.ReconnectExample"/>
  </policy>
  <policy context="default">
    <allow send_destination="org.freedesktop.ReconnectExample"/>
    <allow receive_sender="org.freedesktop.ReconnectExample"/>
  </policy>
</busconfig>
 *
 * Para obtener la propiedad a través de busctl:
 *
 * $ busctl --user get-property org.freedesktop.ReconnectExample \
 *                              /org/freedesktop/ReconnectExample \
 *                              org.freedesktop.ReconnectExample \
 *                              Example
 * s "example"
 */
#include <errno.h>
#include <stdio.h>
#include <stdlib.h>
#include <systemd/sd-bus.h>
#define _cleanup_(f) __attribute__((cleanup(f)))
static int log_error(int r, const char *str) {
  fprintf(stderr, "%s failed: %s\n", str, strerror(-r));
  return r;
}
typedef struct object {
  const char *example;
  sd_bus **bus;
  sd_event **event;
} object;
static int property_get(
                sd_bus *bus,
                const char *path,
                const char *interface,
                const char *property,
                sd_bus_message *reply,
                void *userdata,
                sd_bus_error *error) {
  object *o = userdata;
  if (strcmp(property, "Example") == 0)
    return sd_bus_message_append(reply, "s", o->example);
  return sd_bus_error_setf(error,
                           SD_BUS_ERROR_UNKNOWN_PROPERTY,
                           "Unknown property '%s'",
                           property);
}
/* https://www.freedesktop.org/software/systemd/man/sd_bus_add_object.html */
static const sd_bus_vtable vtable[] = {
  SD_BUS_VTABLE_START(0),
  SD_BUS_PROPERTY(
    "Example", "s",
    property_get,
    0,
    SD_BUS_VTABLE_PROPERTY_CONST),
  SD_BUS_VTABLE_END
};
static int setup(object *o);
static int on_disconnect(sd_bus_message *message, void *userdata, sd_bus_error *ret_error) {
  int r;
  r = setup((object *)userdata);
  if (r < 0) {
    object *o = userdata;
    r = sd_event_exit(*o->event, r);
    if (r < 0)
      return log_error(r, "sd_event_exit()");
  }
  return 1;
}
/* Asegura que el bucle de eventos salga con un error claro si la adquisición del nombre de servicio conocido falla */
static int request_name_callback(sd_bus_message *m, void *userdata, sd_bus_error *ret_error) {
  int r;
  if (!sd_bus_message_is_method_error(m, NULL))
    return 1;
  const sd_bus_error *error = sd_bus_message_get_error(m);
  if (sd_bus_error_has_names(error, SD_BUS_ERROR_TIMEOUT, SD_BUS_ERROR_NO_REPLY))
    return 1; /* El bus no está disponible, inténtalo de nuevo más tarde */
  fprintf(stderr, "Failed to request name: %s\n", error->message);
  object *o = userdata;
  r = sd_event_exit(*o->event, -sd_bus_error_get_errno(error));
  if (r < 0)
    return log_error(r, "sd_event_exit()");
  return 1;
}
static int setup(object *o) {
  int r;
  /* Si nos estamos reconectando, el objeto del bus debe cerrarse, desvincularse
   * del bucle de eventos y recrearse.
   * https://www.freedesktop.org/software/systemd/man/sd_bus_detach_event.html
   * https://www.freedesktop.org/software/systemd/man/sd_bus_close_unref.html
   */
  if (*o->bus) {
    r = sd_bus_detach_event(*o->bus);
    if (r < 0)
      return log_error(r, "sd_bus_detach_event()");
    *o->bus = sd_bus_close_unref(*o->bus);
  }
  /* Configurar un nuevo objeto de bus para el bus del sistema, configurarlo para esperar a que D-Bus
   * esté disponible en lugar de fallar si no lo está, y ponerlo en marcha. Todas las
   * operaciones siguientes son asíncronas y no se bloquearán esperando a que D-Bus
   * esté disponible.
   * https://www.freedesktop.org/software/systemd/man/sd_bus_new.html
   * https://www.freedesktop.org/software/systemd/man/sd_bus_set_address.html
   * https://www.freedesktop.org/software/systemd/man/sd_bus_set_bus_client.html
   * https://www.freedesktop.org/software/systemd/man/sd_bus_negotiate_creds.html
   * https://www.freedesktop.org/software/systemd/man/sd_bus_set_watch_bind.html
   * https://www.freedesktop.org/software/systemd/man/sd_bus_set_connected_signal.html
   * https://www.freedesktop.org/software/systemd/man/sd_bus_start.html
   */
  r 
= sd_bus_new(o->bus);
  if (r < 0)
    return log_error(r, "sd_bus_new()");
  r = sd_bus_set_address(*o->bus, "unix:path=/run/dbus/system_bus_socket");
  if (r < 0)
    return log_error(r, "sd_bus_set_address()");
  r = sd_bus_set_bus_client(*o->bus, 1);
  if (r < 0)
    return log_error(r, "sd_bus_set_bus_client()");
  r = sd_bus_negotiate_creds(*o->bus, 1, SD_BUS_CREDS_UID|SD_BUS_CREDS_EUID|SD_BUS_CREDS_EFFECTIVE_CAPS);
  if (r < 0)
    return log_error(r, "sd_bus_negotiate_creds()");
  r = sd_bus_set_watch_bind(*o->bus, 1);
  if (r < 0)
    return log_error(r, "sd_bus_set_watch_bind()");
  r = sd_bus_start(*o->bus);
  if (r < 0)
    return log_error(r, "sd_bus_start()");
  /* Publicar una interfaz en el bus, especificando nuestra ruta de acceso de objeto conocida
   * y el nombre de la interfaz pública.
   * https://www.freedesktop.org/software/systemd/man/sd_bus_add_object.html
   * https://dbus.freedesktop.org/doc/dbus-tutorial.html
   */
  r = sd_bus_add_object_vtable(*o->bus,
                               NULL,
                               "/org/freedesktop/ReconnectExample",
                               "org.freedesktop.ReconnectExample",
                               vtable,
                               o);
  if (r < 0)
    return log_error(r, "sd_bus_add_object_vtable()");
  /* Por defecto, al servicio se le asigna solo un nombre efímero. También se le añade
   * uno conocido, para que los clientes sepan a quién llamar. Esto necesita ser
   * asíncrono, ya que D-Bus podría aún no estar disponible. La función de devolución de llamada comprobará
   * si el error es esperado o no, en caso de que falle.
   * https://www.freedesktop.org/software/systemd/man/sd_bus_request_name.html
   */
  r = sd_bus_request_name_async(*o->bus,
                                NULL,
                                "org.freedesktop.ReconnectExample",
                                0,
                                request_name_callback,
                                o);
  if (r < 0)
    return log_error(r, "sd_bus_request_name_async()");
  /* Cuando D-Bus se desconecte, se invocará esta función de devolución de llamada, que configurará
   * la conexión de nuevo. Esto necesita ser asíncrono, ya que D-Bus podría aún no
   * estar disponible.
   * https://www.freedesktop.org/software/systemd/man/sd_bus_match_signal_async.html
   */
  r = sd_bus_match_signal_async(*o->bus,
                                NULL,
                                "org.freedesktop.DBus.Local",
                                NULL,
                                "org.freedesktop.DBus.Local",
                                "Disconnected",
                                on_disconnect,
                                NULL,
                                o);
  if (r < 0)
    return log_error(r, "sd_bus_match_signal_async()");
  /* Adjuntar el objeto bus al bucle de eventos para que se procesen las llamadas y las señales.
   * https://www.freedesktop.org/software/systemd/man/sd_bus_attach_event.html
   */
  r = sd_bus_attach_event(*o->bus, *o->event, 0);
  if (r < 0)
    return log_error(r, "sd_bus_attach_event()");
  return 0;
}
int main(int argc, char **argv) {
  /* El bus debe ser liberado antes de que el programa termine. El atributo de limpieza
   * nos permite hacerlo de manera agradable y limpia cada vez que salimos del bloque.
   */
  _cleanup_(sd_bus_flush_close_unrefp) sd_bus *bus = NULL;
  _cleanup_(sd_event_unrefp) sd_event *event = NULL;
  object o = {
    .example = "example",
    .bus = &bus,
    .event = &event,
  };
  int r;
  /* Crear una estructura de datos de bucle de eventos, con parámetros por defecto.
   * https://www.freedesktop.org/software/systemd/man/sd_event_default.html
   */
  r = sd_event_default(&event);
  if (r < 0)
    return log_error(r, "sd_event_default()");
  /* Por defecto, el bucle de eventos terminará cuando todas las fuentes hayan desaparecido,
   * así que tenemos que mantenerlo 'ocupado'. Registrar el manejo de señales para hacerlo.
   * https://www.freedesktop.org/software/systemd/man/sd_event_add_signal.html
   */
  r = sd_event_add_signal(event, NULL, SIGINT|SD_EVENT_SIGNAL_PROCMASK, NULL, NULL);
  if (r < 0)
    return log_error(r, "sd_event_add_signal(SIGINT)");
  r = sd_event_add_signal(event, NULL, SIGTERM|SD_EVENT_SIGNAL_PROCMASK, NULL, NULL);
  if (r < 0)
    return log_error(r, "sd_event_add_signal(SIGTERM)");
  r = setup(&o);
  if (r < 0)
    return EXIT_FAILURE;
  /* Entrar en el bucle principal, solo saldrá en sigint/sigterm.
   * https://www.freedesktop.org/software/systemd/man/sd_event_loop.html
   */
  r = sd_event_loop(event);
  if (r < 0)
    return log_error(r, "sd_event_loop()");
  /* https://www.freedesktop.org/software/systemd/man/sd_bus_release_name.html */
  r = sd_bus_release_name(bus, "org.freedesktop.ReconnectExample");
  if (r < 0)
    return log_error(r, "sd_bus_release_name()");
  return 0;
}
