subroutine sdrvev (integer nsizes, integer, dimension( * ) nn, integer ntypes, logical, dimension( * ) dotype, integer, dimension( 4 ) iseed, real thresh, integer nounit, real, dimension( lda, * ) a, integer lda, real, dimension( lda, * ) h, real, dimension( * ) wr, real, dimension( * ) wi, real, dimension( * ) wr1, real, dimension( * ) wi1, real, dimension( ldvl, * ) vl, integer ldvl, real, dimension( ldvr, * ) vr, integer ldvr, real, dimension( ldlre, * ) lre, integer ldlre, real, dimension( 7 ) result, real, dimension( * ) work, integer nwork, integer, dimension( * ) iwork, integer info)

SDRVEV

Propósito:

!>
!>    SDRVEV  comprueba el controlador del problema de valores propios no simétricos SGEEV.
!>
!>    Cuando se llama a SDRVEV, se especifica un número de tamaños de matriz () y un
!>    número de tipos de matriz. Para cada tamaño ()
!>    y cada tipo de matriz, se generará una matriz y se utilizará
!>    para probar las rutinas de valores propios no simétricos. Para cada matriz, se realizarán 7
!>    pruebas:
!>
!>    (1)     | A * VR - VR * W | / ( n |A| ulp )
!>
!>      Aquí VR es la matriz de vectores propios derecho unitarios.
!>      W es una matriz diagonal por bloques, con un bloque 1x1 para cada
!>      valor propio real y un bloque 2x2 para cada par conjugado complejo.
!>      Si los valores propios j y j+1 son un par conjugado complejo,
!>      entonces WR(j) = WR(j+1) = wr y WI(j) = - WI(j+1) = wi, entonces el
!>      bloque 2 x 2 correspondiente al par será:
!>
!>              (  wr  wi  )
!>              ( -wi  wr  )
!>
!>      Dicho bloque multiplicando una matriz n x 2  ( ur ui ) a la
!>      derecha será lo mismo que multiplicar  ur + i*ui  por  wr + i*wi.
!>
!>    (2)     | A**H * VL - VL * W**H | / ( n |A| ulp )
!>
!>      Aquí VL es la matriz de vectores propios izquierdo unitarios, A**H es la
!>      transpuesta conjugada de A, y W es como se describió anteriormente.
!>
!>    (3)     | |VR(i)| - 1 | / ulp y si el componente más grande es real
!>
!>      VR(i) denota la i-ésima columna de VR.
!>
!>    (4)     | |VL(i)| - 1 | / ulp y si el componente más grande es real
!>
!>      VL(i) denota la i-ésima columna de VL.
!>
!>    (5)     W(completo) = W(parcial)
!>
!>      W(completo) denota los valores propios calculados cuando tanto VR como VL
!>      también se calculan, y W(parcial) denota el resultado
!>      cuando solo se calculan W, solo W y VR, o solo W y VL.
!>
!>    (6)     VR(completo) = VR(parcial)
!>
!>      VR(completo) denota los vectores propios derechos calculados cuando tanto VR
!>      como VL se calculan, y VR(parcial) denota el resultado
!>      cuando solo se calcula VR.
!>
!>     (7)     VL(completo) = VL(parcial)
!>
!>      VL(completo) denota los vectores propios izquierdos calculados cuando tanto VR
!>      como VL también se calculan, y VL(parcial) denota el resultado
!>      cuando solo se calcula VL.
!>
!>    Los  se especifican mediante un array NN(1:NSIZES); el valor de
!>    cada elemento NN(j) especifica un tamaño.
!>    Los  se especifican mediante un array lógico DOTYPE( 1:NTYPES );
!>    si DOTYPE(j) es .TRUE., entonces se generará el tipo de matriz .
!>    Actualmente, la lista de tipos posibles es:
!>
!>    (1)  La matriz cero.
!>    (2)  La matriz identidad.
!>    (3)  Un bloque de Jordan (transpuesto), con 1s en la diagonal.
!>
!>    (4)  Una matriz diagonal con entradas espaciadas uniformemente
!>         1, ..., ULP  y signos aleatorios.
!>         (ULP = (primer número mayor que 1) - 1 )
!>    (5)  Una matriz diagonal con entradas espaciadas geométricamente
!>         1, ..., ULP  y signos aleatorios.
!>    (6)  Una matriz diagonal con  entradas 1, ULP, ..., ULP
!>         y signos aleatorios.
!>
!>    (7)  Igual que (4), pero multiplicado por una constante cercana
!>         al umbral de desbordamiento
!>    (8)  Igual que (4), pero multiplicado por una constante cercana
!>         al umbral de subdesbordamiento
!>
!>    (9)  Una matriz de la forma  U' T U, donde U es ortogonal y
!>         T tiene entradas espaciadas uniformemente 1, ..., ULP con signos
!>         aleatorios en la diagonal y entradas O(1) aleatorias en el
!>         triángulo superior.
!>
!>    (10) Una matriz de la forma  U' T U, donde U es ortogonal y
!>         T tiene entradas espaciadas geométricamente 1, ..., ULP con signos
!>         aleatorios en la diagonal y entradas O(1) aleatorias en el
!>         triángulo superior.
!>
!>    (11) Una matriz de la forma  U' T U, donde U es ortogonal y
!>         T tiene entradas  1, ULP,..., ULP con signos
!>         aleatorios en la diagonal y entradas O(1) aleatorias en el
!>         triángulo superior.
!>
!>    (12) Una matriz de la forma  U' T U, donde U es ortogonal y
!>         T tiene valores propios reales o conjugados complejos elegidos
!>         aleatoriamente de ( ULP, 1 ) y entradas O(1) aleatorias en el
!>         triángulo superior.
!>
!>    (13) Una matriz de la forma  X' T X, donde X tiene condición
!>         SQRT( ULP ) y T tiene entradas espaciadas uniformemente 1, ..., ULP
!>         con signos aleatorios en la diagonal y entradas O(1) aleatorias en el
!>         triángulo superior.
!>
!>    (14) Una matriz de la forma  X' T X, donde X tiene condición
!>         SQRT( ULP ) y T tiene entradas espaciadas geométricamente
!>         1, ..., ULP con signos aleatorios en la diagonal y entradas O(1)
!>         aleatorias en el triángulo superior.
!>
!>    (15) Una matriz de la forma  X' T X, donde X tiene condición
!>         SQRT( ULP ) y T tiene entradas  1, ULP,..., ULP con signos
!>         aleatorios en la diagonal y entradas O(1) aleatorias en el
!>         triángulo superior.
!>
!>    (16) Una matriz de la forma  X' T X, donde X tiene condición
!>         SQRT( ULP ) y T tiene valores propios reales o conjugados complejos
!>         elegidos aleatoriamente de ( ULP, 1 ) y entradas O(1) aleatorias
!>         en el triángulo superior.
!>
!>    (17) Igual que (16), pero multiplicado por una constante cercana
!>         al umbral de desbordamiento
!>    (18) Igual que (16), pero multiplicado por una constante cercana
!>         al umbral de subdesbordamiento
!>
!>    (19) Matriz no simétrica con entradas aleatorias elegidas de (-1,1).
!>         Si N es al menos 4, todas las entradas en la primera y segunda filas y la última
!>         fila, y la primera columna y las dos últimas columnas son cero.
!>    (20) Igual que (19), pero multiplicado por una constante cercana
!>         al umbral de desbordamiento
!>    (21) Igual que (19), pero multiplicado por una constante cercana
!>         al umbral de subdesbordamiento
!>

Parámetros

NSIZES 
!>          NSIZES es INTEGER
!>          El número de tamaños de matrices a utilizar. Si es cero,
!>          SDRVEV no hace nada. Debe ser al menos cero.
!>

NN

!>          NN es un array INTEGER, dimension (NSIZES)
!>          Un array que contiene los tamaños a utilizar para las matrices.
!>          Los valores cero se omitirán. Los valores deben ser al menos
!>          cero.
!>

NTYPES

!>          NTYPES es INTEGER
!>          El número de elementos en DOTYPE. Si es cero, SDRVEV
!>          no hace nada. Debe ser al menos cero. Si es MAXTYP+1
!>          y NSIZES es 1, entonces se define un tipo adicional, MAXTYP+1,
!>          que es usar cualquier matriz que esté en A. Esto
!>          solo es útil si DOTYPE(1:MAXTYP) es .FALSE. y
!!>          DOTYPE(MAXTYP+1) es .TRUE. .
!>

DOTYPE

!>          DOTYPE es un array LOGICAL, dimension (NTYPES)
!>

sdrvev.f(3) - LAPACK

NAME

sdrvev

driver program for computing eigenvalues and eigenvectors of a real general matrix.

SYNOPSIS

SUBROUTINE SDRVEV(
   NSIZES, NN, DOTYPE, ISPEC, SSQES,     &
   THRESH, NOUNIT, A, LDA, H,            &
   WR, WI, WR1, WI1, VL, LDVL, VR, LDVR, &
   LRE, LDLRE, RESULT, WORK, NWORK,     &
   IWORK, INFO )

Arguments:

NSIZES

INTEGER

The number of sizes of matrices to be generated.

NN

INTEGER array, dimension (NSIZES)

The sizes of the matrices to be generated. NN(j) is the dimension of the first matrix to be generated.

DOTYPE

LOGICAL array, dimension (NTYPES)

The matrix type to be generated. DOTYPE(j) = .TRUE. if a matrix of type j will be generated.

ISPEC

INTEGER

Specifies the type of eigenvector problem:

  • 0: Compute eigenvalues only.
  • 1: Compute eigenvalues and right eigenvectors.
  • 2: Compute eigenvalues and left eigenvectors.
  • 3: Compute eigenvalues and both left and right eigenvectors.
SSQES

REAL array, dimension (7)

The values computed by the seven tests.

THRESH

REAL

A test will count as ok if the result is less than or equal to THRESH.

NOUNIT

INTEGER

The FORTRAN unit number for printing out error messages.

A

REAL array, dimension (LDA, max(NN))

Used to hold the matrix whose eigenvalues are to be computed. On exit, A contains the last matrix actually used.

LDA

INTEGER

Leading dimension of A and H.

H

REAL array, dimension (LDA, max(NN))

Another copy of the test matrix A, modified by SGEEV.

WR

REAL array, dimension (max(NN))

WI

REAL array, dimension (max(NN))

The real and imaginary parts of the eigenvalues of A. On exit, WR + WI*i are the eigenvalues of the matrix in A.

WR1

REAL array, dimension (max(NN))

WI1

REAL array, dimension (max(NN))

Like WR, WI, these arrays contain the eigenvalues of A, but those computed when SGEEV only computes a partial eigendecomposition.

VL

REAL array, dimension (LDVL, max(NN))

VL holds the computed left eigenvectors.

LDVL

INTEGER

Leading dimension of VL.

VR

REAL array, dimension (LDVR, max(NN))

VR holds the computed right eigenvectors.

LDVR

INTEGER

Leading dimension of VR.

LRE

REAL array, dimension (LDLRE, max(NN))

LRE holds the computed right or left eigenvectors.

LDLRE

INTEGER

Leading dimension of LRE.

RESULT

REAL array, dimension (7)

The values computed by the seven tests.

WORK

REAL array, dimension (NWORK)

NWORK

INTEGER

The number of entries in WORK.

IWORK

INTEGER array, dimension (max(NN))

INFO

INTEGER

INFO is an integer output argument.

  • If INFO = 0, then the execution has been completed successfully.
  • If INFO < 0, then the i-th argument had an illegal value. A message is printed on unit NOUNIT to indicate which argument.

  • If INFO > 0, then INFO indicates the number of tests that have failed.

Function/Subroutine Documentation

DOTYPE

!>          DOTYPE is LOGICAL array, dimension (NTYPES)
!>          DOTYPE(j) is .TRUE. if a matrix of type j will be generated.
!>          If NTYPES is smaller than the maximum number of types
!>          defined (PARAMETER MAXTYP), then types NTYPES+1 through
!>          MAXTYP will not be generated.  If NTYPES is larger
!>          than MAXTYP, DOTYPE(MAXTYP+1) through DOTYPE(NTYPES)
!>          will be ignored.
!>

ISEED

!>          ISEED is INTEGER array, dimension (4)
!>          On entry ISEED specifies the seed of the random number
!>          generator. The array elements should be between 0 and 4095;
!>          if not they will be reduced mod 4096.  Also, ISEED(4) must
!>          be odd.  The random number generator uses a linear
!>          congruential sequence limited to small integers, and so
!>          should produce machine independent random numbers. The
!>          values of ISEED are changed on exit, and can be used in the
!>          next call to SDRVEV to continue the same random number
!>          sequence.
!>

THRESH

!>          THRESH is REAL
!>          A test will count as OK if the result, computed as
!>          described above, does not exceed THRESH.  Note that the error
!>          is scaled to be O(1), so THRESH should be a reasonably
!>          small multiple of 1, e.g., 10 or 100.  In particular,
!>          it should not depend on the precision (single vs. double)
!>          or the size of the matrix.  It must be at least zero.
!>

NOUNIT

!>          NOUNIT is INTEGER
!>          The FORTRAN unit number for printing out error messages
!>          (e.g., if a routine returns INFO not equal to 0.)
!>

A

!>          A is REAL array, dimension (LDA, max(NN))
!>          Used to hold the matrix whose eigenvalues are to be
!>          computed.  On exit, A contains the last matrix actually used.
!>

LDA

!>          LDA is INTEGER
!>          The leading dimension of A, and H. LDA must be at
!>          least 1 and at least max(NN).
!>

H

!>          H is REAL array, dimension (LDA, max(NN))
!>          Another copy of the test matrix A, modified by SGEEV.
!>

WR

!>          WR is REAL array, dimension (max(NN))
!>

WI

!>          WI is REAL array, dimension (max(NN))
!>
!>          The real and imaginary parts of the eigenvalues of A.
!>          On exit, WR + WI*i are the eigenvalues of the matrix in A.
!>

WR1

!>          WR1 is REAL array, dimension (max(NN))
!>

WI1

!>          WI1 is REAL array, dimension (max(NN))
!>
!>          Like WR, WI, these arrays contain the eigenvalues of A,
!>          but those computed when SGEEV only computes a partial
!>          eigendecomposition, i.e. not the eigenvalues and left
!>          and right eigenvectors.
!>

VL

!>          VL is REAL array, dimension (LDVL, max(NN))
!>          VL holds the computed left eigenvectors.
!>

LDVL

!>          LDVL is INTEGER
!>          Leading dimension of VL. Must be at least max(1,max(NN)).
!>

VR

!>          VR is REAL array, dimension (LDVR, max(NN))
!>          VR holds the computed right eigenvectors.
!>

LDVR

!>          LDVR is INTEGER
!>          Leading dimension of VR. Must be at least max(1,max(NN)).
!>

LRE

!>          LRE is REAL array, dimension (LDLRE,max(NN))
!>          LRE holds the computed right or left eigenvectors.
!>

LDLRE

!>          LDLRE is INTEGER
!>          Leading dimension of LRE. Must be at least max(1,max(NN)).
!>

RESULT

!>          RESULT is REAL array, dimension (7)
!>          The values computed by the seven tests described above.
!>          The values are currently limited to 1/ulp, to avoid overflow.
!>

WORK

!>          WORK is REAL array, dimension (NWORK)
!>

NWORK

!>          NWORK is INTEGER
!>          The number of entries in WORK.  This must be at least
!>          5*NN(j)+2*NN(j)**2 for all j.
!>

IWORK

!>          IWORK is INTEGER array, dimension (max(NN))
!>

INFO

!>          INFO is INTEGER
!>          If 0, then everything ran OK.
!>           -1: NSIZES < 0
!>           -2: Some NN(j) < 0
!>           -3: NTYPES < 0
!>           -6: THRESH < 0
!>           -9: LDA < 1 or LDA < NMAX, where NMAX is max( NN(j) ).
!>          -16: LDVL < 1 or LDVL < NMAX, where NMAX is max( NN(j) ).
!>          -18: LDVR < 1 or LDVR < NMAX, where NMAX is max( NN(j) ).
!>          -20: LDLRE < 1 or LDLRE < NMAX, where NMAX is max( NN(j) ).
!>          -23: NWORK too small.
!>          If  SLATMR, SLATMS, SLATME or SGEEV returns an error code,
!>              the absolute value of it is returned.
!>
!>-----------------------------------------------------------------------
!>
!>     Some Local Variables and Parameters:
!>     ---- ----- --------- --- ----------
!>
!>     ZERO, ONE       Real 0 and 1.
!>     MAXTYP          The number of types defined.
!>     NMAX            Largest value in NN.
!>     NERRS           The number of tests which have exceeded THRESH
!>     COND, CONDS,
!!>     IMODE           Values to be passed to the matrix generators.
!>     ANORM           Norm of A; passed to matrix generators.
!>
!>     OVFL, UNFL      Overflow and underflow thresholds.
!>     ULP, ULPINV     Finest relative precision and its inverse.
!>     RTULP, RTULPI   Square roots of the previous 4 values.
!>
!>             The following four arrays decode JTYPE:
!>     KTYPE(j)        The general type (1-10) for type .
!>     KMODE(j)        The MODE value to be passed to the matrix
!>                     generator for type .
!>     KMAGN(j)        The order of magnitude ( O(1),
!>                     O(overflow^(1/2) ), O(underflow^(1/2) )
!>     KCONDS(j)       Selects whether CONDS is to be 1 or
!>                     1/sqrt(ulp).  (0 means irrelevant.)
!>
