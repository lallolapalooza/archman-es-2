Documentación de función/subrutina

subroutine sdrvpt (logical, dimension( * ) dotype, integer nn, integer, dimension( * ) nval, integer nrhs, real thresh, logical tsterr, real, dimension( * ) a, real, dimension( * ) d, real, dimension( * ) e, real, dimension( * ) b, real, dimension( * ) x, real, dimension( * ) xact, real, dimension( * ) work, real, dimension( * ) rwork, integer nout)

SDRVPT

Propósito:

!>
!> SDRVPT prueba SPTSV y -SVX.
!>

Parámetros

DOTYPE 
!>          DOTYPE es un array LOGICAL, de dimensión (NTYPES)
!>          Los tipos de matriz a utilizar para las pruebas. Las matrices de tipo j
!>          (para 1 <= j <= NTYPES) se utilizan para las pruebas si DOTYPE(j) =
!>          .TRUE.; si DOTYPE(j) = .FALSE., entonces el tipo j no se utiliza.
!>

NN

!>          NN es un ENTERO
!>          El número de valores de N contenidos en el vector NVAL.
!>

NVAL

!>          NVAL es un array de ENTEROS, de dimensión (NN)
!>          Los valores de la dimensión de la matriz N.
!>

NRHS

!>          NRHS es un ENTERO
!>          El número de vectores del lado derecho a generar para
!>          cada sistema lineal.
!>

THRESH

!>          THRESH es REAL
!>          El valor umbral para las relaciones de prueba. Un resultado se
!>          incluye en el archivo de salida si RESULT >= THRESH. Para que
!>          cada relación de prueba se imprima, use THRESH = 0.
!>

TSTERR

!>          TSTERR es LOGICAL
!>          Indicador que señala si las salidas de error deben ser probadas.
!>

A

!>          A es un array REAL, de dimensión (NMAX*2)
!>

D

!>          D es un array REAL, de dimensión (NMAX*2)
!>

E

!>          E es un array REAL, de dimensión (NMAX*2)
!>

B

!>          B es un array REAL, de dimensión (NMAX*NRHS)
!>

X

!>          X es un array REAL, de dimensión (NMAX*NRHS)
!>

XACT

!>          XACT es un array REAL, de dimensión (NMAX*NRHS)
!>

WORK

!>          WORK es un array REAL, de dimensión
!>                      (NMAX*max(3,NRHS))
!>

RWORK

!>          RWORK es un array REAL, de dimensión
!>                      (max(NMAX,2*NRHS))
!>

NOUT

!>          NOUT es un ENTERO
!>          El número de unidad para la salida.
!>
