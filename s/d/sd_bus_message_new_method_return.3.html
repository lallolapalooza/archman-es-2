Paquete: systemd-libs

Licencias: LGPL-2.1-or-later, CC0-1.0, GPL-2.0-or-later WITH Linux-syscall-note

SD_BUS_MESSAGE_NEW_METHOD_CALL(3) sd_bus_message_new_method_call SD_BUS_MESSAGE_NEW_METHOD_CALL(3)

NOMBRE

sd_bus_message_new_method_call, sd_bus_message_new_method_return - Crear un mensaje de llamada a método

SINOPSIS

#include <systemd/sd-bus.h>

int sd_bus_message_new_method_call(sd_bus *bus, sd_bus_message **m, const char *destination, const char *path, const char *interface, const char *member);

int sd_bus_message_new_method_return(sd_bus_message *call, sd_bus_message **m);

DESCRIPCIÓN

La función sd_bus_message_new_method_call() crea un nuevo objeto de mensaje de bus que encapsula una llamada a método D-Bus y lo devuelve en el parámetro de salida m. La llamada se realizará al destino destination, ruta path, en la interfaz interface, miembro member.

En resumen, el destination es un nombre separado por puntos que identifica un servicio conectado al bus. El path es un identificador separado por barras de un objeto dentro del destino que se asemeja a una ruta de sistema de archivos. El significado de esta ruta lo define el destino. La interface es un nombre separado por puntos que se asemeja a un nombre de interfaz de Java que identifica un grupo de métodos y señales soportados por el objeto identificado por la ruta. Los métodos y señales se denominan colectivamente members y se identifican por un nombre simple compuesto por letras ASCII, números y guiones bajos. Consulte el **Tutorial de D-Bus**[1] para una explicación en profundidad.

El parámetro destination puede ser NULL. El parámetro interface puede ser NULL, si el destino tiene solo un miembro con el nombre dado y no hay ambigüedad si se omite el nombre de la interfaz.

Tenga en cuenta que esta es una interfaz de bajo nivel. Consulte sd_bus_call_method(3) para una forma más conveniente de llamar a métodos D-Bus.

La función sd_bus_message_new_method_return() crea un nuevo objeto de mensaje de bus que es una respuesta a la llamada de método call y lo devuelve en el parámetro de salida m. El parámetro call debe ser un mensaje de llamada a método. El remitente de call se utiliza como destino.

VALOR DE RETORNO

En caso de éxito, estas funciones devuelven un entero no negativo. En caso de error, devuelven un código de error negativo de estilo errno.

Errores

Los errores devueltos pueden indicar los siguientes problemas:

-EINVAL

El parámetro de salida m es NULL.

El parámetro destination no es nulo y no es un nombre de servicio D-Bus válido ("org.somewhere.Something"), el parámetro path no es una ruta D-Bus válida ("/an/object/path"), el parámetro interface no es nulo y no es un nombre de interfaz D-Bus válido ("an.interface.name"), o el parámetro member no es un miembro D-Bus válido ("Name").

El parámetro call no es un objeto de llamada a método.

-ENOTCONN

El parámetro de bus bus es NULL o el bus no está conectado.

-ENOMEM

Fallo en la asignación de memoria.

-EPERM

El parámetro call no está sellado.

-EOPNOTSUPP

El mensaje call no tiene una cookie.

NOTAS

Las funciones descritas aquí están disponibles como una biblioteca compartida, que puede compilarse y enlazarse con el archivo pkg-config(1) de libsystemd.

El código descrito aquí utiliza getenv(3), que se declara no ser seguro para subprocesos múltiples. Esto significa que el código que llama a las funciones descritas aquí no debe llamar a setenv(3) desde un hilo paralelo. Se recomienda realizar llamadas a setenv() solo desde una fase temprana del programa cuando no se hayan iniciado otros hilos.

EJEMPLOS

Ejemplo 1. Realizar una llamada a un método D-Bus que acepta un solo parámetro

/* SPDX-License-Identifier: MIT-0 */
/* Esto es equivalente a:
 * busctl call org.freedesktop.systemd1 /org/freedesktop/systemd1 \
 *       org.freedesktop.systemd1.Manager GetUnitByPID $$
 *
 * Compilar con 'cc print-unit-path.c -lsystemd'
 */
#include <errno.h>
#include <stdio.h>
#include <sys/types.h>
#include <unistd.h>
#include <systemd/sd-bus.h>
#define _cleanup_(f) __attribute__((cleanup(f)))
#define DESTINATION "org.freedesktop.systemd1"
#define PATH        "/org/freedesktop/systemd1"
#define INTERFACE   "org.freedesktop.systemd1.Manager"
#define MEMBER      "GetUnitByPID"
static int log_error(int error, const char *message) {
  fprintf(stderr, "%s: %s\n", message, strerror(-error));
  return error;
}
int main(int argc, char **argv) {
  _cleanup_(sd_bus_flush_close_unrefp) sd_bus *bus = NULL;
  _cleanup_(sd_bus_error_free) sd_bus_error error = SD_BUS_ERROR_NULL;
  _cleanup_(sd_bus_message_unrefp) sd_bus_message *reply = NULL, *m = NULL;
  int r;
  r = sd_bus_open_system(&bus);
  if (r < 0)
    return log_error(r, "Failed to acquire bus");
  r = sd_bus_message_new_method_call(bus, &m,
                                     DESTINATION, PATH, INTERFACE, MEMBER);
  if (r < 0)
    return log_error(r, "Failed to create bus message");
  r = sd_bus_message_append(m, "u", (unsigned) getpid());
  if (r < 0)
    return log_error(r, "Failed to append to bus message");
  r = sd_bus_call(bus, m, -1, &error, &reply);
  if (r < 0)
    return log_error(r, MEMBER " call failed");
  const char *ans;
  r = sd_bus_message_read(reply, "o", &ans);
  if (r < 0)
    return log_error(r, "Failed to read reply");
  printf("Unit path is \"%s\".\n", ans);
  return 0;
}

Esto define un programa mínimamente útil que abrirá una conexión al bus, creará un objeto de mensaje, lo enviará, esperará la respuesta y finalmente extraerá e imprimirá la respuesta. Realiza manejo de errores y gestión adecuada de la memoria.

HISTORIA

sd_bus_message_new_method_call() y sd_bus_message_new_method_return() se añadieron en la versión 246.

sd_bus_message_new_method_return

sd_bus_message_new_method_return - Crea un nuevo mensaje de respuesta de método de bus

SECCIÓN

sd-bus.h

SINTAXIS

#include <systemd/sd-bus.h>

int sd_bus_message_new_method_return(sd_bus_message *call, sd_bus_message **reply);
int sd_bus_message_new_method_return_basic(sd_bus_message *call, sd_bus_message **reply, const char *signature, ...);
int sd_bus_message_new_method_return_error(sd_bus_message *call, const char *error_name, const char *error_string);
int sd_bus_message_new_method_return_errorf(sd_bus_message *call, const char *error_name, const char *format, ...);
int sd_bus_message_new_method_return_async(sd_bus_message *call, sd_bus_message **reply);
int sd_bus_message_new_method_return_take(sd_bus_message *call, sd_bus_message **reply);
int sd_bus_message_new_method_return_take_error(sd_bus_message *call, sd_bus_message **reply);

DESCRIPCIÓN

sd_bus_message_new_method_return(), sd_bus_message_new_method_return_basic(), sd_bus_message_new_method_return_error(), sd_bus_message_new_method_return_errorf(), sd_bus_message_new_method_return_async(), sd_bus_message_new_method_return_take() y sd_bus_message_new_method_return_take_error() asignan un nuevo mensaje de respuesta de método de bus en respuesta a un mensaje de llamada de método dado call.

El mensaje de respuesta se devuelve en el parámetro reply. El llamador es responsable de liberar el mensaje de respuesta devuelto con sd_bus_message_unref().

sd_bus_message_new_method_return() crea un mensaje de respuesta de método vacío. La firma del tipo de retorno para un método es la cadena de caracteres de firma del tipo especificada por la interfaz del objeto.

sd_bus_message_new_method_return_basic() crea un mensaje de respuesta de método vacío con la firma de tipo dada signature y los argumentos adicionales proporcionados. Estos argumentos se serializan en el mensaje de respuesta.

sd_bus_message_new_method_return_error() crea un mensaje de respuesta de error para la llamada dada call. El nombre del error se especifica en error_name, y la cadena de caracteres del error opcional se especifica en error_string. Si se proporciona una cadena de error de sólo formato, se recomienda utilizar sd_bus_message_new_method_return_errorf() en su lugar.

sd_bus_message_new_method_return_errorf() es similar a sd_bus_message_new_method_return_error(), pero permite especificar la cadena de error utilizando una cadena de formato y argumentos adicionales, similar a la función printf(). El nombre del error se especifica en error_name.

sd_bus_message_new_method_return_async() se comporta de forma idéntica a sd_bus_message_new_method_return(), pero el mensaje de respuesta se marca como una respuesta asíncrona. Esto es útil para métodos que devuelven un objeto complejo o que requieren una operación de larga duración para completarse.

sd_bus_message_new_method_return_take() se comporta de forma idéntica a sd_bus_message_new_method_return(), pero en lugar de asignar un nuevo mensaje, toma la propiedad de un mensaje de respuesta existente que se pasa como segundo argumento. El mensaje de entrada se establecerá en NULL después de que la llamada tenga éxito.

sd_bus_message_new_method_return_take_error() se comporta de forma idéntica a sd_bus_message_new_method_return_error(), pero en lugar de asignar un nuevo mensaje, toma la propiedad de un mensaje de error existente que se pasa como segundo argumento. El mensaje de entrada se establecerá en NULL después de que la llamada tenga éxito.

VALOR DE RETORNO

En caso de éxito, estas funciones devuelven 0 o un valor positivo. En caso de error, devuelven un número negativo igual al código de error.

errno no se establece por estas funciones.

NOTAS

Estas funciones deben ser utilizadas por los servidores de bus para construir respuestas a las llamadas de método entrantes. La llamada de método de entrada call se utiliza para obtener información sobre la llamada original, como el remitente y el identificador de la transacción, que se utilizan para construir el mensaje de respuesta.

sd_bus_message_new_method_return_basic() es útil cuando la firma del tipo de retorno del método es simple y se puede serializar directamente en el mensaje de respuesta.

Para manejar errores de forma más compleja o para proporcionar información de error adicional, se puede utilizar sd_bus_message_new_method_return_error() o sd_bus_message_new_method_return_errorf().

Las funciones `_take` son útiles cuando un mensaje de respuesta ya ha sido construido por otra parte de la aplicación y se necesita pasar su propiedad a la capa del bus.

EJEMPLOS

// Ejemplo de sd_bus_message_new_method_return()
sd_bus_message *reply = NULL;
int r;

r = sd_bus_message_new_method_return(call, &reply);
if (r < 0) {
    fprintf(stderr, "Error al crear mensaje de respuesta: %s\n", strerror(-r));
    // Manejar el error
}

// Rellenar el mensaje de respuesta con datos si es necesario
// ...

sd_bus_process(bus, NULL); // Enviar el mensaje de respuesta

sd_bus_message_unref(reply); // Liberar el mensaje de respuesta

// Ejemplo de sd_bus_message_new_method_return_basic()
sd_bus_message *reply = NULL;
int r;
int integer_value = 42;
const char *string_value = "hola";

r = sd_bus_message_new_method_return_basic(call, &reply, "is", integer_value, string_value);
if (r < 0) {
    fprintf(stderr, "Error al crear mensaje de respuesta básico: %s\n", strerror(-r));
    // Manejar el error
}

sd_bus_process(bus, NULL); // Enviar el mensaje de respuesta

sd_bus_message_unref(reply); // Liberar el mensaje de respuesta

// Ejemplo de sd_bus_message_new_method_return_error()
sd_bus_message *reply = NULL;
int r;

r = sd_bus_message_new_method_return_error(call, "org.example.Error.NotFound", "El recurso solicitado no fue encontrado.");
if (r < 0) {
    fprintf(stderr, "Error al crear mensaje de error: %s\n", strerror(-r));
    // Manejar el error
}

sd_bus_process(bus, NULL); // Enviar el mensaje de error

sd_bus_message_unref(reply); // Liberar el mensaje de error

// Ejemplo de sd_bus_message_new_method_return_errorf()
sd_bus_message *reply = NULL;
int r;
int error_code = 101;

r = sd_bus_message_new_method_return_errorf(call, "org.example.Error.Custom", "Código de error personalizado: %d", error_code);
if (r < 0) {
    fprintf(stderr, "Error al crear mensaje de error formateado: %s\n", strerror(-r));
    // Manejar el error
}

sd_bus_process(bus, NULL); // Enviar el mensaje de error

sd_bus_message_unref(reply); // Liberar el mensaje de error
