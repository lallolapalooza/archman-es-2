sd_bus_call_method(3)

Paquete: systemd-libs

Licencias: LGPL-2.1-or-later, CC0-1.0, GPL-2.0-or-later WITH Linux-syscall-note

SD_BUS_CALL_METHOD(3) sd_bus_call_method SD_BUS_CALL_METHOD(3)

NOMBRE

sd_bus_call_method, sd_bus_call_methodv, sd_bus_call_method_async, sd_bus_call_method_asyncv - Inicializa un objeto de mensaje de bus e invoca la llamada al método D-Bus correspondiente

SINOPSIS

#include <systemd/sd-bus.h>

typedef int (*sd_bus_message_handler_t)(sd_bus_message *m, void *userdata, sd_bus_error *ret_error);

int sd_bus_call_method(sd_bus *bus, const char *destination, const char *path, const char *interface, const char *member, sd_bus_error *ret_error, sd_bus_message **reply, const char *types, ...);

int sd_bus_call_methodv(sd_bus *bus, const char *destination, const char *path, const char *interface, const char *member, sd_bus_error *ret_error, sd_bus_message **reply, const char *types, va_list ap);

int sd_bus_call_method_async(sd_bus *bus, sd_bus_slot **slot, const char *destination, const char *path, const char *interface, const char *member, sd_bus_message_handler_t callback, void *userdata, const char *types, ...);

int sd_bus_call_method_asyncv(sd_bus *bus, sd_bus_slot **slot, const char *destination, const char *path, const char *interface, const char *member, sd_bus_message_handler_t callback, void *userdata, const char *types, va_list ap);

DESCRIPCIÓN

sd_bus_call_method() es una función de conveniencia para inicializar un objeto de mensaje de bus y llamar al método D-Bus correspondiente. Combina las funciones sd_bus_message_new_method_call(3), sd_bus_message_append(3) y sd_bus_call(3) en una única llamada a función.

sd_bus_call_method_async() es una función de conveniencia para inicializar un objeto de mensaje de bus y llamar al método D-Bus correspondiente de forma asíncrona. Combina las funciones sd_bus_message_new_method_call(3), sd_bus_message_append(3) y sd_bus_call_async(3) en una única llamada a función.

VALOR DE RETORNO

En caso de éxito, estas funciones devuelven un entero no negativo. En caso de fallo, devuelven un código de error negativo estilo errno.

Errores

Consulte las páginas del manual de sd_bus_message_new_method_call(3), sd_bus_message_append(3), sd_bus_call(3) y sd_bus_call_async(3) para obtener una lista de posibles errores.

NOTAS

Las funciones descritas aquí están disponibles como una biblioteca compartida, que se puede compilar y enlazar con el archivo libsystemd de pkg-config(1).

El código descrito aquí utiliza getenv(3), que se declara como no seguro para hilos. Esto significa que el código que llama a las funciones descritas aquí no debe llamar a setenv(3) desde un hilo paralelo. Se recomienda hacer solo llamadas a setenv() desde una fase temprana del programa cuando aún no se han iniciado otros hilos.

EJEMPLOS

Ejemplo 1. Realizar una llamada a un método D-Bus que acepta un único parámetro

/* SPDX-License-Identifier: MIT-0 */
/* Esto es equivalente a:
 * busctl call org.freedesktop.systemd1 /org/freedesktop/systemd1 \
 *       org.freedesktop.systemd1.Manager GetUnitByPID $$
 *
 * Compilar con 'cc print-unit-path-call-method.c -lsystemd'
 */
#include <errno.h>
#include <stdio.h>
#include <sys/types.h>
#include <unistd.h>
#include <systemd/sd-bus.h>
#define _cleanup_(f) __attribute__((cleanup(f)))
#define DESTINATION "org.freedesktop.systemd1"
#define PATH        "/org/freedesktop/systemd1"
#define INTERFACE   "org.freedesktop.systemd1.Manager"
#define MEMBER      "GetUnitByPID"
static int log_error(int error, const char *message) {
  fprintf(stderr, "%s: %s\n", message, strerror(-error));
  return error;
}
int main(int argc, char **argv) {
  _cleanup_(sd_bus_flush_close_unrefp) sd_bus *bus = NULL;
  _cleanup_(sd_bus_error_free) sd_bus_error error = SD_BUS_ERROR_NULL;
  _cleanup_(sd_bus_message_unrefp) sd_bus_message *reply = NULL;
  int r;
  r = sd_bus_open_system(&bus);
  if (r < 0)
    return log_error(r, "Failed to acquire bus");
  r = sd_bus_call_method(bus, DESTINATION, PATH, INTERFACE, MEMBER, &error, &reply, "u", (unsigned) getpid());
  if (r < 0)
    return log_error(r, MEMBER " call failed");
  const char *ans;
  r = sd_bus_message_read(reply, "o", &ans);
  if (r < 0)
    return log_error(r, "Failed to read reply");
  printf("Unit path is \"%s\".\n", ans);
  return 0;
}

Esto define un programa mínimamente útil que abrirá una conexión al bus, llamará a un método, esperará la respuesta y finalmente extraerá e imprimirá la respuesta. Realiza manejo de errores y gestión de memoria adecuada.

sd_bus_call_methodv() y sd_bus_call_method_asyncv() se añadieron en la versión 246.

sd_bus_call_method(3)

Linux manual pages

NAME

sd_bus_call_method, sd_bus_call_methodv, sd_bus_call_method_async, sd_bus_call_method_asyncv - Send a D-Bus method call and process the reply

SYNOPSIS

<syntax highlighting disabled>
#include <systemd/sd-bus.h>

int sd_bus_call_method(sd_bus *bus,
                       const char *destination,
                       const char *path,
                       const char *interface,
                       const char *member,
                       sd_bus_error *ret_error,
                       const char *signature,
                       ...);

int sd_bus_call_methodv(sd_bus *bus,
                        const char *destination,
                        const char *path,
                        const char *interface,
                        const char *member,
                        sd_bus_error *ret_error,
                        const char *signature,
                        va_list ap);

int sd_bus_call_method_async(sd_bus *bus,
                             sd_bus_message **reply,
                             sd_event_source **source,
                             sd_bus_message *call,
                             sd_bus_error *ret_error,
                             sd_bus_callback_t callback,
                             void *userdata);

int sd_bus_call_method_asyncv(sd_bus *bus,
                              sd_bus_message **reply,
                              sd_event_source **source,
                              const char *destination,
                              const char *path,
                              const char *interface,
                              const char *member,
                              sd_bus_error *ret_error,
                              sd_bus_callback_t callback,
                              void *userdata,
                              const char *signature,
                              va_list ap);
</syntax highlighting disabled>

DESCRIPTION

sd_bus_call_method() and sd_bus_call_methodv() may be used to send a D-Bus method call to a service and synchronously wait for the reply. The method call is constructed from the given parameters, and then sent to the bus. The function then enters a blocking loop, waiting for a reply or an error. The arguments to the method call are passed as varargs after the signature string, or via a va_list in the case of sd_bus_call_methodv().

The result of the method call is passed back in the ret_error parameter if an error occurred, or as the return value of the function if the method call succeeded. If the method call succeeded, the return value will be of type 'v' and contains the return value of the method. If the method call failed, it will contain an error structure.

sd_bus_call_method_async() and sd_bus_call_method_asyncv() may be used to send a D-Bus method call asynchronously. Instead of blocking, the function returns immediately after sending the call. The reply is processed later by the event loop, and the callback function is invoked when the reply is received or an error occurs. The call parameter in sd_bus_call_method_async() should be a message object created with sd_bus_message_new_method_call(). sd_bus_call_method_asyncv() constructs the method call message itself from the provided parameters.

The reply parameter in the async functions will be filled with a pointer to the reply message object if the call succeeds. The source parameter will be filled with a pointer to the event source associated with the call, which can be used to cancel the call using sd_event_source_disable().

ARGUMENTS

bus
A valid connection to the D-Bus bus.
destination
The D-Bus name of the destination service, or NULL to send to the bus name of the connection itself.
path
The object path of the object to call the method on.
interface
The interface name of the method to call.
member
The member name (method name) of the method to call.
ret_error
A pointer to an sd_bus_error structure that will be filled with the error information if an error occurs.
signature
The D-Bus type signature of the arguments for the method call, or NULL if no arguments are passed.
...
The arguments for the method call, in order, corresponding to the signature. Each argument must be a pointer to a value of the type specified in the signature. Basic types like integers and booleans can be passed directly.
ap
A va_list containing the arguments for the method call.
reply
A pointer to a pointer to an sd_bus_message object. This will be filled with the reply message if the call succeeds.
source
A pointer to an sd_event_source object. This will be filled with the event source for the asynchronous call.
call
A message object representing the method call, created with sd_bus_message_new_method_call().
callback
The callback function to be invoked when the reply is received or an error occurs.
userdata
User data to be passed to the callback function.

RETURN VALUE

On success, sd_bus_call_method() and sd_bus_call_methodv() return 0. On failure, a negative error number is returned.

On success, sd_bus_call_method_async() and sd_bus_call_method_asyncv() return 0. On failure, a negative error number is returned.

NOTES

The synchronous functions sd_bus_call_method() and sd_bus_call_methodv() will block the calling thread until a reply is received or an error occurs. For applications that need to remain responsive, the asynchronous functions sd_bus_call_method_async() and sd_bus_call_method_asyncv() should be used instead.

The arguments passed to the synchronous functions must be valid for the entire duration of the call. They will be appended to the method call message and then serialized.

EXAMPLE

<syntax highlighting disabled>
#include <stdio.h>
#include <stdlib.h>
#include <systemd/sd-bus.h>

int main(int argc, char *argv[]) {
    sd_bus_error error = SD_BUS_ERROR_NULL;
    sd_bus_message *reply = NULL;
    sd_bus *bus = NULL;
    int r;

    r = sd_bus_open_user(&bus);
    if (r < 0) {
        fprintf(stderr, "Failed to connect to system bus: %s\n", strerror(-r));
        goto finish;
    }

    // Call the "GetHostname" method on the "org.freedesktop.hostname1" interface of the "org.freedesktop.hostname1" service
    r = sd_bus_call_method(bus,
                           "org.freedesktop.hostname1", // destination
                           "/org/freedesktop/hostname1", // path
                           "org.freedesktop.hostname1", // interface
                           "GetHostname", // member
                           &error, // error
                           "s", // signature (reply is a string)
                           "en"); // argument (locale)
    if (r < 0) {
        fprintf(stderr, "Failed to issue method call: %s\n", error.message);
        goto finish;
    }

    // Read the reply
    r = sd_bus_message_read(reply, "s", &hostname);
    if (r < 0) {
        fprintf(stderr, "Failed to parse reply message: %s\n", strerror(-r));
        goto finish;
    }

    printf("Hostname: %s\n", hostname);

    sd_bus_message_unref(reply);

finish:
    sd_bus_unref(bus);
    return r;
}
</syntax highlighting disabled>
