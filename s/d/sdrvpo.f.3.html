subrutina sdrvpo (logical, dimension( * ) dotype, integer nn, integer, dimension( * ) nval, integer nrhs, real thresh, logical tsterr, integer nmax, real, dimension( * ) a, real, dimension( * ) afac, real, dimension( * ) asav, real, dimension( * ) b, real, dimension( * ) bsav, real, dimension( * ) x, real, dimension( * ) xact, real, dimension( * ) s, real, dimension( * ) work, real, dimension( * ) rwork, integer, dimension( * ) iwork, integer nout)

SDRVPO

Propósito:

!>
!> SDRVPO prueba las rutinas de control SPOSV y -SVX.
!>

Parámetros

DOTYPE 
!>          DOTYPE es un array LOGICAL, dimension (NTYPES)
!>          Los tipos de matriz a utilizar para las pruebas. Las matrices de tipo j
!>          (para 1 <= j <= NTYPES) se utilizan para las pruebas si DOTYPE(j) =
!>          .TRUE.; si DOTYPE(j) = .FALSE., entonces el tipo j no se utiliza.
!>

NN

!>          NN es INTEGER
!>          El número de valores de N contenidos en el vector NVAL.
!>

NVAL

!>          NVAL es un array INTEGER, dimension (NN)
!>          Los valores de la dimensión de la matriz N.
!>

NRHS

!>          NRHS es INTEGER
!>          El número de vectores del lado derecho a generar para
!>          cada sistema lineal.
!>

THRESH

!>          THRESH es REAL
!>          El valor umbral para las razones de prueba. Un resultado se
!>          incluye en el archivo de salida si RESULT >= THRESH. Para que
!>          cada razón de prueba se imprima, use THRESH = 0.
!>

TSTERR

!>          TSTERR es LOGICAL
!>          Indicador que señala si se deben probar las salidas de error.
!>

NMAX

!>          NMAX es INTEGER
!>          El valor máximo permitido para N, utilizado en la dimensionamiento de los
!>          arrays de trabajo.
!>

A

!>          A es un array REAL, dimension (NMAX*NMAX)
!>

AFAC

!>          AFAC es un array REAL, dimension (NMAX*NMAX)
!>

ASAV

!>          ASAV es un array REAL, dimension (NMAX*NMAX)
!>

B

!>          B es un array REAL, dimension (NMAX*NRHS)
!>

BSAV

!>          BSAV es un array REAL, dimension (NMAX*NRHS)
!>

X

!>          X es un array REAL, dimension (NMAX*NRHS)
!>

XACT

!>          XACT es un array REAL, dimension (NMAX*NRHS)
!>

S

!>          S es un array REAL, dimension (NMAX)
!>

WORK

!>          WORK es un array REAL, dimension
!>                      (NMAX*max(3,NRHS))
!>

RWORK

!>          RWORK es un array REAL, dimension (NMAX+2*NRHS)
!>

IWORK

!>          IWORK es un array INTEGER, dimension (NMAX)
!>

NOUT

!>          NOUT es INTEGER
!>          La unidad de número para la salida.
!>
