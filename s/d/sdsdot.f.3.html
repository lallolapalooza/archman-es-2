real function sdsdot (integer n, real sb, real, dimension(*) sx, integer incx, real, dimension(*) sy, integer incy)

SDSDOT

Propósito:

!>
!>   Calcula el producto escalar de dos vectores con acumulación de precisión extendida.
!>
!>   Devuelve resultado S.P. con el producto escalar acumulado en D.P.
!>   SDSDOT = SB + suma para I = 0 a N-1 de SX(LX+I*INCX)*SY(LY+I*INCY),
!>   donde LX = 1 si INCX .GE. 0, si no LX = 1+(1-N)*INCX, y LY se
!>   define de manera similar usando INCY.
!>

Parámetros

N 
!>          N es INTEGER
!>          número de elementos en el(los) vector(es) de entrada
!>

SB

!>          SB es REAL
!>          escalar de precisión simple a añadir al producto escalar
!>

SX

!>          SX es array REAL, dimension ( 1 + ( N - 1 )*abs( INCX ) )
!>          vector de precisión simple con N elementos
!>

INCX

!>          INCX es INTEGER
!>          espaciado de almacenamiento entre elementos de SX
!>

SY

!>          SY es array REAL, dimension ( 1 + ( N - 1 )*abs( INCX ) )
!>          vector de precisión simple con N elementos
!>

INCY

!>          INCY es INTEGER
!>          espaciado de almacenamiento entre elementos de SY
!>
