subroutine sdrvgb (logical, dimension( * ) dotype, integer nn, integer, dimension( * ) nval, integer nrhs, real thresh, logical tsterr, real, dimension( * ) a, integer la, real, dimension( * ) afb, integer lafb, real, dimension( * ) asav, real, dimension( * ) b, real, dimension( * ) bsav, real, dimension( * ) x, real, dimension( * ) xact, real, dimension( * ) s, real, dimension( * ) work, real, dimension( * ) rwork, integer, dimension( * ) iwork, integer nout)

SDRVGB

Propósito:

!>
!> SDRVGB prueba las rutinas de controlador SGBSV y -SVX.
!>

Parámetros

DOTYPE 
!>          DOTYPE es un array LOGICAL, de dimensión (NTYPES)
!>          Los tipos de matriz a utilizar para las pruebas. Las matrices de tipo j
!>          (para 1 <= j <= NTYPES) se utilizan para las pruebas si DOTYPE(j) =
!>          .TRUE.; si DOTYPE(j) = .FALSE., entonces el tipo j no se utiliza.
!>

NN

!>          NN es un INTEGER
!>          El número de valores de N contenidos en el vector NVAL.
!>

NVAL

!>          NVAL es un array INTEGER, de dimensión (NN)
!>          Los valores de la dimensión de columna N de la matriz.
!>

NRHS

!>          NRHS es un INTEGER
!>          El número de vectores del lado derecho a generar para
!>          cada sistema lineal.
!>

THRESH

!>          THRESH es un REAL
!>          El valor umbral para las relaciones de prueba. Un resultado se
!>          incluye en el archivo de salida si RESULT >= THRESH. Para que
!>          cada relación de prueba se imprima, use THRESH = 0.
!>

TSTERR

!>          TSTERR es un LOGICAL
!>          Indicador que determina si las salidas de error se probarán.
!>

A

!>          A es un array REAL, de dimensión (LA)
!>

LA

!>          LA es un INTEGER
!>          La longitud del array A. LA >= (2*NMAX-1)*NMAX
!>          donde NMAX es la entrada más grande en NVAL.
!>

AFB

!>          AFB es un array REAL, de dimensión (LAFB)
!>

LAFB

!>          LAFB es un INTEGER
!>          La longitud del array AFB. LAFB >= (3*NMAX-2)*NMAX
!>          donde NMAX es la entrada más grande en NVAL.
!>

ASAV

!>          ASAV es un array REAL, de dimensión (LA)
!>

B

!>          B es un array REAL, de dimensión (NMAX*NRHS)
!>

BSAV

!>          BSAV es un array REAL, de dimensión (NMAX*NRHS)
!>

X

!>          X es un array REAL, de dimensión (NMAX*NRHS)
!>

XACT

!>          XACT es un array REAL, de dimensión (NMAX*NRHS)
!>

S

!>          S es un array REAL, de dimensión (2*NMAX)
!>

WORK

!>          WORK es un array REAL, de dimensión
!>                      (NMAX*max(3,NRHS,NMAX))
!>

RWORK

!>          RWORK es un array REAL, de dimensión
!>                      (NMAX+2*NRHS)
!>

IWORK

!>          IWORK es un array INTEGER, de dimensión (2*NMAX)
!>

NOUT

!>          NOUT es un INTEGER
!>          El número de unidad para la salida.
!>
