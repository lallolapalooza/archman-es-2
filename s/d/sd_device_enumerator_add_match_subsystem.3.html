NOMBRE

#include <systemd/sd-device.h>

int sd_device_enumerator_add_match_parent(sd_device_enumerator *enumerator, sd_device *parent);

int sd_device_enumerator_add_match_property(sd_device_enumerator *enumerator, const char *property, const char *value);

int sd_device_enumerator_add_match_property_required(sd_device_enumerator *enumerator, const char *property, const char *value);

int sd_device_enumerator_add_match_subsystem(sd_device_enumerator *enumerator, const char *subsystem, int match);

int sd_device_enumerator_add_match_sysattr(sd_device_enumerator *enumerator, const char *sysattr, const char *value, int match);

int sd_device_enumerator_add_match_sysname(sd_device_enumerator *enumerator, const char *sysname);

int sd_device_enumerator_add_nomatch_sysname(sd_device_enumerator *enumerator, const char *sysname);

int sd_device_enumerator_add_match_tag(sd_device_enumerator *enumerator, const char *tag);

int sd_device_enumerator_allow_uninitialized(sd_device_enumerator *enumerator);

int sd_device_enumerator_add_all_parents(sd_device_enumerator *enumerator);

DESCRIPCIÓN

La función sd_device_enumerator_add_match_parent() añade un filtro al enumerator para que solo se enumeren los dispositivos bajo el árbol del dispositivo parent especificado. Si se llama varias veces, el dispositivo parent establecido previamente se borra y solo la última llamada tiene efecto.

La función sd_device_enumerator_add_match_property() añade un filtro al enumerator para que solo se enumeren los dispositivos cuya property especificada sea igual al value. Tanto property como value pueden ser patrones glob. Cuando se llama varias veces, se enumeran los dispositivos que tienen al menos una de las propiedades especificadas con valores coincidentes. Es decir, los filtros se combinan con OR.

La función sd_device_enumerator_add_match_property_required() añade un filtro al enumerator para que solo se enumeren los dispositivos cuya property especificada sea igual al value. Esta función es similar a sd_device_enumerator_add_match_property(), pero cuando se llama varias veces, se enumeran los dispositivos que tienen todas las propiedades especificadas con valores coincidentes. Es decir, los filtros se combinan con AND.

La función sd_device_enumerator_add_match_subsystem() añade un filtro al enumerator para que se incluyan todos los dispositivos del subsystem especificado, cuando match es true. Cuando match es false, se enumeran todos los dispositivos excepto los del subsystem especificado. Cuando se llama varias veces, los filtros positivos se combinan con OR y los negativos con AND.

La función sd_device_enumerator_add_match_sysattr() añade un filtro sobre el atributo sysfs sysattr que coincide con value. value puede ser un patrón glob. Si value es NULL, se incluyen los dispositivos que tienen (si match es true) o no tienen (si match es false) el sysattr especificado, independientemente de su valor. Es decir, NULL es prácticamente equivalente a "*". Cuando esta función se llama varias veces, solo se enumeran los dispositivos que coinciden con todos los filtros sysattr especificados. Es decir, estos filtros se combinan con AND.

La función sd_device_enumerator_add_match_sysname() añade un filtro para que solo se enumeren los dispositivos cuyo sysname sea igual a sysname. sysname puede ser un patrón glob. Cuando se llama varias veces, los filtros se combinan con OR.

La función sd_device_enumerator_add_nomatch_sysname() añade un filtro para que los dispositivos cuyo sysname sea igual a sysname se excluyan de la enumeración. Esto es útil para excluir dispositivos específicos del proceso de enumeración. Cuando se llama varias veces, las características se combinan con AND.

La función sd_device_enumerator_add_match_tag() añade un filtro para que solo se enumeren los dispositivos etiquetados con tag. Cuando se llama varias veces, los filtros se combinan con OR.

La función sd_device_enumerator_allow_uninitialized() permite que los dispositivos que aún no han sido inicializados por udev se incluyan en la enumeración.

La función sd_device_enumerator_add_all_parents() enumera todos los dispositivos padres de los dispositivos coincidentes. Esto es útil para casos en los que se desean incluir todos los dispositivos padres en la enumeración, como cuando se está interesado en el árbol completo de dispositivos que conduce a un dispositivo específico.

VALOR DE RETORNO

Todas las funciones devuelven 0 o un entero positivo en caso de éxito, o un código de error negativo de estilo errno en caso de fallo.

Errores

Los errores devueltos pueden indicar los siguientes problemas:

-ENOMEM

Fallo en la asignación de memoria.

-EINVAL

Uno de los argumentos es inválido.

EJEMPLOS

Ejemplo 1. Detectar Dispositivos USB Extraíbles (Usando Coincidencia y Exclusión)

/* SPDX-License-Identifier: MIT-0 */
#include <stdbool.h>
#include <stdio.h>
#include <systemd/sd-device.h>
int main(void) {
    __attribute__((cleanup(sd_device_enumerator_unrefp))) sd_device_enumerator *enumerator = NULL;
    sd_device *device;
    int r;
    /* Crear un nuevo enumerador de dispositivos */
    r = sd_device_enumerator_new(&enumerator);
    if (r < 0) {
        fprintf(stderr, "Fallo al crear el enumerador de dispositivos: %s\n", strerror(-r));
        return 1;
    }
    /* Incluir solo dispositivos del subsistema "usb" */
    r = sd_device_enumerator_add_match_subsystem(enumerator, "usb", true);
    if (r < 0) {
        fprintf(stderr, "Fallo al añadir la coincidencia de subsistema: %s\n", strerror(-r));
        return 1;
    }
    /*
     * Excluir dispositivos donde el atributo sysfs "removable" sea "0"
     * Estos suelen ser dispositivos no extraíbles como interfaces USB integradas
     */
    r = sd_device_enumerator_add_match_sysattr(enumerator, "removable", "0", false);
    if (r < 0) {
        fprintf(stderr, "Fallo al añadir la coincidencia de sysattr: %s\n", strerror(-r));
        return 1;
    }
    /* Empezar a enumerar los dispositivos coincidentes */
    for (device = sd_device_enumerator_get_device_first(enumerator);
         device;
         device = sd_device_enumerator_get_device_next(enumerator)) {
        const char *syspath;
        /* Obtener la ruta sysfs del dispositivo */
        if (sd_device_get_syspath(device, &syspath) >= 0)
            printf("Dispositivo USB extraíble encontrado: %s\n", syspath);
    }
    return 0;
}

sd_device_enumerator_add_all_parents(3) — systemd-libs

NAME

sd_device_enumerator_add_all_parents - Add all parents of a device to the enumerator filter

SYNOPSIS 

#include <systemd/sd-device.h>

int
sd_device_enumerator_add_all_parents(sd_device_enumerator *enumerator,
                                     sd_device *device);

DESCRIPTION

This function is similar to sd_device_enumerator_add_parent(3), but instead of adding only the immediate parent of device, it adds all of its ancestors in the device tree.

Use this function to list a device and all of its parent devices up to the root.

RETURN VALUE

On success, this function returns a non-negative return value. On failure, it returns a negative errno error code.

EXAMPLES 

#include <stdio.h>
#include <stdlib.h>
#include <systemd/sd-device.h>

int main(int argc, char *argv[]) {

    sd_device_enumerator *enumerator = NULL;
    sd_device *device = NULL;
    sd_device *parent_device = NULL;
    sd_device_enumerator *parent_enumerator = NULL;
    int r;

    r = sd_device_enumerator_new(&enumerator);
    if (r < 0) {
        fprintf(stderr, "Failed to create device enumerator: %s\n", strerror(-r));
        goto finish;
    }

    r = sd_device_get_from_syspath(&device, "/sys/devices/pci0000:00/0000:00:1f.3/sound/card0");
    if (r < 0) {
        fprintf(stderr, "Failed to get device: %s\n", strerror(-r));
        goto finish;
    }

    r = sd_device_enumerator_add_all_parents(enumerator, device);
    if (r < 0) {
        fprintf(stderr, "Failed to add parent devices: %s\n", strerror(-r));
        goto finish;
    }

    r = sd_device_enumerator_new(&parent_enumerator);
    if (r < 0) {
        fprintf(stderr, "Failed to create parent enumerator: %s\n", strerror(-r));
        goto finish;
    }

    r = sd_device_enumerator_add_all_parents(parent_enumerator, device);
    if (r < 0) {
        fprintf(stderr, "Failed to add parent devices: %s\n", strerror(-r));
        goto finish;
    }

    r = sd_device_enumerator_get_device_first(parent_enumerator, &parent_device);
    if (r < 0) {
        fprintf(stderr, "Failed to get first parent device: %s\n", strerror(-r));
        goto finish;
    }

    printf("Parent device found: %s\n", sd_device_get_syspath(parent_device));

finish:
    sd_device_unref(device);
    sd_device_unref(parent_device);
    sd_device_enumerator_unref(enumerator);
    sd_device_enumerator_unref(parent_enumerator);

    return EXIT_SUCCESS;
}

HISTORY

The function sd_device_enumerator_add_all_parents() was added in version 258.
