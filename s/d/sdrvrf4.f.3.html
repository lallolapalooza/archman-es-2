Documentación de función/subrutina

subroutine sdrvrf4 (integer nout, integer nn, integer, dimension( nn ) nval, real thresh, real, dimension( ldc, * ) c1, real, dimension( ldc, *) c2, integer ldc, real, dimension( * ) crf, real, dimension( lda, * ) a, integer lda, real, dimension( * ) s_work_slange)

SDRVRF4

Propósito:

!>
!> SDRVRF4 prueba las rutinas RFP de LAPACK:
!>     SSFRK
!>

Parámetros

NOUT 
!>          NOUT es ENTERO
!>                El número de unidad para la salida.
!>

NN

!>          NN es ENTERO
!>                El número de valores de N contenidos en el vector NVAL.
!>

NVAL

!>          NVAL es un array de ENTEROS, de dimensión (NN)
!>                Los valores de la dimensión de la matriz N.
!>

THRESH

!>          THRESH es REAL
!>                El valor umbral para las ratios de prueba. Un resultado se
!>                incluye en el archivo de salida si RESULT >= THRESH. Para
!>                imprimir todas las ratios de prueba, use THRESH = 0.
!>

C1

!>          C1 es un array REAL,
!>                de dimensión (LDC,NMAX)
!>

C2

!>          C2 es un array REAL,
!>                de dimensión (LDC,NMAX)
!>

LDC

!>          LDC es ENTERO
!>                La dimensión principal del array A.
!>                LDA >= max(1,NMAX).
!>

CRF

!>          CRF es un array REAL,
!>                de dimensión ((NMAX*(NMAX+1))/2).
!>

A

!>          A es un array REAL,
!>                de dimensión (LDA,NMAX)
!>

LDA

!>          LDA es ENTERO
!>                La dimensión principal del array A. LDA >= max(1,NMAX).
!>

S_WORK_SLANGE

!>          S_WORK_SLANGE es un array REAL, de dimensión (NMAX)
!>
