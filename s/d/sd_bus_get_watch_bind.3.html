SD_BUS_SET_WATCH_BIND(3) sd_bus_set_watch_bind SD_BUS_SET_WATCH_BIND(3)

NOMBRE

sd_bus_set_watch_bind, sd_bus_get_watch_bind - Controlar la observación de la vinculación de sockets en conexiones de bus

SINOPSIS

#include <systemd/sd-bus.h>

int sd_bus_set_watch_bind(sd_bus *bus, int b);

int sd_bus_get_watch_bind(sd_bus *bus);

DESCRIPCIÓN

sd_bus_set_watch_bind() se puede utilizar para habilitar o deshabilitar la observación del lado del cliente de la vinculación de sockets del servidor para un objeto de conexión de bus. Si b es verdadero, la función se habilita; de lo contrario, se deshabilita (que es el comportamiento predeterminado). Cuando está habilitada, y la dirección del bus seleccionada se refiere a un socket AF_UNIX en el sistema de archivos que no existe mientras se realiza el intento de conexión, se instala un observador inotify(7) sobre él, esperando que aparezca el socket. Tan pronto como aparezca el socket, se establece la conexión. Esta funcionalidad es particularmente útil en programas de arranque temprano que necesitan ejecutarse antes de que el bus del sistema esté disponible, pero que desean conectarse a él en el instante en que sea posible.

sd_bus_get_watch_bind() se puede utilizar para consultar la configuración actual de esta función. Devuelve cero cuando la función está deshabilitada y un valor positivo si está habilitada.

Tenga en cuenta que no se aplica ningún tiempo de espera mientras esperamos que aparezca el socket. Esto significa que cualquier operación remota síncrona (como sd_bus_call(3), sd_bus_add_match(3) o sd_bus_request_name(3)), que se utiliza en una conexión con esta función habilitada que aún no se ha establecido, podría bloquearse indefinidamente si el socket nunca se crea. Sin embargo, las operaciones remotas asíncronas (como sd_bus_send(3), sd_bus_call_async(3), sd_bus_add_match_async(3)) no se bloquean en este caso y ponen en cola de forma segura las operaciones solicitadas para que se distribuyan en el instante en que se establece la conexión.

Utilice sd_bus_is_ready(3) para determinar si la conexión está completamente establecida, es decir, si el socket del par ha sido vinculado, conectado y autenticado. Utilice sd_bus_set_connected_signal(3) para ser notificado cuando la conexión esté completamente establecida.

VALOR DE RETORNO

En caso de éxito, estas funciones devuelven 0 o un entero positivo. En caso de error, devuelven un código de error negativo del tipo errno.

Errores

Los errores devueltos pueden indicar los siguientes problemas:

-ECHILD

La conexión de bus se ha creado en un proceso, biblioteca o instancia de módulo diferente.

NOTAS

Las funciones descritas aquí están disponibles como una biblioteca compartida, que se puede compilar y enlazar con el archivo pkg-config(1) de libsystemd.

El código descrito aquí utiliza getenv(3), que se declara como no seguro para subprocesos. Esto significa que el código que llama a las funciones descritas aquí no debe llamar a setenv(3) desde un hilo paralelo. Se recomienda hacer llamadas a setenv() solo desde una fase temprana del programa cuando no se hayan iniciado otros hilos.

EJEMPLO

Ejemplo 1. Crear un servicio simple del sistema que publica una propiedad en el bus del sistema y puede reconectarse cuando D-Bus se desconecta y se vuelve a conectar

/* SPDX-License-Identifier: MIT-0 */
/* Un servicio D-Bus que se reconecta automáticamente cuando se reinicia el bus del sistema.
 *
 * Compilar con 'cc sd_bus_service_reconnect.c $(pkg-config --libs --cflags libsystemd)'
 *
 * Para permitir que el programa tome posesión del nombre 'org.freedesktop.ReconnectExample',
 * agregue lo siguiente como /etc/dbus-1/system.d/org.freedesktop.ReconnectExample.conf
 * y luego recargue el agente con 'systemctl reload dbus':
<?xml version="1.0"?> <!--*-nxml-*-->
<!DOCTYPE busconfig PUBLIC "-//freedesktop//DTD D-BUS Bus Configuration 1.0//EN"
  "http://www.freedesktop.org/standards/dbus/1.0/busconfig.dtd">
<busconfig>
  <policy user="root">
    <allow own="org.freedesktop.ReconnectExample"/>
    <allow send_destination="org.freedesktop.ReconnectExample"/>
    <allow receive_sender="org.freedesktop.ReconnectExample"/>
  </policy>
  <policy context="default">
    <allow send_destination="org.freedesktop.ReconnectExample"/>
    <allow receive_sender="org.freedesktop.ReconnectExample"/>
  </policy>
</busconfig>
 *
 * Para obtener la propiedad a través de busctl:
 *
 * $ busctl --user get-property org.freedesktop.ReconnectExample \
 *                              /org/freedesktop/ReconnectExample \
 *                              org.freedesktop.ReconnectExample \
 *                              Example
 * s "example"
 */
#include <errno.h>
#include <stdio.h>
#include <stdlib.h>
#include <systemd/sd-bus.h>
#define _cleanup_(f) __attribute__((cleanup(f)))
static int log_error(int r, const char *str) {
  fprintf(stderr, "%s failed: %s\n", str, strerror(-r));
  return r;
}
typedef struct object {
  const char *example;
  sd_bus **bus;
  sd_event **event;
} object;
static int property_get(
                sd_bus *bus,
                const char *path,
                const char *interface,
                const char *property,
                sd_bus_message *reply,
                void *userdata,
                sd_bus_error *error) {
  object *o = userdata;
  if (strcmp(property, "Example") == 0)
    return sd_bus_message_append(reply, "s", o->example);
  return sd_bus_error_setf(error,
                           SD_BUS_ERROR_UNKNOWN_PROPERTY,
                           "Unknown property '%s'",
                           property);
}
/* https://www.freedesktop.org/software/systemd/man/sd_bus_add_object.html */
static const sd_bus_vtable vtable[] = {
  SD_BUS_VTABLE_START(0),
  SD_BUS_PROPERTY(
    "Example", "s",
    property_get,
    0,
    SD_BUS_VTABLE_PROPERTY_CONST),
  SD_BUS_VTABLE_END
};
static int setup(object *o);
static int on_disconnect(sd_bus_message *message, void *userdata, sd_bus_error *ret_error) {
  int r;
  r = setup((object *)userdata);
  if (r < 0) {
    object *o = userdata;
    r = sd_event_exit(*o->event, r);
    if (r < 0)
      return log_error(r, "sd_event_exit()");
  }
  return 1;
}
/* Asegurarse de que el bucle de eventos salga con un error claro si la adquisición del nombre de servicio conocido falla */
static int request_name_callback(sd_bus_message *m, void *userdata, sd_bus_error *ret_error) {
  int r;
  if (!sd_bus_message_is_method_error(m, NULL))
    return 1;
  const sd_bus_error *error = sd_bus_message_get_error(m);
  if (sd_bus_error_has_names(error, SD_BUS_ERROR_TIMEOUT, SD_BUS_ERROR_NO_REPLY))
    return 1; /* El bus no está disponible, intente de nuevo más tarde */
  fprintf(stderr, "Fallo al solicitar el nombre: %s\n", error->message);
  object *o = userdata;
  r = sd_event_exit(*o->event, -sd_bus_error_get_errno(error));
  if (r < 0)
    return log_error(r, "sd_event_exit()");
  return 1;
}
static int setup(object *o) {
  int r;
  /* Si nos estamos reconectando, el objeto de bus debe cerrarse, desvincularse
   * del bucle de eventos y recrearse.
   * https://www.freedesktop.org/software/systemd/man/sd_bus_detach_event.html
   * https://www.freedesktop.org/software/systemd/man/sd_bus_close_unref.html
   */
  if (*o->bus) {
    r = sd_bus_detach_event(*o->bus);
    if (r < 0)
      return log_error(r, "sd_bus_detach_event()");
    *o->bus = sd_bus_close_unref(*o->bus);
  }
  /* Configurar un nuevo objeto de bus para el bus del sistema, configurarlo para esperar a que D-Bus
   * esté disponible en lugar de fallar si no lo está, e iniciarlo. Todas las
   * operaciones siguientes son asíncronas y no se bloquearán esperando a que D-Bus
   * esté disponible.
   * https://www.freedesktop.org/software/systemd/man/sd_bus_new.html
   * https://www.freedesktop.org/software/systemd/man/sd_bus_set_address.html
   * https://www.freedesktop.org/software/systemd/man/sd_bus_set_bus_client.html
   * https://www.freedesktop.org/software/systemd/man/sd_bus_negotiate_creds.html
   * https://www.freedesktop.org/software/systemd/man/sd_bus_set_watch_bind.html
   * https://www.freedesktop.org/software/systemd/man/sd_bus_set_connected_signal.html
   * https://www.freedesktop.org/software/systemd/man/sd_bus_start.html
   */
  r = sd_bus_new(o->bus);
  if (r < 0)
    return log_error(r, "sd_bus_new()");
  r = sd_bus_set_bus_client(*o->bus, true);
  if (r < 0)
    return log_error(r, "sd_bus_set_bus_client()");
  r = sd_bus_set_address(*o->bus, "unix:path=/run/user/1000/bus");
  if (r < 0)
    return log_error(r, "sd_bus_set_address()");
  r = sd_bus_negotiate_creds(*o->bus, SD_BUS_CREDS_UNIX_UID | SD_BUS_CREDS_CAP_PEER, NULL);
  if (r < 0)
    return log_error(r, "sd_bus_negotiate_creds()");
  r = sd_bus_set_watch_bind(*o->bus, true);
  if (r < 0)
    return log_error(r, "sd_bus_set_watch_bind()");
  r = sd_bus_set_connected_signal(*o->bus, NULL, on_disconnect, o);
  if (r < 0)
    return log_error(r, "sd_bus_set_connected_signal()");
  r = sd_bus_start(*o->bus);
  if (r < 0)
    return log_error(r, "sd_bus_start()");
  /* Reenviar el nombre */
  r = sd_bus_call_method(*o->bus,
                         "org.freedesktop.DBus",
                         "/org/freedesktop/DBus",
                         "org.freedesktop.DBus",
                         "RequestName",
                         "su",
                         0, /* UID */
                         1, /* Flags */
                         "s",
                         "org.freedesktop.ReconnectExample",
                         NULL);
  if (r < 0) {
    if (sd_bus_error_has_names(&error, SD_BUS_ERROR_TIMEOUT, SD_BUS_ERROR_NO_REPLY))
      return 1; /* El bus aún no está listo, intente de nuevo más tarde */
    return log_error(r, "sd_bus_call_method()");
  }
  /* Configurar el objeto */
  r = sd_bus_add_object(*o->bus,
                        "/org/freedesktop/ReconnectExample",
                        vtable,
                        o);
  if (r < 0)
    return log_error(r, "sd_bus_add_object()");
  return 0;
}
int main(int argc, char *argv[]) {
  int r;
  sd_event *event = NULL;
  sd_bus *bus = NULL;
  object o = {
    .example = "example",
    .bus = &bus,
    .event = &event,
  };
  /* Ponerse en cola para un bucle de eventos que se reinicia automáticamente
   * cuando el bus del sistema deja de estar disponible y regresa.
   * https://www.freedesktop.org/software/systemd/man/sd_event_new.html
   * https://www.freedesktop.org/software/systemd/man/sd_event_add_defer.html
   */
  r = sd_event_new(&event);
  if (r < 0)
    return log_error(r, "sd_event_new()");
  r = sd_event_add_defer(event, NULL, setup, &o);
  if (r < 0)
    return log_error(r, "sd_event_add_defer()");
  /* Iniciar el bucle de eventos */
  r = sd_event_loop(event);
  /* Limpieza */
  sd_bus_close_unref(bus);
  sd_event_unref(event);
  return r < 0 ? EXIT_FAILURE : EXIT_SUCCESS;
}
