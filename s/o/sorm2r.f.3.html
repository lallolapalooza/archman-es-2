subroutine sorm2r (character side, character trans, integer m, integer n, integer k, real, dimension( lda, * ) a, integer lda, real, dimension( * ) tau, real, dimension( ldc, * ) c, integer ldc, real, dimension( * ) work, integer info)

SORM2R multiplica una matriz general por la matriz ortogonal de una factorización QR determinada por sgeqrf (algoritmo no bloqueado).

Propósito:

!>
!> SORM2R sobrescribe la matriz general real m por n C con
!>
!>       Q * C  si SIDE = 'L' y TRANS = 'N', o
!>
!>       Q**T* C  si SIDE = 'L' y TRANS = 'T', o
!>
!>       C * Q  si SIDE = 'R' y TRANS = 'N', o
!>
!>       C * Q**T si SIDE = 'R' y TRANS = 'T',
!>
!> donde Q es una matriz ortogonal real definida como el producto de k
!>   reflectores elementales
!>
!>       Q = H(1) H(2) . . . H(k)
!>
!> como lo devuelve SGEQRF. Q es de orden m si SIDE = 'L' y de orden n
!>   si SIDE = 'R'.
!>

Parámetros

SIDE 
!>          SIDE es CHARACTER*1
!>          = 'L': aplicar Q o Q**T desde la Izquierda
!>          = 'R': aplicar Q o Q**T desde la Derecha
!>

TRANS

!>          TRANS es CHARACTER*1
!>          = 'N': aplicar Q  (Sin transpuesta)
!>          = 'T': aplicar Q**T (Transpuesta)
!>

M

!>          M es INTEGER
!>          El número de filas de la matriz C. M >= 0.
!>

N

!>          N es INTEGER
!>          El número de columnas de la matriz C. N >= 0.
!>

K

!>          K es INTEGER
!>          El número de reflectores elementales cuyo producto define
!>          la matriz Q.
!>          Si SIDE = 'L', M >= K >= 0;
!>          si SIDE = 'R', N >= K >= 0.
!>

A

!>          A es una matriz REAL, dimensión (LDA,K)
!>          La i-ésima columna debe contener el vector que define el
!>          reflector elemental H(i), para i = 1,2,...,k, como lo devuelve
!>          SGEQRF en las primeras k columnas de su argumento de matriz A.
!>          A es modificada por la rutina pero restaurada al salir.
!>

LDA

!>          LDA es INTEGER
!>          La dimensión principal de la matriz A.
!>          Si SIDE = 'L', LDA >= max(1,M);
!>          si SIDE = 'R', LDA >= max(1,N).
!>

TAU

!>          TAU es una matriz REAL, dimensión (K)
!>          TAU(i) debe contener el factor escalar del reflector
!>          elemental H(i), como lo devuelve SGEQRF.
!>

C

!>          C es una matriz REAL, dimensión (LDC,N)
!>          En la entrada, la matriz m por n C.
!>          En la salida, C es sobrescrita por Q*C o Q**T*C o C*Q**T o C*Q.
!>

LDC

!>          LDC es INTEGER
!>          La dimensión principal de la matriz C. LDC >= max(1,M).
!>

WORK

!>          WORK es una matriz REAL, dimensión
!>                                   (N) si SIDE = 'L',
!>                                   (M) si SIDE = 'R'
!>

INFO

!>          INFO es INTEGER
!>          = 0: salida exitosa
!>          < 0: si INFO = -i, el i-ésimo argumento tenía un valor ilegal
!>
