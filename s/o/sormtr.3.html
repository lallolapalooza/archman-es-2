subroutine sormtr (character side, character uplo, character trans, integer m, integer n, real, dimension( lda, * ) a, integer lda, real, dimension( * ) tau, real, dimension( ldc, * ) c, integer ldc, real, dimension( * ) work, integer lwork, integer info)

SORMTR

Propósito:

!>
!> SORMTR sobrescribe la matriz real general M por N, C, con
!>
!>                 SIDE = 'L'     SIDE = 'R'
!> TRANS = 'N':      Q * C          C * Q
!> TRANS = 'T':      Q**T * C       C * Q**T
!>
!> donde Q es una matriz ortogonal real de orden nq, con nq = m si
!> SIDE = 'L' y nq = n si SIDE = 'R'. Q se define como el producto de
!> nq-1 reflectores elementales, como se devuelve por SSYTRD:
!>
!> si UPLO = 'U', Q = H(nq-1) . . . H(2) H(1);
!>
!> si UPLO = 'L', Q = H(1) H(2) . . . H(nq-1).
!>

Parámetros

SIDE 
!>          SIDE es CHARACTER*1
!>          = 'L': aplicar Q o Q**T desde la Izquierda;
!>          = 'R': aplicar Q o Q**T desde la Derecha.
!>

UPLO

!>          UPLO es CHARACTER*1
!>          = 'U': el triángulo superior de A contiene los reflectores elementales
!>                 de SSYTRD;
!>          = 'L': el triángulo inferior de A contiene los reflectores elementales
!>                 de SSYTRD.
!>

TRANS

!>          TRANS es CHARACTER*1
!>          = 'N':  Sin transponer, aplicar Q;
!>          = 'T':  Transpuesta, aplicar Q**T.
!>

M

!>          M es INTEGER
!>          El número de filas de la matriz C. M >= 0.
!>

N

!>          N es INTEGER
!>          El número de columnas de la matriz C. N >= 0.
!>

A

!>          A es array REAL, dimensión
!>                               (LDA,M) si SIDE = 'L'
!>                               (LDA,N) si SIDE = 'R'
!>          Los vectores que definen los reflectores elementales, como
!>          se devuelve por SSYTRD.
!>

LDA

!>          LDA es INTEGER
!>          La dimensión principal del array A.
!>          LDA >= max(1,M) si SIDE = 'L'; LDA >= max(1,N) si SIDE = 'R'.
!>

TAU

!>          TAU es array REAL, dimensión
!>                               (M-1) si SIDE = 'L'
!>                               (N-1) si SIDE = 'R'
!>          TAU(i) debe contener el factor escalar del reflector elemental
!>          H(i), como se devuelve por SSYTRD.
!>

C

!>          C es array REAL, dimensión (LDC,N)
!>          En la entrada, la matriz C de M por N.
!>          En la salida, C se sobrescribe con Q*C o Q**T*C o C*Q**T o C*Q.
!>

LDC

!>          LDC es INTEGER
!>          La dimensión principal del array C. LDC >= max(1,M).
!>

WORK

!>          WORK es array REAL, dimensión (MAX(1,LWORK))
!>          En la salida, si INFO = 0, WORK(1) devuelve el LWORK óptimo.
!>

LWORK

!>          LWORK es INTEGER
!>          La dimensión del array WORK.
!>          Si SIDE = 'L', LWORK >= max(1,N);
!>          si SIDE = 'R', LWORK >= max(1,M).
!>          Para un rendimiento óptimo LWORK >= N*NB si SIDE = 'L', y
!>          LWORK >= M*NB si SIDE = 'R', donde NB es el tamaño de bloque óptimo.
!>
!>          Si LWORK = -1, entonces se asume una consulta de espacio de trabajo; la rutina
!>          solo calcula el tamaño óptimo del array WORK, devuelve
!>          este valor como la primera entrada del array WORK, y XERBLA no emite ningún mensaje de error
!>          relacionado con LWORK.
!>

INFO

!>          INFO es INTEGER
!>          = 0:  salida exitosa
!>          < 0:  si INFO = -i, el i-ésimo argumento tenía un valor ilegal
!>
