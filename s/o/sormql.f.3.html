subroutine sormql (character side, character trans, integer m, integer n, integer k, real, dimension( lda, * ) a, integer lda, real, dimension( * ) tau, real, dimension( ldc, * ) c, integer ldc, real, dimension( * ) work, integer lwork, integer info)

SORMQL

Propósito:

!>
!> SORMQL sobrescribe la matriz real general M por N, C, con
!>
!>                 SIDE = 'L'     SIDE = 'R'
!> TRANS = 'N':      Q * C          C * Q
!> TRANS = 'T':      Q**T * C       C * Q**T
!>
!> donde Q es una matriz ortogonal real definida como el producto de k
!> reflectores elementales
!>
!>       Q = H(k) . . . H(2) H(1)
!>
!> como se devuelve por SGEQLF. Q es de orden M si SIDE = 'L' y de orden N
!> si SIDE = 'R'.
!>

Parámetros

SIDE 
!>          SIDE es CHARACTER*1
!>          = 'L': aplicar Q o Q**T desde la Izquierda;
!>          = 'R': aplicar Q o Q**T desde la Derecha.
!>

TRANS

!>          TRANS es CHARACTER*1
!>          = 'N':  Sin transpuesta, aplicar Q;
!>          = 'T':  Transpuesta, aplicar Q**T.
!>

M

!>          M es INTEGER
!>          El número de filas de la matriz C. M >= 0.
!>

N

!>          N es INTEGER
!>          El número de columnas de la matriz C. N >= 0.
!>

K

!>          K es INTEGER
!>          El número de reflectores elementales cuyo producto define
!>          la matriz Q.
!>          Si SIDE = 'L', M >= K >= 0;
!>          si SIDE = 'R', N >= K >= 0.
!>

A

!>          A es array REAL, dimension (LDA,K)
!>          La columna i debe contener el vector que define el reflector
!>          elemental H(i), para i = 1,2,...,k, como se devuelve por
!>          SGEQLF en las últimas k columnas de su array argumento A.
!>

LDA

!>          LDA es INTEGER
!>          La dimensión principal del array A.
!>          Si SIDE = 'L', LDA >= max(1,M);
!>          si SIDE = 'R', LDA >= max(1,N).
!>

TAU

!>          TAU es array REAL, dimension (K)
!>          TAU(i) debe contener el factor escalar del reflector
!>          elemental H(i), como se devuelve por SGEQLF.
!>

C

!>          C es array REAL, dimension (LDC,N)
!>          En la entrada, la matriz C de M por N.
!>          En la salida, C se sobrescribe con Q*C o Q**T*C o C*Q**T o C*Q.
!>

LDC

!>          LDC es INTEGER
!>          La dimensión principal del array C. LDC >= max(1,M).
!>

WORK

!>          WORK es array REAL, dimension (MAX(1,LWORK))
!>          En la salida, si INFO = 0, WORK(1) devuelve el LWORK óptimo.
!>

LWORK

!>          LWORK es INTEGER
!>          La dimensión del array WORK.
!>          Si SIDE = 'L', LWORK >= max(1,N);
!>          si SIDE = 'R', LWORK >= max(1,M).
!>          Para un buen rendimiento, LWORK debería ser generalmente mayor.
!>
!>          Si LWORK = -1, entonces se asume una consulta de espacio de trabajo; la rutina
!>          solo calcula el tamaño óptimo del array WORK, devuelve
!>          este valor como la primera entrada del array WORK, y ninguna
!>          mensaje de error relacionado con LWORK es emitido por XERBLA.
!>

INFO

!>          INFO es INTEGER
!>          = 0:  salida exitosa
!>          < 0:  si INFO = -i, el i-ésimo argumento tuvo un valor ilegal
!>
