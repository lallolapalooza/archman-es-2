subroutine sorhr_col02 (integer m, integer n, integer mb1, integer nb1, integer nb2, real, dimension(6) resultado)

SORHR_COL02

Propósito:

!>
!> SORHR_COL02 prueba SORGTSQR_ROW y SORHR_COL dentro de SGETSQRHRT
!> (que llama a SLATSQR, SORGTSQR_ROW y SORHR_COL) usando SGEMQRT.
!> Por lo tanto, SLATSQR (parte de SGEQR), SGEMQRT (parte de SGEMQR)
!> deben ser probados antes de esta prueba.
!>
!>

Parámetros

M 
!>          M es INTEGER
!>          Número de filas en la matriz de prueba.
!>

N

!>          N es INTEGER
!>          Número de columnas en la matriz de prueba.
!>

MB1

!>          MB1 es INTEGER
!>          Número de filas en el bloque de filas en una matriz de prueba de entrada.
!>

NB1

!>          NB1 es INTEGER
!>          Número de columnas en el bloque de columnas de una matriz de prueba de entrada.
!>

NB2

!>          NB2 es INTEGER
!>          Número de columnas en el bloque de columnas en una matriz de salida.
!>

RESULTADO

!>          RESULTADO es un array REAL, de dimensión (6)
!>          Resultados de cada una de las seis pruebas a continuación.
!>
!>            A es una matriz de entrada de prueba m-por-n a factorizar.
!>            tal que A = Q_gr * ( R )
!>                               ( 0 ),
!>
!>            Q_qr es una matriz Q ortogonal implícita m-por-m, el resultado
!>            de la factorización en representación WY bloqueada,
!>            almacenada en formato de salida SGEQRT.
!>
!>            R es una matriz triangular superior n-por-n,
!>
!>            0 es una matriz cero (m-n)-por-n,
!>
!>            Q es una matriz ortogonal explícita m-por-m Q = Q_gr * I
!>
!>            C es una matriz aleatoria m-por-n,
!>
!>            D es una matriz aleatoria n-por-m.
!>
!>            Las seis pruebas son:
!>
!>            RESULTADO(1) = |R - (Q**H) * A| / ( eps * m * |A| )
!>              es equivalente a la prueba de | A - Q * R | / (eps * m * |A|),
!>
!>            RESULTADO(2) = |I - (Q**H) * Q| / ( eps * m ),
!>
!>            RESULTADO(3) = | Q_qr * C - Q * C | / (eps * m * |C|),
!>
!>            RESULTADO(4) = | (Q_gr**H) * C - (Q**H) * C | / (eps * m * |C|)
!>
!>            RESULTADO(5) = | D * Q_qr - D * Q | / (eps * m * |D|)
!>
!>            RESULTADO(6) = | D * (Q_qr**H) - D * (Q**H) | / (eps * m * |D|),
!>
!>            donde:
!>              Q_qr * C, (Q_gr**H) * C, D * Q_qr, D * (Q_qr**H) son
!>              calculados usando SGEMQRT,
!>
!>              Q * C, (Q**H) * C, D * Q, D * (Q**H) son
!>              calculados usando SGEMM.
!>
