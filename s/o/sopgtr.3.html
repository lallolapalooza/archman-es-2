subroutine sopgtr (character uplo, integer n, real, dimension( * ) ap, real, dimension( * ) tau, real, dimension( ldq, * ) q, integer ldq, real, dimension( * ) work, integer info)

SOPGTR

Propósito:

!>
!> SOPGTR genera una matriz ortogonal real Q que se define como el
!> producto de n-1  reflectores elementales H(i) de orden n, como se devuelve
!> por SSPTRD usando almacenamiento empaquetado:
!>
!> si UPLO = 'U', Q = H(n-1) . . . H(2) H(1),
!>
!> si UPLO = 'L', Q = H(1) H(2) . . . H(n-1).
!>

Parámetros

UPLO 
!>          UPLO es CHARACTER*1
!>          = 'U': Almacenamiento empaquetado triangular superior utilizado en la
!>                 llamada previa a SSPTRD;
!>          = 'L': Almacenamiento empaquetado triangular inferior utilizado en la
!>                 llamada previa a SSPTRD.
!>

N

!>          N es INTEGER
!>          El orden de la matriz Q. N >= 0.
!>

AP

!>          AP es un array REAL, dimension (N*(N+1)/2)
!>          Los vectores que definen los  reflectores elementales, como
!>          se devuelve por SSPTRD.
!>

TAU

!>          TAU es un array REAL, dimension (N-1)
!>          TAU(i) debe contener el factor escalar del reflector
!>          elemental H(i), como se devuelve por SSPTRD.
!>

Q

!>          Q es un array REAL, dimension (LDQ,N)
!>          La matriz ortogonal N-por-N, Q.
!>

LDQ

!>          LDQ es INTEGER
!>          La dimensión principal del array Q. LDQ >= max(1,N).
!>

WORK

!>          WORK es un array REAL, dimension (N-1)
!>

INFO

!>          INFO es INTEGER
!>          = 0:  salida exitosa
!>          < 0:  si INFO = -i, el i-ésimo argumento tenía un valor ilegal
!>
