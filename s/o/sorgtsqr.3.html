subroutine sorgtsqr (integer m, integer n, integer mb, integer nb, real, dimension( lda, * ) a, integer lda, real, dimension( ldt, * ) t, integer ldt, real, dimension( * ) work, integer lwork, integer info)

SORGTSQR

Propósito:

!>
!> SORGTSQR genera una matriz real M por N Q_out con columnas ortonormales,
!> que son las primeras N columnas de un producto de matrices ortogonales reales
!> de orden M que son devueltas por SLATSQR
!>
!>      Q_out = primeras_N_columnas_de( Q(1)_in * Q(2)_in * ... * Q(k)_in ).
!>
!> Ver la documentación de SLATSQR.
!>

Parámetros

M 
!>          M es ENTERO
!>          El número de filas de la matriz A.  M >= 0.
!>

N

!>          N es ENTERO
!>          El número de columnas de la matriz A. M >= N >= 0.
!>

MB

!>          MB es ENTERO
!>          El tamaño del bloque de filas utilizado por SLATSQR para devolver
!>          las matrices A y T. MB > N.
!>          (Nota: si MB > M, entonces M se usa en lugar de MB
!>          como el tamaño del bloque de filas).
!>

NB

!>          NB es ENTERO
!>          El tamaño del bloque de columnas utilizado por SLATSQR para devolver
!>          las matrices A y T. NB >= 1.
!>          (Nota: si NB > N, entonces N se usa en lugar de NB
!>          como el tamaño del bloque de columnas).
!>

A

!>          A es matriz REAL, dimensión (LDA,N)
!>
!>          En la entrada:
!>
!>             Los elementos en y por encima de la diagonal no se acceden.
!>             Los elementos por debajo de la diagonal representan la matriz
!>             inferior-trapezoidal en bloque unitaria V calculada por SLATSQR
!>             que define las matrices de entrada Q_in(k) (los unos en la
!>             diagonal no se almacenan) (mismo formato que la salida A
!>             por debajo de la diagonal en SLATSQR).
!>
!>          En la salida:
!>
!>             La matriz A contiene una matriz ortonormal M por N, Q_out,
!>             es decir, las columnas de A son vectores unitarios ortogonales.
!>

LDA

!>          LDA es ENTERO
!>          La dimensión principal de la matriz A. LDA >= max(1,M).
!>

T

!>          T es matriz REAL,
!>          dimensión (LDT, N * NIRB)
!>          donde NIRB = Número_de_bloques_de_filas_de_entrada
!>                     = MAX( 1, CEIL((M-N)/(MB-N)) )
!>          Sea NICB = Número_de_bloques_de_columnas_de_entrada
!>                   = CEIL(N/NB)
!>
!>          Los reflectores de bloque triangulares superiores utilizados para definir las
!>          matrices de entrada Q_in(k), k=(1:NIRB*NICB). Los reflectores de bloque
!>          se almacenan en forma compacta en NIRB secuencias de reflectores de bloque.
!>          Cada una de las NIRB secuencias de reflectores de bloque se almacena
!>          en un bloque de columna NB por N más grande de T y consta de NICB
!>          bloques de columna triangulares superiores más pequeños de NB por NB.
!>          (mismo formato que la salida T en SLATSQR).
!>

LDT

!>          LDT es ENTERO
!>          La dimensión principal de la matriz T.
!>          LDT >= max(1,min(NB1,N)).
!>

WORK

!>          (espacio de trabajo) matriz REAL, dimensión (MAX(2,LWORK))
!>          En la salida, si INFO = 0, WORK(1) devuelve el LWORK óptimo.
!>

LWORK

!>          LWORK es ENTERO
!>          La dimensión de la matriz WORK. LWORK >= (M+NB)*N.
!>          Si LWORK = -1, entonces se asume una consulta de espacio de trabajo.
!>          La rutina solo calcula el tamaño óptimo de la matriz WORK,
!>          devuelve este valor como la primera entrada de la matriz WORK,
!>          y XERBLA no emite ningún mensaje de error relacionado con LWORK.
!>

INFO

!>          INFO es ENTERO
!>          = 0:  salida exitosa
!>          < 0:  si INFO = -i, el argumento i-ésimo tenía un valor ilegal
!>
