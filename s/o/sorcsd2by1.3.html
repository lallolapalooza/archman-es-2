subroutine sorcsd2by1 (character jobu1, character jobu2, character jobv1t, integer m, integer p, integer q, real, dimension(ldx11,*) x11, integer ldx11, real, dimension(ldx21,*) x21, integer ldx21, real, dimension(*) theta, real, dimension(ldu1,*) u1, integer ldu1, real, dimension(ldu2,*) u2, integer ldu2, real, dimension(ldv1t,*) v1t, integer ldv1t, real, dimension(*) work, integer lwork, integer, dimension(*) iwork, integer info)

SORCSD2BY1

Propósito:

!>
!> SORCSD2BY1 calcula la descomposición CS de una matriz X de M por Q con
!> columnas ortonormales que ha sido particionada en una estructura de 2 bloques por 1:
!>
!>                                [  I1 0  0 ]
!>                                [  0  C  0 ]
!>          [ X11 ]   [ U1 |    ] [  0  0  0 ]
!>      X = [-----] = [---------] [----------] V1**T .
!>          [ X21 ]   [    | U2 ] [  0  0  0 ]
!>                                [  0  S  0 ]
!>                                [  0  0  I2]
!>
!> X11 es P-por-Q. Las matrices ortogonales U1, U2 y V1 son P-por-P,
!> (M-P)-por-(M-P) y Q-por-Q respectivamente. C y S son matrices diagonales no negativas R-por-R
!> tales que C^2 + S^2 = I, donde R = MIN(P,M-P,Q,M-Q). I1 es una matriz identidad K1-por-K1 y I2 es una
!> matriz identidad K2-por-K2, donde K1 = MAX(Q+P-M,0), K2 = MAX(Q-P,0).
!>

Parámetros

JOBU1 
!>          JOBU1 es CHARACTER
!>          = 'Y':      se calcula U1;
!>          en caso contrario:  U1 no se calcula.
!>

JOBU2

!>          JOBU2 es CHARACTER
!>          = 'Y':      se calcula U2;
!>          en caso contrario:  U2 no se calcula.
!>

JOBV1T

!>          JOBV1T es CHARACTER
!>          = 'Y':      se calcula V1T;
!>          en caso contrario:  V1T no se calcula.
!>

M

!>          M es INTEGER
!>          El número de filas de X.
!>

P

!>          P es INTEGER
!>          El número de filas de X11. 0 <= P <= M.
!>

Q

!>          Q es INTEGER
!>          El número de columnas de X11 y X21. 0 <= Q <= M.
!>

X11

!>          X11 es array REAL, dimension (LDX11,Q)
!>          A la entrada, parte de la matriz ortogonal cuya CSD se desea.
!>

LDX11

!>          LDX11 es INTEGER
!>          La dimensión principal de X11. LDX11 >= MAX(1,P).
!>

X21

!>          X21 es array REAL, dimension (LDX21,Q)
!>          A la entrada, parte de la matriz ortogonal cuya CSD se desea.
!>

LDX21

!>          LDX21 es INTEGER
!>           La dimensión principal de X21. LDX21 >= MAX(1,M-P).
!>

THETA

!>          THETA es array REAL, dimension (R), donde R =
!>          MIN(P,M-P,Q,M-Q).
!>          C = DIAG( COS(THETA(1)), ... , COS(THETA(R)) ) y
!>          S = DIAG( SIN(THETA(1)), ... , SIN(THETA(R)) ).
!>

U1

!>          U1 es array REAL, dimension (P)
!>          Si JOBU1 = 'Y', U1 contiene la matriz ortogonal P-por-P U1.
!>

LDU1

!>          LDU1 es INTEGER
!>          La dimensión principal de U1. Si JOBU1 = 'Y', LDU1 >=
!>          MAX(1,P).
!>

U2

!>          U2 es array REAL, dimension (M-P)
!>          Si JOBU2 = 'Y', U2 contiene la matriz ortogonal (M-P)-por-(M-P)
!>          U2.
!>

LDU2

!>          LDU2 es INTEGER
!>          La dimensión principal de U2. Si JOBU2 = 'Y', LDU2 >=
!>          MAX(1,M-P).
!>

V1T

!>          V1T es array REAL, dimension (Q)
!>          Si JOBV1T = 'Y', V1T contiene la matriz ortogonal Q-por-Q
!>          V1**T.
!>

LDV1T

!>          LDV1T es INTEGER
!>          La dimensión principal de V1T. Si JOBV1T = 'Y', LDV1T >=
!>          MAX(1,Q).
!>

WORK

!>          WORK es array REAL, dimension (MAX(1,LWORK))
!>          A la salida, si INFO = 0, WORK(1) devuelve el LWORK óptimo.
!>          Si INFO > 0 a la salida, WORK(2:R) contiene los valores PHI(1),
!>          ..., PHI(R-1) que, junto con THETA(1), ..., THETA(R),
!>          definen la matriz en forma bidiagonal-bloque intermedia
!>          que queda tras la no convergencia. INFO especifica el número
!>          de PHI's distintos de cero.
!>

LWORK

!>          LWORK es INTEGER
!>          La dimensión del array WORK.
!>
!>          Si LWORK = -1, entonces se asume una consulta de espacio de
!>          trabajo; la rutina solo calcula el tamaño óptimo del array WORK,
!>          devuelve este valor como la primera entrada del array work, y
!>          XERBLA no emite ningún mensaje de error relacionado con LWORK.
!>

IWORK

!>          IWORK es array INTEGER, dimension (M-MIN(P,M-P,Q,M-Q))
!>

INFO

!>          INFO es INTEGER
!>          = 0:  salida exitosa.
!>          < 0:  si INFO = -i, el i-ésimo argumento tenía un valor ilegal.
!>          > 0:  SBBCSD no convergió. Véase la descripción de WORK
!>                arriba para más detalles.
!>

Referencias:

[1] Brian D. Sutton. Computing the complete CS decomposition. Numer. Algorithms, 50(1):33-65, 2009.
