subroutine shgeqz (character job, character compq, character compz, integer n, integer ilo, integer ihi, real, dimension( ldh, * ) h, integer ldh, real, dimension( ldt, * ) t, integer ldt, real, dimension( * ) alphar, real, dimension( * ) alphai, real, dimension( * ) beta, real, dimension( ldq, * ) q, integer ldq, real, dimension( ldz, * ) z, integer ldz, real, dimension( * ) work, integer lwork, integer info)

SHGEQZ

Propósito:

!>
!> SHGEQZ calcula los valores propios de un par de matrices reales (H,T),
!> donde H es una matriz de Hessenberg superior y T es triangular superior,
!> utilizando el método QZ de doble desplazamiento.
!> Los pares de matrices de este tipo se producen por la reducción a
!> la forma de Hessenberg generalizada de un par de matrices reales (A,B):
!>
!>    A = Q1*H*Z1**T,  B = Q1*T*Z1**T,
!>
!> como se calcula por SGGHRD.
!>
!> Si JOB='S', entonces el par de Hessenberg-triangular (H,T) se
!> reduce también a la forma de Schur generalizada,
!>
!>    H = Q*S*Z**T,  T = Q*P*Z**T,
!>
!> donde Q y Z son matrices ortogonales, P es una matriz triangular superior
!> y S es una matriz cuasi-triangular con bloques diagonales de 1x1 y 2x2.
!>
!> Los bloques de 1x1 corresponden a valores propios reales del par de matrices
!> (H,T) y los bloques de 2x2 corresponden a pares conjugados complejos de
!> valores propios.
!>
!> Adicionalmente, los bloques diagonales triangulares superiores de 2x2 de P
!> correspondientes a los bloques de 2x2 de S se reducen a forma diagonal positiva,
!> es decir, si S(j+1,j) no es cero, entonces P(j+1,j) = P(j,j+1) = 0,
!> P(j,j) > 0, y P(j+1,j+1) > 0.
!>
!> Opcionalmente, la matriz ortogonal Q de la factorización de Schur
!> generalizada puede ser postmultiplicada en una matriz de entrada Q1, y la
!> matriz ortogonal Z puede ser postmultiplicada en una matriz de entrada Z1.
!> Si Q1 y Z1 son las matrices ortogonales de SGGHRD que redujeron
!> el par de matrices (A,B) a la forma de Hessenberg generalizada, entonces las
!> matrices de salida Q1*Q y Z1*Z son los factores ortogonales de la
!> factorización de Schur generalizada de (A,B):
!>
!>    A = (Q1*Q)*S*(Z1*Z)**T,  B = (Q1*Q)*P*(Z1*Z)**T.
!>
!> Para evitar el desbordamiento, los valores propios del par de matrices
!> (H,T) (equivalente, de (A,B)) se calculan como un par de valores (alpha,beta),
!> donde alpha es complejo y beta real.
!> Si beta no es cero, lambda = alpha / beta es un valor propio del problema
!> de valores propios no simétricos generalizados (GNEP)
!>    A*x = lambda*B*x
!> y si alpha no es cero, mu = beta / alpha es un valor propio de la
!> forma alternativa del GNEP
!>    mu*A*y = B*y.
!> Los valores propios reales se pueden leer directamente de la forma de Schur
!> generalizada:
!>   alpha = S(i,i), beta = P(i,i).
!>
!> Ref: C.B. Moler & G.W. Stewart, , SIAM J. Numer. Anal., 10(1973),
!>      pp. 241--256.
!>

Parámetros

JOB 
!>          JOB es CHARACTER*1
!>          = 'E': Calcular solo los valores propios;
!>          = 'S': Calcular los valores propios y la forma de Schur.
!>

COMPQ

!>          COMPQ es CHARACTER*1
!>          = 'N': Los vectores de Schur izquierdos (Q) no se calculan;
!>          = 'I': Q se inicializa a la matriz unidad y se devuelve la matriz Q
!>                 de vectores de Schur izquierdos de (H,T);
!>          = 'V': Q debe contener una matriz ortogonal Q1 en la entrada y
!>                 se devuelve el producto Q1*Q.
!>

COMPZ

!>          COMPZ es CHARACTER*1
!>          = 'N': Los vectores de Schur derechos (Z) no se calculan;
!>          = 'I': Z se inicializa a la matriz unidad y se devuelve la matriz Z
!>                 de vectores de Schur derechos de (H,T);
!>          = 'V': Z debe contener una matriz ortogonal Z1 en la entrada y
!>                 se devuelve el producto Z1*Z.
!>

N

!>          N es INTEGER
!>          El orden de las matrices H, T, Q e Z.  N >= 0.
!>

ILO

!>          ILO es INTEGER
!>

IHI

!>          IHI es INTEGER
!>          ILO e IHI marcan las filas y columnas de H que están en
!>          forma de Hessenberg. Se asume que A ya es triangular superior
!>          en las filas y columnas 1:ILO-1 y IHI+1:N.
!>          Si N > 0, 1 <= ILO <= IHI <= N; si N = 0, ILO=1 e IHI=0.
!>

H

!>          H es array REAL, dimension (LDH, N)
!>          En la entrada, la matriz H de Hessenberg superior de NxN.
!>          En la salida, si JOB = 'S', H contiene la matriz cuasi-triangular
!>          superior S de la factorización de Schur generalizada.
!>          Si JOB = 'E', los bloques diagonales de H coinciden con los de S, pero
!>          el resto de H no está especificado.
!>

LDH

!>          LDH es INTEGER
!>          La dimensión principal del array H.  LDH >= max( 1, N ).
!>

T

!>          T es array REAL, dimension (LDT, N)
!>          En la entrada, la matriz triangular superior T de NxN.
!>          En la salida, si JOB = 'S', T contiene la matriz triangular
!>          superior P de la factorización de Schur generalizada;
!>          los bloques diagonales de 2x2 de P correspondientes a los bloques de 2x2 de S
!>          se reducen a forma diagonal positiva, es decir, si H(j+1,j) no es
!>          cero, entonces T(j+1,j) = T(j,j+1) = 0, T(j,j) > 0, y
!>          T(j+1,j+1) > 0.
!>          Si JOB = 'E', los bloques diagonales de T coinciden con los de P, pero
!>          el resto de T no está especificado.
!>

LDT

!>          LDT es INTEGER
!>          La dimensión principal del array T.  LDT >= max( 1, N ).
!>

ALPHAR

!>          ALPHAR es array REAL, dimension (N)
!>          Las partes reales de cada escalar alpha que define un valor propio
!>          del GNEP.
!>

ALPHAI

!>          ALPHAI es array REAL, dimension (N)
!>          Las partes imaginarias de cada escalar alpha que define un
!>          valor propio del GNEP.
!>          Si ALPHAI(j) es cero, entonces el j-ésimo valor propio es real; si
!>          es positivo, entonces los valores propios j-ésimo y (j+1)-ésimo son un
!>          par conjugado complejo, con ALPHAI(j+1) = -ALPHAI(j).
!>

BETA

!>          BETA es array REAL, dimension (N)
!>          Los escalares beta que definen los valores propios del GNEP.
!>          Juntos, las cantidades alpha = (ALPHAR(j),ALPHAI(j)) y
!>          beta = BETA(j) representan el j-ésimo valor propio del
!>          par de matrices (A,B), en una de las formas lambda = alpha/beta o
!>          mu = beta/alpha. Dado que lambda o mu pueden desbordarse,
!>          no deben, en general, calcularse.
!>

Q

!>          Q es array REAL, dimension (LDQ, N)
!>          En la entrada, si COMPQ = 'V', la matriz ortogonal Q1 utilizada en
!>          la reducción de (A,B) a forma de Hessenberg generalizada.
!>          En la salida, si COMPQ = 'I', la matriz ortogonal de vectores de Schur
!>          izquierdos de (H,T), y si COMPQ = 'V', la matriz ortogonal
!>          de vectores de Schur izquierdos de (A,B).
!>          No se referencia si COMPQ = 'N'.
!>

LDQ

!>          LDQ es INTEGER
!>          La dimensión principal del array Q.  LDQ >= 1.
!>          Si COMPQ='V' o 'I', entonces LDQ >= N.
!>

Z

!>          Z es array REAL, dimension (LDZ, N)
!>          En la entrada, si COMPZ = 'V', la matriz ortogonal Z1 utilizada en
!>          la reducción de (A,B) a forma de Hessenberg generalizada.
!>          En la salida, si COMPZ = 'I', la matriz ortogonal de
!>          vectores de Schur derechos de (H,T)
