DOCUMENTACIÓN DE FUNCIÓN/SUBRUTINA

subroutine strmv (character uplo, character trans, character diag, integer n, real, dimension(lda,*) a, integer lda, real, dimension(*) x, integer incx)

STRMV

Propósito:

!>
!> STRMV  realiza una de las operaciones de matriz-vector
!>
!>    x := A*x,   o   x := A**T*x,
!>
!> donde x es un vector de n elementos y A es una matriz triangular superior o inferior
!> unitaria o no unitaria de n por n.
!>

Parámetros

UPLO 
!>          UPLO es CHARACTER*1
!>           A la entrada, UPLO especifica si la matriz es una matriz triangular superior o
!>           inferior de la siguiente manera:
!>
!>              UPLO = 'U' o 'u'   A es una matriz triangular superior.
!>
!>              UPLO = 'L' o 'l'   A es una matriz triangular inferior.
!>

TRANS

!>          TRANS es CHARACTER*1
!>           A la entrada, TRANS especifica la operación a realizar de la siguiente manera:
!>
!>              TRANS = 'N' o 'n'   x := A*x.
!>
!>              TRANS = 'T' o 't'   x := A**T*x.
!>
!>              TRANS = 'C' o 'c'   x := A**T*x.
!>

DIAG

!>          DIAG es CHARACTER*1
!>           A la entrada, DIAG especifica si A es unitaria o no
!>           triangular de la siguiente manera:
!>
!>              DIAG = 'U' o 'u'   Se asume que A es unitaria triangular.
!>
!>              DIAG = 'N' o 'n'   No se asume que A es unitaria
!>                                  triangular.
!>

N

!>          N es INTEGER
!>           A la entrada, N especifica el orden de la matriz A.
!>           N debe ser al menos cero.
!>

A

!>          A es un array REAL, dimension ( LDA, N )
!>           Antes de la entrada con UPLO = 'U' o 'u', la parte triangular superior
!>           principal de n por n del array A debe contener la matriz triangular superior
!>           y la parte estrictamente triangular inferior de A no se referencia.
!>           Antes de la entrada con UPLO = 'L' o 'l', la parte triangular inferior
!>           principal de n por n del array A debe contener la matriz triangular inferior
!>           y la parte estrictamente triangular superior de A no se referencia.
!>           Tenga en cuenta que cuando DIAG = 'U' o 'u', los elementos diagonales de
!>           A tampoco se referencian, pero se asumen como unidad.
!>

LDA

!>          LDA es INTEGER
!>           A la entrada, LDA especifica la primera dimensión de A tal como está declarada
!>           en el programa llamante (subrutina). LDA debe ser al menos
!>           max( 1, n ).
!>

X

!>          X es un array REAL, dimension al menos
!>           ( 1 + ( n - 1 )*abs( INCX ) ).
!>           Antes de la entrada, el array X incrementado debe contener el vector x
!>           de n elementos. A la salida, X se sobrescribe con el vector x transformado.
!>

INCX

!>          INCX es INTEGER
!>           A la entrada, INCX especifica el incremento para los elementos de
!>           X. INCX no debe ser cero.
!>
