subroutine stbt06 (real rcond, real rcondc, character uplo, character diag, integer n, integer kd, real, dimension( ldab, * ) ab, integer ldab, real, dimension( * ) work, real rat)

STBT06

Propósito

!>
!> STBT06 calcula una relación de prueba que compara RCOND (el recíproco
!> del número de condición de una matriz triangular A) y RCONDC, la estimación
!> calculada por STBCON. La información sobre la matriz triangular A se
!> utiliza si una estimación es cero y la otra no es cero para decidir si
!> el subdesbordamiento en la estimación está justificado.
!>

Parámetros

RCOND 
!>          RCOND es REAL
!>          La estimación del recíproco del número de condición obtenida
!>          calculando la inversa explícita de la matriz A y calculando
!>          RCOND = 1/( norm(A) * norm(inv(A)) ).
!>

RCONDC

!>          RCONDC es REAL
!>          La estimación del recíproco del número de condición calculada
!>          por STBCON.
!>

UPLO

!>          UPLO es CHARACTER
!>          Especifica si la matriz A es triangular superior o inferior.
!>          = 'U':  Triangular superior
!>          = 'L':  Triangular inferior
!>

DIAG

!>          DIAG es CHARACTER
!>          Especifica si la matriz A es triangular unitaria o no.
!>          = 'N':  Triangular no unitaria
!>          = 'U':  Triangular unitaria
!>

N

!>          N es INTEGER
!>          El orden de la matriz A.  N >= 0.
!>

KD

!>          KD es INTEGER
!>          El número de superdiagonales o subdiagonales de la
!>          matriz de banda triangular A.  KD >= 0.
!>

AB

!>          AB es un array REAL, dimensión (LDAB,N)
!>          La matriz de banda triangular superior o inferior A, almacenada
!>          en las primeras kd+1 filas del array. La j-ésima columna de A
!>          se almacena en la j-ésima columna del array AB de la siguiente
!>          manera:
!>          si UPLO = 'U', AB(kd+1+i-j,j) = A(i,j) para max(1,j-kd)<=i<=j;
!>          si UPLO = 'L', AB(1+i-j,j)    = A(i,j) para j<=i<=min(n,j+kd).
!>

LDAB

!>          LDAB es INTEGER
!>          La dimensión principal del array AB.  LDAB >= KD+1.
!>

WORK

!>          WORK es un array REAL, dimensión (N)
!>

RAT

!>          RAT es REAL
!>          La relación de prueba.  Si tanto RCOND como RCONDC son distintos
!>          de cero,
!>             RAT = MAX( RCOND, RCONDC )/MIN( RCOND, RCONDC ) - 1.
!>          Si RAT = 0, las dos estimaciones son exactamente las mismas.
!>
