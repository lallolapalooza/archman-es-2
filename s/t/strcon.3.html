subroutine strcon (character norm, character uplo, character diag, integer n, real, dimension( lda, * ) a, integer lda, real rcond, real, dimension( * ) work, integer, dimension( * ) iwork, integer info)

STRCON

Propósito:

!>
!> STRCON estima el recíproco del número de condición de una matriz
!> triangular A, en la norma 1 o en la norma infinito.
!>
!> Se calcula la norma de A y se obtiene una estimación de
!> norm(inv(A)), luego se calcula el recíproco del número de condición
!> como
!>    RCOND = 1 / ( norm(A) * norm(inv(A)) ).
!>

Parámetros

NORM 
!>          NORM es CHARACTER*1
!>          Especifica si se requiere el número de condición de la norma 1 o
!>          el número de condición de la norma infinito:
!>          = '1' o 'O':  Norma 1;
!>          = 'I':         Norma Infinito.
!>

UPLO

!>          UPLO es CHARACTER*1
!>          = 'U':  A es triangular superior;
!>          = 'L':  A es triangular inferior.
!>

DIAG

!>          DIAG es CHARACTER*1
!>          = 'N':  A es triangular no unitaria;
!>          = 'U':  A es triangular unitaria.
!>

N

!>          N es INTEGER
!>          El orden de la matriz A.  N >= 0.
!>

A

!>          A es un array REAL, de dimensión (LDA,N)
!>          La matriz triangular A.  Si UPLO = 'U', la parte triangular
!>          superior de N por N de la cabecera del array A contiene la
!>          matriz triangular superior, y la parte estrictamente triangular
!>          inferior de A no se referencia.  Si UPLO = 'L', la parte
!>          triangular inferior de N por N de la cabecera del array A
!>          contiene la matriz triangular inferior, y la parte estrictamente
!>          triangular superior de A no se referencia.  Si DIAG = 'U', los
!>          elementos diagonales de A tampoco se referencian y se asumen
!>          que son 1.
!>

LDA

!>          LDA es INTEGER
!>          La dimensión principal del array A.  LDA >= max(1,N).
!>

RCOND

!>          RCOND es REAL
!>          El recíproco del número de condición de la matriz A,
!>          calculado como RCOND = 1/(norm(A) * norm(inv(A))).
!>

WORK

!>          WORK es un array REAL, de dimensión (3*N)
!>

IWORK

!>          IWORK es un array INTEGER, de dimensión (N)
!>

INFO

!>          INFO es INTEGER
!>          = 0:  salida exitosa
!>          < 0:  si INFO = -i, el i-ésimo argumento tenía un valor ilegal
!>
