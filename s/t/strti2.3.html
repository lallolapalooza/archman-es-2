subroutine strti2 (character uplo, character diag, integer n, real, dimension( lda, * ) a, integer lda, integer info)

STRTI2 calcula la inversa de una matriz triangular real.

Propósito:

!>
!> STRTI2 calcula la inversa de una matriz triangular real superior o inferior.
!>
!> Esta es la versión de Nivel 2 BLAS del algoritmo.
!>

Parámetros

UPLO 
!>          UPLO es CHARACTER*1
!>          Especifica si la matriz A es triangular superior o inferior.
!>          = 'U':  Triangular superior
!>          = 'L':  Triangular inferior
!>

DIAG

!>          DIAG es CHARACTER*1
!>          Especifica si la matriz A es unitaria triangular o no.
!>          = 'N':  No unitaria triangular
!>          = 'U':  Unitaria triangular
!>

N

!>          N es INTEGER
!>          El orden de la matriz A.  N >= 0.
!>

A

!>          A es un array REAL, de dimensión (LDA,N)
!>          En la entrada, la matriz triangular A.  Si UPLO = 'U', la
!>          parte triangular superior principal de n por n del array A contiene
!>          la matriz triangular superior, y la parte estrictamente triangular
!>          inferior de A no se referencia. Si UPLO = 'L', la
!>          parte triangular inferior principal de n por n del array A contiene
!>          la matriz triangular inferior, y la parte estrictamente triangular
!>          superior de A no se referencia. Si DIAG = 'U', los
!>          elementos diagonales de A tampoco se referencian y se
!>          asumen que son 1.
!>
!>          En la salida, la inversa (triangular) de la matriz original, en
!>          el mismo formato de almacenamiento.
!>

LDA

!>          LDA es INTEGER
!>          La dimensión principal del array A.  LDA >= max(1,N).
!>

INFO

!>          INFO es INTEGER
!>          = 0: salida exitosa
!>          < 0: si INFO = -k, el k-ésimo argumento tenía un valor ilegal
!>
