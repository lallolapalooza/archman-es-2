subroutine stgsna (character job, character howmny, logical, dimension( * ) select, integer n, real, dimension( lda, * ) a, integer lda, real, dimension( ldb, * ) b, integer ldb, real, dimension( ldvl, * ) vl, integer ldvl, real, dimension( ldvr, * ) vr, integer ldvr, real, dimension( * ) s, real, dimension( * ) dif, integer mm, integer m, real, dimension( * ) work, integer lwork, integer, dimension( * ) iwork, integer info)

STGSNA

Propósito:

!>
!> STGSNA estima los números recíprocos de condición para valores propios y/o
!> vectores propios especificados de un par de matrices (A, B) en forma
!> canónica de Schur generalizada real (o de cualquier par de matrices
!> (Q*A*Z**T, Q*B*Z**T) con matrices ortogonales Q y Z, donde Z**T denota la
!> transpuesta de Z.
!>
!> (A, B) debe estar en forma de Schur generalizada real (como la devuelta por SGGES),
!> es decir, A es una matriz cuasi-triangular superior con bloques diagonales de
!> 1x1 y 2x2. B es una matriz triangular superior.
!>
!>

Parámetros

JOB 
!>          JOB es CHARACTER*1
!>          Especifica si se requieren números de condición para
!>          valores propios (S) o vectores propios (DIF):
!>          = 'E': solo para valores propios (S);
!>          = 'V': solo para vectores propios (DIF);
!>          = 'B': para ambos valores propios y vectores propios (S y DIF).
!>

HOWMNY

!>          HOWMNY es CHARACTER*1
!>          = 'A': calcular números de condición para todos los pares de
!>                 valores propios;
!>          = 'S': calcular números de condición para pares de valores propios
!>                 seleccionados especificados por el array SELECT.
!>

SELECT

!>          SELECT es un array LOGICAL, de dimensión (N)
!>          Si HOWMNY = 'S', SELECT especifica los pares de valores propios para los
!>          cuales se requieren números de condición. Para seleccionar números de
!>          condición para el par de valores propios correspondiente a un valor
!>          propio real w(j), SELECT(j) debe establecerse a .TRUE.. Para
!>          seleccionar números de condición correspondientes a un par conjugado
!>          complejo de valores propios w(j) y w(j+1), bien SELECT(j) o
!>          SELECT(j+1) o ambos, deben establecerse a .TRUE..
!>          Si HOWMNY = 'A', SELECT no se referencia.
!>

N

!>          N es INTEGER
!>          El orden del par de matrices cuadradas (A, B). N >= 0.
!>

A

!>          A es un array REAL, de dimensión (LDA,N)
!>          La matriz cuasi-triangular superior A en el par (A,B).
!>

LDA

!>          LDA es INTEGER
!>          La dimensión principal del array A. LDA >= max(1,N).
!>

B

!>          B es un array REAL, de dimensión (LDB,N)
!>          La matriz triangular superior B en el par (A,B).
!>

LDB

!>          LDB es INTEGER
!>          La dimensión principal del array B. LDB >= max(1,N).
!>

VL

!>          VL es un array REAL, de dimensión (LDVL,M)
!>          Si JOB = 'E' o 'B', VL debe contener los vectores propios izquierdos
!>          de (A, B), correspondientes a los pares de valores propios especificados
!>          por HOWMNY y SELECT. Los vectores propios deben almacenarse en
!>          columnas consecutivas de VL, como las devuelve STGEVC.
!>          Si JOB = 'V', VL no se referencia.
!>

LDVL

!>          LDVL es INTEGER
!>          La dimensión principal del array VL. LDVL >= 1.
!>          Si JOB = 'E' o 'B', LDVL >= N.
!>

VR

!>          VR es un array REAL, de dimensión (LDVR,M)
!>          Si JOB = 'E' o 'B', VR debe contener los vectores propios derechos
!>          de (A, B), correspondientes a los pares de valores propios especificados
!>          por HOWMNY y SELECT. Los vectores propios deben almacenarse en
!>          columnas consecutivas de VR, como las devuelve STGEVC.
!>          Si JOB = 'V', VR no se referencia.
!>

LDVR

!>          LDVR es INTEGER
!>          La dimensión principal del array VR. LDVR >= 1.
!>          Si JOB = 'E' o 'B', LDVR >= N.
!>

S

!>          S es un array REAL, de dimensión (MM)
!>          Si JOB = 'E' o 'B', los números recíprocos de condición de los
!>          valores propios seleccionados, almacenados en elementos
!>          consecutivos del array. Para un par conjugado complejo de valores
!>          propios, dos elementos consecutivos de S se establecen al mismo
!>          valor. Por lo tanto, S(j), DIF(j), y las columnas j de VL y VR
!>          corresponden todos al mismo par de valores propios (pero no,
!>          en general, al par de valores propios j, a menos que se
!>          seleccionen todos los pares de valores propios).
!>          Si JOB = 'V', S no se referencia.
!>

DIF

!>          DIF es un array REAL, de dimensión (MM)
!>          Si JOB = 'V' o 'B', los números recíprocos de condición estimados de
!>          los vectores propios seleccionados, almacenados en elementos
!>          consecutivos del array. Para un vector propio complejo, dos
!>          elementos consecutivos de DIF se establecen al mismo valor. Si los
!>          valores propios no pueden ser reordenados para calcular DIF(j),
!>          DIF(j) se establece a 0; esto solo puede ocurrir cuando el valor
!>          real sería muy pequeño de todos modos.
!>          Si JOB = 'E', DIF no se referencia.
!>

MM

!>          MM es INTEGER
!>          El número de elementos en los arrays S y DIF. MM >= M.
!>

M

!>          M es INTEGER
!>          El número de elementos de los arrays S y DIF utilizados para
!>          almacenar los números de condición especificados; para cada valor
!>          propio real seleccionado se utiliza un elemento, y para cada par
!>          conjugado complejo de valores propios seleccionados, se utilizan
!>          dos elementos.
!>          Si HOWMNY = 'A', M se establece a N.
!>

WORK

!>          WORK es un array REAL, de dimensión (MAX(1,LWORK))
!>          Al salir, si INFO = 0, WORK(1) devuelve el LWORK óptimo.
!>

LWORK

!>          LWORK es INTEGER
!>          La dimensión del array WORK. LWORK >= max(1,N).
!>          Si JOB = 'V' o 'B', LWORK >= 2*N*(N+2)+16.
!>
!>          Si LWORK = -1, entonces se asume una consulta de espacio de trabajo;
!>          la rutina solo calcula el tamaño óptimo del array WORK, devuelve
!>          este valor como la primera entrada del array WORK, y XERBLA no
!>          emite ningún mensaje de error relacionado con LWORK.
!>

IWORK

!>          IWORK es un array INTEGER, de dimensión (N + 6)
!>          Si JOB = 'E', IWORK no se referencia.
!>

INFO

!>          INFO es INTEGER
!>          =0: Salida exitosa
!>          <0: Si INFO = -i, el i-ésimo argumento tenía un valor ilegal
!>

Autor

Detalles adicionales:

!>
!>  El recíproco del número de condición de un valor propio generalizado
!>  w = (a, b) se define como
!>
!>       S(w) = (|u**TAv|**2 + |u**TBv|**2)**(1/2) / (norm(u)*norm(v))
!>
!>  donde u y v son los vectores propios izquierdos y derechos de (A, B)
!>  correspondientes a w; |z| denota el valor absoluto del número complejo,
!>  y norm(u) denota la norma 2 del vector u.
!>  El par (a, b) corresponde a un valor propio w = a/b (= u**TAv/u**TBv)
!>  del par de matrices (A, B). Si tanto a como b son iguales a cero, entonces
!>  (A B) es singular y se devuelve S(I) = -1.
!

NAME

STGSNA — Generalized Singular Value Decomposition (GSVD) condition estimators

Function/Subroutine Documentation

!>
!>  Purpose
!>  =======
!>
!>  STGSNA estimates the separation between two pairs of generalized
!>  eigenvectors (or generalized Schur vectors) of a matrix pair (A, B).
!>
!>  Input arguments:
!>  ================
!>
!>  JOB      (character(*)) CHARACTER*1.
!>           Specifies the computation to be performed:
!>           = 'E': compute the reciprocal condition numbers for the
!>                  generalized eigenvalues only.
!>           = 'V': compute the reciprocal condition numbers for the
!>                  generalized eigenvectors only.
!>           = 'B': compute the reciprocal condition numbers for the
!>                  generalized eigenvalues and eigenvectors.
!>
!>  HOWMANY  (character(*)) CHARACTER*1.
!>           Specifies how many linear estimators are to be computed:
!>           = 'A': All ! THE separation is computed for each of the
!>                  n generalized eigenvalues (under debate).
!>           = 'S': Some ! The separation is computed for the
!>                  differences specified by the array SELECT.
!>
!>  SELECT   (integer(*)) INTEGER array, dimension (N).
!>           On entry, the dimension of SELECT is (N). If SELECT(I) = TRUE,
!>           then the I-th generalized eigenvalue/vector will be chosen
!>           for which the reciprocal condition number will be computed.
!>           If SELECT(I) = FALSE, then the I-th generalized eigenvalue/vector
!>           will be passed over.
!>           Only used when HOWMANY = 'S'.
!>
!>  N        (integer) INTEGER.
!>           The order of the matrices A and B. N >= 0.
!>
!>  A        (real, dimension (LDA, N)) REAL array.
!>           The upper triangular matrix A.
!>
!>  LDA      (integer) INTEGER.
!>           The leading dimension of A. LDA >= max(1, N).
!>
!>  B        (real, dimension (LDB, N)) REAL array.
!>           The upper triangular matrix B.
!>
!>  LDB      (integer) INTEGER.
!>           The leading dimension of B. LDB >= max(1, N).
!>
!>  VL       (real, dimension (LDVL, N)) REAL array.
!>           On entry, VL contains the left generalized Schur vectors in
!>           rows 1 to N. If JOB = 'E', VL is not referenced.
!>           On exit, if JOB = 'V' or 'B', VL contains the selected
!>           approximate left generalized eigenvectors.
!>
!>  LDVL     (integer) INTEGER.
!>           The leading dimension of VL. LDVL >= N.
!>
!>  VR       (real, dimension (LDVR, N)) REAL array.
!>           On entry, VR contains the right generalized Schur vectors in
!>           rows 1 to N. If JOB = 'E', VR is not referenced.
!>           On exit, if JOB = 'V' or 'B', VR contains the selected
!>           approximate right generalized eigenvectors.
!>
!>  LDVR     (integer) INTEGER.
!>           The leading dimension of VR. LDVR >= N.
!>
!>  TAU      (real, dimension (N)) REAL array.
!>           The diagonal elements of the generalized Schur form (S, T)
!>           of the pair (A, B). T(I,I) is stored in TAU(I).
!>
!>  LWORK    (integer) INTEGER.
!>           The length of the array WORK.
!>           LWORK >= 2*N.
!>
!>  WORK     (real, dimension (LWORK)) REAL array.
!>           The workspace array.
!>
!>  IWORK    (integer, dimension (N)) INTEGER array.
!>           The workspace array for column indexing.
!>
!>  INFO     (integer) INTEGER.
!>           = 0:  successful exit
!>           < 0:  if INFO = -i, the i-th argument had an illegal value.
!>           > 0:  if INFO = i, the i-th eigenvalue/vector could not be
!>                  processed.
!>
!>  Details
!>  =======
!>
!>  The generalized eigenvalue problem A*x = lambda*B*x is the standard
!>  eigenvalue problem for the matrix pair (A, B).
!>
!>  This routine computes the reciprocal condition numbers for the
!>  generalized eigenvalues and/or the generalized eigenvectors of a
!>  regular generalized eigenvalue problem A*x = lambda*B*x.
!>
!>  The pair (A, B) must be such that the generalized Schur decomposition
!>  is available. That is, there exist orthogonal matrices V and U such
!>  that
!>
!>       Q**T*A*Z = T (upper triangular)
!>       Q**T*B*Z = S (upper triangular)
!>
!>  where Q and Z are orthogonal matrices.
!>
!>  The generalized eigenvalues are then the ratios lambda_i = T(i,i) / S(i,i),
!>  where S(i,i) must be non-zero.
!>
!>  The generalized Schur vectors VL and VR are the columns of Z and Q
!>  respectively.
!>
!>  An approximate error bound for the i-th computed generalized eigenvalue
!>  w = tau(i) of the pair (A, B) is given by
!>
!>       chord(w, lambda) <= EPS * norm(A, B) / S(I)
!>
!>  where EPS is the machine precision. S(I) is the reciprocal condition
!>  number of the i-th generalized eigenvalue.
!>
!>  The reciprocal of the condition number DIF(i) of right eigenvector u
!>  and left eigenvector v corresponding to the generalized eigenvalue w
!>  is defined as follows:
!>
!>  a) If the i-th eigenvalue w = (a,b) is real
!>
!>     Suppose U and V are orthogonal transformations such that
!>
!>              U**T*(A, B)*V  = (S, T) = ( a   *  ) ( b  *  )  1
!>                                        ( 0  S22 ),( 0 T22 )  n-1
!>                                          1  n-1     1 n-1
!>
!>     Then the reciprocal condition number DIF(i) is
!>
!>                Difl((a, b), (S22, T22)) = sigma-min( Zl ),
!>
!>     where sigma-min(Zl) denotes the smallest singular value of the
!>     2(n-1)-by-2(n-1) matrix
!>
!>         Zl = [ kron(a, In-1)  -kron(1, S22) ]
!>              [ kron(b, In-1)  -kron(1, T22) ] .
!>
!>     Here In-1 is the identity matrix of size n-1. kron(X, Y) is the
!>     Kronecker product between the matrices X and Y.
!>
!>     Note that if the default method for computing DIF(i) is wanted
!>     (see SLATDF), then the parameter DIFDRI (see below) should be
!>     changed from 3 to 4 (routine SLATDF(IJOB = 2 will be used)).
!>     See STGSYL for more details.
!>
!>  b) If the i-th and (i+1)-th eigenvalues are complex conjugate pair,
!>
!>     Suppose U and V are orthogonal transformations such that
!>
!>              U**T*(A, B)*V = (S, T) = ( S11  *   ) ( T11  *  )  2
!>                                       ( 0    S22 ),( 0    T22) n-2
!>                                         2    n-2     2    n-2
!>
!>     and (S11, T11) corresponds to the complex conjugate eigenvalue
!>     pair (w, conjg(w)). There exist unitary matrices U1 and V1 such
!>     that
!>
!>       U1**T*S11*V1 = ( s11 s12 ) and U1**T*T11*V1 = ( t11 t12 )
!>                      (  0  s22 )                    (  0  t22 )
!>
!>     where the generalized eigenvalues w = s11/t11 and
!>     conjg(w) = s22/t22.
!>
!>     Then the reciprocal condition number DIF(i) is bounded by
!>
!>         min( d1, max( 1, |real(s11)/real(s22)| )*d2 )
!>
!>     where, d1 = Difl((s11, t11), (s22, t22)) = sigma-min(Z1), where
!>     Z1 is the complex 2-by-2 matrix
!>
!>              Z1 =  [ s11  -s22 ]
!>                    [ t11  -t22 ],
!>
!>     This is done by computing (using real arithmetic) the
!>     roots of the characteristical polynomial det(Z1**T * Z1 - lambda I),
!>     where Z1**T denotes the transpose of Z1 and det(X) denotes
!>     the determinant of X.
!>
!>     and d2 is an upper bound on Difl((S11, T11), (S22, T22)), i.e. an
!>     upper bound on sigma-min(Z2), where Z2 is (2n-2)-by-(2n-2)
!>
!>              Z2 = [ kron(S11**T, In-2)  -kron(I2, S22) ]
!>                   [ kron(T11**T, In-2)  -kron(I2, T22) ]
!>
!>     Note that if the default method for computing DIF is wanted (see
!>     SLATDF), then the parameter DIFDRI (see below) should be changed
!>     from 3 to 4 (routine SLATDF(IJOB = 2 will be used)). See STGSYL
!>     for more details.
!>
!>  For each eigenvalue/vector specified by SELECT, DIF stores a
!>  Frobenius norm-based estimate of Difl.
!>
!>  An approximate error bound for the i-th computed eigenvector VL(i) or
!>  VR(i) is given by
!>
!>             EPS * norm(A, B) / DIF(i).
!>
!>  See ref. [2-3] for more details and further references.
!>
