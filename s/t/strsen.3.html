subroutine strsen (character job, character compq, logical, dimension( * ) select, integer n, real, dimension( ldt, * ) t, integer ldt, real, dimension( ldq, * ) q, integer ldq, real, dimension( * ) wr, real, dimension( * ) wi, integer m, real s, real sep, real, dimension( * ) work, integer lwork, integer, dimension( * ) iwork, integer liwork, integer info)

STRSEN

Propósito:

!>
!> STRSEN reordena la factorización de Schur real de una matriz real
!> A = Q*T*Q**T, para que un clúster seleccionado de autovalores aparezca en
!> los bloques diagonales principales de la matriz cuasi-triangular superior T,
!> y las columnas principales de Q formen una base ortonormal del
!> subespacio invariante derecho correspondiente.
!>
!> Opcionalmente, la rutina calcula los números de condición recíprocos de
!> el clúster de autovalores y/o el subespacio invariante.
!>
!> T debe estar en forma canónica de Schur (como la devuelta por SHSEQR), es decir,
!> triangular por bloques con bloques diagonales de 1x1 y 2x2; cada
!> bloque diagonal de 2x2 tiene sus elementos diagonales iguales y sus
!> elementos fuera de la diagonal de signo opuesto.
!>

Parámetros

JOB 
!>          JOB es CHARACTER*1
!>          Especifica si se requieren números de condición para el
!>          clúster de autovalores (S) o el subespacio invariante (SEP):
!>          = 'N': ninguno;
!>          = 'E': solo para autovalores (S);
!>          = 'V': solo para subespacio invariante (SEP);
!>          = 'B': para autovalores y subespacio invariante (S y
!>                 SEP).
!>

COMPQ

!>          COMPQ es CHARACTER*1
!>          = 'V': actualizar la matriz Q de vectores de Schur;
!>          = 'N': no actualizar Q.
!>

SELECT

!>          SELECT es un array LOGICAL, de dimensión (N)
!>          SELECT especifica los autovalores en el clúster seleccionado. Para
!>          seleccionar un autovalor real w(j), SELECT(j) debe establecerse a
!>          .TRUE.. Para seleccionar un par conjugado complejo de autovalores
!>          w(j) y w(j+1), correspondiente a un bloque diagonal de 2x2,
!>          se debe establecer SELECT(j) o SELECT(j+1) o ambos a
!>          .TRUE.; un par conjugado complejo de autovalores debe ser
!>          o bien incluido en el clúster o bien excluido.
!>

N

!>          N es INTEGER
!>          El orden de la matriz T. N >= 0.
!>

T

!>          T es un array REAL, de dimensión (LDT,N)
!>          En la entrada, la matriz cuasi-triangular superior T, en forma canónica
!>          de Schur.
!>          En la salida, T se sobrescribe con la matriz reordenada T, de nuevo en
!>          forma canónica de Schur, con los autovalores seleccionados en los
!>          bloques diagonales principales.
!>

LDT

!>          LDT es INTEGER
!>          La dimensión principal del array T. LDT >= max(1,N).
!>

Q

!>          Q es un array REAL, de dimensión (LDQ,N)
!>          En la entrada, si COMPQ = 'V', la matriz Q de vectores de Schur.
!>          En la salida, si COMPQ = 'V', Q ha sido postmultiplicada por la
!>          matriz de transformación ortogonal que reordena T; las
!>          M columnas principales de Q forman una base ortonormal para
!>          el subespacio invariante especificado.
!>          Si COMPQ = 'N', Q no se referencia.
!>

LDQ

!>          LDQ es INTEGER
!>          La dimensión principal del array Q.
!>          LDQ >= 1; y si COMPQ = 'V', LDQ >= N.
!>

WR

!>          WR es un array REAL, de dimensión (N)
!>

WI

!>          WI es un array REAL, de dimensión (N)
!>
!>          Las partes real e imaginaria, respectivamente, de los autovalores
!>          reordenados de T. Los autovalores se almacenan en el mismo
!>          orden que en la diagonal de T, con WR(i) = T(i,i) y, si
!>          T(i:i+1,i:i+1) es un bloque diagonal de 2x2, WI(i) > 0 y
!>          WI(i+1) = -WI(i). Nótese que si un autovalor complejo está
!>          suficientemente mal condicionado, su valor puede diferir
!>          significativamente de su valor antes de la reordenación.
!>

M

!>          M es INTEGER
!>          La dimensión del subespacio invariante especificado.
!>          0 < = M <= N.
!>

S

!>          S es REAL
!>          Si JOB = 'E' o 'B', S es un límite inferior del número de condición recíproco
!>          para el clúster seleccionado de autovalores.
!>          S no puede subestimar el número de condición recíproco verdadero
!>          en más de un factor de sqrt(N). Si M = 0 o N, S = 1.
!>          Si JOB = 'N' o 'V', S no se referencia.
!>

SEP

!>          SEP es REAL
!>          Si JOB = 'V' o 'B', SEP es el número de condición recíproco estimado
!>          del subespacio invariante especificado. Si
!>          M = 0 o N, SEP = norm(T).
!>          Si JOB = 'N' o 'E', SEP no se referencia.
!>

WORK

!>          WORK es un array REAL, de dimensión (MAX(1,LWORK))
!>          En la salida, si INFO = 0, WORK(1) devuelve el LWORK óptimo.
!>

LWORK

!>          LWORK es INTEGER
!>          La dimensión del array WORK.
!>          Si JOB = 'N', LWORK >= max(1,N);
!>          si JOB = 'E', LWORK >= max(1,M*(N-M));
!>          si JOB = 'V' o 'B', LWORK >= max(1,2*M*(N-M)).
!>
!>          Si LWORK = -1, entonces se asume una consulta de espacio de trabajo; la rutina
!>          solo calcula el tamaño óptimo del array WORK, devuelve
!>          este valor como la primera entrada del array WORK, y no se emite ningún mensaje de error
!>          relacionado con LWORK por XERBLA.
!>

IWORK

!>          IWORK es un array INTEGER, de dimensión (MAX(1,LIWORK))
!>          En la salida, si INFO = 0, IWORK(1) devuelve el LIWORK óptimo.
!>

LIWORK

!>          LIWORK es INTEGER
!>          La dimensión del array IWORK.
!>          Si JOB = 'N' o 'E', LIWORK >= 1;
!>          si JOB = 'V' o 'B', LIWORK >= max(1,M*(N-M)).
!>
!>          Si LIWORK = -1, entonces se asume una consulta de espacio de trabajo; la
!>          rutina solo calcula el tamaño óptimo del array IWORK,
!>          devuelve este valor como la primera entrada del array IWORK, y
!>          no se emite ningún mensaje de error relacionado con LIWORK por XERBLA.
!>

INFO

!>          INFO es INTEGER
!>          = 0: finalización exitosa
!>          < 0: si INFO = -i, el i-ésimo argumento tenía un valor ilegal
!>          = 1: la reordenación de T falló porque algunos autovalores están demasiado
!>               cerca para separarse (el problema está muy mal condicionado);
!>               T puede haber sido reordenada parcialmente, y WR y WI
!>               contienen los autovalores en el mismo orden que en T; S y
!>               SEP (si se solicitaron) se establecen a cero.
!>

Detalles Adicionales:

!>
!>  STRSEN primero recopila los autovalores seleccionados calculando una
!>  transformación ortogonal Z para moverlos a la esquina superior izquierda de T.
!>  En otras palabras, los autovalores seleccionados son los autovalores de T11
!>  en:
!>
!>          Z**T * T * Z = ( T11 T12 ) n1
!>                         (  0  T22 ) n2
!>                            n1  n2
!>
!>  donde N = n1+n2 y Z**T significa la transpuesta de Z. Las primeras n1 columnas
!>  de Z abarcan el subespacio invar

#include <stdbool.h>
!
!*===========================================================================*
!* *
!* RESIDUAL ESTIMATES FOR EIGENSYSTEM PROBLEM (REAL) *
!*===========================================================================*
!
! Purpose
! =======
! STRSEN computes residual error bounds for the Schur decomposition of
! a matrix A and for the problem of computing the
! eigenvectors of A.
!
! A SCHUR decomposition of A is an orthogonal transformation
! Z**T*A*Z = T such that T is in real Schur form (upper quasi-triangular
! with 1-by-1 and 2-by-2 blocks).
!
! STRSEN also provides an estimate of the reciprocal condition number
! of the average of the eigenvalues of the block T11, and the
! reciprocal condition number of the right invariant subspace spanned
! by the eigenvectors of T11.
!
! Arguments
! ========
!
! JOB (input) CHARACTER*1
! Specifies whether to compute the average of the eigenvalues of
! T11 and/or the reciprocal condition number of the invariant
! subspace.
! = 'E': Compute AVG(EI) and SEP concerning T11.
! = 'V': Compute AVG(EI) and the RECIPROCAL condition number S of
! the invariant subspace T11.
! = 'B': Compute AVG(EI), S and SEP concerning T11.
!
! COMPQ (input) CHARACTER*1
! Specifies whether to compute the Schur vectors which are used
! to transform A into Schur canonical form.
! = 'N': Do not compute the Schur vectors.
! = 'I': Compute the Schur vectors (updated).
! = 'V': Compute the Schur vectors (updated).
!
! SELECT (input) LOGICAL array, dimension (N)
! Defines the subset of eigenvalues which are to be included in the
! left (upper) part of the Schur canonical form.
! In other words, the eigenvalues which satisfy SELECT(I) = .TRUE.
! are the ones to be moved.
!
! N (input) INTEGER
! The order of the matrix A. N >= 0.
!
! A (input/output) REAL array, dimension (LDA, N)
! On entry, the N-by-N matrix A in Schur canonical form.
! On exit, A is overwritten by a matrix Z*A*Z**T, where Z is an
! orthogonal matrix such that A(I, J) = 0 for J < I and
! SELECT(I) = .TRUE. for I = 1, ..., N1,
! and SELECT(I) = .FALSE. for I = N1 + 1, ..., N.
!
! LDA (input) INTEGER
! The leading dimension of the array A. LDA >= max(1, N).
!
! LDZ (input) INTEGER
! The leading dimension of the array Z. LDZ >= max(1, N).
!
! Z (output) REAL array, dimension (LDZ, N)
! If COMPQ = 'V', Z is an orthogonal matrix which transforms A
! into the Schur canonical form Z*A*Z**T.
! If COMPQ = 'N', Z may be any matrix.
! If COMPQ = 'I', Z is updated to the matrix Z(old)*Z,
! where Z is the orthogonal matrix in the previous step.
!
! NCONV (output) INTEGER
! The number of eigenvalues which are not selected. This is the
! number of columns of Z, and the dimension of the block T11.
!
! SL (output) REAL
! SR (output) REAL
! SL and SR are real and imaginary parts of the average
! eigenvalue AVG(EI) of the block T11, where EI are the
! eigenvalues of T11.
!
! S (output) REAL
! S is the reciprocal condition number of the invariant subspace
! associated with T11.
!
! SEP (output) REAL
! SEP is the reciprocal condition number of the average of the
! eigenvalues of T11.
!
! INFO (output) INTEGER
! = 0: successful exit
! < 0: if INFO = -i, the i-th argument had an illegal value.
! > 0: if INFO = i, the QR algorithm failed to compute the
! transformation matrix.
! Specifically, INFO = N+1 indicates that the Schur vectors
! could not be computed.
!
! Details
! =======
!
! The accuracy of the output is stated in terms of the machine
! precision EPS.
!
! The average eigenvalue AVG(EI) is computed as follows. Let E be the
! diagonal elements of T. AVG(EI) = SUM(EI) / NCONV, where EI are the
! eigenvalues corresponding to the selected eigenvalues.
!
! The reciprocal condition number of the average of the eigenvalues of
! T11 may be returned in S. S lies between 0 (very badly conditioned)
! and 1 (very well conditioned). It is computed as follows. First we
! compute R so that
!
! P = ( I R ) n1
! ( 0 0 ) n2
! n1 n2
! is the projector on the invariant subspace associated with T11.
! R is the solution of the Sylvester equation:
!
! T11*R - R*T22 = T12.
!
! Let F-norm(M) denote the Frobenius-norm of M and 2-norm(M) denote
! the two-norm of M. Then S is computed as the lower bound
!
! (1 + F-norm(R)**2)**(-1/2)
!
! on the reciprocal of 2-norm(P), the true reciprocal condition number.
! S cannot underestimate 1 / 2-norm(P) by more than a factor of
! sqrt(N).
!
! An approximate error bound for the computed average of the
! eigenvalues of T11 is
!
! EPS * norm(T) / S
!
! where EPS is the machine precision.
!
! The reciprocal condition number of the right invariant subspace
<
! spanned by the first n1 columns of Z (or of Q*Z) is returned in SEP.
! SEP is defined as the separation of T11 and T22:
!
! sep( T11, T22 ) = sigma-min( C )
!
! where sigma-min(C) is the smallest singular value of the
! n1*n2-by-n1*n2 matrix
!
! C = kprod( I(n2), T11 ) - kprod( transpose(T22), I(n1) )
!
! I(m) is an m by m identity matrix, and kprod denotes the Kronecker
! product. We estimate sigma-min(C) by the reciprocal of an estimate of
! the 1-norm of inverse(C). The true reciprocal 1-norm of inverse(C)
! cannot differ from sigma-min(C) by more than a factor of sqrt(n1*n2).
!
! When SEP is small, small changes in T can cause large changes in
! the invariant subspace. An approximate bound on the maximum angular
! error in the computed right invariant subspace is
!
! EPS * norm(T) / SEP
!
!
