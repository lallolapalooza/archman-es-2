subroutine stpt03 (character uplo, character trans, character diag, integer n, integer nrhs, real, dimension( * ) ap, real scale, real, dimension( * ) cnorm, real tscal, real, dimension( ldx, * ) x, integer ldx, real, dimension( ldb, * ) b, integer ldb, real, dimension( * ) work, real resid)

STPT03

Propósito:

!>
!> STPT03 calcula el residuo para la solución de un sistema triangular escalado
!> de ecuaciones A*x = s*b  o  A'*x = s*b  cuando la matriz triangular
!> A se almacena en formato empaquetado. Aquí A' es la transpuesta de A,
!> s es un escalar, y x e b son matrices N por NRHS. La relación de prueba es
!> el máximo sobre el número de lados derechos de
!>    norm(s*b - op(A)*x) / ( norm(op(A)) * norm(x) * EPS ),
!> donde op(A) denota A o A' y EPS es el epsilon de la máquina.
!>

Parámetros

UPLO 
!>          UPLO es CHARACTER*1
!>          Especifica si la matriz A es triangular superior o inferior.
!>          = 'U':  Triangular superior
!>          = 'L':  Triangular inferior
!>

TRANS

!>          TRANS es CHARACTER*1
!>          Especifica la operación aplicada a A.
!>          = 'N':  A *x = s*b  (Sin transponer)
!>          = 'T':  A'*x = s*b  (Transpuesta)
!>          = 'C':  A'*x = s*b  (Transpuesta conjugada = Transpuesta)
!>

DIAG

!>          DIAG es CHARACTER*1
!>          Especifica si la matriz A es unitaria o no.
!>          = 'N':  Triangular no unitaria
!>          = 'U':  Triangular unitaria
!>

N

!>          N es INTEGER
!>          El orden de la matriz A.  N >= 0.
!>

NRHS

!>          NRHS es INTEGER
!>          El número de lados derechos, es decir, el número de columnas
!>          de las matrices X e B.  NRHS >= 0.
!>

AP

!>          AP es un array REAL, dimension (N*(N+1)/2)
!>          La matriz triangular superior o inferior A, empaquetada por
!>          columnas en un array lineal. La j-ésima columna de A se almacena
!>          en el array AP de la siguiente manera:
!>          si UPLO = 'U', AP((j-1)*j/2 + i) = A(i,j) para 1<=i<=j;
!>          si UPLO = 'L',
!>             AP((j-1)*(n-j) + j*(j+1)/2 + i-j) = A(i,j) para j<=i<=n.
!>

SCALE

!>          SCALE es REAL
!>          El factor de escala s utilizado al resolver el sistema triangular.
!>

CNORM

!>          CNORM es un array REAL, dimension (N)
!>          Las normas 1 de las columnas de A, sin contar la diagonal.
!>

TSCAL

!>          TSCAL es REAL
!>          El factor de escala utilizado al calcular las normas 1 en CNORM.
!>          CNORM contiene realmente las normas de columna de TSCAL*A.
!>

X

!>          X es un array REAL, dimension (LDX,NRHS)
!>          Los vectores de solución calculados para el sistema de ecuaciones
!>          lineales.
!>

LDX

!>          LDX es INTEGER
!>          La dimensión principal del array X.  LDX >= max(1,N).
!>

B

!>          B es un array REAL, dimension (LDB,NRHS)
!>          Los vectores del lado derecho para el sistema de ecuaciones
!>          lineales.
!>

LDB

!>          LDB es INTEGER
!>          La dimensión principal del array B.  LDB >= max(1,N).
!>

WORK

!>          WORK es un array REAL, dimension (N)
!>

RESID

!>          RESID es REAL
!>          El máximo sobre el número de lados derechos de
!>          norm(op(A)*x - s*b) / ( norm(op(A)) * norm(x) * EPS ).
!>
