subroutine stprfs (character uplo, character trans, character diag, integer n, integer nrhs, real, dimension( * ) ap, real, dimension( ldb, * ) b, integer ldb, real, dimension( ldx, * ) x, integer ldx, real, dimension( * ) ferr, real, dimension( * ) berr, real, dimension( * ) work, integer, dimension( * ) iwork, integer info)

STPRFS

Propósito:

!>
!> STPRFS proporciona límites de error y estimaciones de error hacia atrás para
!> la solución de un sistema de ecuaciones lineales con una matriz de
!> coeficientes triangular empaquetada.
!>
!> La matriz solución X debe ser calculada por STPTRS o por algún otro medio
!> antes de entrar en esta rutina. STPRFS no realiza refinamiento iterativo
!> porque hacerlo no puede mejorar el error hacia atrás.
!>

Parámetros

UPLO 
!>          UPLO es CHARACTER*1
!>          = 'U':  A es triangular superior;
!>          = 'L':  A es triangular inferior.
!>

TRANS

!>          TRANS es CHARACTER*1
!>          Especifica la forma del sistema de ecuaciones:
!>          = 'N':  A * X = B  (Sin transpuesta)
!>          = 'T':  A**T * X = B  (Transpuesta)
!>          = 'C':  A**H * X = B  (Transpuesta conjugada = Transpuesta)
!>

DIAG

!>          DIAG es CHARACTER*1
!>          = 'N':  A es triangular no unitaria;
!>          = 'U':  A es triangular unitaria.
!>

N

!>          N es INTEGER
!>          El orden de la matriz A. N >= 0.
!>

NRHS

!>          NRHS es INTEGER
!>          El número de lados derechos, es decir, el número de columnas
!>          de las matrices B y X. NRHS >= 0.
!>

AP

!>          AP es un array REAL, de dimensión (N*(N+1)/2)
!>          La matriz triangular superior o inferior A, empaquetada por
!>          columnas en un array lineal. La j-ésima columna de A se almacena
!>          en el array AP de la siguiente manera:
!>          si UPLO = 'U', AP(i + (j-1)*j/2) = A(i,j) para 1<=i<=j;
!>          si UPLO = 'L', AP(i + (j-1)*(2*n-j)/2) = A(i,j) para j<=i<=n.
!>          Si DIAG = 'U', los elementos diagonales de A no se referencian
!>          y se asumen que son 1.
!>

B

!>          B es un array REAL, de dimensión (LDB,NRHS)
!>          La matriz del lado derecho B.
!>

LDB

!>          LDB es INTEGER
!>          La dimensión principal del array B. LDB >= max(1,N).
!>

X

!>          X es un array REAL, de dimensión (LDX,NRHS)
!>          La matriz solución X.
!>

LDX

!>          LDX es INTEGER
!>          La dimensión principal del array X. LDX >= max(1,N).
!>

FERR

!>          FERR es un array REAL, de dimensión (NRHS)
!>          El límite de error hacia adelante estimado para cada vector solución
!>          X(j) (la j-ésima columna de la matriz solución X).
!>          Si XTRUE es la solución verdadera correspondiente a X(j), FERR(j)
!>          es un límite superior estimado para la magnitud del elemento más
!>          grande en (X(j) - XTRUE) dividido por la magnitud del elemento más
!>          grande en X(j). La estimación es tan confiable como
!>          la estimación para RCOND, y es casi siempre una ligera
!>          sobreestimación del error verdadero.
!>

BERR

!>          BERR es un array REAL, de dimensión (NRHS)
!>          El error hacia atrás relativo componente a componente de cada vector solución
!>          X(j) (es decir, el cambio relativo más pequeño en
!>          cualquier elemento de A o B que haga que X(j) sea una solución exacta).
!>

WORK

!>          WORK es un array REAL, de dimensión (3*N)
!>

IWORK

!>          IWORK es un array INTEGER, de dimensión (N)
!>

INFO

!>          INFO es INTEGER
!>          = 0:  salida exitosa
!>          < 0:  si INFO = -i, el i-ésimo argumento tenía un valor ilegal
!>
