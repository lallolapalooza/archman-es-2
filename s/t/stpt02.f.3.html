subroutine stpt02 (character uplo, character trans, character diag, integer n, integer nrhs, real, dimension( * ) ap, real, dimension( ldx, * ) x, integer ldx, real, dimension( ldb, * ) b, integer ldb, real, dimension( * ) work, real resid)

STPT02

Propósito:

!>
!> STPT02 calcula el residuo para la solución calculada de un
!> sistema de ecuaciones lineales triangular op(A)*X = B, cuando la
!> matriz triangular A se almacena en formato empaquetado. La relación de prueba es
!> el máximo de
!>    norm(b - op(A)*x) / ( ||op(A)||_1 * norm(x) * EPS ),
!> donde op(A) = A o A**T, b es la columna de B, x es el vector solución,
!> y EPS es el epsilon de la máquina.
!> La norma utilizada es la norma 1.
!>

Parámetros

UPLO 
!>          UPLO es CHARACTER*1
!>          Especifica si la matriz A es triangular superior o inferior.
!>          = 'U':  Triangular superior
!>          = 'L':  Triangular inferior
!>

TRANS

!>          TRANS es CHARACTER*1
!>          Especifica la operación aplicada a A.
!>          = 'N':  A    * X = B  (Sin transpuesta)
!>          = 'T':  A**T * X = B  (Transpuesta)
!>          = 'C':  A**H * X = B  (Transpuesta conjugada = Transpuesta)
!>

DIAG

!>          DIAG es CHARACTER*1
!>          Especifica si la matriz A es unitriangular o no.
!>          = 'N':  Triangular no unitaria
!>          = 'U':  Triangular unitaria
!>

N

!>          N es INTEGER
!>          El orden de la matriz A.  N >= 0.
!>

NRHS

!>          NRHS es INTEGER
!>          El número de lados derechos, es decir, el número de columnas
!>          de las matrices X e B.  NRHS >= 0.
!>

AP

!>          AP es un array REAL, dimension (N*(N+1)/2)
!>          La matriz triangular superior o inferior A, empaquetada por columnas en
!>          un array lineal. La j-ésima columna de A se almacena en el array
!>          AP de la siguiente manera:
!>          si UPLO = 'U', AP((j-1)*j/2 + i) = A(i,j) para 1<=i<=j;
!>          si UPLO = 'L',
!>             AP((j-1)*(n-j) + j*(j+1)/2 + i-j) = A(i,j) para j<=i<=n.
!>

X

!>          X es un array REAL, dimension (LDX,NRHS)
!>          Los vectores solución calculados para el sistema de
!>          ecuaciones lineales.
!>

LDX

!>          LDX es INTEGER
!>          La dimensión principal del array X.  LDX >= max(1,N).
!>

B

!>          B es un array REAL, dimension (LDB,NRHS)
!>          Los vectores del lado derecho para el sistema de
!>          ecuaciones lineales.
!>

LDB

!>          LDB es INTEGER
!>          La dimensión principal del array B.  LDB >= max(1,N).
!>

WORK

!>          WORK es un array REAL, dimension (N)
!>

RESID

!>          RESID es REAL
!>          El máximo sobre el número de lados derechos de
!>          norm(op(A)*X - B) / ( norm(op(A)) * norm(X) * EPS ).
!>
