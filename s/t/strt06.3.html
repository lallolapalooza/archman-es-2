subroutine strt06 (real rcond, real rcondc, character uplo, character diag, integer n, real, dimension( lda, * ) a, integer lda, real, dimension( * ) work, real rat)

STRT06

Propósito:

!>
!> STRT06 calcula una relación de prueba que compara RCOND (el recíproco
!> del número de condición de una matriz triangular A) y RCONDC, la estimación
!> calculada por STRCON. Se utiliza información sobre la matriz triangular A
!> si una estimación es cero y la otra no cero para decidir si
!> el subdesbordamiento en la estimación está justificado.
!>

Parámetros

RCOND 
!>          RCOND es REAL
!>          La estimación del recíproco del número de condición obtenida
!>          formando la inversa explícita de la matriz A y calculando
!>          RCOND = 1/( norma(A) * norma(inv(A)) ).
!>

RCONDC

!>          RCONDC es REAL
!>          La estimación del recíproco del número de condición calculada por
!>          STRCON.
!>

UPLO

!>          UPLO es CHARACTER
!>          Especifica si la matriz A es superior o inferior triangular.
!>          = 'U':  Triangular superior
!>          = 'L':  Triangular inferior
!>

DIAG

!>          DIAG es CHARACTER
!>          Especifica si la matriz A es unitriangular o no.
!>          = 'N':  Triangular no unitaria
!>          = 'U':  Triangular unitaria
!>

N

!>          N es INTEGER
!>          El orden de la matriz A.  N >= 0.
!>

A

!>          A es un array REAL, dimensión (LDA,N)
!>          La matriz triangular A.  Si UPLO = 'U', la parte superior triangular
!>          principal de n por n de la matriz A contiene la matriz triangular
!>          superior, y la parte triangular estrictamente inferior de A no se
!>          referencia.  Si UPLO = 'L', la parte inferior triangular principal
!>          de n por n de la matriz A contiene la matriz triangular inferior,
!>          y la parte triangular estrictamente superior de A no se referencia.
!>          Si DIAG = 'U', los elementos diagonales de A tampoco se referencian y
!>          se asumen como 1.
!>

LDA

!>          LDA es INTEGER
!>          La dimensión principal del array A.  LDA >= max(1,N).
!>

WORK

!>          WORK es un array REAL, dimensión (N)
!>

RAT

!>          RAT es REAL
!>          La relación de prueba.  Si tanto RCOND como RCONDC son distintos de
!>          cero,
!>             RAT = MAX( RCOND, RCONDC )/MIN( RCOND, RCONDC ) - 1.
!>          Si RAT = 0, las dos estimaciones son exactamente las mismas.
!>
