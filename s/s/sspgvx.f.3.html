subroutine sspgvx (integer itype, character jobz, character range, character uplo, integer n, real, dimension( * ) ap, real, dimension( * ) bp, real vl, real vu, integer il, integer iu, real abstol, integer m, real, dimension( * ) w, real, dimension( ldz, * ) z, integer ldz, real, dimension( * ) work, integer, dimension( * ) iwork, integer, dimension( * ) ifail, integer info)

SSPGVX

Propósito:

!>
!> SSPGVX calcula valores propios seleccionados y, opcionalmente, vectores propios
!> de un problema de valores propios generalizado simétrico-definido real, de la forma
!> A*x=(lambda)*B*x,  A*Bx=(lambda)*x,  o B*A*x=(lambda)*x.  Aquí A
!> y B se asumen simétricas, almacenadas en almacenamiento empaquetado, y B
!> también es definida positiva.  Los valores propios y vectores propios pueden
!> seleccionarse especificando ya sea un rango de valores o un rango de índices
!> para los valores propios deseados.
!>

Parámetros

ITYPE 
!>          ITYPE es INTEGER
!>          Especifica el tipo de problema a resolver:
!>          = 1:  A*x = (lambda)*B*x
!>          = 2:  A*B*x = (lambda)*x
!>          = 3:  B*A*x = (lambda)*x
!>

JOBZ

!>          JOBZ es CHARACTER*1
!>          = 'N':  Calcular solo valores propios;
!>          = 'V':  Calcular valores propios y vectores propios.
!>

RANGE

!>          RANGE es CHARACTER*1
!>          = 'A': se encontrarán todos los valores propios.
!>          = 'V': se encontrarán todos los valores propios en el intervalo semiabierto (VL,VU]
!>                 .
!>          = 'I': se encontrarán los valores propios desde el IL-ésimo hasta el IU-ésimo.
!>

UPLO

!>          UPLO es CHARACTER*1
!>          = 'U':  Se almacena el triángulo superior de A y B;
!>          = 'L':  Se almacena el triángulo inferior de A y B.
!>

N

!>          N es INTEGER
!>          El orden del haz de matrices (A,B).  N >= 0.
!>

AP

!>          AP es un arreglo REAL, dimensión (N*(N+1)/2)
!>          Al entrar, el triángulo superior o inferior de la matriz simétrica
!>          A, empaquetado por columnas en un arreglo lineal.  La j-ésima columna de A
!>          se almacena en el arreglo AP de la siguiente manera:
!>          si UPLO = 'U', AP(i + (j-1)*j/2) = A(i,j) para 1<=i<=j;
!>          si UPLO = 'L', AP(i + (j-1)*(2*n-j)/2) = A(i,j) para j<=i<=n.
!>
!>          Al salir, el contenido de AP se destruye.
!>

BP

!>          BP es un arreglo REAL, dimensión (N*(N+1)/2)
!>          Al entrar, el triángulo superior o inferior de la matriz simétrica
!>          B, empaquetado por columnas en un arreglo lineal.  La j-ésima columna de B
!>          se almacena en el arreglo BP de la siguiente manera:
!>          si UPLO = 'U', BP(i + (j-1)*j/2) = B(i,j) para 1<=i<=j;
!>          si UPLO = 'L', BP(i + (j-1)*(2*n-j)/2) = B(i,j) para j<=i<=n.
!>
!>          Al salir, el factor triangular U o L de la factorización
!>          de Cholesky B = U**T*U o B = L*L**T, en el mismo formato
!>          de almacenamiento que B.
!>

VL

!>          VL es REAL
!>
!>          Si RANGE='V', el límite inferior del intervalo a
!>          buscar para valores propios. VL < VU.
!>          No referenciado si RANGE = 'A' o 'I'.
!>

VU

!>          VU es REAL
!>
!>          Si RANGE='V', el límite superior del intervalo a
!>          buscar para valores propios. VL < VU.
!>          No referenciado si RANGE = 'A' o 'I'.
!>

IL

!>          IL es INTEGER
!>
!>          Si RANGE='I', el índice del
!>          valor propio más pequeño a devolver.
!>          1 <= IL <= IU <= N, si N > 0; IL = 1 y IU = 0 si N = 0.
!>          No referenciado si RANGE = 'A' o 'V'.
!>

IU

!>          IU es INTEGER
!>
!>          Si RANGE='I', el índice del
!>          valor propio más grande a devolver.
!>          1 <= IL <= IU <= N, si N > 0; IL = 1 y IU = 0 si N = 0.
!>          No referenciado si RANGE = 'A' o 'V'.
!>

ABSTOL

!>          ABSTOL es REAL
!>          La tolerancia de error absoluto para los valores propios.
!>          Un valor propio aproximado se acepta como convergido
!>          cuando se determina que se encuentra en un intervalo [a,b]
!>          de ancho menor o igual a
!>
!>                  ABSTOL + EPS *   max( |a|,|b| ) ,
!>
!>          donde EPS es la precisión de la máquina.  Si ABSTOL es menor
!>          o igual a cero, entonces se usará EPS*|T| en su lugar,
!>          donde |T| es la norma 1 de la matriz tridiagonal obtenida
!>          al reducir A a forma tridiagonal.
!>
!>          Los valores propios se calcularán con mayor precisión cuando ABSTOL se
!>          establece en el doble del umbral de subdesbordamiento 2*SLAMCH('S'), no cero.
!>          Si esta rutina regresa con INFO>0, indicando que algunos
!>          vectores propios no convergieron, intente establecer ABSTOL en
!>          2*SLAMCH('S').
!>

M

!>          M es INTEGER
!>          El número total de valores propios encontrados.  0 <= M <= N.
!>          Si RANGE = 'A', M = N, y si RANGE = 'I', M = IU-IL+1.
!>

W

!>          W es un arreglo REAL, dimensión (N)
!>          En salida normal, los primeros M elementos contienen los valores propios
!>          seleccionados en orden ascendente.
!>

Z

!>          Z es un arreglo REAL, dimensión (LDZ, max(1,M))
!>          Si JOBZ = 'N', entonces Z no es referenciado.
!>          Si JOBZ = 'V', entonces si INFO = 0, las primeras M columnas de Z
!>          contienen los vectores propios ortonormales de la matriz A
!>          correspondientes a los valores propios seleccionados, con la i-ésima
!>          columna de Z conteniendo el vector propio asociado con W(i).
!>          Los vectores propios se normalizan de la siguiente manera:
!>          si ITYPE = 1 o 2, Z**T*B*Z = I;
!>          si ITYPE = 3, Z**T*inv(B)*Z = I.
!>
!>          Si un vector propio no converge, entonces esa columna de Z
!>          contiene la última aproximación al vector propio, y el
!>          índice del vector propio se devuelve en IFAIL.
!>          Nota: el usuario debe asegurar que al menos max(1,M) columnas sean
!>          proporcionadas en el arreglo Z; si RANGE = 'V', el valor exacto de M
!>          no se conoce de antemano y se debe usar una cota superior.
!>

LDZ

!>          LDZ es INTEGER
!>          La dimensión principal del arreglo Z.  LDZ >= 1, y si
!>          JOBZ = 'V', LDZ >= max(1,N).
!>

WORK

!>          WORK es un arreglo REAL, dimensión (8*N)
!>

IWORK

!>          IWORK es un arreglo INTEGER, dimensión (5*N)
!>

IFAIL

!>          IFAIL es un arreglo INTEGER, dimensión (N)
!>          Si JOBZ = 'V', entonces si INFO = 0, los primeros M elementos de
!>          IFAIL son cero.  Si INFO > 0, entonces IFAIL contiene los
!>          índices de los vectores propios que no convergieron.
!>          Si JOBZ = 'N', entonces IFAIL no es referenciado.
!>

INFO

!>          INFO es INTEGER
!>          = 0:  salida exitosa
!>          < 0:  si INFO = -i, el i-ésimo argumento tenía un valor ilegal
!>          > 0:  SPPTRF o SSPEVX devolvieron un código de error:
!>             <= N:  si INFO = i, SSPEVX no convergió;
!>                    i vectores propios no convergieron.  Sus índices
!>                    se almacenan en el arreglo IFAIL.
!>             > N:   si INFO = N + i, para 1 <= i <= N, entonces el menor
!>                    principal dominante de orden i de B no es positivo.
!>                    La factorización de B no pudo completarse y
!>                    no se calcularon valores propios ni vectores propios.
!>
