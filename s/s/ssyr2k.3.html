subroutine ssyr2k (character uplo, character trans, integer n, integer k, real alpha, real, dimension(lda,*) a, integer lda, real, dimension(ldb,*) b, integer ldb, real beta, real, dimension(ldc,*) c, integer ldc)

SSYR2K

Propósito:

!>
!> SSYR2K  realiza una de las operaciones simétricas de rango 2k
!>
!>    C := alpha*A*B**T + alpha*B*A**T + beta*C,
!>
!> o
!>
!>    C := alpha*A**T*B + alpha*B**T*A + beta*C,
!>
!> donde  alpha y beta  son escalares, C es una matriz simétrica de  n x n
!> y  A y B  son matrices de  n x k  en el primer caso y matrices de  k x n
!> en el segundo caso.
!>

Parámetros

UPLO 
!>          UPLO es CHARACTER*1
!>           En la entrada,  UPLO  especifica si la parte  triangular  superior o
!>           inferior del array C  se referencia como:
!>
!>              UPLO = 'U' o 'u'   Sólo se referencia la parte triangular superior de C.
!>
!>              UPLO = 'L' o 'l'   Sólo se referencia la parte triangular inferior de C.
!>

TRANS

!>          TRANS es CHARACTER*1
!>           En la entrada,  TRANS  especifica la operación a realizar como:
!>
!>              TRANS = 'N' o 'n'   C := alpha*A*B**T + alpha*B*A**T +
!>                                        beta*C.
!>
!>              TRANS = 'T' o 't'   C := alpha*A**T*B + alpha*B**T*A +
!>                                        beta*C.
!>
!>              TRANS = 'C' o 'c'   C := alpha*A**T*B + alpha*B**T*A +
!>                                        beta*C.
!>

N

!>          N es INTEGER
!>           En la entrada,  N especifica el orden de la matriz C.  N debe ser
!>           al menos cero.
!>

K

!>          K es INTEGER
!>           En la entrada con  TRANS = 'N' o 'n',  K  especifica el número
!>           de columnas  de las matrices  A y B,  y en la entrada con
!>           TRANS = 'T' o 't' o 'C' o 'c',  K  especifica el número
!>           de filas de las matrices  A y B.  K debe ser al menos  cero.
!>

ALPHA

!>          ALPHA es REAL
!>           En la entrada, ALPHA especifica el escalar alpha.
!>

A

!>          A es un array REAL, de dimensión ( LDA, ka ), donde ka es
!>           k  cuando  TRANS = 'N' o 'n',  y es  n  en caso contrario.
!>           Antes de la entrada con  TRANS = 'N' o 'n',  la parte  principal
!>           de n x k  del array  A  debe contener la matriz  A; en caso
!>           contrario, la parte principal de  k x n  del array  A  debe
!>           contener la matriz A.
!>

LDA

!>          LDA es INTEGER
!>           En la entrada, LDA especifica la primera dimensión de A tal como
!>           se declaró en el programa (sub)llamador.  Cuando TRANS = 'N' o 'n',
!>           LDA debe ser al menos  max( 1, n ), en caso contrario LDA debe
!>           ser al menos  max( 1, k ).
!>

B

!>          B es un array REAL, de dimensión ( LDB, kb ), donde kb es
!>           k  cuando  TRANS = 'N' o 'n',  y es  n  en caso contrario.
!>           Antes de la entrada con  TRANS = 'N' o 'n',  la parte  principal
!>           de n x k  del array  B  debe contener la matriz  B; en caso
!>           contrario, la parte principal de  k x n  del array  B  debe
!>           contener la matriz B.
!>

LDB

!>          LDB es INTEGER
!>           En la entrada, LDB especifica la primera dimensión de B tal como
!>           se declaró en el programa (sub)llamador.  Cuando TRANS = 'N' o 'n',
!>           LDB debe ser al menos  max( 1, n ), en caso contrario LDB debe
!>           ser al menos  max( 1, k ).
!>

BETA

!>          BETA es REAL
!>           En la entrada, BETA especifica el escalar beta.
!>

C

!>          C es un array REAL, de dimensión ( LDC, N )
!>           Antes de la entrada con  UPLO = 'U' o 'u',  la parte  superior
!>           de n x n  del array C  debe contener la parte triangular  superior
!>           de la matriz simétrica, y la parte estrictamente triangular inferior
!>           de C no se referencia.  A la salida, la parte triangular superior
!>           del array C  se sobrescribe con la parte triangular superior de la
!>           matriz actualizada.
!>           Antes de la entrada con  UPLO = 'L' o 'l',  la parte  inferior
!>           de n x n  del array C  debe contener la parte triangular inferior
!>           de la matriz simétrica, y la parte estrictamente triangular superior
!>           de C no se referencia.  A la salida, la parte triangular inferior
!>           del array C  se sobrescribe con la parte triangular inferior de la
!>           matriz actualizada.
!>

LDC

!>          LDC es INTEGER
!>           En la entrada, LDC especifica la primera dimensión de C tal como
!>           se declaró en el programa (sub)llamador.  LDC debe ser al menos
!>           max( 1, n ).
!>
