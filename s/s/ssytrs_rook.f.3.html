subroutine ssytrs_rook (character uplo, integer n, integer nrhs, real, dimension( lda, * ) a, integer lda, integer, dimension( * ) ipiv, real, dimension( ldb, * ) b, integer ldb, integer info)

SSYTRS_ROOK

Propósito:

!>
!> SSYTRS_ROOK resuelve un sistema de ecuaciones lineales A*X = B con
!> una matriz simétrica real A utilizando la factorización A = U*D*U**T o
!> A = L*D*L**T calculada por SSYTRF_ROOK.
!>

Parámetros

UPLO 
!>          UPLO es CHARACTER*1
!>          Especifica si los detalles de la factorización se almacenan
!>          como una matriz triangular superior o inferior.
!>          = 'U':  Triangular superior, la forma es A = U*D*U**T;
!>          = 'L':  Triangular inferior, la forma es A = L*D*L**T.
!>

N

!>          N es INTEGER
!>          El orden de la matriz A.  N >= 0.
!>

NRHS

!>          NRHS es INTEGER
!>          El número de lados derechos, es decir, el número de columnas
!>          de la matriz B.  NRHS >= 0.
!>

A

!>          A es array REAL, dimension (LDA,N)
!>          La matriz diagonal de bloques D y los multiplicadores utilizados para
!>          obtener el factor U o L como lo calcula SSYTRF_ROOK.
!>

LDA

!>          LDA es INTEGER
!>          La dimensión principal del array A.  LDA >= max(1,N).
!>

IPIV

!>          IPIV es array INTEGER, dimension (N)
!>          Detalles de los intercambios y la estructura de bloques de D
!>          según lo determinado por SSYTRF_ROOK.
!>

B

!>          B es array REAL, dimension (LDB,NRHS)
!>          En la entrada, la matriz del lado derecho B.
!>          En la salida, la matriz solución X.
!>

LDB

!>          LDB es INTEGER
!>          La dimensión principal del array B.  LDB >= max(1,N).
!>

INFO

!>          INFO es INTEGER
!>          = 0:  salida exitosa
!>          < 0:  si INFO = -i, el i-ésimo argumento tenía un valor ilegal
!>
