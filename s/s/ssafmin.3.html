NAME

la_constants - LAPACK constants module

SYNOPSIS

module la_constants
  implicit none

  public

  !     -----------------------------------------------------------------------
  !     Machine constants for single precision
  !     -----------------------------------------------------------------------
  integer,   parameter :: sp = kind(1.e0)
  real(sp),  parameter :: szero = 0.0_sp
  real(sp),  parameter :: sone = 1.0_sp
  real(sp),  parameter :: stwo = 2.0_sp
  real(sp),  parameter :: sthree = 3.0_sp
  real(sp),  parameter :: sfour = 4.0_sp
  real(sp),  parameter :: seight = 8.0_sp
  real(sp),  parameter :: sten = 10.0_sp

  real(sp),  parameter :: shalf = 0.5_sp

  !     Machine epsilon
  real(sp),  parameter :: sulp = epsilon(0._sp)
  !     Smallest number such that 1.0 + eps > 1.0
  real(sp),  parameter :: seps = sulp * 0.5_sp

  !     Smallest positive number
  !     ssafmin = smallest positive number such that 1/ssafmin is representable
  !     ssafmin = radix(0._sp)**(minexponent(0._sp) - 1)
  real(sp),  parameter :: ssafmin =
       real(radix(0._sp), sp)**max( minexponent(0._sp)-1, 1-maxexponent(0._sp) )
  !     ssmlnum = ssafmin / sulp
  real(sp),  parameter :: ssmlnum = ssafmin / sulp
  !     Smallest number such that 1.0/ssmlnum is representable
  real(sp),  parameter :: srtmin = sqrt(ssmlnum)

  !     Largest positive number
  !     ssafmax = largest positive number such that 1/ssafmax is representable
  !     ssafmax = (1-sulp)/sulp * radix(0._sp)**minexponent(0._sp)
  real(sp),  parameter :: ssafmax = sone / ssafmin
  !     slgmax = ssafmax * sulp
  real(sp),  parameter :: slgmax = ssafmax * sulp
  !     Largest number such that 1/slgmax is representable
  real(sp),  parameter :: srtmax = sqrt(ssafmax)

  !     Largest number such that 1/ssafmin is representable

  !     Number of decimal digits in the mantissa
  !     digits(0._sp)
  integer, parameter :: sdig = digits(0._sp)

  !     Maximum exponent
  !     maxexponent(0._sp)
  integer, parameter :: sexp = maxexponent(0._sp)

  !     Minimum exponent
  !     minexponent(0._sp)
  integer, parameter :: sminexp = minexponent(0._sp)

  !     Radix
  !     radix(0._sp)
  integer, parameter :: sradix = radix(0._sp)

  !     Largest representable number in magnitude
  !     HUGE(0._sp)
  real(sp), parameter :: shuge = huge(0._sp)

  !     Number of bits in the machine
  !     nbits
  integer, parameter :: snbits = bit_size(0._sp)

  !     Rounding mode
  !     round(0._sp)
  integer, parameter :: sround = round(0._sp)

  !     Precision
  !     precision(0._sp)
  integer, parameter :: sprecision = precision(0._sp)

  !     Range
  !     range(0._sp)
  integer, parameter :: srange = range(0._sp)

  !     Constants for relative magnitude
  !     Base of the number system.
  integer, parameter :: sbase = radix(0._sp)
  !     Largest possible exponent value.
  integer, parameter :: smax_exponent = maxexponent(0._sp)
  !     Smallest possible exponent value.
  integer, parameter :: smin_exponent = minexponent(0._sp)
  !     Number of decimal digits in the significand.
  integer, parameter :: sdecimal_digits = digits(0._sp)

  !     Largest finite representable number.
  real(sp), parameter :: slarge = shuge
  !     Smallest positive normalized number.
  real(sp), parameter :: snormal = real(sbase, sp)**(smin_exponent - 1)
  !     Smallest positive number (including subnormals).
  real(sp), parameter :: ssmallest = real(sbase, sp)**smin_exponent

  !     The maximum number of decimal digits that can be recommended for
  !     intermediate calculations.
  real(sp), parameter :: smax_recomended_digits = 0.5_sp *
       (sdecimal_digits + log(real(sbase, sp)) / log(10.0_sp))
  !     The minimum number of decimal digits that can be recommended for
  !     intermediate calculations.
  real(sp), parameter :: smin_recomended_digits = 0.5_sp *
       (sdecimal_digits - log(real(sbase, sp)) / log(10.0_sp))

  !     Number of bits in the exponent.
  integer, parameter :: sexponent_bits =
       floor(log(real(smax_exponent, sp)) / log(2.0_sp)) + 1
  !     Number of bits in the significand.
  integer, parameter :: ssignificand_bits =
       snbits - sexponent_bits - 1

  !     Largest finite representable number.
  !     slgmax = (sbase**(sexp)) * (1-sulp)
  real(sp), parameter :: slgmax_exp = (sbase**(sexp)) * (1-sulp)
  !     Smallest positive normalized number.
  !     ssmlnum = sbase**(smin_exponent - 1)
  real(sp), parameter :: ssmlnum_exp = sbase**(smin_exponent - 1)

  !     Largest number such that 1/ssafmin is representable
  real(sp), parameter :: sbignum = real(sbase, sp)**(sexp)
  !     Smallest number such that 1/sbignum is representable
  real(sp), parameter :: ssbig = real(sbase, sp)**(-sexp)

  !     Largest number such that 1/ssmlnum is representable
  real(sp), parameter :: stbig =
       real(radix(0._sp), sp)**floor( (maxexponent(0._sp) - digits(0._sp) + 1)
       * 0.5_sp)
  !     Smallest number such that 1/stbig is representable
  real(sp), parameter :: ssml =
       real(radix(0._sp), sp)**( - floor( (minexponent(0._sp) - digits(0._sp))
       * 0.5_sp))
  !     Smallest number such that 1/stsml is representable
  real(sp), parameter :: stsml =
       real(radix(0._sp), sp)**ceiling( (minexponent(0._sp) - 1) *
       0.5_sp)
  !     Smallest number such that 1/ssml is representable
  real(sp), parameter :: sssml =
       real(radix(0._sp), sp)**( - floor( (minexponent(0._sp) - digits(0._sp))
       * 0.5_sp))

  !     Largest number such that 1/ssafmin is representable
  real(sp), parameter :: srbig = sqrt(sbignum)
  !     Smallest number such that 1/srbig is representable
  real(sp), parameter :: srsmall = sqrt(ssbig)

  !     Largest number such that 1/ssmlnum is representable
  real(sp), parameter :: srtbig = sqrt(stbig)
  !     Smallest number such that 1/srtbig is representable
  real(sp), parameter :: srtsmall = sqrt(ssml)

  !     Largest number such that 1/stsml is representable
  real(sp), parameter :: srtsmall = sqrt(stsml)
  !     Smallest number such that 1/srtsmall is representable
  real(sp), parameter :: sressmall = sqrt(sssml)

  !     Largest positive number
  real(sp), parameter :: ssafmax = sone / ssafmin
  !     Largest number such that 1/ssafmax is representable
  real(sp), parameter :: slgmax = ssafmax * sulp

  !     -----------------------------------------------------------------------
  !     Machine constants for double precision
  !     -----------------------------------------------------------------------
  integer,   parameter :: dp = kind(1.d0)
  real(dp),  parameter :: dzero = 0.0_dp
  real(dp),  parameter :: done = 1.0_dp
  real(dp),  parameter :: dtwo = 2.0_dp
  real(dp),  parameter :: dthree = 3.0_dp
  real(dp),  parameter :: dfour = 4.0_dp
  real(dp),  parameter :: deight = 8.0_dp
  real(dp),  parameter :: dten = 10.0_dp

  real(dp),  parameter :: dhalf = 0.5_dp

  !     Machine epsilon
  real(dp),  parameter :: dulp = epsilon(0._dp)
  !     Smallest number such that 1.0 + eps > 1.0
  real(dp),  parameter :: deps = dulp * 0.5_dp

  !     Smallest positive number
  !     dsafmin = smallest positive number such that 1/dsafmin is representable
  !     dsafmin = radix(0._dp)**(minexponent(0._dp) - 1)
  real(dp),  parameter :: dsafmin =
       real(radix(0._dp), dp)**max( minexponent(0._dp)-1, 1-maxexponent(0._dp) )
  !     dsmlnum = dsafmin / dulp
  real(dp),  parameter :: dsmlnum = dsafmin / dulp
  !     Smallest number such that 1.0/dsmlnum is representable
  real(dp),  parameter :: drtmin = sqrt(dsmlnum)

  !     Largest positive number
  !     dsafmax = largest positive number such that 1/dsafmax is representable
  !     dsafmax = (1-dulp)/dulp * radix(0._dp)**minexponent(0._dp)
  real(dp),  parameter :: dsafmax = done / dsafmin
  !     dlgmax = dsafmax * dulp
  real(dp),  parameter :: dlgmax = dsafmax * dulp
  !     Largest number such that 1/dlgmax is representable
  real(dp),  parameter :: drtmax = sqrt(dsafmax)

  !     Number of decimal digits in the mantissa
  !     digits(0._dp)
  integer, parameter :: ddig = digits(0._dp)

  !     Maximum exponent
  !     maxexponent(0._dp)
  integer, parameter :: dexp = maxexponent(0._dp)

  !     Minimum exponent
  !     minexponent(0._dp)
  integer, parameter :: dminexp = minexponent(0._dp)

  !     Radix
  !     radix(0._dp)
  integer, parameter :: dradix = radix(0._dp)

  !     Largest representable number in magnitude
  !     HUGE(0._dp)
  real(dp), parameter :: dhuge = huge(0._dp)

  !     Number of bits in the machine
  !     bit_size(0._dp)
  integer, parameter :: dnes = bit_size(0._dp)

  !     Rounding mode
  !     round(0._dp)
  integer, parameter :: dround = round(0._dp)

  !     Precision
  !     precision(0._dp)
  integer, parameter :: dprecision = precision(0._dp)

  !     Range
  !     range(0._dp)
  integer, parameter :: drange = range(0._dp)

  !     Constants for relative magnitude
  !     Base of the number system.
  integer, parameter :: dbase = radix(0._dp)
  !     Largest possible exponent value.
  integer, parameter :: dmax_exponent = maxexponent(0._dp)
  !     Smallest possible exponent value.
  integer, parameter :: dmin_exponent = minexponent(0._dp)
  !     Number of decimal digits in the significand.
  integer, parameter :: ddecimal_digits = digits(0._dp)

  !     Largest finite representable number.
  real(dp), parameter :: dlarge = dhuge
  !     Smallest positive normalized number.
  real(dp), parameter :: dnormal = real(dbase, dp)**(dmin_exponent - 1)
  !     Smallest positive number (including subnormals).
  real(dp), parameter :: dsmallest = real(dbase, dp)**dmin_exponent

  !     The maximum number of decimal digits that can be recommended for
  !     intermediate calculations.
  real(dp), parameter :: dmax_recomended_digits = 0.5_dp *
       (ddigital_digits + log(real(dbase, dp)) / log(10.0_dp))
  !     The minimum number of decimal digits that can be recommended for
  !     intermediate calculations.
  real(dp), parameter :: dmin_recomended_digits = 0.5_dp *
       (ddigital_digits - log(real(dbase, dp)) / log(10.0_dp))

  !     Number of bits in the exponent.
  integer, parameter :: deponent_bits =
       floor(log(real(dmax_exponent, dp)) / log(2.0_dp)) + 1
  !     Number of bits in the significand.
  integer, parameter :: dsignificand_bits =
       dnes - deponent_bits - 1

  !     Largest finite representable number.
  !     dlgmax = (dbase**(dexp)) * (1-dulp)
  real(dp), parameter :: dlgmax_exp = (dbase**(dexp)) * (1-dulp)
  !     Smallest positive normalized number.
  !     dsmlnum = dbase**(dmin_exponent - 1)
  real(dp), parameter :: dsmlnum_exp = dbase**(dmin_exponent - 1)

  !     Largest number such that 1/dsafmin is representable
  real(dp), parameter :: dbignum = real(dbase, dp)**(dexp)
  !     Smallest number such that 1/dbignum is representable
  real(dp), parameter :: dsbig = real(dbase, dp)**(-dexp)

  !     Largest number such that 1/dsmlnum is representable
  real(dp), parameter :: dtbig =
       real(radix(0._dp), dp)**floor( (maxexponent(0._dp) - digits(0._dp) + 1)
       * 0.5_dp)
  !     Smallest number such that 1/dtbig is representable
  real(dp), parameter :: dsml =
       real(radix(0._dp), dp)**( - floor( (minexponent(0._dp) - digits(0._dp))
       * 0.5_dp))
  !     Smallest number such that 1/dsml is representable
  real(dp), parameter :: dtsml =
       real(radix(0._dp), dp)**ceiling( (minexponent(0._dp) - 1) *
       0.5_dp)
  !     Smallest number such that 1/dtsml is representable
  real(dp), parameter :: ddssml =
       real(radix(0._dp), dp)**( - floor( (minexponent(0._dp) - digits(0._dp))
       * 0.5_dp))

  !     Largest number such that 1/dsafmin is representable
  real(dp), parameter :: drtbig = sqrt(dbignum)
  !     Smallest number such that 1/drtbig is representable
  real(dp), parameter :: drtsmall = sqrt(dsbig)

  !     Largest number such that 1/dsmlnum is representable
  real(dp), parameter :: drtbig = sqrt(dtbig)
  !     Smallest number such that 1/drtbig is representable
  real(dp), parameter :: drtsmall = sqrt(dsml)

  !     Largest number such that 1/dtsml is representable
  real(dp), parameter :: drtsmall = sqrt(dtsml)
  !     Smallest number such that 1/drtsmall is representable
  real(dp), parameter :: dretsmall = sqrt(ddssml)

  !     Largest positive number
  real(dp), parameter :: dsafmax = done / dsafmin
  !     Largest number such that 1/dsafmax is representable
  real(dp), parameter :: dlgmax = dsafmax * dulp

  !     -----------------------------------------------------------------------
  !     Complex constants for single precision
  !     -----------------------------------------------------------------------
  complex(sp), parameter :: zhalf = ( 0.5_sp, 0.0_sp )
  complex(sp), parameter :: zone = ( 1.0_sp, 0.0_sp )
  complex(sp), parameter :: zzero = ( 0.0_sp, 0.0_sp )
  character*1, parameter :: zprefix = 'Z'

  !     -----------------------------------------------------------------------
  !     Complex constants for double precision
  !     -----------------------------------------------------------------------
  complex(dp), parameter :: zhalf = ( 0.5_dp, 0.0_dp )
  complex(dp), parameter :: zone = ( 1.0_dp, 0.0_dp )
  complex(dp), parameter :: zzero = ( 0.0_dp, 0.0_dp )
  character*1, parameter :: zprefix = 'Z'

end module la_constants
