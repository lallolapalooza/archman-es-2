subroutine sspgv (integer itype, character jobz, character uplo, integer n, real, dimension( * ) ap, real, dimension( * ) bp, real, dimension( * ) w, real, dimension( ldz, * ) z, integer ldz, real, dimension( * ) work, integer info)

SSPGV

Propósito:

!>
!> SSPGV calcula todos los valores propios y, opcionalmente, los vectores propios
!> de un problema de eigenvalor generalizado simétrico-definido real, de la forma
!> A*x=(lambda)*B*x,  A*Bx=(lambda)*x,  o B*A*x=(lambda)*x.
!> Aquí A y B se asumen simétricas, almacenadas en formato empaquetado,
!> y B también es positiva definida.
!>

Parámetros

ITYPE 
!>          ITYPE es INTEGER
!>          Especifica el tipo de problema a resolver:
!>          = 1:  A*x = (lambda)*B*x
!>          = 2:  A*B*x = (lambda)*x
!>          = 3:  B*A*x = (lambda)*x
!>

JOBZ

!>          JOBZ es CHARACTER*1
!>          = 'N':  Calcular solo los valores propios;
!>          = 'V':  Calcular valores propios y vectores propios.
!>

UPLO

!>          UPLO es CHARACTER*1
!>          = 'U':  Se almacenan los triángulos superiores de A y B;
!>          = 'L':  Se almacenan los triángulos inferiores de A y B.
!>

N

!>          N es INTEGER
!>          El orden de las matrices A y B.  N >= 0.
!>

AP

!>          AP es array REAL, dimensión (N*(N+1)/2)
!>          En la entrada, el triángulo superior o inferior de la matriz simétrica
!>          A, empaquetado por columnas en un array lineal. La j-ésima columna de A
!>          se almacena en el array AP de la siguiente manera:
!>          si UPLO = 'U', AP(i + (j-1)*j/2) = A(i,j) para 1<=i<=j;
!>          si UPLO = 'L', AP(i + (j-1)*(2*n-j)/2) = A(i,j) para j<=i<=n.
!>
!>          En la salida, el contenido de AP se destruye.
!>

BP

!>          BP es array REAL, dimensión (N*(N+1)/2)
!>          En la entrada, el triángulo superior o inferior de la matriz simétrica
!>          B, empaquetado por columnas en un array lineal. La j-ésima columna de B
!>          se almacena en el array BP de la siguiente manera:
!>          si UPLO = 'U', BP(i + (j-1)*j/2) = B(i,j) para 1<=i<=j;
!>          si UPLO = 'L', BP(i + (j-1)*(2*n-j)/2) = B(i,j) para j<=i<=n.
!>
!>          En la salida, el factor triangular U o L de la factorización de Cholesky
!>          B = U**T*U o B = L*L**T, en el mismo formato de almacenamiento
!>          que B.
!>

W

!>          W es array REAL, dimensión (N)
!>          Si INFO = 0, los valores propios en orden ascendente.
!>

Z

!>          Z es array REAL, dimensión (LDZ, N)
!>          Si JOBZ = 'V', entonces si INFO = 0, Z contiene la matriz Z de
!>          vectores propios. Los vectores propios se normalizan de la siguiente
!>          manera:
!>          si ITYPE = 1 o 2, Z**T*B*Z = I;
!>          si ITYPE = 3, Z**T*inv(B)*Z = I.
!>          Si JOBZ = 'N', entonces Z no se referencia.
!>

LDZ

!>          LDZ es INTEGER
!>          La dimensión principal del array Z.  LDZ >= 1, y si
!>          JOBZ = 'V', LDZ >= max(1,N).
!>

WORK

!>          WORK es array REAL, dimensión (3*N)
!>

INFO

!>          INFO es INTEGER
!>          = 0:  salida exitosa
!>          < 0:  si INFO = -i, el i-ésimo argumento tuvo un valor ilegal
!>          > 0:  SPPTRF o SSPEV devolvieron un código de error:
!>             <= N:  si INFO = i, SSPEV falló en converger;
!>                    i elementos fuera de la diagonal de una forma tridiagonal
!>                    intermedia no convergeron a cero.
!>             > N:   si INFO = n + i, para 1 <= i <= n, entonces el menor
!>                    principal líder de orden i de B no es positivo.
!>                    La factorización de B no pudo completarse y
!>                    no se calcularon valores propios ni vectores propios.
!>
