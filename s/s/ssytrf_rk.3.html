subroutine ssytrf_rk (character uplo, integer n, real, dimension( lda, * ) a, integer lda, real, dimension( * ) e, integer, dimension( * ) ipiv, real, dimension( * ) work, integer lwork, integer info)

SSYTRF_RK calcula la factorización de una matriz real simétrica indefinida utilizando el método de pivoteo diagonal de Bunch-Kaufman acotado (rook) (algoritmo bloqueado BLAS3).

Propósito:

!> SSYTRF_RK calcula la factorización de una matriz real simétrica A
!> utilizando el método de pivoteo diagonal de Bunch-Kaufman acotado (rook):
!>
!>    A = P*U*D*(U**T)*(P**T) o A = P*L*D*(L**T)*(P**T),
!>
!> donde U (o L) es una matriz triangular unitaria superior (o inferior),
!> U**T (o L**T) es la transpuesta de U (o L), P es una matriz de permutación,
!> P**T es la transpuesta de P, y D es simétrica y diagonal por bloques con
!> bloques diagonales de 1x1 y 2x2.
!>
!> Esta es la versión bloqueada del algoritmo, que llama a BLAS de Nivel 3.
!> Para más información, consulte la sección de Detalles Adicionales.
!>

Parámetros

UPLO 
!>          UPLO es CHARACTER*1
!>          Especifica si la parte triangular superior o inferior de la
!>          matriz simétrica A está almacenada:
!>          = 'U': Triangular superior
!>          = 'L': Triangular inferior
!>

N

!>          N es INTEGER
!>          El orden de la matriz A.  N >= 0.
!>

A

!>          A es array REAL, dimension (LDA,N)
!>          En la entrada, la matriz simétrica A.
!>            Si UPLO = 'U': la parte triangular superior N-por-N líder
!>            de A contiene la parte triangular superior de la matriz A,
!>            y la parte estrictamente triangular inferior de A no se
!>            referencia.
!>
!>            Si UPLO = 'L': la parte triangular inferior N-por-N líder
!>            de A contiene la parte triangular inferior de la matriz A,
!>            y la parte estrictamente triangular superior de A no se
!>            referencia.
!>
!>          En la salida, contiene:
!>            a) SOLAMENTE los elementos diagonales de la matriz simétrica
!>               diagonal por bloques D en la diagonal de A, es decir,
!>               D(k,k) = A(k,k);
!>               (los elementos superdiagonales (o subdiagonales) de D
!>                se almacenan en la salida en el array E), y
!>            b) Si UPLO = 'U': factor U en la parte superdiagonal de A.
!>               Si UPLO = 'L': factor L en la parte subdiagonal de A.
!>

LDA

!>          LDA es INTEGER
!>          La dimensión líder del array A.  LDA >= max(1,N).
!>

E

!>          E es array REAL, dimension (N)
!>          En la salida, contiene los elementos superdiagonales (o
!>          subdiagonales) de la matriz simétrica diagonal por bloques D
!>          con bloques diagonales de 1x1 o 2x2, donde
!>          Si UPLO = 'U': E(i) = D(i-1,i), i=2:N, E(1) se establece a 0;
!>          Si UPLO = 'L': E(i) = D(i+1,i), i=1:N-1, E(N) se establece a 0.
!>
!>          NOTA: Para un bloque diagonal de 1x1 D(k), donde
!>          1 <= k <= N, el elemento E(k) se establece a 0 en ambos
!>          casos UPLO = 'U' o UPLO = 'L'.
!>

IPIV

!>          IPIV es array INTEGER, dimension (N)
!>          IPIV describe la matriz de permutación P en la factorización
!>          de la matriz A de la siguiente manera. El valor absoluto de IPIV(k)
!>          representa el índice de la fila y columna que fueron
!>          intercambiados con la fila y columna k. El valor de UPLO
!>          describe el orden en que se aplicaron los intercambios.
!>          Además, el signo de IPIV representa la estructura de bloque
!>          de la matriz simétrica diagonal por bloques D con bloques
!>          diagonales de 1x1 o 2x2 que corresponden a 1 o 2 intercambios
!>          en cada paso de factorización. Para más información, consulte
!>          la sección de Detalles Adicionales.
!>
!>          Si UPLO = 'U',
!>          (en orden de factorización, k disminuye de N a 1):
!>            a) Una única entrada positiva IPIV(k) > 0 significa:
!>               D(k,k) es un bloque diagonal de 1x1.
!>               Si IPIV(k) != k, las filas y columnas k e IPIV(k) fueron
!>               intercambiadas en la matriz A(1:N,1:N);
!>               Si IPIV(k) = k, no ocurrió ningún intercambio.
!>
!>            b) Un par de entradas negativas consecutivas
!>               IPIV(k) < 0 e IPIV(k-1) < 0 significa:
!>               D(k-1:k,k-1:k) es un bloque diagonal de 2x2.
!>               (NOTA: las entradas negativas en IPIV aparecen SOLAMENTE en pares).
!>               1) Si -IPIV(k) != k, las filas y columnas
!>                  k y -IPIV(k) fueron intercambiadas
!>                  en la matriz A(1:N,1:N).
!>                  Si -IPIV(k) = k, no ocurrió ningún intercambio.
!>               2) Si -IPIV(k-1) != k-1, las filas y columnas
!>                  k-1 y -IPIV(k-1) fueron intercambiadas
!>                  en la matriz A(1:N,1:N).
!>                  Si -IPIV(k-1) = k-1, no ocurrió ningún intercambio.
!>
!>            c) En ambos casos a) y b), siempre ABS( IPIV(k) ) <= k.
!>
!>            d) NOTA: Cualquier entrada IPIV(k) es siempre NO NULA en la salida.
!>
!>          Si UPLO = 'L',
!>          (en orden de factorización, k aumenta de 1 a N):
!>            a) Una única entrada positiva IPIV(k) > 0 significa:
!>               D(k,k) es un bloque diagonal de 1x1.
!>               Si IPIV(k) != k, las filas y columnas k e IPIV(k) fueron
!>               intercambiadas en la matriz A(1:N,1:N).
!>               Si IPIV(k) = k, no ocurrió ningún intercambio.
!>
!>            b) Un par de entradas negativas consecutivas
!>               IPIV(k) < 0 e IPIV(k+1) < 0 significa:
!>               D(k:k+1,k:k+1) es un bloque diagonal de 2x2.
!>               (NOTA: las entradas negativas en IPIV aparecen SOLAMENTE en pares).
!>               1) Si -IPIV(k) != k, las filas y columnas
!>                  k y -IPIV(k) fueron intercambiadas
!>                  en la matriz A(1:N,1:N).
!>                  Si -IPIV(k) = k, no ocurrió ningún intercambio.
!>               2) Si -IPIV(k+1) != k+1, las filas y columnas
!>                  k-1 y -IPIV(k-1) fueron intercambiadas
!>                  en la matriz A(1:N,1:N).
!>                  Si -IPIV(k+1) = k+1, no ocurrió ningún intercambio.
!>
!>            c) En ambos casos a) y b), siempre ABS( IPIV(k) ) >= k.
!>
!>            d) NOTA: Cualquier entrada IPIV(k) es siempre NO NULA en la salida.
!>

WORK

!>          WORK es array REAL, dimension (MAX(1,LWORK)).
!>          En la salida, si INFO = 0, WORK(1) devuelve el LWORK óptimo.
!>

LWORK

!>          LWORK es INTEGER
!>          La longitud de WORK.  LWORK >= 1.  Para un rendimiento óptimo
!>          LWORK >= N*NB, donde NB es el tamaño del bloque devuelto
!>          por ILAENV.
!>
!>          Si LWORK = -1, entonces se asume una consulta de espacio de
!>          trabajo; la rutina solo calcula el tamaño óptimo del array
!>          WORK, devuelve este valor como la primera entrada del array
!>          WORK, y XERBLA no emite ningún mensaje de error relacionado
!>          con LWORK.
!>

INFO

!>          INFO es INTEGER
!>          = 0: salida exitosa
!>
!>          < 0: Si INFO = -k, el k-ésimo argumento tenía un valor ilegal
!>
!>          > 0: Si INFO = k, la matriz A es singular, porque:
!>                 Si UPLO = 'U': la columna k en la parte
!>                 triangular superior de A contiene todos ceros.
!>                 Si UPLO = 'L': la columna k en la parte
!>                 triangular inferior de A contiene todos ceros.
!>
!>               Por lo tanto, D(k,k) es exactamente cero, y los elementos
!>               superdiagonales de la columna k de U (o subdiagonales de
!>               la columna k de L) son todos ceros. La factorización se ha
!>               completado, pero la matriz diagonal por bloques D
!>               es exactamente singular, y se producirá una división por cero si
!>               se utiliza para resolver un sistema de ecuaciones.

Function/Subroutine Documentation

• UPLO
CHARACTER * 1. • INPUT • On entry, UPLO specifies whether the upper or lower triangular part of the array A is to be referenced as follows: 
      UPLO = 'U' => Upper triangle
      UPLO = 'L' => Lower triangle
• UNCHANGED.
• N
INTEGER. • INPUT • The order of the matrix A. • N ≥ 0.
• A
REAL , DIMENSION( LDA, * ). • INPUT/OUTPUT • On entry, the symmetric (or triangular) matrix A. • On exit, the factorization has been completed, and the solution A = S * T * S**T where S is a block upper triangular matrix and T is a block diagonal matrix is available if INFO = 0. • On exit, if UPLO = 'U', A contains the upper triangle of the matrix S and the diagonal blocks T. If UPLO = 'L', A contains the lower triangle of the matrix S and the diagonal blocks T.
• LDA
INTEGER. • INPUT • The leading dimension of the array A. • LDA ≥ max( 1, N ).
• IPVT
INTEGER , DIMENSION( MIN( N, NB ) ). • OUTPUT • Details of the block factorization, a sequence of block indices indicating the sequence of pivot blocks. • The output depends on the value of INFO.
• INFO
INTEGER. • OUTPUT • • If INFO = 0, the factorization has been completed. • If INFO > 0, the factorization has been completed, but IPVT(INFO) indicates a singularity. INFO is the order of the leading minor which is singular. • If INFO < 0, then the i-th argument had an illegal value. It is indicated also by the exception algorithm: 1 ≤ i ≤ 7.

Algorithm Description:

!> The algorithm used is described in "Block algorithms for symmetric
!> factorization", Nicolas J. Higham, 2007.
!> This routine uses block partition of the matrix, where the block
!> size is the minimum of NB and N, where NB is the block size.
!> The algorithm is:
!>
!> Algorithm SSYTRF_RK(A)
!>
!> Input: symmetric matrix A of order N
!> Output: factorization A = S * T * S**T
!>
!>   if N == 0 then return
!>
!>   NB = max(1, iint(A(1,1)))  ! Block size, derived from array A
!>   if N <= NB then
!>     call SSYTRF(UPLO, N, A, LDA, IPVT, INFO) ! Use standard algorithm
!>   else
!>     call SSYTRF_RK_1(UPLO, N, A, LDA, IPVT, INFO) ! Use block algorithm
!>   endif
!>
!> The implementation of SSYTRF_RK_1 is as follows:
!>
!>   Let A = [ A11 A12 ]
!>           [ A21 A22 ]
!>   where A11 is NB x NB
!>
!>   if UPLO == 'U' then
!>     call SSYTRF(UPLO, NB, A11, LDA, IPVT, INFO) ! Factorize A11
!>     call STRT_update(NB, NB+1, N, A, LDA)     ! Update A12
!>     call SSYRK(UPLO, TRANSPOSE, N-NB, NB, -1.0, A(1, NB+1), LDA, &
!>               1.0, A(NB+1, NB+1), LDA)      ! Update A22
!>     call SSYTRF_RK_1(UPLO, N-NB, A(NB+1, NB+1), LDA, IPVT(NB+1), INFO) ! Factorize A22
!>   else ! UPLO == 'L'
!>     call SSYTRF(UPLO, NB, A11, LDA, IPVT, INFO) ! Factorize A11
!>     call STRT_update(NB+1, 1, N, A, LDA)     ! Update A21
!>     call SSYRK(UPLO, TRANSPOSE, N-NB, NB, -1.0, A(NB+1, 1), LDA, &
!>               1.0, A(NB+1, NB+1), LDA)      ! Update A22
!>     call SSYTRF_RK_1(UPLO, N-NB, A(NB+1, NB+1), LDA, IPVT(NB+1), INFO) ! Factorize A22
!>   endif
!>
!> Note: INFO is the first entry in IPVT that indicates a singularity.
!>       The factorization always completes, but the resulting matrix may
!>       be singular.
!>
!> NOTE: INFO only stores the first occurrence of
!>       a singularity, any subsequent occurrence of singularity
!>       is not stored in INFO even though the factorization
!>       always completes.
!>
