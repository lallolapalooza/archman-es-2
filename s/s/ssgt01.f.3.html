subroutine ssgt01 (integer itype, character uplo, integer n, integer m, real, dimension( lda, * ) a, integer lda, real, dimension( ldb, * ) b, integer ldb, real, dimension( ldz, * ) z, integer ldz, real, dimension( * ) d, real, dimension( * ) work, real, dimension( * ) result)

SSGT01

Propósito:

!>
!> SSGT01 comprueba una descomposición de la forma
!>
!>    A Z   =  B Z D o
!>    A B Z =  Z D o
!>    B A Z =  Z D
!>
!> donde A es una matriz simétrica, B es
!> simétrica definida positiva, Z es ortogonal y D es diagonal.
!>
!> Se calcula uno de los siguientes ratios de prueba:
!>
!> ITYPE = 1:  RESULT(1) = | A Z - B Z D | / ( |A| |Z| n ulp )
!>
!> ITYPE = 2:  RESULT(1) = | A B Z - Z D | / ( |A| |Z| n ulp )
!>
!> ITYPE = 3:  RESULT(1) = | B A Z - Z D | / ( |A| |Z| n ulp )
!>

Parámetros

ITYPE 
!>          ITYPE es INTEGER
!>          La forma del problema de valores propios generalizado simétrico.
!>          = 1:  A*z = (lambda)*B*z
!>          = 2:  A*B*z = (lambda)*z
!>          = 3:  B*A*z = (lambda)*z
!>

UPLO

!>          UPLO es CHARACTER*1
!>          Especifica si la parte triangular superior o inferior de las
!>          matrices simétricas A y B está almacenada.
!>          = 'U':  Triangular superior
!>          = 'L':  Triangular inferior
!>

N

!>          N es INTEGER
!>          El orden de la matriz A.  N >= 0.
!>

M

!>          M es INTEGER
!>          El número de valores propios encontrados.  0 <= M <= N.
!>

A

!>          A es un array REAL, dimensión (LDA, N)
!>          La matriz simétrica original A.
!>

LDA

!>          LDA es INTEGER
!>          La dimensión principal del array A.  LDA >= max(1,N).
!>

B

!>          B es un array REAL, dimensión (LDB, N)
!>          La matriz simétrica definida positiva original B.
!>

LDB

!>          LDB es INTEGER
!>          La dimensión principal del array B.  LDB >= max(1,N).
!>

Z

!>          Z es un array REAL, dimensión (LDZ, M)
!>          Los vectores propios calculados del problema de valores propios generalizado.
!>

LDZ

!>          LDZ es INTEGER
!>          La dimensión principal del array Z.  LDZ >= max(1,N).
!>

D

!>          D es un array REAL, dimensión (M)
!>          Los valores propios calculados del problema de valores propios generalizado.
!>

WORK

!>          WORK es un array REAL, dimensión (N*N)
!>

RESULT

!>          RESULT es un array REAL, dimensión (1)
!>          El ratio de prueba como se describe arriba.
!>
