subroutine ssb2st_kernels (character uplo, logical wantz, integer ttype, integer st, integer ed, integer sweep, integer n, integer nb, integer ib, real, dimension( lda, * ) a, integer lda, real, dimension( * ) v, real, dimension( * ) tau, integer ldvt, real, dimension( * ) work)

SSB2ST_KERNELS

Propósito:

!>
!> SSB2ST_KERNELS es una rutina interna utilizada por la subrutina SSYTRD_SB2ST
!>

Parámetros

UPLO 
!>          UPLO es CHARACTER*1
!>

WANTZ

!>          WANTZ es LOGICAL que indica si se solicitan valores propios o ambos
!>          valores propios/vectores propios.
!>

TTYPE

!>          TTYPE es INTEGER
!>

ST

!>          ST es INTEGER
!>          parámetro interno para índices.
!>

ED

!>          ED es INTEGER
!>          parámetro interno para índices.
!>

SWEEP

!>          SWEEP es INTEGER
!>          parámetro interno para índices.
!>

N

!>          N es INTEGER. El orden de la matriz A.
!>

NB

!>          NB es INTEGER. El tamaño de la banda.
!>

IB

!>          IB es INTEGER.
!>

A

!>          A es un array REAL. Un puntero a la matriz A.
!>

LDA

!>          LDA es INTEGER. La dimensión principal de la matriz A.
!>

V

!>          V es un array REAL, de dimensión 2*n si solo se solicitan
!>          valores propios o se consultan vectores.
!>

TAU

!>          TAU es un array REAL, de dimensión (2*n).
!>          Los factores escalares de los reflectores de Householder se almacenan
!>          en este array.
!>

LDVT

!>          LDVT es INTEGER.
!>

WORK

!>          WORK es un array REAL. Espacio de trabajo de tamaño nb.
!>
