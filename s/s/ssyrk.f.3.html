DOCUMENTACIÓN DE FUNCIÓN/SUBRUTINA

subroutine ssyrk (character uplo, character trans, integer n, integer k, real alpha, real, dimension(lda,*) a, integer lda, real beta, real, dimension(ldc,*) c, integer ldc)

SSYRK

Propósito:

!>
!> SSYRK  realiza una de las operaciones de rango k simétricas
!>
!>    C := alpha*A*A**T + beta*C,
!>
!> o
!>
!>    C := alpha*A**T*A + beta*C,
!>
!> donde  alpha y beta  son escalares, C es una matriz simétrica de n por n
!> y  A  es una matriz de n por k  en el primer caso y una matriz de k por n
!> en el segundo caso.
!>

Parámetros

UPLO 
!>          UPLO es CHARACTER*1
!>           Al entrar,   UPLO  especifica  si  la  parte  triangular  superior  o
!>           inferior  del  array  C  debe  ser  referenciada  como
!>           sigue:
!>
!>              UPLO = 'U' o 'u'   Solo se referencia la parte triangular superior de C.
!>
!>              UPLO = 'L' o 'l'   Solo se referencia la parte triangular inferior de C.
!>

TRANS

!>          TRANS es CHARACTER*1
!>           Al entrar,  TRANS  especifica la operación a realizar como
!>           sigue:
!>
!>              TRANS = 'N' o 'n'   C := alpha*A*A**T + beta*C.
!>
!>              TRANS = 'T' o 't'   C := alpha*A**T*A + beta*C.
!>
!>              TRANS = 'C' o 'c'   C := alpha*A**T*A + beta*C.
!>

N

!>          N es INTEGER
!>           Al entrar,  N  especifica el orden de la matriz C.  N debe ser
!>           al menos cero.
!>

K

!>          K es INTEGER
!>           Al entrar con  TRANS = 'N' o 'n',  K  especifica el número
!>           de  columnas   de la  matriz   A,   y  al  entrar  con
!>           TRANS = 'T' o 't' o 'C' o 'c',  K  especifica el  número
!>           de filas de la matriz  A.  K debe ser al menos cero.
!>

ALPHA

!>          ALPHA es REAL
!>           Al entrar, ALPHA especifica el escalar alpha.
!>

A

!>          A es array REAL, dimension ( LDA, ka ), donde ka es
!>           k  cuando  TRANS = 'N' o 'n',  y es  n  en caso contrario.
!>           Antes de entrar con  TRANS = 'N' o 'n',  la parte  superior
!>           principal de n por k  del array  A  debe contener la matriz  A,
!>           en caso contrario, la parte superior principal de k por n
!>           del array  A  debe contener la matriz A.
!>

LDA

!>          LDA es INTEGER
!>           Al entrar, LDA especifica la primera dimensión de A tal como
!>           se declara en el  programa (sub)llamador.   Cuando  TRANS = 'N' o 'n'
!>           entonces  LDA debe ser al menos  max( 1, n ), en caso contrario
!>           LDA debe ser al menos  max( 1, k ).
!>

BETA

!>          BETA es REAL
!>           Al entrar, BETA especifica el escalar beta.
!>

C

!>          C es array REAL, dimension ( LDC, N )
!>           Antes de entrar con  UPLO = 'U' o 'u',  la parte superior
!>           principal de n por n  del array C debe contener la parte
!>           triangular superior  de la matriz simétrica  y la parte estrictamente
!>           triangular inferior de C no se referencia.  Al salir, la
!>           parte triangular superior del array  C se sobrescribe con la
!>           parte triangular superior de la matriz actualizada.
!>           Antes de entrar con  UPLO = 'L' o 'l',  la parte inferior
!>           principal de n por n  del array C debe contener la parte
!>           triangular inferior  de la matriz simétrica  y la parte estrictamente
!>           triangular superior de C no se referencia.  Al salir, la
!>           parte triangular inferior del array  C se sobrescribe con la
!>           parte triangular inferior de la matriz actualizada.
!>

LDC

!>          LDC es INTEGER
!>           Al entrar, LDC especifica la primera dimensión de C tal como
!>           se declara en el  programa (sub)llamador.   LDC  debe ser
!>           al menos  max( 1, n ).
!>
