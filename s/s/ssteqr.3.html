subroutine ssteqr (character compz, integer n, real, dimension( * ) d, real, dimension( * ) e, real, dimension( ldz, * ) z, integer ldz, real, dimension( * ) work, integer info)

SSTEQR

Propósito:

!>
!> SSTEQR calcula todos los valores propios y, opcionalmente, los vectores propios de una
!> matriz tridiagonal simétrica utilizando el método QL o QR implícito.
!> Los vectores propios de una matriz simétrica completa o de banda también se pueden encontrar
!> si se han utilizado SSYTRD o SSPTRD o SSBTRD para reducir esta matriz a
!> forma tridiagonal.
!>

Parámetros

COMPZ 
!>          COMPZ es CHARACTER*1
!>          = 'N':  Solo calcula los valores propios.
!>          = 'V':  Calcula los valores propios y los vectores propios de la matriz
!>                  simétrica original. En la entrada, Z debe contener la
!>                  matriz ortogonal utilizada para reducir la matriz original
!>                  a forma tridiagonal.
!>          = 'I':  Calcula los valores propios y los vectores propios de la
!>                  matriz tridiagonal. Z se inicializa a la matriz identidad.
!>

N

!>          N es INTEGER
!>          El orden de la matriz. N >= 0.
!>

D

!>          D es un array REAL de dimensión (N)
!>          En la entrada, los elementos diagonales de la matriz tridiagonal.
!>          En la salida, si INFO = 0, los valores propios en orden ascendente.
!>

E

!>          E es un array REAL de dimensión (N-1)
!>          En la entrada, los (n-1) elementos subdiagonales de la matriz
!>          tridiagonal.
!>          En la salida, E ha sido destruido.
!>

Z

!>          Z es un array REAL de dimensión (LDZ, N)
!>          En la entrada, si COMPZ = 'V', entonces Z contiene la matriz
!>          ortogonal utilizada en la reducción a forma tridiagonal.
!>          En la salida, si INFO = 0, entonces si COMPZ = 'V', Z contiene los
!>          vectores propios ortonormales de la matriz simétrica original,
!>          y si COMPZ = 'I', Z contiene los vectores propios
!>          ortonormales de la matriz tridiagonal simétrica.
!>          Si COMPZ = 'N', entonces Z no se referencia.
!>

LDZ

!>          LDZ es INTEGER
!>          La dimensión principal del array Z. LDZ >= 1, y si
!>          se desean vectores propios, entonces LDZ >= max(1,N).
!>

WORK

!>          WORK es un array REAL de dimensión (max(1,2*N-2))
!>          Si COMPZ = 'N', entonces WORK no se referencia.
!>

INFO

!>          INFO es INTEGER
!>          = 0:  salida exitosa
!>          < 0:  si INFO = -i, el i-ésimo argumento tuvo un valor ilegal
!>          > 0:  el algoritmo ha fallado en encontrar todos los valores propios en
!>                un total de 30*N iteraciones; si INFO = i, entonces i
!>                elementos de E no han convergido a cero; en la salida, D
!>                y E contienen los elementos de una matriz tridiagonal
!>                simétrica que es ortogonalmente similar a la matriz
!>                original.
!>
