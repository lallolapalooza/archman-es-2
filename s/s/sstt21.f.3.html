subroutine sstt21 (integer n, integer kband, real, dimension( * ) ad, real, dimension( * ) ae, real, dimension( * ) sd, real, dimension( * ) se, real, dimension( ldu, * ) u, integer ldu, real, dimension( * ) work, real, dimension( 2 ) result)

SSTT21

Propósito:

!>
!> SSTT21 comprueba una descomposición de la forma
!>
!>    A = U S U'
!>
!> donde ' significa transpuesta, A es simétrica tridiagonal, U es ortogonal,
!> y S es diagonal (si KBAND=0) o simétrica tridiagonal (si KBAND=1).
!> Se realizan dos pruebas:
!>
!>    RESULT(1) = | A - U S U' | / ( |A| n ulp )
!>
!>    RESULT(2) = | I - UU' | / ( n ulp )
!>

Parámetros

N 
!>          N es INTEGER
!>          El tamaño de la matriz. Si es cero, SSTT21 no hace nada.
!>          Debe ser al menos cero.
!>

KBAND

!>          KBAND es INTEGER
!>          El ancho de banda de la matriz S. Sólo puede ser cero o uno.
!>          Si es cero, entonces S es diagonal y SE no se referencia. Si
!>          es uno, entonces S es tri-diagonal simétrica.
!>

AD

!>          AD es REAL array, dimension (N)
!>          La diagonal de la matriz A original (no factorizada). Se
!>          supone que A es simétrica tridiagonal.
!>

AE

!>          AE es REAL array, dimension (N-1)
!>          La fuera de diagonal de la matriz A original (no factorizada). Se
!>          supone que A es simétrica tridiagonal. AE(1) es el elemento (1,2)
!>          y (2,1), AE(2) es el elemento (2,3) y (3,2), etc.
!>

SD

!>          SD es REAL array, dimension (N)
!>          La diagonal de la matriz S (tri-) diagonal (simétrica).
!>

SE

!>          SE es REAL array, dimension (N-1)
!>          La fuera de diagonal de la matriz S (tri-) diagonal (simétrica).
!>          No se referencia si KBSND=0. Si KBAND=1, entonces AE(1) es el
!>          elemento (1,2) y (2,1), SE(2) es el elemento (2,3) y (3,2),
!>          etc.
!>

U

!>          U es REAL array, dimension (LDU, N)
!>          La matriz ortogonal en la descomposición.
!>

LDU

!>          LDU es INTEGER
!>          La dimensión principal de U. LDU debe ser al menos N.
!>

WORK

!>          WORK es REAL array, dimension (N*(N+1))
!>

RESULT

!>          RESULT es REAL array, dimension (2)
!>          Los valores calculados por las dos pruebas descritas
!>          anteriormente. Los valores están actualmente limitados a 1/ulp,
!>          para evitar desbordamiento. RESULT(1) se modifica siempre.
!>
