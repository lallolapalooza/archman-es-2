subroutine sspmv (character uplo, integer n, real alpha, real, dimension(*) ap, real, dimension(*) x, integer incx, real beta, real, dimension(*) y, integer incy)

SSPMV

Propósito:

!>
!> SSPMV  realiza la operación matriz-vector
!>
!>    y := alpha*A*x + beta*y,
!>
!> donde alpha y beta son escalares, x e y son vectores de n elementos y
!> A es una matriz simétrica de n por n, suministrada en forma empaquetada.
!>

Parámetros

UPLO 
!>          UPLO es CHARACTER*1
!>           En la entrada, UPLO especifica si la parte triangular superior o inferior de la matriz A se suministra en el arreglo AP de la siguiente manera:
!>
!>              UPLO = 'U' o 'u'   Se suministra la parte triangular superior de A en AP.
!>
!>              UPLO = 'L' o 'l'   Se suministra la parte triangular inferior de A en AP.
!>

N

!>          N es INTEGER
!>           En la entrada, N especifica el orden de la matriz A.
!>           N debe ser mayor o igual a cero.
!>

ALPHA

!>          ALPHA es REAL
!>           En la entrada, ALPHA especifica el escalar alpha.
!>

AP

!>          AP es un arreglo REAL, de dimensión al menos
!>           ( ( n*( n + 1 ) )/2 ).
!>           Antes de la entrada con UPLO = 'U' o 'u', el arreglo AP debe
!>           contener la parte triangular superior de la matriz simétrica
!>           empaquetada secuencialmente, columna por columna, de modo que AP( 1 )
!>           contenga a( 1, 1 ), AP( 2 ) y AP( 3 ) contengan a( 1, 2 )
!>           y a( 2, 2 ) respectivamente, y así sucesivamente.
!>           Antes de la entrada con UPLO = 'L' o 'l', el arreglo AP debe
!>           contener la parte triangular inferior de la matriz simétrica
!>           empaquetada secuencialmente, columna por columna, de modo que AP( 1 )
!>           contenga a( 1, 1 ), AP( 2 ) y AP( 3 ) contengan a( 2, 1 )
!>           y a( 3, 1 ) respectivamente, y así sucesivamente.
!>

X

!>          X es un arreglo REAL, de dimensión al menos
!>           ( 1 + ( n - 1 )*abs( INCX ) ).
!>           Antes de la entrada, el arreglo incrementado X debe contener el vector x de n elementos.
!>

INCX

!>          INCX es INTEGER
!>           En la entrada, INCX especifica el incremento para los elementos de
!>           X. INCX no debe ser cero.
!>

BETA

!>          BETA es REAL
!>           En la entrada, BETA especifica el escalar beta. Cuando BETA se
!>           suministra como cero, Y no necesita ser configurado en la entrada.
!>

Y

!>          Y es un arreglo REAL, de dimensión al menos
!>           ( 1 + ( n - 1 )*abs( INCY ) ).
!>           Antes de la entrada, el arreglo incrementado Y debe contener el vector y de n elementos. En la salida, Y se sobrescribe con el vector y actualizado.
!>

INCY

!>          INCY es INTEGER
!>           En la entrada, INCY especifica el incremento para los elementos de
!>           Y. INCY no debe ser cero.
!>
