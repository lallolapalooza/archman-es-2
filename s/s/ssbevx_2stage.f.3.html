subroutine ssbevx_2stage (character jobz, character range, character uplo, integer n, integer kd, real, dimension( ldab, * ) ab, integer ldab, real, dimension( ldq, * ) q, integer ldq, real vl, real vu, integer il, integer iu, real abstol, integer m, real, dimension( * ) w, real, dimension( ldz, * ) z, integer ldz, real, dimension( * ) work, integer lwork, integer, dimension( * ) iwork, integer, dimension( * ) ifail, integer info)

SSBEVX_2STAGE calcula los valores y, opcionalmente, los vectores propios izquierdos y/o derechos para OTRAS matrices

Propósito:

!>
!> SSBEVX_2STAGE calcula los valores propios seleccionados y, opcionalmente, los
!> vectores propios de una matriz de banda simétrica real A utilizando la técnica
!> de 2 etapas para la reducción a tridiagonal. Los valores y vectores propios
!> pueden seleccionarse especificando un rango de valores o un rango de índices
!> para los valores propios deseados.
!>

Parámetros

JOBZ 
!>          JOBZ es CHARACTER*1
!>          = 'N':  Calcular solo los valores propios;
!>          = 'V':  Calcular valores y vectores propios.
!>                  No disponible en esta versión.
!>

RANGE

!>          RANGE es CHARACTER*1
!>          = 'A': se encontrarán todos los valores propios;
!>          = 'V': se encontrarán todos los valores propios en el intervalo semiabierto (VL,VU]
!>                 ;
!>          = 'I': se encontrarán los valores propios del IL-ésimo al IU-ésimo.
!>

UPLO

!>          UPLO es CHARACTER*1
!>          = 'U':  Se almacena el triángulo superior de A;
!>          = 'L':  Se almacena el triángulo inferior de A.
!>

N

!>          N es INTEGER
!>          El orden de la matriz A.  N >= 0.
!>

KD

!>          KD es INTEGER
!>          El número de superdiagonales de la matriz A si UPLO = 'U',
!>          o el número de subdiagonales si UPLO = 'L'.  KD >= 0.
!>

AB

!>          AB es un array REAL, dimensión (LDAB, N)
!>          En la entrada, el triángulo superior o inferior de la matriz de banda
!>          simétrica A, almacenado en las primeras KD+1 filas del array. La
!>          j-ésima columna de A se almacena en la j-ésima columna del array AB
!>          de la siguiente manera:
!>          si UPLO = 'U', AB(kd+1+i-j,j) = A(i,j) para max(1,j-kd)<=i<=j;
!>          si UPLO = 'L', AB(1+i-j,j)    = A(i,j) para j<=i<=min(n,j+kd).
!>
!>          En la salida, AB se sobrescribe con valores generados durante la
!>          reducción a forma tridiagonal. Si UPLO = 'U', la primera
!>          superdiagonal y la diagonal de la matriz tridiagonal T
!>          se devuelven en las filas KD y KD+1 de AB, y si UPLO = 'L',
!>          la diagonal y la primera subdiagonal de T se devuelven en las
!>          primeras dos filas de AB.
!>

LDAB

!>          LDAB es INTEGER
!>          La dimensión principal del array AB.  LDAB >= KD + 1.
!>

Q

!>          Q es un array REAL, dimensión (LDQ, N)
!>          Si JOBZ = 'V', la matriz ortogonal N-por-N utilizada en la
!>                         reducción a tridiagonal.
!>          Si JOBZ = 'N', el array Q no se referencia.
!>

LDQ

!>          LDQ es INTEGER
!>          La dimensión principal del array Q.  Si JOBZ = 'V', entonces
!>          LDQ >= max(1,N).
!>

VL

!>          VL es REAL
!>          Si RANGE='V', el límite inferior del intervalo a
!>          buscar para los valores propios. VL < VU.
!>          No se referencia si RANGE = 'A' o 'I'.
!>

VU

!>          VU es REAL
!>          Si RANGE='V', el límite superior del intervalo a
!>          buscar para los valores propios. VL < VU.
!>          No se referencia si RANGE = 'A' o 'I'.
!>

IL

!>          IL es INTEGER
!>          Si RANGE='I', el índice del
!>          valor propio más pequeño a devolver.
!>          1 <= IL <= IU <= N, si N > 0; IL = 1 y IU = 0 si N = 0.
!>          No se referencia si RANGE = 'A' o 'V'.
!>

IU

!>          IU es INTEGER
!>          Si RANGE='I', el índice del
!>          valor propio más grande a devolver.
!>          1 <= IL <= IU <= N, si N > 0; IL = 1 y IU = 0 si N = 0.
!>          No se referencia si RANGE = 'A' o 'V'.
!>

ABSTOL

!>          ABSTOL es REAL
!>          La tolerancia de error absoluto para los valores propios.
!>          Se acepta un valor propio aproximado como convergente
!>          cuando se determina que se encuentra en un intervalo [a,b]
!>          de anchura menor o igual a
!>
!>                  ABSTOL + EPS *   max( |a|,|b| ) ,
!>
!>          donde EPS es la precisión de la máquina. Si ABSTOL es menor o
!>          igual que cero, entonces se usará EPS*|T| en su lugar,
!>          donde |T| es la norma 1 de la matriz tridiagonal obtenida
!>          al reducir AB a forma tridiagonal.
!>
!>          Los valores propios se calcularán con la mayor precisión cuando ABSTOL
!>          se establezca en el doble del umbral de subdesbordamiento 2*SLAMCH('S'),
!>          no en cero.
!>          Si esta rutina retorna con INFO>0, indicando que algunos
!>          vectores propios no convergieron, intente establecer ABSTOL a
!>          2*SLAMCH('S').
!>
!>          Ver  por Demmel y
!>          Kahan, Nota de Trabajo de LAPACK #3.
!>

M

!>          M es INTEGER
!>          El número total de valores propios encontrados.  0 <= M <= N.
!>          Si RANGE = 'A', M = N, y si RANGE = 'I', M = IU-IL+1.
!>

W

!>          W es un array REAL, dimensión (N)
!>          Los primeros M elementos contienen los valores propios seleccionados en
!>          orden ascendente.
!>

Z

!>          Z es un array REAL, dimensión (LDZ, max(1,M))
!>          Si JOBZ = 'V', entonces si INFO = 0, las primeras M columnas de Z
!>          contienen los vectores propios ortonormales de la matriz A
!>          correspondientes a los valores propios seleccionados, con la i-ésima
!>          columna de Z conteniendo el vector propio asociado a W(i).
!>          Si un vector propio falla en converger, entonces esa columna de Z
!>          contiene la última aproximación al vector propio, y el
!>          índice del vector propio se devuelve en IFAIL.
!>          Si JOBZ = 'N', entonces Z no se referencia.
!>          Nota: el usuario debe asegurarse de que se suministran al menos max(1,M) columnas
!>          en el array Z; si RANGE = 'V', el valor exacto de M
!>          no se conoce de antemano y se debe usar un límite superior.
!>

LDZ

!>          LDZ es INTEGER
!>          La dimensión principal del array Z.  LDZ >= 1, y si
!>          JOBZ = 'V', LDZ >= max(1,N).
!>

WORK

!>          WORK es un array REAL, dimensión (LWORK)
!>

LWORK

!>          LWORK es INTEGER
!>          La longitud del array WORK. LWORK >= 1, cuando N <= 1;
!>          en caso contrario
!>          Si JOBZ = 'N' y N > 1, LWORK debe ser consultado.
!>                                   LWORK = MAX(1, 7*N, dimension) donde
!>                                   dimension = (2KD+1)*N + KD*NTHREADS + 2*N
!>                                   donde KD es el tamaño de la banda.
!>

NAME

ssbevx_2stage - ssbevx_2stage

SYNOPSIS

    call ssbevx_2stage(JOBZ, RANGE, UPLO, N, KD, AB, LDAB, AP, LWORK, IWORK, BWORK,
                      INFO)

ssbevx_2stage requires:

    subroutine ssbevx_2stage(JOBZ, RANGE, UPLO, N, KD, AB, LDAB, AP, LWORK, IWORK,
                             BWORK, INFO)
    character JOBZ, RANGE, UPLO
    integer N, KD, LDAB, LWORK, INFO
    real AB(LDAB, *), AP(*)
    integer IWORK(*)
    logical BWORK(*)

Function/Subroutine Documentation

ssbevx_2stage computes the eigenvalues and eigenvectors of a real symmetric banded matrix.

Arguments:

JOBZ

!>          JOBZ = 'N':  Compute eigenvalues only;
!>          JOBZ = 'V':  Compute eigenvalues and eigenvectors;

RANGE

!>          RANGE = 'A': all eigenvalues will be found.
!>          RANGE = 'I': all eigenvalues in the interval (VL, VU] will be
!>                     found;
!>          RANGE = 'S': all eigenvalues in the interval [VL, VU] that are
!>                     closest to VL will be found;

UPLO

!>          UPLO = 'U': Upper triangle of AB and AB are used and AP stored.
!>          UPLO = 'L': Lower triangle of AB and AB are used and AP stored.

N

!>          N is INTEGER
!>          The order of the matrix AB.  N >= 0.

KD

!>          KD is INTEGER
!>          The number of super- or sub-diagonals of the matrix AB. KD >= 0.

AB

!>          AB is REAL array, dimension (LDAB,N)
!>          On entry, the K-th diagonal of the matrix A. The K-th diagonal is
!>          stored in the K-th column of AB if UPLO = 'U' and K < 0, and in
!>          the K-th row of AB if UPLO = 'L' and K < 0.
!>          If UPLO = 'U', ab(kd+1+i-j,j) and ab(i,j) for i <= kd+1.
!>          If UPLO = 'L', ab(i,j) for i >= kd+1, and ab(i,j) for i <= kd+1.

LDAB

!>          LDAB is INTEGER, dimension of the array AB. LDAB >= kd + 1.

AP

!>          AP is REAL array, dimension (N*(KD+1))
!>          On entry, AP stores the diagonal entries of the matrix A.
!>          Specifically, if UPLO = 'U', ap(i+(j-1)*(kd+1)) stores a(i,j)
!>          for 1<=i<=kd+1 and kd+1<j<=n.
!>          If UPLO = 'L', ap(i+(j-1)*(kd+1)) stores a(i,j) for kd+1<=i<=n
!>          and 1<=j<=kd+1.

LWORK

!>          LWORK is INTEGER
!>          The dimension of the array WORK.
!>          If LWORK = -1, then the optimal dimension is returned in WORK(1).
!>          LWORK must be at least 2*N.
!>
!>          NTHREADS is the number of threads used when
!>          openMP compilation is enabled, otherwise =1.
!>          If JOBZ = 'V' and N > 1, LWORK must be queried. Not yet available
!>
!>          If LWORK = -1, then a workspace query is assumed; the routine
!>          only calculates the optimal size of the WORK array, returns
!>          this value as the first entry of the WORK array, and no error
!>          message related to LWORK is issued by XERBLA.
!>

IWORK

!>          IWORK is INTEGER array, dimension (5*N)
!>

IFAIL

!>          IFAIL is INTEGER array, dimension (N)
!>          If JOBZ = 'V', then if INFO = 0, the first M elements of
!>          IFAIL are zero.  If INFO > 0, then IFAIL contains the
!>          indices of the eigenvectors that failed to converge.
!>          If JOBZ = 'N', then IFAIL is not referenced.
!>

INFO

!>          INFO is INTEGER
!>          = 0:  successful exit.
!>          < 0:  if INFO = -i, the i-th argument had an illegal value.
!>          > 0:  if INFO = i, then i eigenvectors failed to converge.
!>                Their indices are stored in array IFAIL.
!>
