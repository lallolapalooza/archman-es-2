subroutine ssyequb (character uplo, integer n, real, dimension( lda, * ) a, integer lda, real, dimension( * ) s, real scond, real amax, real, dimension( * ) work, integer info)

SSYEQUB

Propósito:

!>
!> SSYEQUB calcula los escalados de fila y columna destinados a equilibrar una
!> matriz simétrica A (con respecto a la norma euclidiana) y reducir
!> su número de condición. Los factores de escala S se calculan mediante el algoritmo
!> BIN (ver referencias) de modo que la matriz escalada B con elementos
!> B(i,j) = S(i)*A(i,j)*S(j) tenga un número de condición dentro de un factor N de
!> el menor número de condición posible sobre todos los escalados diagonales
!> posibles.
!>

Parámetros

UPLO 
!>          UPLO es CHARACTER*1
!>          = 'U':  El triángulo superior de A está almacenado;
!>          = 'L':  El triángulo inferior de A está almacenado.
!>

N

!>          N es INTEGER
!>          El orden de la matriz A. N >= 0.
!>

A

!>          A es un array REAL, de dimensión (LDA,N)
!>          La matriz simétrica N por N cuyos factores de escala se van a
!>          calcular.
!>

LDA

!>          LDA es INTEGER
!>          La dimensión principal del array A. LDA >= max(1,N).
!>

S

!>          S es un array REAL, de dimensión (N)
!>          Si INFO = 0, S contiene los factores de escala para A.
!>

SCOND

!>          SCOND es REAL
!>          Si INFO = 0, S contiene la relación entre el S(i) más pequeño y
!>          el S(i) más grande. Si SCOND >= 0.1 y AMAX no es ni demasiado
!>          grande ni demasiado pequeño, no vale la pena escalar con S.
!>

AMAX

!>          AMAX es REAL
!>          El valor absoluto más grande de cualquier elemento de la matriz. Si AMAX está
!>          muy cerca del desbordamiento (overflow) o muy cerca del subdesbordamiento (underflow),
!>          la matriz debe ser escalada.
!>

WORK

!>          WORK es un array REAL, de dimensión (2*N)
!>

INFO

!>          INFO es INTEGER
!>          = 0:  salida exitosa
!>          < 0:  si INFO = -i, el i-ésimo argumento tenía un valor ilegal
!>          > 0:  si INFO = i, el i-ésimo elemento diagonal no es positivo.
!>
