subroutine sspcon (character uplo, integer n, real, dimension( * ) ap, integer, dimension( * ) ipiv, real anorm, real rcond, real, dimension( * ) work, integer, dimension( * ) iwork, integer info)

SSPCON

Propósito:

!>
!> SSPCON estima la inversa del número de condición (en la norma 1)
!> de una matriz simétrica real empaquetada A utilizando la factorización
!> A = U*D*U**T o A = L*D*L**T calculada por SSPTRF.
!>
!> Se obtiene una estimación de norm(inv(A)), y la inversa del número de
!> condición se calcula como RCOND = 1 / (ANORM * norm(inv(A))).
!>

Parámetros

UPLO 
!>          UPLO es CHARACTER*1
!>          Especifica si los detalles de la factorización se almacenan
!>          como una matriz triangular superior o inferior.
!>          = 'U':  Triangular superior, la forma es A = U*D*U**T;
!>          = 'L':  Triangular inferior, la forma es A = L*D*L**T.
!>

N

!>          N es INTEGER
!>          El orden de la matriz A.  N >= 0.
!>

AP

!>          AP es array REAL, dimension (N*(N+1)/2)
!>          La matriz diagonal de bloques D y los multiplicadores utilizados para
!>          obtener el factor U o L como lo calcula SSPTRF, almacenados como
!>          una matriz triangular empaquetada.
!>

IPIV

!>          IPIV es array INTEGER, dimension (N)
!>          Detalles de los intercambios y la estructura de bloques de D
!>          determinada por SSPTRF.
!>

ANORM

!>          ANORM es REAL
!>          La norma 1 de la matriz original A.
!>

RCOND

!>          RCOND es REAL
!>          La inversa del número de condición de la matriz A,
!>          calculada como RCOND = 1/(ANORM * AINVNM), donde AINVNM es una
!>          estimación de la norma 1 de inv(A) calculada en esta rutina.
!>

WORK

!>          WORK es array REAL, dimension (2*N)
!>

IWORK

!>          IWORK es array INTEGER, dimension (N)
!>

INFO

!>          INFO es INTEGER
!>          = 0:  salida exitosa
!>          < 0:  si INFO = -i, el i-ésimo argumento tenía un valor ilegal
!>
