subroutine ssyr2 (character uplo, integer n, real alpha, real, dimension(*) x, integer incx, real, dimension(*) y, integer incy, real, dimension(lda,*) a, integer lda)

SSYR2

Propósito:

!>
!> SSYR2  realiza la operación simétrica de rango 2
!>
!>    A := alpha*x*y**T + alpha*y*x**T + A,
!>
!> donde alpha es un escalar, x e y son vectores de n elementos y A es una matriz
!> simétrica de n por n.
!>

Parámetros

UPLO 
!>          UPLO es CHARACTER*1
!>           A la entrada, UPLO especifica si la parte triangular superior o
!>           inferior de la matriz A se va a referenciar como
!>           sigue:
!>
!>              UPLO = 'U' o 'u'   Solo se hará referencia a la parte triangular
!>                                  superior de A.
!>
!>              UPLO = 'L' o 'l'   Solo se hará referencia a la parte triangular
!>                                  inferior de A.
!>

N

!>          N es INTEGER
!>           A la entrada, N especifica el orden de la matriz A.
!>           N debe ser mayor o igual a cero.
!>

ALPHA

!>          ALPHA es REAL
!>           A la entrada, ALPHA especifica el escalar alpha.
!>

X

!>          X es una matriz REAL, de dimensión al menos
!>           ( 1 + ( n - 1 )*abs( INCX ) ).
!>           Antes de la entrada, la matriz X incrementada debe contener el
!>           vector x de n elementos.
!>

INCX

!>          INCX es INTEGER
!>           A la entrada, INCX especifica el incremento para los elementos de
!>           X. INCX no debe ser cero.
!>

Y

!>          Y es una matriz REAL, de dimensión al menos
!>           ( 1 + ( n - 1 )*abs( INCY ) ).
!>           Antes de la entrada, la matriz Y incrementada debe contener el
!>           vector y de n elementos.
!>

INCY

!>          INCY es INTEGER
!>           A la entrada, INCY especifica el incremento para los elementos de
!>           Y. INCY no debe ser cero.
!>

A

!>          A es una matriz REAL, de dimensión ( LDA, N )
!>           Antes de la entrada con UPLO = 'U' o 'u', la parte superior
!>           del triángulo principal de n por n de la matriz A debe contener
!>           la parte triangular superior de la matriz simétrica y la parte
!>           estrictamente inferior del triángulo de A no se referencia. A la
!>           salida, la parte triangular superior de la matriz A se sobrescribe
!>           con la parte triangular superior de la matriz actualizada.
!>           Antes de la entrada con UPLO = 'L' o 'l', la parte inferior
!>           del triángulo principal de n por n de la matriz A debe contener
!>           la parte triangular inferior de la matriz simétrica y la parte
!>           estrictamente superior del triángulo de A no se referencia. A la
!>           salida, la parte triangular inferior de la matriz A se sobrescribe
!>           con la parte triangular inferior de la matriz actualizada.
!>

LDA

!>          LDA es INTEGER
!>           A la entrada, LDA especifica la primera dimensión de A tal como
!>           se declara en el programa (sub)llamador. LDA debe ser al menos
!>           max( 1, n ).
!>
