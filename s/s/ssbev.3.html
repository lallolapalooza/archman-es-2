subroutine ssbev (character jobz, character uplo, integer n, integer kd, real, dimension( ldab, * ) ab, integer ldab, real, dimension( * ) w, real, dimension( ldz, * ) z, integer ldz, real, dimension( * ) work, integer info)

SSBEV calcula los valores propios y, opcionalmente, los vectores propios izquierdos y/o derechos para matrices SIMÉTRICAS DE BANDA REALES

Propósito:

!>
!> SSBEV calcula todos los valores propios y, opcionalmente, los vectores propios
!> de una matriz de banda simétrica real A.
!>

Parámetros

JOBZ 
!>          JOBZ es CHARACTER*1
!>          = 'N':  Calcula solo los valores propios;
!>          = 'V':  Calcula los valores propios y los vectores propios.
!>

UPLO

!>          UPLO es CHARACTER*1
!>          = 'U':  Se almacena el triángulo superior de A;
!>          = 'L':  Se almacena el triángulo inferior de A.
!>

N

!>          N es INTEGER
!>          El orden de la matriz A.  N >= 0.
!>

KD

!>          KD es INTEGER
!>          El número de superdiagonales de la matriz A si UPLO = 'U',
!>          o el número de subdiagonales si UPLO = 'L'.  KD >= 0.
!>

AB

!>          AB es un array REAL, de dimensión (LDAB, N)
!>          En la entrada, el triángulo superior o inferior de la matriz de banda
!>          simétrica A, almacenado en las primeras KD+1 filas del array. La
!>          columna j de A se almacena en la columna j del array AB
!>          de la siguiente manera:
!>          si UPLO = 'U', AB(kd+1+i-j,j) = A(i,j) para max(1,j-kd)<=i<=j;
!>          si UPLO = 'L', AB(1+i-j,j)    = A(i,j) para j<=i<=min(n,j+kd).
!>
!>          En la salida, AB se sobrescribe con los valores generados durante la
!>          reducción a forma tridiagonal. Si UPLO = 'U', la primera
!>          superdiagonal y la diagonal de la matriz tridiagonal T
!>          se devuelven en las filas KD y KD+1 de AB, y si UPLO = 'L',
!>          la diagonal y la primera subdiagonal de T se devuelven en las
!>          primeras dos filas de AB.
!>

LDAB

!>          LDAB es INTEGER
!>          La dimensión principal del array AB.  LDAB >= KD + 1.
!>

W

!>          W es un array REAL, de dimensión (N)
!>          Si INFO = 0, los valores propios en orden ascendente.
!>

Z

!>          Z es un array REAL, de dimensión (LDZ, N)
!>          Si JOBZ = 'V', entonces si INFO = 0, Z contiene los vectores propios
!>          ortonormales de la matriz A, con la columna i de Z
!>          conteniendo el vector propio asociado a W(i).
!>          Si JOBZ = 'N', entonces Z no se referencia.
!>

LDZ

!>          LDZ es INTEGER
!>          La dimensión principal del array Z.  LDZ >= 1, y si
!>          JOBZ = 'V', LDZ >= max(1,N).
!>

WORK

!>          WORK es un array REAL, de dimensión (max(1,3*N-2))
!>

INFO

!>          INFO es INTEGER
!>          = 0:  salida exitosa
!>          < 0:  si INFO = -i, el i-ésimo argumento tenía un valor ilegal
!>          > 0:  si INFO = i, el algoritmo falló en converger; i
!>                elementos fuera de la diagonal de una forma tridiagonal
!>                intermedia no convergieron a cero.
!>
