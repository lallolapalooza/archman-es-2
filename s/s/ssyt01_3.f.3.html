subroutine ssyt01_3 (character uplo, integer n, real, dimension( lda, * ) a, integer lda, real, dimension( ldafac, * ) afac, integer ldafac, real, dimension( * ) e, integer, dimension( * ) ipiv, real, dimension( ldc, * ) c, integer ldc, real, dimension( * ) rwork, real resid)

SSYT01_3

Propósito:

!>
!> SSYT01_3 reconstruye una matriz simétrica indefinida A a partir de su
!> factorización en bloques L*D*L' o U*D*U' calculada por SSYTRF_RK
!> (o SSYTRF_BK) y calcula el residuo
!>    norm( C - A ) / ( N * norm(A) * EPS ),
!> donde C es la matriz reconstruida y EPS es la epsilon de la máquina.
!>

Parámetros

UPLO 
!>          UPLO es CHARACTER*1
!>          Especifica si la parte triangular superior o inferior de la
!>          matriz simétrica A está almacenada:
!>          = 'U':  Triangular superior
!>          = 'L':  Triangular inferior
!>

N

!>          N es INTEGER
!>          El número de filas y columnas de la matriz A.  N >= 0.
!>

A

!>          A es una matriz DOUBLE PRECISION, dimensionada (LDA,N)
!>          La matriz simétrica original A.
!>

LDA

!>          LDA es INTEGER
!>          La dimensión principal del array A.  LDA >= max(1,N)
!>

AFAC

!>          AFAC es una matriz DOUBLE PRECISION, dimensionada (LDAFAC,N)
!>          Diagonal de la matriz de bloques diagonales D y factores U o L
!>          calculados por SSYTRF_RK y SSYTRF_BK:
!>            a) SÓLO los elementos diagonales de la matriz de bloques
!>               diagonales simétrica D en la diagonal de A, es decir,
!>               D(k,k) = A(k,k);
!>               (los elementos de la superdiagonal (o subdiagonal) de D
!>                deben proporcionarse en la entrada en el array E), y
!>            b) Si UPLO = 'U': factor U en la parte de superdiagonal de A.
!>               Si UPLO = 'L': factor L en la parte de subdiagonal de A.
!>

LDAFAC

!>          LDAFAC es INTEGER
!>          La dimensión principal del array AFAC.
!>          LDAFAC >= max(1,N).
!>

E

!>          E es una matriz DOUBLE PRECISION, dimensionada (N)
!>          En la entrada, contiene los elementos de la superdiagonal (o subdiagonal)
!>          de la matriz de bloques diagonales simétrica D
!>          con bloques diagonales de 1x1 o 2x2, donde
!>          Si UPLO = 'U': E(i) = D(i-1,i),i=2:N, E(1) no referenciado;
!>          Si UPLO = 'L': E(i) = D(i+1,i),i=1:N-1, E(N) no referenciado.
!>

IPIV

!>          IPIV es un array INTEGER, dimensionado (N)
!>          Los índices de pivote de SSYTRF_RK (o SSYTRF_BK).
!>

C

!>          C es una matriz DOUBLE PRECISION, dimensionada (LDC,N)
!>

LDC

!>          LDC es INTEGER
!>          La dimensión principal del array C.  LDC >= max(1,N).
!>

RWORK

!>          RWORK es una matriz DOUBLE PRECISION, dimensionada (N)
!>

RESID

!>          RESID es DOUBLE PRECISION
!>          Si UPLO = 'L', norm(L*D*L' - A) / ( N * norm(A) * EPS )
!>          Si UPLO = 'U', norm(U*D*U' - A) / ( N * norm(A) * EPS )
!>
