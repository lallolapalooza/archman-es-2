subroutine ssyr (character uplo, integer n, real alpha, real, dimension(*) x, integer incx, real, dimension(lda,*) a, integer lda)

SSYR

Propósito:

!>
!> SSYR   realiza la operación simétrica de rango 1
!>
!>    A := alpha*x*x**T + A,
!>
!> donde alpha es un escalar real, x es un vector de n elementos y A es una
!> matriz simétrica de n por n.
!>

Parámetros

UPLO 
!>          UPLO es CHARACTER*1
!>           A la entrada, UPLO especifica si la parte triangular superior o inferior
!>           del array A debe ser referenciada como
!>           sigue:
!>
!>              UPLO = 'U' o 'u'   Solo la parte triangular superior de A
!>                                  será referenciada.
!>
!>              UPLO = 'L' o 'l'   Solo la parte triangular inferior de A
!>                                  será referenciada.
!>

N

!>          N es INTEGER
!>           A la entrada, N especifica el orden de la matriz A.
!>           N debe ser al menos cero.
!>

ALPHA

!>          ALPHA es REAL
!>           A la entrada, ALPHA especifica el escalar alpha.
!>

X

!>          X es un array REAL, de dimensión al menos
!>           ( 1 + ( n - 1 )*abs( INCX ) ).
!>           Antes de la entrada, el array X incrementado debe contener el vector
!>           x de n elementos.
!>

INCX

!>          INCX es INTEGER
!>           A la entrada, INCX especifica el incremento para los elementos de
!>           X. INCX no debe ser cero.
!>

A

!>          A es un array REAL, de dimensión ( LDA, N )
!>           Antes de la entrada con UPLO = 'U' o 'u', la parte superior principal
!>           de n por n del array A debe contener la parte triangular superior
!>           de la matriz simétrica y la parte estrictamente triangular inferior de A
!>           no se referencia. A la salida, la parte triangular superior del array A
!>           se sobrescribe con la parte triangular superior de la matriz actualizada.
!>           Antes de la entrada con UPLO = 'L' o 'l', la parte inferior principal
!>           de n por n del array A debe contener la parte triangular inferior
!>           de la matriz simétrica y la parte estrictamente triangular superior de A
!>           no se referencia. A la salida, la parte triangular inferior del array A
!>           se sobrescribe con la parte triangular inferior de la matriz actualizada.
!>

LDA

!>          LDA es INTEGER
!>           A la entrada, LDA especifica la primera dimensión de A tal como se declara
!>           en el programa (sub)llamador. LDA debe ser al menos
!>           max( 1, n ).
!>
