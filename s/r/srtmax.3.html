NAME

la_constants.f90 - LAPACK constants

SYNOPSIS

! File: la_constants.f90

! Module for LAPACK constants.
module la_constants

  implicit none

  private

  ! Exported types
  public    :: dp, sp

  ! Exported constants
  public    :: dbig, dhalf, dnegten, dnegone, dnone, &
                dsafmax, dsafmin, dsbig, dscale, dsmlnum, &
                dssml, dtbig, dten, dthree, dtsml, &
                dtwo, dulp, dzero, sbignum, seight, &
                seps, sfour, shalf, sone, sprefix, &
                srtmax, srtmin, ssafmax, ssafmin, &
                ssbig, ssmlnum, sssml, stbig, &
                sten, sthree, stsml, stwo, sulp, &
                szero, zhalf, zone, zprefix, zzero

  ! Precision
  integer, parameter :: dp = kind(1.d0)
  integer, parameter :: sp = kind(1.e0)

  ! Constants for single precision
  real(sp), parameter :: szero = 0.0_sp
  real(sp), parameter :: shalf = 0.5_sp
  real(sp), parameter :: sone = 1.0_sp
  real(sp), parameter :: stwo = 2.0_sp
  real(sp), parameter :: sthree = 3.0_sp
  real(sp), parameter :: sfour = 4.0_sp
  real(sp), parameter :: seight = 8.0_sp
  real(sp), parameter :: sten = 10.0_sp
  character*1, parameter :: sprefix = 'S'
  real(sp), parameter :: sulp = epsilon(0.0_sp)
  real(sp), parameter :: seps = sulp * 0.5_sp
  real(sp), parameter :: ssafmin = real(radix(0.0_sp), sp)**max( minexponent(0.0_sp)-1, 1-maxexponent(0.0_sp) )
  real(sp), parameter :: ssmlnum = ssafmin / sulp
  real(sp), parameter :: ssafmax = sone / ssafmin
  real(sp), parameter :: srtmin = sqrt(ssmlnum)
  real(sp), parameter :: srtmax = sqrt(sbignum)
  real(sp), parameter :: sbignum = ssafmax * sulp
  real(sp), parameter :: stsml = real(radix(0.0_sp), sp)**ceiling( (minexponent(0.0_sp) - 1) * 0.5_sp )
  real(sp), parameter :: sssml = real(radix(0.0_sp), sp)**( - floor( (minexponent(0.0_sp) - digits(0.0_sp)) * 0.5_sp ) )
  real(sp), parameter :: stbig = real(radix(0.0_sp), sp)**floor( (maxexponent(0.0_sp) - digits(0.0_sp) + 1) * 0.5_sp )
  real(sp), parameter :: ssbig = real(radix(0.0_sp), sp)**( - ceiling( (maxexponent(0.0_sp) + digits(0.0_sp) - 1) * 0.5_sp ) )

  ! Constants for double precision
  real(dp), parameter :: dzero = 0.0_dp
  real(dp), parameter :: dhalf = 0.5_dp
  real(dp), parameter :: dnegone = -1.0_dp
  real(dp), parameter :: dnone = 1.0_dp
  real(dp), parameter :: dtwo = 2.0_dp
  real(dp), parameter :: dthree = 3.0_dp
  real(dp), parameter :: dten = 10.0_dp
  real(dp), parameter :: dnegten = -10.0_dp
  real(dp), parameter :: dulp = epsilon(0.0_dp)
  real(dp), parameter :: dsafmin = real(radix(0.0_dp), dp)**max( minexponent(0.0_dp)-1, 1-maxexponent(0.0_dp) )
  real(dp), parameter :: dsmlnum = dsafmin / dulp
  real(dp), parameter :: dsafmax = dnone / dsafmin
  real(dp), parameter :: dtsml = real(radix(0.0_dp), dp)**ceiling( (minexponent(0.0_dp) - 1) * 0.5_dp )
  real(dp), parameter :: dssml = real(radix(0.0_dp), dp)**( - floor( (minexponent(0.0_dp) - digits(0.0_dp)) * 0.5_dp ) )
  real(dp), parameter :: dtbig = real(radix(0.0_dp), dp)**floor( (maxexponent(0.0_dp) - digits(0.0_dp) + 1) * 0.5_dp )
  real(dp), parameter :: dbig = real(radix(0.0_dp), dp)**( - ceiling( (maxexponent(0.0_dp) + digits(0.0_dp) - 1) * 0.5_dp ) )
  real(dp), parameter :: dsbig = dsafmax * dulp
  real(dp), parameter :: dscale = dtsml / dsbig

  ! Constants for complex double precision
  complex(dp), parameter :: zzero = ( 0.0_dp, 0.0_dp )
  complex(dp), parameter :: zhalf = ( 0.5_dp, 0.0_dp )
  complex(dp), parameter :: zone = ( 1.0_dp, 0.0_dp )
  character*1, parameter :: zprefix = 'Z'

end module la_constants
