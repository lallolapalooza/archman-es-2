subroutine srqt01 (integer m, integer n, real, dimension( lda, * ) a, real, dimension( lda, * ) af, real, dimension( lda, * ) q, real, dimension( lda, * ) r, integer lda, real, dimension( * ) tau, real, dimension( lwork ) work, integer lwork, real, dimension( * ) rwork, real, dimension( * ) result)

SRQT01

Propósito:

!>
!> SRQT01 prueba SGERQF, que calcula la factorización RQ de una matriz m-por-n
!> A, y prueba parcialmente SORGRQ que forma la matriz ortogonal n-por-n
!> Q.
!>
!> SRQT01 compara R con A*Q', y comprueba que Q es ortogonal.
!>

Parámetros

M 
!>          M es INTEGER
!>          El número de filas de la matriz A.  M >= 0.
!>

N

!>          N es INTEGER
!>          El número de columnas de la matriz A.  N >= 0.
!>

A

!>          A es array REAL, dimension (LDA,N)
!>          La matriz m-por-n A.
!>

AF

!>          AF es array REAL, dimension (LDA,N)
!>          Detalles de la factorización RQ de A, devueltos por SGERQF.
!>          Ver SGERQF para más detalles.
!>

Q

!>          Q es array REAL, dimension (LDA,N)
!>          La matriz ortogonal n-por-n Q.
!>

R

!>          R es array REAL, dimension (LDA,max(M,N))
!>

LDA

!>          LDA es INTEGER
!>          La dimensión principal de los arrays A, AF, Q y L.
!>          LDA >= max(M,N).
!>

TAU

!>          TAU es array REAL, dimension (min(M,N))
!>          Los factores escalares de los reflectores elementales, devueltos
!>          por SGERQF.
!>

WORK

!>          WORK es array REAL, dimension (LWORK)
!>

LWORK

!>          LWORK es INTEGER
!>          La dimensión del array WORK.
!>

RWORK

!>          RWORK es array REAL, dimension (max(M,N))
!>

RESULT

!>          RESULT es array REAL, dimension (2)
!>          Las relaciones de prueba:
!>          RESULT(1) = norm( R - A*Q' ) / ( N * norm(A) * EPS )
!>          RESULT(2) = norm( I - Q*Q' ) / ( N * EPS )
!>
