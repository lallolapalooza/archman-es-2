subroutine srotm (integer n, real, dimension(*) sx, integer incx, real, dimension(*) sy, integer incy, real, dimension(5) sparam)

SROTM

Propósito:

!>
!>    APLICA LA TRANSFORMACIÓN DE GIVENS MODIFICADA, H, A LA MATRIZ DE 2 POR N
!>
!>    (SX**T) , DONDE **T INDICA TRANSPUESTA. LOS ELEMENTOS DE SX ESTÁN EN
!>    (SX**T)
!>
!>    SX(LX+I*INCX), I = 0 A N-1, DONDE LX = 1 SI INCX .GE. 0, DE LO CONTRARIO
!>    LX = (-INCX)*N, Y DE FORMA SIMILAR PARA SY USANDO LY E INCY.
!>    CON SPARAM(1)=SFLAG, H TIENE UNA DE LAS SIGUIENTES FORMAS..
!>
!>    SFLAG=-1.E0     SFLAG=0.E0        SFLAG=1.E0     SFLAG=-2.E0
!>
!>      (SH11  SH12)    (1.E0  SH12)    (SH11  1.E0)    (1.E0  0.E0)
!>    H=(          )    (          )    (          )    (          )
!>      (SH21  SH22),   (SH21  1.E0),   (-1.E0 SH22),   (0.E0  1.E0).
!>    CONSULTA SROTMG PARA UNA DESCRIPCIÓN DEL ALMACENAMIENTO DE DATOS EN SPARAM.
!>
!>

Parámetros

N 
!>          N es INTEGER
!>         número de elementos en el/los vector(es) de entrada
!>

SX

!>          SX es array REAL, dimensión ( 1 + ( N - 1 )*abs( INCX ) )
!>

INCX

!>          INCX es INTEGER
!>         espaciado de almacenamiento entre elementos de SX
!>

SY

!>          SY es array REAL, dimensión ( 1 + ( N - 1 )*abs( INCY ) )
!>

INCY

!>          INCY es INTEGER
!>         espaciado de almacenamiento entre elementos de SY
!>

SPARAM

!>          SPARAM es array REAL, dimensión (5)
!>     SPARAM(1)=SFLAG
!>     SPARAM(2)=SH11
!>     SPARAM(3)=SH21
!>     SPARAM(4)=SH12
!>     SPARAM(5)=SH22
!>
