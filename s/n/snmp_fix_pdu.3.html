NOMBRE

snmp_pdu_create, snmp_clone_pdu, snmp_fix_pdu, snmp_free_pdu - funciones de la API netsnmp_pdu

SINOPSIS

#include <net-snmp/pdu_api.h>

netsnmp_pdu *snmp_pdu_create( inttipo);

netsnmp_pdu *snmp_clone_pdu( netsnmp_pdu*pdu);

netsnmp_pdu *snmp_free_pdu( netsnmp_pdu*pdu);
