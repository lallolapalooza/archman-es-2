subroutine sbdt03 (character uplo, integer n, integer kd, real, dimension( * ) d, real, dimension( * ) e, real, dimension( ldu, * ) u, integer ldu, real, dimension( * ) s, real, dimension( ldvt, * ) vt, integer ldvt, real, dimension( * ) work, real resid)

SBDT03

Propósito:

!>
!> SBDT03 reconstruye una matriz bidimensional B a partir de su SVD:
!>    S = U' * B * V
!> donde U y V son matrices ortogonales y S es diagonal.
!>
!> La razón de prueba para probar la descomposición en valores singulares es
!>    RESID = norm( B - U * S * VT ) / ( n * norm(B) * EPS )
!> donde VT = V' y EPS es la precisión de la máquina.
!>

Parámetros

UPLO 
!>          UPLO es CHARACTER*1
!>          Especifica si la matriz B es superior o inferior bidimensional.
!>          = 'U':  Superior bidimensional
!>          = 'L':  Inferior bidimensional
!>

N

!>          N es INTEGER
!>          El orden de la matriz B.
!>

KD

!>          KD es INTEGER
!>          El ancho de banda de la matriz bidimensional B. Si KD = 1, la
!>          matriz B es bidimensional, y si KD = 0, B es diagonal y E no se
!>          referencia. Si KD es mayor que 1, se asume que es
!>          1, y si KD es menor que 0, se asume que es 0.
!>

D

!>          D es REAL array, dimension (N)
!>          Los n elementos diagonales de la matriz bidimensional B.
!>

E

!>          E es REAL array, dimension (N-1)
!>          Los n-1 elementos superdiagonales de la matriz bidimensional B
!>          si UPLO = 'U', o los n-1 elementos subdiagonales de B si
!>          UPLO = 'L'.
!>

U

!>          U es REAL array, dimension (LDU,N)
!>          La matriz ortogonal n por n U en la reducción B = U'*A*P.
!>

LDU

!>          LDU es INTEGER
!>          La dimensión principal del array U. LDU >= max(1,N)
!>

S

!>          S es REAL array, dimension (N)
!>          Los valores singulares de la SVD de B, ordenados en orden
!>          decreciente.
!>

VT

!>          VT es REAL array, dimension (LDVT,N)
!>          La matriz ortogonal n por n V' en la reducción
!>          B = U * S * V'.
!>

LDVT

!>          LDVT es INTEGER
!>          La dimensión principal del array VT.
!>

WORK

!>          WORK es REAL array, dimension (2*N)
!>

RESID

!>          RESID es REAL
!>          La razón de prueba: norm(B - U * S * V') / ( n * norm(A) * EPS )
!>
