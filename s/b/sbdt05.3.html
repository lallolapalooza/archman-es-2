Documentación de Función/Subrutina

subroutine sbdt05 (integer m, integer n, real, dimension( lda, * ) a, integer lda, real, dimension( * ) s, integer ns, real, dimension( ldu, * ) u, integer ldu, real, dimension( ldvt, * ) vt, integer ldvt, real, dimension( * ) work, real resid)

SBDT05

Propósito:

!>
!> SBDT05 reconstruye una matriz bidiagonal B a partir de su SVD (parcial):
!>    S = U' * B * V
!> donde U y V son matrices ortogonales y S es diagonal.
!>
!> La razón de prueba para probar la descomposición de valores singulares es
!>    RESID = norm( S - U' * B * V ) / ( n * norm(B) * EPS )
!> donde VT = V' y EPS es la precisión de la máquina.
!>

Parámetros

M 
!>          M es INTEGER
!>          El número de filas de las matrices A y U.
!>

N

!>          N es INTEGER
!>          El número de columnas de las matrices A y VT.
!>

A

!>          A es array REAL, dimension (LDA,N)
!>          La matriz A de m por n.
!>

LDA

!>          LDA es INTEGER
!>          La dimensión principal del array A. LDA >= max(1,M).
!>

S

!>          S es array REAL, dimension (NS)
!>          Los valores singulares de la SVD (parcial) de B, ordenados en
!>          orden decreciente.
!>

NS

!>          NS es INTEGER
!>          El número de valores/vectores singulares de la SVD (parcial)
!>          de B.
!>

U

!>          U es array REAL, dimension (LDU,NS)
!>          La matriz ortogonal U de n por ns en S = U' * B * V.
!>

LDU

!>          LDU es INTEGER
!>          La dimensión principal del array U. LDU >= max(1,N)
!>

VT

!>          VT es array REAL, dimension (LDVT,N)
!>          La matriz ortogonal VT de n por ns en S = U' * B * V.
!>

LDVT

!>          LDVT es INTEGER
!>          La dimensión principal del array VT.
!>

WORK

!>          WORK es array REAL, dimension (M,N)
!>

RESID

!>          RESID es REAL
!>          La razón de prueba: norm(S - U' * A * V) / ( n * norm(A) * EPS )
!>
