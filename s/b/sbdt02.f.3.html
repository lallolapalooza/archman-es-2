subroutine sbdt02 (integer m, integer n, real, dimension( ldb, * ) b, integer ldb, real, dimension( ldc, * ) c, integer ldc, real, dimension( ldu, * ) u, integer ldu, real, dimension( * ) work, real resid)

SBDT02

Propósito:

!>
!> SBDT02 prueba el cambio de base C = U**H * B calculando el
!> residuo
!>
!>    RESID = norm(B - U * C) / ( max(m,n) * norm(B) * EPS ),
!>
!> donde B y C son matrices M por N, U es una matriz ortogonal M por M,
!> y EPS es la precisión de la máquina.
!>

Parámetros

M 
!>          M es ENTERO
!>          El número de filas de las matrices B y C y el orden de
!>          la matriz Q.
!>

N

!>          N es ENTERO
!>          El número de columnas de las matrices B y C.
!>

B

!>          B es un array REAL, dimensión (LDB,N)
!>          La matriz B de m por n.
!>

LDB

!>          LDB es ENTERO
!>          La dimensión principal del array B.  LDB >= max(1,M).
!>

C

!>          C es un array REAL, dimensión (LDC,N)
!>          La matriz C de m por n, que se asume contiene U**H * B.
!>

LDC

!>          LDC es ENTERO
!>          La dimensión principal del array C.  LDC >= max(1,M).
!>

U

!>          U es un array REAL, dimensión (LDU,M)
!>          La matriz ortogonal U de m por m.
!>

LDU

!>          LDU es ENTERO
!>          La dimensión principal del array U.  LDU >= max(1,M).
!>

WORK

!>          WORK es un array REAL, dimensión (M)
!>

RESID

!>          RESID es REAL
!>          RESID = norm(B - U * C) / ( max(m,n) * norm(B) * EPS ),
!>
