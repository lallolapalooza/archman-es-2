subroutine sgebak (character job, character side, integer n, integer ilo, integer ihi, real, dimension( * ) scale, integer m, real, dimension( ldv, * ) v, integer ldv, integer info)

SGEBAK

Propósito:

!>
!> SGEBAK forma los   vectores propios (eigenvectors) derechos o izquierdos de una matriz   general real por transformación hacia atrás sobre los vectores propios calculados de
!> la matriz balanceada producida por SGEBAL.
!>

Parámetros

JOB 
!>          JOB es CHARACTER*1
!>          Especifica el tipo de transformación hacia atrás requerida:
!>          = 'N': no hacer nada, regresar inmediatamente;
!>          = 'P': hacer transformación hacia atrás solo para permutación;
!>          = 'S': hacer transformación hacia atrás solo para escalado;
!>          = 'B': hacer transformaciones hacia atrás tanto para permutación como
!>                 para escalado.
!>          JOB debe ser el mismo que el argumento JOB suministrado a SGEBAL.
!>

SIDE

!>          SIDE es CHARACTER*1
!>          = 'R':  V contiene vectores propios derechos;
!>          = 'L':  V contiene vectores propios izquierdos.
!>

N

!>          N es INTEGER
!>          El número de filas de la matriz V.  N >= 0.
!>

ILO

!>          ILO es INTEGER
!>

IHI

!>          IHI es INTEGER
!>          Los enteros ILO e IHI determinados por SGEBAL.
!>          1 <= ILO <= IHI <= N, si N > 0; ILO=1 y IHI=0, si N=0.
!>

SCALE

!>          SCALE es un array REAL, dimensión (N)
!>          Detalles de la permutación y los factores de escalado, tal como fueron
!>          devueltos por SGEBAL.
!>

M

!>          M es INTEGER
!>          El número de columnas de la matriz V.  M >= 0.
!>

V

!>          V es un array REAL, dimensión (LDV,M)
!>          En la entrada, la matriz de vectores propios derechos o izquierdos a ser
!>          transformada, tal como fue devuelta por SHSEIN o STREVC.
!>          En la salida, V es sobrescrita por los vectores propios transformados.
!>

LDV

!>          LDV es INTEGER
!>          La dimensión principal del array V. LDV >= max(1,N).
!>

INFO

!>          INFO es INTEGER
!>          = 0:  salida exitosa
!>          < 0:  si INFO = -i, el i-ésimo argumento tenía un valor ilegal.
!>
