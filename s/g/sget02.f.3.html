subroutine sget02 (character trans, integer m, integer n, integer nrhs, real, dimension( lda, * ) a, integer lda, real, dimension( ldx, * ) x, integer ldx, real, dimension( ldb, * ) b, integer ldb, real, dimension( * ) rwork, real resid)

SGET02

Propósito:

!>
!> SGET02 calcula el residuo para una solución de un sistema de ecuaciones
!> lineales op(A)*X = B:
!>    RESID = norm(B - op(A)*X) / ( norm(op(A)) * norm(X) * EPS ),
!> donde op(A) = A o A**T, dependiendo de TRANS, y EPS es la
!> épsilon de la máquina.
!> La norma utilizada es la 1-norma.
!>

Parámetros

TRANS 
!>          TRANS es CHARACTER*1
!>          Especifica la forma del sistema de ecuaciones:
!>          = 'N':  A    * X = B  (Sin transposición)
!>          = 'T':  A**T * X = B  (Transpuesta)
!>          = 'C':  A**H * X = B  (Transpuesta conjugada = Transpuesta)
!>

M

!>          M es INTEGER
!>          El número de filas de la matriz A.  M >= 0.
!>

N

!>          N es INTEGER
!>          El número de columnas de la matriz A.  N >= 0.
!>

NRHS

!>          NRHS es INTEGER
!>          El número de columnas de B, la matriz de los lados derechos.
!>          NRHS >= 0.
!>

A

!>          A es un array REAL, de dimensión (LDA,N)
!>          La matriz original M x N, A.
!>

LDA

!>          LDA es INTEGER
!>          La dimensión principal del array A.  LDA >= max(1,M).
!>

X

!>          X es un array REAL, de dimensión (LDX,NRHS)
!>          A la entrada, los vectores de solución computados para el sistema de
!>          ecuaciones lineales.
!>

LDX

!>          LDX es INTEGER
!>          La dimensión principal del array X.  Si TRANS = 'N',
!>          LDX >= max(1,N); si TRANS = 'T' o 'C', LDX >= max(1,M).
!>

B

!>          B es un array REAL, de dimensión (LDB,NRHS)
!>          A la entrada, los vectores del lado derecho para el sistema de
!>          ecuaciones lineales.
!>          A la salida, B se sobrescribe con la diferencia B - op(A)*X.
!>

LDB

!>          LDB es INTEGER
!>          La dimensión principal del array B.  Si TRANS = 'N',
!>          LDB >= max(1,M); si TRANS = 'T' o 'C', LDB >= max(1,N).
!>

RWORK

!>          RWORK es un array REAL, de dimensión (M)
!>

RESID

!>          RESID es REAL
!>          El máximo sobre el número de lados derechos de
!>          norm(B - op(A)*X) / ( norm(op(A)) * norm(X) * EPS ).
!>
