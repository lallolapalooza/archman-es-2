Documentación de Función/Subrutina

subroutine sgeqrs (integer m, integer n, integer nrhs, real, dimension( lda, * ) a, integer lda, real, dimension( * ) tau, real, dimension( ldb, * ) b, integer ldb, real, dimension( lwork ) work, integer lwork, integer info)

SGEQRS

Propósito:

!>
!> Resuelve el problema de mínimos cuadrados
!>     min || A*X - B ||
!> usando la factorización QR
!>     A = Q*R
!> calculada por SGEQRF.
!>

Parámetros

M 
!>          M es INTEGER
!>          El número de filas de la matriz A.  M >= 0.
!>

N

!>          N es INTEGER
!>          El número de columnas de la matriz A.  M >= N >= 0.
!>

NRHS

!>          NRHS es INTEGER
!>          El número de columnas de B.  NRHS >= 0.
!>

A

!>          A es array REAL, dimension (LDA,N)
!>          Detalles de la factorización QR de la matriz original A como
!>          devuelto por SGEQRF.
!>

LDA

!>          LDA es INTEGER
!>          La dimensión principal del array A.  LDA >= M.
!>

TAU

!>          TAU es array REAL, dimension (N)
!>          Detalles de la matriz ortogonal Q.
!>

B

!>          B es array REAL, dimension (LDB,NRHS)
!>          En la entrada, la matriz B del lado derecho de n rhs columnas.
!>          En la salida, la matriz solución X de n rhs columnas.
!>

LDB

!>          LDB es INTEGER
!>          La dimensión principal del array B. LDB >= M.
!>

WORK

!>          WORK es array REAL, dimension (LWORK)
!>

LWORK

!>          LWORK es INTEGER
!>          La longitud del array WORK.  LWORK debe ser al menos NRHS,
!>          y debe ser al menos NRHS*NB, donde NB es el tamaño del bloque
!>          para este entorno.
!>

INFO

!>          INFO es INTEGER
!>          = 0: salida exitosa
!>          < 0: si INFO = -i, el i-ésimo argumento tenía un valor ilegal
!>
