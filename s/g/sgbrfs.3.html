subroutine sgbrfs (character trans, integer n, integer kl, integer ku, integer nrhs, real, dimension( ldab, * ) ab, integer ldab, real, dimension( ldafb, * ) afb, integer ldafb, integer, dimension( * ) ipiv, real, dimension( ldb, * ) b, integer ldb, real, dimension( ldx, * ) x, integer ldx, real, dimension( * ) ferr, real, dimension( * ) berr, real, dimension( * ) work, integer, dimension( * ) iwork, integer info)

SGBRFS

Propósito:

!>
!> SGBRFS mejora la solución computada de un sistema de ecuaciones lineales
!> cuando la matriz de coeficientes es de banda, y proporciona
!> cotas de error y estimaciones de error hacia atrás para la solución.
!>

Parámetros

TRANS 
!>          TRANS es CHARACTER*1
!>          Especifica la forma del sistema de ecuaciones:
!>          = 'N':  A * X = B     (Sin transpuesta)
!>          = 'T':  A**T * X = B  (Transpuesta)
!>          = 'C':  A**H * X = B  (Transpuesta conjugada = Transpuesta)
!>

N

!>          N es INTEGER
!>          El orden de la matriz A.  N >= 0.
!>

KL

!>          KL es INTEGER
!>          El número de subdiagonales dentro de la banda de A.  KL >= 0.
!>

KU

!>          KU es INTEGER
!>          El número de superdiagonales dentro de la banda de A.  KU >= 0.
!>

NRHS

!>          NRHS es INTEGER
!>          El número de lados derechos, es decir, el número de columnas
!>          de las matrices B y X.  NRHS >= 0.
!>

AB

!>          AB es array REAL, dimension (LDAB,N)
!>          La matriz de banda original A, almacenada en las filas 1 a KL+KU+1.
!>          La j-ésima columna de A se almacena en la j-ésima columna del
!>          array AB de la siguiente manera:
!>          AB(ku+1+i-j,j) = A(i,j) para max(1,j-ku)<=i<=min(n,j+kl).
!>

LDAB

!>          LDAB es INTEGER
!>          La dimensión principal del array AB.  LDAB >= KL+KU+1.
!>

AFB

!>          AFB es array REAL, dimension (LDAFB,N)
!>          Detalles de la factorización LU de la matriz de banda A, como
!>          calculado por SGBTRF.  U se almacena como una matriz de banda triangular superior
!>          con KL+KU superdiagonales en las filas 1 a KL+KU+1, y
!>          los multiplicadores utilizados durante la factorización se almacenan en
!>          las filas KL+KU+2 a 2*KL+KU+1.
!>

LDAFB

!>          LDAFB es INTEGER
!>          La dimensión principal del array AFB.  LDAFB >= 2*KL*KU+1.
!>

IPIV

!>          IPIV es array INTEGER, dimension (N)
!>          Los índices de pivote de SGBTRF; para 1<=i<=N, la fila i de la
!>          matriz fue intercambiada con la fila IPIV(i).
!>

B

!>          B es array REAL, dimension (LDB,NRHS)
!>          La matriz del lado derecho B.
!>

LDB

!>          LDB es INTEGER
!>          La dimensión principal del array B.  LDB >= max(1,N).
!>

X

!>          X es array REAL, dimension (LDX,NRHS)
!>          En la entrada, la matriz solución X, como calculada por SGBTRS.
!>          En la salida, la matriz solución mejorada X.
!>

LDX

!>          LDX es INTEGER
!>          La dimensión principal del array X.  LDX >= max(1,N).
!>

FERR

!>          FERR es array REAL, dimension (NRHS)
!>          La cota de error hacia adelante estimada para cada vector solución
!>          X(j) (la j-ésima columna de la matriz solución X).
!>          Si XTRUE es la solución verdadera correspondiente a X(j), FERR(j)
!>          es una cota superior estimada para la magnitud del elemento más grande
!>          en (X(j) - XTRUE) dividido por la magnitud del elemento más grande
!>          en X(j). La estimación es tan fiable como
!>          la estimación para RCOND, y es casi siempre una ligera
!>          sobreestimación del error verdadero.
!>

BERR

!>          BERR es array REAL, dimension (NRHS)
!>          El error hacia atrás relativo componente a componente de cada vector solución
!>          X(j) (es decir, el cambio relativo más pequeño en
!>          cualquier elemento de A o B que hace que X(j) sea una solución exacta).
!>

WORK

!>          WORK es array REAL, dimension (3*N)
!>

IWORK

!>          IWORK es array INTEGER, dimension (N)
!>

INFO

!>          INFO es INTEGER
!>          = 0:  salida exitosa
!>          < 0:  si INFO = -i, el i-ésimo argumento tenía un valor ilegal
!>

Parámetros internos:

!>  ITMAX es el número máximo de pasos de refinamiento iterativo.
!>
