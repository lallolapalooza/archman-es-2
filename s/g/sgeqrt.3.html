subroutine sgeqrt (integer m, integer n, integer nb, real, dimension( lda, * ) a, integer lda, real, dimension( ldt, * ) t, integer ldt, real, dimension( * ) work, integer info)

SGEQRT

Propósito:

!>
!> SGEQRT calcula una factorización QR bloqueada de una matriz real M por N, A,
!> utilizando la representación WY compacta de Q.
!>

Parámetros

M 
!>          M es INTEGER
!>          El número de filas de la matriz A.  M >= 0.
!>

N

!>          N es INTEGER
!>          El número de columnas de la matriz A.  N >= 0.
!>

NB

!>          NB es INTEGER
!>          El tamaño del bloque a utilizar en la QR bloqueada.  MIN(M,N) >= NB >= 1.
!>

A

!>          A es un array REAL, dimension (LDA,N)
!>          En la entrada, la matriz A de M por N.
!>          En la salida, los elementos en y por encima de la diagonal del array
!>          contienen la matriz trapezoidal superior R de min(M,N) por N (R es
!>          triangular superior si M >= N); los elementos por debajo de la diagonal
!>          son las columnas de V.
!>

LDA

!>          LDA es INTEGER
!>          La dimensión principal del array A.  LDA >= max(1,M).
!>

T

!>          T es un array REAL, dimension (LDT,MIN(M,N))
!>          Los bloques reflectores triangulares superiores almacenados en forma compacta
!>          como una secuencia de bloques triangulares superiores. Ver más abajo
!>          para más detalles.
!>

LDT

!>          LDT es INTEGER
!>          La dimensión principal del array T.  LDT >= NB.
!>

WORK

!>          WORK es un array REAL, dimension (NB*N)
!>

INFO

!>          INFO es INTEGER
!>          = 0:  salida exitosa
!>          < 0:  si INFO = -i, el i-ésimo argumento tenía un valor ilegal
!>

Más detalles:

!>
!>  La matriz V almacena los reflectores elementales H(i) en la i-ésima columna
!>  por debajo de la diagonal. Por ejemplo, si M=5 y N=3, la matriz V es
!>
!>               V = (  1       )
!>                   ( v1  1    )
!>                   ( v1 v2  1 )
!>                   ( v1 v2 v3 )
!>                   ( v1 v2 v3 )
!>
!>  donde los vi's representan los vectores que definen H(i), los cuales son devueltos
!>  en la matriz A. Los 1's a lo largo de la diagonal de V no se almacenan en A.
!>
!>  Sea K=MIN(M,N). El número de bloques es B = techo(K/NB), donde cada
!>  bloque es de orden NB excepto el último bloque, que es de orden
!>  IB = K - (B-1)*NB. Para cada uno de los B bloques, se calcula un factor de
!>  reflector de bloque triangular superior: T1, T2, ..., TB. Los T de NB por NB
!>  (y de IB por IB para el último bloque) se almacenan en la matriz T de NB por K como
!>
!>               T = (T1 T2 ... TB).
!>
