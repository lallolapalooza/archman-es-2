subroutine sgttrs (character trans, integer n, integer nrhs, real, dimension( * ) dl, real, dimension( * ) d, real, dimension( * ) du, real, dimension( * ) du2, integer, dimension( * ) ipiv, real, dimension( ldb, * ) b, integer ldb, integer info)

SGTTRS

Propósito:

!>
!> SGTTRS resuelve uno de los sistemas de ecuaciones
!>    A*X = B  o  A**T*X = B,
!> con una matriz tridiagonal A utilizando la factorización LU calculada
!> por SGTTRF.
!>

Parámetros

TRANS 
!>          TRANS es CHARACTER*1
!>          Especifica la forma del sistema de ecuaciones.
!>          = 'N':  A * X = B  (Sin traspuesta)
!>          = 'T':  A**T* X = B  (Traspuesta)
!>          = 'C':  A**T* X = B  (Traspuesta conjugada = Traspuesta)
!>

N

!>          N es INTEGER
!>          El orden de la matriz A.
!>

NRHS

!>          NRHS es INTEGER
!>          El número de lados derechos, es decir, el número de columnas
!>          de la matriz B.  NRHS >= 0.
!>

DL

!>          DL es un array REAL, dimensión (N-1)
!>          Los (n-1) multiplicadores que definen la matriz L de la
!>          factorización LU de A.
!>

D

!>          D es un array REAL, dimensión (N)
!>          Los n elementos diagonales de la matriz triangular superior U de
!>          la factorización LU de A.
!>

DU

!>          DU es un array REAL, dimensión (N-1)
!>          Los (n-1) elementos de la superdiagonal superior de U.
!>

DU2

!>          DU2 es un array REAL, dimensión (N-2)
!>          Los (n-2) elementos de la segunda superdiagonal de U.
!>

IPIV

!>          IPIV es un array INTEGER, dimensión (N)
!>          Los índices de pivote; para 1 <= i <= n, la fila i de la matriz fue
!>          intercambiada con la fila IPIV(i).  IPIV(i) siempre será i o
!>          i+1; IPIV(i) = i indica que no se requirió un intercambio de fila.
!>

B

!>          B es un array REAL, dimensión (LDB,NRHS)
!>          En la entrada, la matriz de vectores del lado derecho B.
!>          En la salida, B se sobrescribe con los vectores solución X.
!>

LDB

!>          LDB es INTEGER
!>          La dimensión principal del array B.  LDB >= max(1,N).
!>

INFO

!>          INFO es INTEGER
!>          = 0:  salida exitosa
!>          < 0:  si INFO = -i, el argumento i-ésimo tuvo un valor ilegal
!>
