subroutine sget22 (character transa, character transe, character transw, integer n, real, dimension( lda, * ) a, integer lda, real, dimension( lde, * ) e, integer lde, real, dimension( * ) wr, real, dimension( * ) wi, real, dimension( * ) work, real, dimension( 2 ) result)

SGET22

Propósito:

!>
!> SGET22 realiza una comprobación de autovectores.
!>
!> La comprobación básica es:
!>
!>    RESULT(1) = | A E  -  E W | / ( |A| |E| ulp )
!>
!> usando la norma 1. También comprueba la normalización de E:
!>
!>    RESULT(2) = max | m-norm(E(j)) - 1 | / ( n ulp )
!>                 j
!>
!> donde E(j) es el j-ésimo autovector, y m-norm es la norma máxima de un
!> vector. Si un autovector es complejo, como se determina por WI(j)
!> distinto de cero, entonces la norma máxima del vector ( er + i*ei ) es el máximo
!> de
!>    |er(1)| + |ei(1)|, ... , |er(n)| + |ei(n)|
!>
!> W es una matriz diagonal por bloques, con un bloque de 1 por 1 para cada autovalor real
!> y un bloque de 2 por 2 para cada par conjugado complejo.
!> Si los autovalores j y j+1 son un par conjugado complejo, de modo que
!> WR(j) = WR(j+1) = wr y WI(j) = - WI(j+1) = wi, entonces el bloque de 2 por 2
!> correspondiente al par será:
!>
!>    (  wr  wi  )
!>    ( -wi  wr  )
!>
!> Dicho bloque multiplicando una matriz de n por 2 ( ur ui ) por la derecha
!> será lo mismo que multiplicar ur + i*ui por wr + i*wi.
!>
!> Para manejar varios esquemas de almacenamiento de autovectores izquierdos, existen
!> opciones para usar la transpuesta de A en lugar de A, la transpuesta de E en lugar de E,
!> y/o la transpuesta de W en lugar de W.
!>

Parámetros

TRANSA 
!>          TRANSA es CHARACTER*1
!>          Especifica si A se transpone o no.
!>          = 'N':  Sin transponer
!>          = 'T':  Transpuesta
!>          = 'C':  Transpuesta conjugada (= Transpuesta)
!>

TRANSE

!>          TRANSE es CHARACTER*1
!>          Especifica si E se transpone o no.
!>          = 'N':  Sin transponer, los autovectores están en las columnas de E
!>          = 'T':  Transpuesta, los autovectores están en las filas de E
!>          = 'C':  Transpuesta conjugada (= Transpuesta)
!>

TRANSW

!>          TRANSW es CHARACTER*1
!>          Especifica si W se transpone o no.
!>          = 'N':  Sin transponer
!>          = 'T':  Transpuesta, usa -WI(j) en lugar de WI(j)
!>          = 'C':  Transpuesta conjugada, usa -WI(j) en lugar de WI(j)
!>

N

!>          N es INTEGER
!>          El orden de la matriz A.  N >= 0.
!>

A

!>          A es un array REAL, de dimensión (LDA,N)
!>          La matriz cuyos autovectores están en E.
!>

LDA

!>          LDA es INTEGER
!>          La dimensión principal del array A.  LDA >= max(1,N).
!>

E

!>          E es un array REAL, de dimensión (LDE,N)
!>          La matriz de autovectores. Si TRANSE = 'N', los autovectores
!>          están almacenados en las columnas de E, si TRANSE = 'T' o 'C',
!>          los autovectores están almacenados en las filas de E.
!>

LDE

!>          LDE es INTEGER
!>          La dimensión principal del array E.  LDE >= max(1,N).
!>

WR

!>          WR es un array REAL, de dimensión (N)
!>

WI

!>          WI es un array REAL, de dimensión (N)
!>
!>          Las partes real e imaginaria de los autovalores de A.
!>          Los autovalores puramente reales se indican con WI(j) = 0.
!>          Los pares conjugados complejos se indican con WR(j)=WR(j+1) y
!>          WI(j) = - WI(j+1) distinto de cero; la parte real se asume
!>          almacenada en la fila/columna j y la parte imaginaria en
!>          la fila/columna (j+1).
!>

WORK

!>          WORK es un array REAL, de dimensión (N*(N+1))
!>

RESULT

!>          RESULT es un array REAL, de dimensión (2)
!>          RESULT(1) = | A E  -  E W | / ( |A| |E| ulp )
!>          RESULT(2) = max | m-norm(E(j)) - 1 | / ( n ulp )
!>                       j
!>
