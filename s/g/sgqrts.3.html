Documentación de Función/Subrutina

subroutine sgqrts (integer n, integer m, integer p, real, dimension( lda, * ) a, real, dimension( lda, * ) af, real, dimension( lda, * ) q, real, dimension( lda, * ) r, integer lda, real, dimension( * ) taua, real, dimension( ldb, * ) b, real, dimension( ldb, * ) bf, real, dimension( ldb, * ) z, real, dimension( ldb, * ) t, real, dimension( ldb, * ) bwk, integer ldb, real, dimension( * ) taub, real, dimension( lwork ) work, integer lwork, real, dimension( * ) rwork, real, dimension( 4 ) result)

SGQRTS

Propósito:

!>
!> SGQRTS prueba SGGQRF, que calcula la factorización GQR de una
!> matriz A de N por M y una matriz B de N por P: A = Q*R y B = Q*T*Z.
!>

Parámetros

N 
!>          N es INTEGER
!>          El número de filas de las matrices A y B.  N >= 0.
!>

M

!>          M es INTEGER
!>          El número de columnas de la matriz A.  M >= 0.
!>

P

!>          P es INTEGER
!>          El número de columnas de la matriz B.  P >= 0.
!>

A

!>          A es un array REAL, dimension (LDA,M)
!>          La matriz A de N por M.
!>

AF

!>          AF es un array REAL, dimension (LDA,N)
!>          Detalles de la factorización GQR de A y B, según lo devuelto
!>          por SGGQRF, véase SGGQRF para más detalles.
!>

Q

!>          Q es un array REAL, dimension (LDA,N)
!>          La matriz ortogonal Q de M por M.
!>

R

!>          R es un array REAL, dimension (LDA,MAX(M,N))
!>

LDA

!>          LDA es INTEGER
!>          La dimensión principal de los arrays A, AF, R y Q.
!>          LDA >= max(M,N).
!>

TAUA

!>          TAUA es un array REAL, dimension (min(M,N))
!>          Los factores escalares de los reflectores elementales, según lo devuelto
!>          por SGGQRF.
!>

B

!>          B es un array REAL, dimension (LDB,P)
!>          En la entrada, la matriz B de N por P.
!>

BF

!>          BF es un array REAL, dimension (LDB,N)
!>          Detalles de la factorización GQR de A y B, según lo devuelto
!>          por SGGQRF, véase SGGQRF para más detalles.
!>

Z

!>          Z es un array REAL, dimension (LDB,P)
!>          La matriz ortogonal Z de P por P.
!>

T

!>          T es un array REAL, dimension (LDB,max(P,N))
!>

BWK

!>          BWK es un array REAL, dimension (LDB,N)
!>

LDB

!>          LDB es INTEGER
!>          La dimensión principal de los arrays B, BF, Z y T.
!>          LDB >= max(P,N).
!>

TAUB

!>          TAUB es un array REAL, dimension (min(P,N))
!>          Los factores escalares de los reflectores elementales, según lo devuelto
!>          por SGGRQF.
!>

WORK

!>          WORK es un array REAL, dimension (LWORK)
!>

LWORK

!>          LWORK es INTEGER
!>          La dimensión del array WORK, LWORK >= max(N,M,P)**2.
!>

RWORK

!>          RWORK es un array REAL, dimension (max(N,M,P))
!>

RESULT

!>          RESULT es un array REAL, dimension (4)
!>          Los ratios de prueba:
!>            RESULT(1) = norm( R - Q'*A ) / ( MAX(M,N)*norm(A)*ULP)
!>            RESULT(2) = norm( T*Z - Q'*B ) / (MAX(P,N)*norm(B)*ULP)
!>            RESULT(3) = norm( I - Q'*Q ) / ( M*ULP )
!>            RESULT(4) = norm( I - Z'*Z ) / ( P*ULP )
!>
