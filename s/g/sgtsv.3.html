subroutine sgtsv (integer n, integer nrhs, real, dimension( * ) dl, real, dimension( * ) d, real, dimension( * ) du, real, dimension( ldb, * ) b, integer ldb, integer info)

SGTSV calcula la solución al sistema de ecuaciones lineales A * X = B para matrices GT

Propósito:

!>
!> SGTSV  resuelve la ecuación
!>
!>    A*X = B,
!>
!> donde A es una matriz tridiagonal de n por n, mediante eliminación gaussiana con
!> pivoteo parcial.
!>
!> Tenga en cuenta que la ecuación  A**T*X = B  puede resolverse intercambiando el
!> orden de los argumentos DU y DL.
!>

Parámetros

N 
!>          N es INTEGER
!>          El orden de la matriz A.  N >= 0.
!>

NRHS

!>          NRHS es INTEGER
!>          El número de lados derechos, es decir, el número de columnas
!>          de la matriz B.  NRHS >= 0.
!>

DL

!>          DL es un array REAL, de dimensión (N-1)
!>          En la entrada, DL debe contener los elementos sub-diagonales (n-1) de
!>          A.
!>
!>          En la salida, DL se sobrescribe con los elementos (n-2) de la
!>          segunda superdiagonal de la matriz triangular superior U de
!>          la factorización LU de A, en DL(1), ..., DL(n-2).
!>

D

!>          D es un array REAL, de dimensión (N)
!>          En la entrada, D debe contener los elementos diagonales de A.
!>
!>          En la salida, D se sobrescribe con los n elementos diagonales de U.
!>

DU

!>          DU es un array REAL, de dimensión (N-1)
!>          En la entrada, DU debe contener los elementos super-diagonales (n-1)
!>          de A.
!>
!>          En la salida, DU se sobrescribe con los elementos (n-1) de la primera
!>          superdiagonal de U.
!>

B

!>          B es un array REAL, de dimensión (LDB,NRHS)
!>          En la entrada, la matriz n por NRHS del lado derecho B.
!>          En la salida, si INFO = 0, la matriz solución n por NRHS X.
!>

LDB

!>          LDB es INTEGER
!>          La dimensión principal del array B.  LDB >= max(1,N).
!>

INFO

!>          INFO es INTEGER
!>          = 0: salida exitosa
!>          < 0: si INFO = -i, el i-ésimo argumento tenía un valor ilegal
!>          > 0: si INFO = i, U(i,i) es exactamente cero, y la solución
!>               no se ha calculado. La factorización no se ha
!>               completado a menos que i = N.
!>
