DOCUMENTACIÓN DE LA FUNCIÓN/SUBRUTINA

subroutine sget54 (integer n, real, dimension( lda, * ) a, integer lda, real, dimension( ldb, * ) b, integer ldb, real, dimension( lds, * ) s, integer lds, real, dimension( ldt, * ) t, integer ldt, real, dimension( ldu, * ) u, integer ldu, real, dimension( ldv, * ) v, integer ldv, real, dimension( * ) work, real result)

SGET54

Propósito:

!>
!> SGET54 comprueba una descomposición generalizada de la forma
!>
!>          A = U*S*V'  y B = U*T* V'
!>
!> donde ' significa transpuesta y U y V son ortogonales.
!>
!> Específicamente,
!>
!>  RESULT = ||( A - U*S*V', B - U*T*V' )|| / (||( A, B )||*n*ulp )
!>

Parámetros

N 
!>          N es ENTERO
!>          El tamaño de la matriz. Si es cero, SGET54 no hace nada.
!>          Debe ser al menos cero.
!>

A

!>          A es un array REAL, dimension (LDA, N)
!>          La matriz A original (sin factorizar).
!>

LDA

!>          LDA es ENTERO
!>          La dimensión principal de A. Debe ser al menos 1
!>          y al menos N.
!>

B

!>          B es un array REAL, dimension (LDB, N)
!>          La matriz B original (sin factorizar).
!>

LDB

!>          LDB es ENTERO
!>          La dimensión principal de B. Debe ser al menos 1
!>          y al menos N.
!>

S

!>          S es un array REAL, dimension (LDS, N)
!>          La matriz factorizada S.
!>

LDS

!>          LDS es ENTERO
!>          La dimensión principal de S. Debe ser al menos 1
!>          y al menos N.
!>

T

!>          T es un array REAL, dimension (LDT, N)
!>          La matriz factorizada T.
!>

LDT

!>          LDT es ENTERO
!>          La dimensión principal de T. Debe ser al menos 1
!>          y al menos N.
!>

U

!>          U es un array REAL, dimension (LDU, N)
!>          La matriz ortogonal en el lado izquierdo de la
!>          descomposición.
!>

LDU

!>          LDU es ENTERO
!>          La dimensión principal de U. LDU debe ser al menos N y
!>          al menos 1.
!>

V

!>          V es un array REAL, dimension (LDV, N)
!>          La matriz ortogonal en el lado izquierdo de la
!>          descomposición.
!>

LDV

!>          LDV es ENTERO
!>          La dimensión principal de V. LDV debe ser al menos N y
!>          al menos 1.
!>

WORK

!>          WORK es un array REAL, dimension (3*N**2)
!>

RESULT

!>          RESULT es REAL
!>          El valor RESULT, actualmente limitado a 1/ulp, para
!>          evitar desbordamiento. Los errores se marcan con RESULT=10/ulp.
!>
