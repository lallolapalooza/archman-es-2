Documentación de la función/subrutina

subroutine sgeqls (integer m, integer n, integer nrhs, real, dimension( lda, * ) a, integer lda, real, dimension( * ) tau, real, dimension( ldb, * ) b, integer ldb, real, dimension( lwork ) work, integer lwork, integer info)

SGEQLS

Propósito:

!>
!> Resolver el problema de mínimos cuadrados
!>     min || A*X - B ||
!> usando la factorización QL
!>     A = Q*L
!> calculada por SGEQLF.
!>

Parámetros

M 
!>          M es INTEGER
!>          El número de filas de la matriz A.  M >= 0.
!>

N

!>          N es INTEGER
!>          El número de columnas de la matriz A.  M >= N >= 0.
!>

NRHS

!>          NRHS es INTEGER
!>          El número de columnas de B.  NRHS >= 0.
!>

A

!>          A es un array REAL, de dimensión (LDA,N)
!>          Detalles de la factorización QL de la matriz original A,
!>          tal como se devuelve por SGEQLF.
!>

LDA

!>          LDA es INTEGER
!>          La dimensión principal del array A.  LDA >= M.
!>

TAU

!>          TAU es un array REAL, de dimensión (N)
!>          Detalles de la matriz ortogonal Q.
!>

B

!>          B es un array REAL, de dimensión (LDB,NRHS)
!>          En la entrada, la matriz B del lado derecho de m por nrhs.
!>          En la salida, la matriz solución X de n por nrhs, almacenada
!>          en las filas m-n+1:m.
!>

LDB

!>          LDB es INTEGER
!>          La dimensión principal del array B. LDB >= M.
!>

WORK

!>          WORK es un array REAL, de dimensión (LWORK)
!>

LWORK

!>          LWORK es INTEGER
!>          La longitud del array WORK.  LWORK debe ser al menos NRHS,
!>          y debería ser al menos NRHS*NB, donde NB es el tamaño del
!>          bloque para este entorno.
!>

INFO

!>          INFO es INTEGER
!>          = 0: salida exitosa
!>          < 0: si INFO = -i, el i-ésimo argumento tuvo un valor ilegal
!>
