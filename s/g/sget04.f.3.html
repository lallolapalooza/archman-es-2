subroutine sget04 (integer n, integer nrhs, real, dimension( ldx, * ) x, integer ldx, real, dimension( ldxact, * ) xact, integer ldxact, real rcond, real resid)

SGET04

Propósito:

!>
!> SGET04 calcula la diferencia entre una solución calculada y la
!> solución verdadera de un sistema de ecuaciones lineales.
!>
!> RESID =  ( norm(X-XACT) * RCOND ) / ( norm(XACT) * EPS ),
!> donde RCOND es el recíproco del número de condición y EPS es la
!> épsilon de máquina.
!>

Parámetros

N 
!>          N es INTEGER
!>          El número de filas de las matrices X y XACT.  N >= 0.
!>

NRHS

!>          NRHS es INTEGER
!>          El número de columnas de las matrices X y XACT.  NRHS >= 0.
!>

X

!>          X es un array REAL, de dimensión (LDX,NRHS)
!>          Los vectores de solución calculados. Cada vector se almacena como
!>          una columna de la matriz X.
!>

LDX

!>          LDX es INTEGER
!>          La dimensión principal del array X.  LDX >= max(1,N).
!>

XACT

!>          XACT es un array REAL, de dimensión( LDX, NRHS )
!>          Los vectores de solución exactos. Cada vector se almacena como
!>          una columna de la matriz XACT.
!>

LDXACT

!>          LDXACT es INTEGER
!>          La dimensión principal del array XACT.  LDXACT >= max(1,N).
!>

RCOND

!>          RCOND es REAL
!>          El recíproco del número de condición de la matriz de coeficientes
!>          en el sistema de ecuaciones.
!>

RESID

!>          RESID es REAL
!>          El máximo sobre los NRHS vectores de solución de
!>          ( norm(X-XACT) * RCOND ) / ( norm(XACT) * EPS )
!>
