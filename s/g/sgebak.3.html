subroutine sgebak (character job, character side, integer n, integer ilo, integer ihi, real, dimension( * ) scale, integer m, real, dimension( ldv, * ) v, integer ldv, integer info)

SGEBAK

Propósito:

!>
!> SGEBAK forma los autovectores derechos o izquierdos de una matriz general real
!> por transformación inversa sobre los autovectores calculados de
!> la matriz balanceada producida por SGEBAL.
!>

Parámetros

JOB 
!>          JOB es CHARACTER*1
!>          Especifica el tipo de transformación inversa requerida:
!>          = 'N': no hacer nada, retornar inmediatamente;
!>          = 'P': hacer la transformación inversa solo para la permutación;
!>          = 'S': hacer la transformación inversa solo para el escalado;
!>          = 'B': hacer las transformaciones inversas tanto para la permutación
!>                 como para el escalado.
!>          JOB debe ser el mismo que el argumento JOB suministrado a SGEBAL.
!>

SIDE

!>          SIDE es CHARACTER*1
!>          = 'R':  V contiene autovectores derechos;
!>          = 'L':  V contiene autovectores izquierdos.
!>

N

!>          N es INTEGER
!>          El número de filas de la matriz V.  N >= 0.
!>

ILO

!>          ILO es INTEGER
!>

IHI

!>          IHI es INTEGER
!>          Los enteros ILO e IHI determinados por SGEBAL.
!>          1 <= ILO <= IHI <= N, si N > 0; ILO=1 e IHI=0, si N=0.
!>

SCALE

!>          SCALE es un array REAL, dimension (N)
!>          Detalles de la permutación y los factores de escalado, como los devuelve
!>          SGEBAL.
!>

M

!>          M es INTEGER
!>          El número de columnas de la matriz V.  M >= 0.
!>

V

!>          V es un array REAL, dimension (LDV,M)
!>          En la entrada, la matriz de autovectores derechos o izquierdos a transformar,
!>          como la devuelve SHSEIN o STREVC.
!>          En la salida, V se sobrescribe con los autovectores transformados.
!>

LDV

!>          LDV es INTEGER
!>          La dimensión principal del array V. LDV >= max(1,N).
!>

INFO

!>          INFO es INTEGER
!>          = 0:  salida exitosa
!>          < 0:  si INFO = -i, el i-ésimo argumento tenía un valor ilegal.
!>
