DOCUMENTACIÓN DE FUNCIÓN/SUBRUTINA

subroutine sget03 (integer n, real, dimension( lda, * ) a, integer lda, real, dimension( ldainv, * ) ainv, integer ldainv, real, dimension( ldwork, * ) work, integer ldwork, real, dimension( * ) rwork, real rcond, real resid)

SGET03

Propósito:

!>
!> SGET03 calcula el residuo de una matriz general multiplicada por su inversa:
!>    norm( I - AINV*A ) / ( N * norm(A) * norm(AINV) * EPS ),
!> donde EPS es el epsilon de la máquina.
!>

Parámetros

N 
!>          N es INTEGER
!>          El número de filas y columnas de la matriz A.  N >= 0.
!>

A

!>          A es array REAL, dimension (LDA,N)
!>          La matriz original N x N, A.
!>

LDA

!>          LDA es INTEGER
!>          La dimensión principal del array A.  LDA >= max(1,N).
!>

AINV

!>          AINV es array REAL, dimension (LDAINV,N)
!>          La inversa de la matriz A.
!>

LDAINV

!>          LDAINV es INTEGER
!>          La dimensión principal del array AINV.  LDAINV >= max(1,N).
!>

WORK

!>          WORK es array REAL, dimension (LDWORK,N)
!>

LDWORK

!>          LDWORK es INTEGER
!>          La dimensión principal del array WORK.  LDWORK >= max(1,N).
!>

RWORK

!>          RWORK es array REAL, dimension (N)
!>

RCOND

!>          RCOND es REAL
!>          El recíproco del número de condición de A, calculado como
!>          ( 1/norm(A) ) / norm(AINV).
!>

RESID

!>          RESID es REAL
!>          norm(I - AINV*A) / ( N * norm(A) * norm(AINV) * EPS )
!>
