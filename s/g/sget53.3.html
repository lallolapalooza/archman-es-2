subroutine sget53 (real, dimension( lda, * ) a, integer lda, real, dimension( ldb, * ) b, integer ldb, real scale, real wr, real wi, real result, integer info)

SGET53

Propósito:

!>
!> SGET53 comprueba los valores propios generalizados calculados por SLAG2.
!>
!> La prueba básica para un valor propio es:
!>
!>                              | det( s A - w B ) |
!>     RESULT =  ---------------------------------------------------
!>               ulp max( s norm(A), |w| norm(B) )*norm( s A - w B )
!>
!> Se realizan dos pruebas:
!>
!> (1)  ulp*max( s*norm(A), |w|*norm(B) )  debe ser al menos
!>      safe_minimum. Esto asegura que la prueba realizada no es esencialmente
!>      det(0*A + 0*B)=0.
!>
!> (2)  s*norm(A) + |w|*norm(B) debe ser menor que 1/safe_minimum.
!>      Esto asegura que  s*A - w*B  no provocará un desbordamiento.
!>
!> Si estas pruebas no se cumplen, entonces  s  y  w  se escalan y
!> se prueban de todos modos, si esto es posible.
!>

Parámetros

A 
!>          A es un array REAL, dimension (LDA, 2)
!>          La matriz 2x2 A.
!>

LDA

!>          LDA es un INTEGER
!>          La dimensión principal de A. Debe ser al menos 2.
!>

B

!>          B es un array REAL, dimension (LDB, N)
!>          La matriz triangular superior 2x2 B.
!>

LDB

!>          LDB es un INTEGER
!>          La dimensión principal de B. Debe ser al menos 2.
!>

SCALE

!>          SCALE es un REAL
!>          La escala  s  en la fórmula  s A - w B .  Se
!>          asume que es no negativa.
!>

WR

!>          WR es un REAL
!>          La parte real del valor propio  w  en la fórmula
!>          s A - w B .
!>

WI

!>          WI es un REAL
!>          La parte imaginaria del valor propio  w  en la fórmula
!>          s A - w B .
!>

RESULT

!>          RESULT es un REAL
!>          Si INFO es 2 o menos, el valor calculado por la prueba
!>             descrita arriba.
!>          Si INFO=3, esto será simplemente 1/ulp.
!>

INFO

!>          INFO es un INTEGER
!>          =0:  Los datos de entrada pasan las pruebas.
!>          =1:  s*norm(A) + |w|*norm(B) > 1/safe_minimum.
!>          =2:  ulp*max( s*norm(A), |w|*norm(B) ) < safe_minimum
!>          =3:  igual que INFO=2, pero  s  y  w  no pudieron escalarse para
!>               calcular la prueba.
!>
