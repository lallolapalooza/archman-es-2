subroutine sgbbrd (character vect, integer m, integer n, integer ncc, integer kl, integer ku, real, dimension( ldab, * ) ab, integer ldab, real, dimension( * ) d, real, dimension( * ) e, real, dimension( ldq, * ) q, integer ldq, real, dimension( ldpt, * ) pt, integer ldpt, real, dimension( ldc, * ) c, integer ldc, real, dimension( * ) work, integer info)

SGBBRD

Propósito:

!>
!> SGBBRD reduce una matriz de banda general real m por n A a forma bidiagonal
!> superior B mediante una transformación ortogonal: Q**T * A * P = B.
!>
!> La rutina calcula B y, opcionalmente, forma Q o P**T, o calcula
!> Q**T*C para una matriz C dada.
!>

Parámetros

VECT 
!>          VECT es CHARACTER*1
!>          Especifica si las matrices Q y P**T deben formarse o no.
!>          = 'N': no formar Q ni P**T;
!>          = 'Q': formar solo Q;
!>          = 'P': formar solo P**T;
!>          = 'B': formar ambas.
!>

M

!>          M es INTEGER
!>          El número de filas de la matriz A. M >= 0.
!>

N

!>          N es INTEGER
!>          El número de columnas de la matriz A. N >= 0.
!>

NCC

!>          NCC es INTEGER
!>          El número de columnas de la matriz C. NCC >= 0.
!>

KL

!>          KL es INTEGER
!>          El número de subdiagonales de la matriz A. KL >= 0.
!>

KU

!>          KU es INTEGER
!>          El número de superdiagonales de la matriz A. KU >= 0.
!>

AB

!>          AB es array REAL, dimension (LDAB,N)
!>          En la entrada, la matriz de banda A de m por n, almacenada en las
!>          filas 1 a KL+KU+1. La j-ésima columna de A se almacena en la j-ésima
!>          columna de la matriz AB como sigue:
!>          AB(ku+1+i-j,j) = A(i,j) para max(1,j-ku) <= i <= min(m,j+kl).
!>          En la salida, A se sobrescribe con valores generados durante la
!>          reducción.
!>

LDAB

!>          LDAB es INTEGER
!>          La dimensión principal de la matriz A. LDAB >= KL+KU+1.
!>

D

!>          D es array REAL, dimension (min(M,N))
!>          Los elementos diagonales de la matriz bidiagonal B.
!>

E

!>          E es array REAL, dimension (min(M,N)-1)
!>          Los elementos superdiagonales de la matriz bidiagonal B.
!>

Q

!>          Q es array REAL, dimension (LDQ,M)
!>          Si VECT = 'Q' o 'B', la matriz ortogonal Q de m por m.
!>          Si VECT = 'N' o 'P', la matriz Q no se referencia.
!>

LDQ

!>          LDQ es INTEGER
!>          La dimensión principal de la matriz Q.
!>          LDQ >= max(1,M) si VECT = 'Q' o 'B'; de lo contrario, LDQ >= 1.
!>

PT

!>          PT es array REAL, dimension (LDPT,N)
!>          Si VECT = 'P' o 'B', la matriz ortogonal P' de n por n.
!>          Si VECT = 'N' o 'Q', la matriz PT no se referencia.
!>

LDPT

!>          LDPT es INTEGER
!>          La dimensión principal de la matriz PT.
!>          LDPT >= max(1,N) si VECT = 'P' o 'B'; de lo contrario, LDPT >= 1.
!>

C

!>          C es array REAL, dimension (LDC,NCC)
!>          En la entrada, una matriz C de m por ncc.
!>          En la salida, C se sobrescribe con Q**T*C.
!>          C no se referencia si NCC = 0.
!>

LDC

!>          LDC es INTEGER
!>          La dimensión principal de la matriz C.
!>          LDC >= max(1,M) si NCC > 0; LDC >= 1 si NCC = 0.
!>

WORK

!>          WORK es array REAL, dimension (2*max(M,N))
!>

INFO

!>          INFO es INTEGER
!>          = 0:  salida exitosa.
!>          < 0:  si INFO = -i, el i-ésimo argumento tenía un valor ilegal.
!>
