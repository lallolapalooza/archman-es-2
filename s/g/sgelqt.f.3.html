subroutine sgelqt (integer m, integer n, integer mb, real, dimension( lda, * ) a, integer lda, real, dimension( ldt, * ) t, integer ldt, real, dimension( * ) work, integer info)

SGELQT

Propósito:

!>
!> SGELQT calcula una factorización LQ bloqueada de una matriz real M por N, A,
!> utilizando la representación WY compacta de Q.
!>

Parámetros

M 
!>          M es INTEGER
!>          El número de filas de la matriz A. M >= 0.
!>

N

!>          N es INTEGER
!>          El número de columnas de la matriz A. N >= 0.
!>

MB

!>          MB es INTEGER
!>          El tamaño del bloque que se utilizará en la QR bloqueada. MIN(M,N) >= MB >= 1.
!>

A

!>          A es array REAL, dimension (LDA,N)
!>          En la entrada, la matriz M por N, A.
!>          En la salida, los elementos en y debajo de la diagonal del array
!>          contienen la matriz trapezoidal inferior L M por MIN(M,N) (L es
!>          triangular inferior si M <= N); los elementos por encima de la diagonal
!>          son las filas de V.
!>

LDA

!>          LDA es INTEGER
!>          La dimensión principal del array A. LDA >= max(1,M).
!>

T

!>          T es array REAL, dimension (LDT,MIN(M,N))
!>          Los reflejos de bloque triangulares superiores almacenados en forma compacta
!>          como una secuencia de bloques triangulares superiores. Ver abajo
!>          para más detalles.
!>

LDT

!>          LDT es INTEGER
!>          La dimensión principal del array T. LDT >= MB.
!>

WORK

!>          WORK es array REAL, dimension (MB*N)
!>

INFO

!>          INFO es INTEGER
!>          = 0: salida exitosa
!>          < 0: si INFO = -i, el i-ésimo argumento tenía un valor ilegal
!>

Detalles adicionales:

!>
!>  La matriz V almacena los reflejos elementales H(i) en la fila i
!>  por encima de la diagonal. Por ejemplo, si M=5 y N=3, la matriz V es
!>
!>               V = (  1  v1 v1 v1 v1 )
!>                   (     1  v2 v2 v2 )
!>                   (         1 v3 v3 )
!>
!>
!>  donde los vi representan los vectores que definen H(i), que se devuelven
!>  en la matriz A. Los 1's a lo largo de la diagonal de V no se almacenan en A.
!>  Sea K=MIN(M,N). El número de bloques es B = techo(K/MB), donde cada
!>  bloque es de orden MB excepto el último bloque, que es de orden
!>  IB = K - (B-1)*MB. Para cada uno de los B bloques, se calcula un factor de
!>  reflejo de bloque triangular superior: T1, T2, ..., TB. Los T de MB por MB
!>  (y IB por IB para el último bloque) se almacenan en la matriz T de MB por K como
!>
!>               T = (T1 T2 ... TB).
!>
