subroutine sget01 (integer m, integer n, real, dimension( lda, * ) a, integer lda, real, dimension( ldafac, * ) afac, integer ldafac, integer, dimension( * ) ipiv, real, dimension( * ) rwork, real resid)

SGET01

Propósito:

!>
!> SGET01 reconstruye una matriz A a partir de su factorización L*U y
!> calcula el residuo
!>    norm(L*U - A) / ( N * norm(A) * EPS ),
!> donde EPS es la epsilon de la máquina.
!>

Parámetros

M 
!>          M es INTEGER
!>          El número de filas de la matriz A.  M >= 0.
!>

N

!>          N es INTEGER
!>          El número de columnas de la matriz A.  N >= 0.
!>

A

!>          A es un array REAL, dimension (LDA,N)
!>          La matriz original M x N, A.
!>

LDA

!>          LDA es INTEGER
!>          La dimensión principal del array A.  LDA >= max(1,M).
!>

AFAC

!>          AFAC es un array REAL, dimension (LDAFAC,N)
!>          La forma factorizada de la matriz A.  AFAC contiene los factores
!>          L y U de la factorización L*U calculada por SGETRF.
!>          Se sobrescribe con la matriz reconstruida y, a continuación, con
!>          la diferencia L*U - A.
!>

LDAFAC

!>          LDAFAC es INTEGER
!>          La dimensión principal del array AFAC.  LDAFAC >= max(1,M).
!>

IPIV

!>          IPIV es un array INTEGER, dimension (N)
!>          Los índices de pivote de SGETRF.
!>

RWORK

!>          RWORK es un array REAL, dimension (M)
!>

RESID

!>          RESID es REAL
!>          norm(L*U - A) / ( N * norm(A) * EPS )
!>
