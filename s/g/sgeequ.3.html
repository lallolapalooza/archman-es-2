subroutine sgeequ (integer m, integer n, real, dimension( lda, * ) a, integer lda, real, dimension( * ) r, real, dimension( * ) c, real rowcnd, real colcnd, real amax, integer info)

SGEEQU

Propósito:

!>
!> SGEEQU calcula factores de escala de fila y columna destinados a equilibrar una
!> matriz A de M por N y a reducir su número de condición. R devuelve los factores de escala de fila
!> y C los factores de escala de columna, elegidos para intentar hacer
!> que el elemento más grande de cada fila y columna de la matriz B con
!> elementos B(i,j)=R(i)*A(i,j)*C(j) tenga un valor absoluto de 1.
!>
!> R(i) y C(j) están restringidos a estar entre SMLNUM = el número seguro más pequeño
!> y BIGNUM = el número seguro más grande. El uso de estos factores de escala
!> no garantiza la reducción del número de condición de A, pero
!> funciona bien en la práctica.
!>

Parámetros

M 
!>          M es INTEGER
!>          El número de filas de la matriz A.  M >= 0.
!>

N

!>          N es INTEGER
!>          El número de columnas de la matriz A.  N >= 0.
!>

A

!>          A es un array REAL, de dimensión (LDA,N)
!>          La matriz M por N cuyos factores de equilibrado se
!>          han de calcular.
!>

LDA

!>          LDA es INTEGER
!>          La dimensión principal del array A.  LDA >= max(1,M).
!>

R

!>          R es un array REAL, de dimensión (M)
!>          Si INFO = 0 o INFO > M, R contiene los factores de escala de fila
!>          para A.
!>

C

!>          C es un array REAL, de dimensión (N)
!>          Si INFO = 0,  C contiene los factores de escala de columna para A.
!>

ROWCND

!>          ROWCND es REAL
!>          Si INFO = 0 o INFO > M, ROWCND contiene la relación del
!>          R(i) más pequeño con el R(i) más grande.  Si ROWCND >= 0.1 y
!>          AMAX no es ni demasiado grande ni demasiado pequeño, no vale la pena
!>          escalar con R.
!>

COLCND

!>          COLCND es REAL
!>          Si INFO = 0, COLCND contiene la relación del
!>          C(i) más pequeño con el C(i) más grande.  Si COLCND >= 0.1, no vale la pena
!>          escalar con C.
!>

AMAX

!>          AMAX es REAL
!>          Valor absoluto del elemento de matriz más grande.  Si AMAX está muy
!>          cerca del desbordamiento o muy cerca del subdesbordamiento, la matriz
!>          debería escalarse.
!>

INFO

!>          INFO es INTEGER
!>          = 0:  salida exitosa
!>          < 0:  si INFO = -i, el i-ésimo argumento tenía un valor ilegal
!>          > 0:  si INFO = i,  y i es
!>                <= M:  la i-ésima fila de A es exactamente cero
!>                >  M:  la (i-M)-ésima columna de A es exactamente cero
!>
