subroutine sgeqrt2 (integer m, integer n, real, dimension( lda, * ) a, integer lda, real, dimension( ldt, * ) t, integer ldt, integer info)

SGEQRT2 calcula una factorización QR de una matriz real o compleja general utilizando la representación WY compacta de Q.

Propósito:

!>
!> SGEQRT2 calcula una factorización QR de una matriz real M por N, A,
!> utilizando la representación WY compacta de Q.
!>

Parámetros

M 
!>          M es INTEGER
!>          El número de filas de la matriz A.  M >= N.
!>

N

!>          N es INTEGER
!>          El número de columnas de la matriz A.  N >= 0.
!>

A

!>          A es un array REAL, de dimensión (LDA,N)
!>          En la entrada, los elementos de la matriz real M por N A. En la salida,
!>          los elementos en y por encima de la diagonal contienen la matriz
!>          triangular superior R de N por N; los elementos por debajo de la
!>          diagonal son las columnas de V. Ver más abajo para más detalles.
!>

LDA

!>          LDA es INTEGER
!>          La dimensión principal del array A.  LDA >= max(1,M).
!>

T

!>          T es un array REAL, de dimensión (LDT,N)
!>          La matriz triangular superior R de N por N del reflector de bloque.
!>          Los elementos en y por encima de la diagonal contienen el reflector
!>          de bloque T; los elementos por debajo de la diagonal no se utilizan.
!>          Ver más abajo para más detalles.
!>

LDT

!>          LDT es INTEGER
!>          La dimensión principal del array T.  LDT >= max(1,N).
!>

INFO

!>          INFO es INTEGER
!>          = 0: salida exitosa
!>          < 0: si INFO = -i, el i-ésimo argumento tenía un valor ilegal
!>

Detalles adicionales:

!>
!>  La matriz V almacena los reflectores elementales H(i) en la i-ésima
!>  columna por debajo de la diagonal. Por ejemplo, si M=5 y N=3, la matriz V es
!>
!>               V = (  1       )
!>                   ( v1  1    )
!>                   ( v1 v2  1 )
!>                   ( v1 v2 v3 )
!>                   ( v1 v2 v3 )
!>
!>  donde los vi's representan los vectores que definen H(i), que se devuelven
!>  en la matriz A. Los 1's a lo largo de la diagonal de V no se almacenan en A.
!>  El reflector de bloque H se da entonces por
!>
!>               H = I - V * T * V**T
!>
!>  donde V**T es la transpuesta de V.
!>
