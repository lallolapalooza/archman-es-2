subroutine sgbrfsx (character trans, character equed, integer n, integer kl, integer ku, integer nrhs, real, dimension( ldab, * ) ab, integer ldab, real, dimension( ldafb, * ) afb, integer ldafb, integer, dimension( * ) ipiv, real, dimension( * ) r, real, dimension( * ) c, real, dimension( ldb, * ) b, integer ldb, real, dimension( ldx , * ) x, integer ldx, real rcond, real, dimension( * ) berr, integer n_err_bnds, real, dimension( nrhs, * ) err_bnds_norm, real, dimension( nrhs, * ) err_bnds_comp, integer nparams, real, dimension( * ) params, real, dimension( * ) work, integer, dimension( * ) iwork, integer info)

SGBRFSX

Propósito:

!>
!>    SGBRFSX mejora la solución calculada de un sistema de ecuaciones
!>    lineales y proporciona límites de error y estimaciones de error
!>    hacia atrás para la solución. Además del límite de error normwise,
!>    el código proporciona un límite de error componente a componente máximo
!>    si es posible. Ver los comentarios de ERR_BNDS_NORM y ERR_BNDS_COMP
!>    para detalles sobre los límites de error.
!>
!>    El sistema original de ecuaciones lineales puede haber sido
!>    equilibrado antes de llamar a esta rutina, como se describe por
!>    los argumentos EQUED, R y C a continuación. En este caso, la
!>    solución y los límites de error devueltos son para el sistema
!>    original no equilibrado.
!> 
!>     Algunos parámetros opcionales se agrupan en el array PARAMS. Estas
!>     configuraciones determinan cómo se realiza el refinamiento, pero a
!>     menudo los valores predeterminados son aceptables. Si los valores
!>     predeterminados son aceptables, los usuarios pueden pasar NPARAMS = 0
!>     lo que evita que el código fuente acceda al argumento PARAMS.
!>

Parámetros

TRANS 
!>          TRANS is CHARACTER*1
!>     Especifica la forma del sistema de ecuaciones:
!>       = 'N':  A * X = B     (Sin transpuesta)
!>       = 'T':  A**T * X = B  (Transpuesta)
!>       = 'C':  A**H * X = B  (Transpuesta conjugada = Transpuesta)
!>

EQUED

!>          EQUED is CHARACTER*1
!>     Especifica la forma de equilibrado que se realizó en A
!>     antes de llamar a esta rutina. Esto es necesario para calcular
!>     la solución y los límites de error correctamente.
!>       = 'N':  Sin equilibrado
!>       = 'R':  Equilibrado de filas, es decir, A ha sido premultiplicado
!>               por diag(R).
!>       = 'C':  Equilibrado de columnas, es decir, A ha sido postmultiplicado
!>               por diag(C).
!>       = 'B':  Equilibrado tanto de filas como de columnas, es decir, A ha sido
!>               reemplazado por diag(R) * A * diag(C).
!>               El lado derecho B se ha modificado en consecuencia.
!>

N

!>          N is INTEGER
!>     El orden de la matriz A.  N >= 0.
!>

KL

!>          KL is INTEGER
!>     El número de subdiagonales dentro de la banda de A.  KL >= 0.
!>

KU

!>          KU is INTEGER
!>     El número de superdiagonales dentro de la banda de A.  KU >= 0.
!>

NRHS

!>          NRHS is INTEGER
!>     El número de lados derechos, es decir, el número de columnas
!>     de las matrices B y X.  NRHS >= 0.
!>

AB

!>          AB is REAL array, dimension (LDAB,N)
!>     La matriz de banda original A, almacenada en las filas 1 a KL+KU+1.
!>     La j-ésima columna de A se almacena en la j-ésima columna del
!>     array AB de la siguiente manera:
!>     AB(ku+1+i-j,j) = A(i,j) para max(1,j-ku)<=i<=min(n,j+kl).
!>

LDAB

!>          LDAB is INTEGER
!>     La dimensión principal del array AB.  LDAB >= KL+KU+1.
!>

AFB

!>          AFB is REAL array, dimension (LDAFB,N)
!>     Detalles de la factorización LU de la matriz de banda A, como
!>     calculado por SGBTRF. U se almacena como una matriz triangular
!>     superior de banda con KL+KU superdiagonales en las filas 1 a
!>     KL+KU+1, y los multiplicadores utilizados durante la
!>     factorización se almacenan en las filas KL+KU+2 a 2*KL+KU+1.
!>

LDAFB

!>          LDAFB is INTEGER
!>     La dimensión principal del array AFB.  LDAFB >= 2*KL*KU+1.
!>

IPIV

!>          IPIV is INTEGER array, dimension (N)
!>     Los índices de pivote de SGETRF; para 1<=i<=N, la fila i de la
!>     matriz se intercambió con la fila IPIV(i).
!>

R

!>          R is REAL array, dimension (N)
!>     Los factores de escala de fila para A. Si EQUED = 'R' o 'B', A se
!>     multiplica a la izquierda por diag(R); si EQUED = 'N' o 'C', R
!>     no se accede. R es un argumento de entrada si FACT = 'F';
!>     de lo contrario, R es un argumento de salida. Si FACT = 'F' y
!>     EQUED = 'R' o 'B', cada elemento de R debe ser positivo.
!>     Si R es de salida, cada elemento de R es una potencia de la base.
!>     Si R es de entrada, cada elemento de R debe ser una potencia de la base
!>     para garantizar una solución y estimaciones de error fiables. El
!>     escalado por potencias de la base no causa errores de redondeo
!>     a menos que el resultado sea un subdesbordamiento o
!>     desbordamiento. Los errores de redondeo durante el escalado
!>     conducen a refinar con una matriz que no es equivalente a la
!>     matriz de entrada, produciendo estimaciones de error que pueden no ser
!>     fiables.
!>

C

!>          C is REAL array, dimension (N)
!>     Los factores de escala de columna para A. Si EQUED = 'C' o 'B', A se
!>     multiplica a la derecha por diag(C); si EQUED = 'N' o 'R', C
!>     no se accede. C es un argumento de entrada si FACT = 'F';
!>     de lo contrario, C es un argumento de salida. Si FACT = 'F' y
!>     EQUED = 'C' o 'B', cada elemento de C debe ser positivo.
!>     Si C es de salida, cada elemento de C es una potencia de la base.
!>     Si C es de entrada, cada elemento de C debe ser una potencia de la base
!>     para garantizar una solución y estimaciones de error fiables. El
!>     escalado por potencias de la base no causa errores de redondeo
!>     a menos que el resultado sea un subdesbordamiento o
!>     desbordamiento. Los errores de redondeo durante el escalado
!>     conducen a refinar con una matriz que no es equivalente a la
!>     matriz de entrada, produciendo estimaciones de error que pueden no ser
!>     fiables.
!>

B

!>          B is REAL array, dimension (LDB,NRHS)
!>     La matriz del lado derecho B.
!>

LDB

!>          LDB is INTEGER
!>     La dimensión principal del array B.  LDB >= max(1,N).
!>

X

!>          X is REAL array, dimension (LDX,NRHS)
!>     En la entrada, la matriz solución X, como la calculó SGETRS.
!>     En la salida, la matriz solución mejorada X.
!>

LDX

!>          LDX is INTEGER
!>     La dimensión principal del array X.  LDX >= max(1,N).
!>

RCOND

!>          RCOND is REAL
!>     Condición número escalado recíproco. Esta es una estimación del
!>     número de condición de Skeel recíproco de la matriz A después de
!>     el equilibrado (si se realizó). Si este es menor que la precisión
!>     de la máquina (en particular, si es cero), la matriz es singular
!>     a la precisión de trabajo. Tenga en cuenta que el error puede ser
!>     todavía pequeño incluso si este número

SUBROUTINE SGBRFSX(

Parameters

UPLO
CHARACTER*1. Specifies whether the matrix stored in AB is upper or lower triangular.
INTEGER. The number of columns of the matrix AB.
INTEGER. The number of subdiagonals within the bandwidth of AB.
INTEGER. The number of superdiagonals within the bandwidth of AB.
INTEGER. The number of right hand sides, i.e., the number of columns of the matrices B, X and Z.
REAL array, dimension (LDAB,N)
On entry, the N-by-N general band matrix A, stored columnwise. The width of A is ML + MU + 1. If UPLO ='L' or 'l', the elements A(i,j) for i > j+ML and j > i+MU are not used.
On exit, the LU factorization of A, stored columnwise. The diagonal elements of U are stored in AB(ML+1,:).
INTEGER. The leading dimension of the array AB. LDAB >= ML + MU + 1.
REAL array, dimension (LDB,NRHS)
On entry, the right-hand side matrix B, dimension (N,NRHS).
INTEGER. The leading dimension of the array B. LDB >= max(N,N).
REAL array, dimension (LDX,NRHS)
On exit, the solution matrix X, dimension (N,NRHS).
INTEGER. The leading dimension of the array X. LDX >= max(N,N).
REAL array, dimension (NRHS)
On exit, the estimated forward error bound for each respective solution vector in X. If INFO = 0, then FERR(j) is an estimate of the relative forward error in the j-th solution vector X(j), and is bounded by 10*FERR(j) by approximately 100 percent of the random vectors, i.e., safeguarded downward. If UPLO ='L' or 'l', then FERR(j) is computed using the LU factorization of A. If UPLO ='U' or 'u', then FERR(j) is computed using the LU factorization of A.
REAL array, dimension (NRHS)
On exit, the componentwise relative backward error of each solution vector X(j) (i.e., the smallest relative change in any element of A or B that makes X(j) an exact solution).
REAL array, dimension (LDZ,NRHS)
On exit, the triangular factor U of the LU factorization of A, stored columnwise. The diagonal elements of U are in AB(ML+1,:).
INTEGER. The leading dimension of the array Z. LDZ >= N.
INTEGER array, dimension (NRHS)
Input flag indicating whether the system is to be solved.
INTEGER array, dimension (NRHS)
Output flag indicating whether the system was solved.
REAL array, dimension (3)
Specifies algorithm parameters.
PARAMS(1)
Whether to perform iterative refinement or not.
PARAMS(2)
Maximum number of residual computations allowed.
PARAMS(3)
Flag determining if the code will attempt to find a solution with small componentwise relative error.
INTEGER
Output flag.

Function/Subroutine Documentation

UPLO

!> UPLO is CHARACTER*1
!> On entry, UPLO specifies whether the matrix stored in AB is upper
!> or lower triangular, as follows:
!>    UPLO = 'U' or 'u'   Only the upper triangular part of AB is used.
!>    UPLO = 'L' or 'l'   Only the lower triangular part of AB is used.
!>
!> Unchanged on exit.
!>

N

!> N is INTEGER
!> The number of columns of the matrix AB.  N must be at least zero.
!> Unchanged on exit.
!>

ML

!> ML is INTEGER
!> The number of subdiagonals within the bandwidth of AB.
!> Unchanged on exit.
!>

MU

!> MU is INTEGER
!> The number of superdiagonals within the bandwidth of AB.
!> Unchanged on exit.
!>

NRHS

!> NRHS is INTEGER
!> The number of right hand sides, i.e., the number of columns of the
!> matrices B, X, and Z. NRHS must be at least zero.
!> Unchanged on exit.
!>

AB

!> AB is REAL array, dimension (LDAB,N)
!> On entry, the N-by-N general band matrix A, stored columnwise.  The
!> width of A is ML + MU + 1.
!>
!> If UPLO = 'L' or 'l', the entries A(i,j) for i < j - ML or j < i - ML
!> are not used.
!> If UPLO = 'U' or 'u', the entries A(i,j) for i > j + MU or j > i + MU
!> are not used.
!>
!> On exit, AB is overwritten by values generated during the factorization.
!>

LDAB

!> LDAB is INTEGER
!> The leading dimension of the array AB.  LDAB must be at least
!> ML + MU + 1.
!> Unchanged on exit.
!>

B

!> B is REAL array, dimension (LDB,NRHS)
!> On entry, the right hand side matrix B, dimension (N,NRHS).
!>
!> On exit, the N-by-NRHS right hand side matrix B may be overwritten by
!> the N-by-NRHS triangular factor U of the LU factorization of A, stored
!> columnwise. The diagonal elements of U are in AB(ML+1,:).
!>

LDB

!> LDB is INTEGER
!> The leading dimension of the array B. LDB must be at least
!> max(1,N).
!> Unchanged on exit.
!>

X

!> X is REAL array, dimension (LDX,NRHS)
!> On exit, the N-by-NRHS solution matrix X, dimension (N,NRHS).
!>

LDX

!> LDX is INTEGER
!> The leading dimension of the array X. LDX must be at least
!> max(1,N).
!> Unchanged on exit.
!>

FERR

!> FERR is REAL array, dimension (NRHS)
!> On exit, the estimated forward error bound for each respective solution
!> vector in X. If INFO = 0, then FERR(j) is an estimate of the relative
!> forward error in the j-th solution vector X(j), and is bounded by
!> 10*FERR(j) by approximately 100 percent of the random vectors, i.e.,
!> safeguarded downward.  If UPLO = 'L' or 'l', then FERR(j) is computed using
!> the LU factorization of A. If UPLO = 'U' or 'u', then FERR(j) is computed
!> using the LU factorization of A.
!>

BERR

!> BERR is REAL array, dimension (NRHS)
!> Componentwise relative backward error. This is the
!> componentwise relative backward error of each solution vector X(j)
!> (i.e., the smallest relative change in any element of A or B that
!> makes X(j) an exact solution).
!>

N_ERR_BNDS

!> N_ERR_BNDS is INTEGER
!> Number of error bounds to return for each right hand side
!> and each type (normwise or componentwise). See ERR_BNDS_NORM and
!> ERR_BNDS_COMP below.
!>

ERR_BNDS_NORM

!> ERR_BNDS_NORM is REAL array, dimension (NRHS, N_ERR_BNDS)
!> For each right-hand side, this array contains information about
!> various error bounds and condition numbers corresponding to the
!> normwise relative error, which is defined as follows:
!>
!> Normwise relative error in the ith solution vector:
!> max_j (abs(XTRUE(j,i) - X(j,i)))
!> ------------------------------
!> max_j abs(X(j,i))
!>
!> The array is indexed by the type of error information as described
!> below. There currently are up to three pieces of information
!> returned.
!>
!> The first index in ERR_BNDS_NORM(i,:) corresponds to the ith
!> right-hand side.
!>
!> The second index in ERR_BNDS_NORM(:,err) contains the following
!> three fields:
!> err = 1 boolean. Trust the answer if the
!> reciprocal condition number is less than the threshold
!> sqrt(n) * slamch('Epsilon').
!>
!> err = 2 error bound: The estimated forward error,
!> almost certainly within a factor of 10 of the true error
!> so long as the next entry is greater than the threshold
!> sqrt(n) * slamch('Epsilon'). This error bound should only
!> be trusted if the previous boolean is true.
!>
!> err = 3 Reciprocal condition number: Estimated normwise
!> reciprocal condition number. Compared with the threshold
!> sqrt(n) * slamch('Epsilon') to determine if the error
!> estimate is . These reciprocal condition
!> numbers are 1 / (norm(Z^{-1},inf) * norm(Z,inf)) for some
!> appropriately scaled matrix Z.
!> Let Z = S*A, where S scales each row by a power of the
!> radix so all absolute row sums of Z are approximately 1.
!>
!> See Lapack Working Note 165 for further details and extra
!> cautions.
!>

ERR_BNDS_COMP

!> ERR_BNDS_COMP is REAL array, dimension (NRHS, N_ERR_BNDS)
!> For each right-hand side, this array contains information about
!> various error bounds and condition numbers corresponding to the
!> componentwise relative error, which is defined as follows:
!>
!> Componentwise relative error in the ith solution vector:
!> abs(XTRUE(j,i) - X(j,i))
!> max_j ----------------------
!> abs(X(j,i))
!>
!> The array is indexed by the right-hand side i (on which the
!> componentwise relative error depends), and the type of error
!> information as described below. There currently are up to three
!> pieces of information returned for each right-hand side. If
!> componentwise accuracy is not requested (PARAMS(3) = 0.0), then
!> ERR_BNDS_COMP is not accessed. If N_ERR_BNDS < 3, then at most
!> the first (:,N_ERR_BNDS) entries are returned.
!>
!> The first index in ERR_BNDS_COMP(i,:) corresponds to the ith
!> right-hand side.
!>
!> The second index in ERR_BNDS_COMP(:,err) contains the following
!> three fields:
!> err = 1 boolean. Trust the answer if the
!> reciprocal condition number is less than the threshold
!> sqrt(n) * slamch('Epsilon').
!>
!> err = 2 error bound: The estimated forward error,
!> almost certainly within a factor of 10 of the true error
!> so long as the next entry is greater than the threshold
!> sqrt(n) * slamch('Epsilon'). This error bound should only
!> be trusted if the previous boolean is true.
!>
!> err = 3 Reciprocal condition number: Estimated componentwise
!> reciprocal condition number. Compared with the threshold
!> sqrt(n) * slamch('Epsilon') to determine if the error
!> estimate is . These reciprocal condition
!> numbers are 1 / (norm(Z^{-1},inf) * norm(Z,inf)) for some
!> appropriately scaled matrix Z.
!> Let Z = S*(A*diag(x)), where x is the solution for the
!> current right-hand side and S scales each row of
!> A*diag(x) by a power of the radix so all absolute row
!> sums of Z are approximately 1.
!>
!> See Lapack Working Note 165 for further details and extra
!> cautions.
!>

NPARAMS

!> NPARAMS is INTEGER
!> Specifies the number of parameters set in PARAMS. If <= 0, the
!> PARAMS array is never referenced and default values are used.
!>

PARAMS

!> PARAMS is REAL array, dimension NPARAMS
!> Specifies algorithm parameters. If an entry is < 0.0, then
!> that entry will be filled with default value used for that
!> parameter. Only positions up to NPARAMS are accessed; defaults
!> are used for higher-numbered parameters.
!>
!>   PARAMS(LA_LINRX_ITREF_I = 1) : Whether to perform iterative
!>        refinement or not.
!>     Default: 1.0
!>       = 0.0: No refinement is performed, and no error bounds are
!>             computed.
!>       = 1.0: Use the double-precision refinement algorithm,
!>             possibly with doubled-single computations if the
!>             compilation environment does not support DOUBLE
!>             PRECISION.
!>       (other values are reserved for future use)
!>
!>   PARAMS(LA_LINRX_ITHRESH_I = 2) : Maximum number of residual
!>        computations allowed for refinement.
!>     Default: 10
!>     Aggressive: Set to 100 to permit convergence using approximate
!>                 factorizations or factorizations other than LU. If
!>                 the factorization uses a technique other than
!>                 Gaussian elimination, the guarantees in
!>                 err_bnds_norm and err_bnds_comp may no longer be
!>                 trustworthy.
!>
!>   PARAMS(LA_LINRX_CWISE_I = 3) : Flag determining if the code
!>        will attempt to find a solution with small componentwise
!>        relative error in the double-precision algorithm. Positive
!>        is true, 0.0 is false.
!>     Default: 1.0 (attempt componentwise convergence)
!>

WORK

!> WORK is REAL array, dimension (4*N)
!>

IWORK

!> IWORK is INTEGER array, dimension (N)
!>

INFO

!> INFO is INTEGER
!> = 0: Successful exit. The solution to every right-hand side is
!> guaranteed.
!> < 0: If INFO = -i, the i-th argument had an illegal value
!> > 0 and <= N: U(INFO,INFO) is exactly zero. The factorization
!> has been completed, but the factor U is exactly singular, so
!> the solution and error bounds could not be computed. RCOND = 0
!> is returned.
!> = N+J: The solution corresponding to the Jth right-hand side is
!> not guaranteed. The solutions corresponding to other right-
!> hand sides K with K > J may not be guaranteed as well, but
!> only the first such right-hand side is reported. If a small
!> componentwise error is not requested (PARAMS(3) = 0.0) then
!> the Jth right-hand side is the first with a normwise error
!> bound that is not guaranteed (the smallest J such
!> that ERR_BNDS_NORM(J,1) = 0.0). By default (PARAMS(3) = 1.0)
!> the Jth right-hand side is the first with either a normwise or
!> componentwise error bound that is not guaranteed (the smallest
!> J such that either ERR_BNDS_NORM(J,1) = 0.0 or
!> ERR_BNDS_COMP(J,1) = 0.0). See the definition of
!> ERR_BNDS_NORM(:,1) and ERR_BNDS_COMP(:,1). To get information
!> about all of the right-hand sides check ERR_BNDS_NORM or
!> ERR_BNDS_COMP.
!>
