Documentación de la función/subrutina

subroutine sgelqs (integer m, integer n, integer nrhs, real, dimension( lda, * ) a, integer lda, real, dimension( * ) tau, real, dimension( ldb, * ) b, integer ldb, real, dimension( lwork ) work, integer lwork, integer info)

SGELQS

Propósito:

!>
!> Calcula una solución de norma mínima
!>     min || A*X - B ||
!> usando la factorización LQ
!>     A = L*Q
!> calculada por SGELQF.
!>

Parámetros

M 
!>          M es INTEGER
!>          El número de filas de la matriz A.  M >= 0.
!>

N

!>          N es INTEGER
!>          El número de columnas de la matriz A.  N >= M >= 0.
!>

NRHS

!>          NRHS es INTEGER
!>          El número de columnas de B.  NRHS >= 0.
!>

A

!>          A es array REAL, dimension (LDA,N)
!>          Detalles de la factorización LQ de la matriz original A como
!>          devuelto por SGELQF.
!>

LDA

!>          LDA es INTEGER
!>          La dimensión principal del array A.  LDA >= M.
!>

TAU

!>          TAU es array REAL, dimension (M)
!>          Detalles de la matriz ortogonal Q.
!>

B

!>          B es array REAL, dimension (LDB,NRHS)
!>          En la entrada, la matriz B de mano derecha de m por nrhs.
!>          En la salida, la matriz solución X de n por nrhs.
!>

LDB

!>          LDB es INTEGER
!>          La dimensión principal del array B. LDB >= N.
!>

WORK

!>          WORK es array REAL, dimension (LWORK)
!>

LWORK

!>          LWORK es INTEGER
!>          La longitud del array WORK.  LWORK debe ser al menos NRHS,
!>          y debe ser al menos NRHS*NB, donde NB es el tamaño del bloque
!>          para este entorno.
!>

INFO

!>          INFO es INTEGER
!>          = 0: salida exitosa
!>          < 0: si INFO = -i, el i-ésimo argumento tenía un valor ilegal
!>
