subroutine sgerfs (character trans, integer n, integer nrhs, real, dimension( lda, * ) a, integer lda, real, dimension( ldaf, * ) af, integer ldaf, integer, dimension( * ) ipiv, real, dimension( ldb, * ) b, integer ldb, real, dimension( ldx, * ) x, integer ldx, real, dimension( * ) ferr, real, dimension( * ) berr, real, dimension( * ) work, integer, dimension( * ) iwork, integer info)

SGERFS

Propósito:

!>
!> SGERFS mejora la solución calculada de un sistema de ecuaciones lineales
!> y proporciona cotas de error y estimaciones de error hacia atrás para
!> la solución.
!>

Parámetros

TRANS 
!>          TRANS es CHARACTER*1
!>          Especifica la forma del sistema de ecuaciones:
!>          = 'N':  A * X = B     (Sin transpuesta)
!>          = 'T':  A**T * X = B  (Transpuesta)
!>          = 'C':  A**H * X = B  (Transpuesta conjugada = Transpuesta)
!>

N

!>          N es INTEGER
!>          El orden de la matriz A.  N >= 0.
!>

NRHS

!>          NRHS es INTEGER
!>          El número de lados derechos, es decir, el número de columnas
!>          de las matrices B y X.  NRHS >= 0.
!>

A

!>          A es un array REAL, de dimensión (LDA,N)
!>          La matriz original A de N por N.
!>

LDA

!>          LDA es INTEGER
!>          La dimensión principal del array A.  LDA >= max(1,N).
!>

AF

!>          AF es un array REAL, de dimensión (LDAF,N)
!>          Los factores L y U de la factorización A = P*L*U
!>          calculada por SGETRF.
!>

LDAF

!>          LDAF es INTEGER
!>          La dimensión principal del array AF.  LDAF >= max(1,N).
!>

IPIV

!>          IPIV es un array INTEGER, de dimensión (N)
!>          Los índices de pivote de SGETRF; para 1<=i<=N, la fila i de la
!>          matriz se intercambió con la fila IPIV(i).
!>

B

!>          B es un array REAL, de dimensión (LDB,NRHS)
!>          La matriz del lado derecho B.
!>

LDB

!>          LDB es INTEGER
!>          La dimensión principal del array B.  LDB >= max(1,N).
!>

X

!>          X es un array REAL, de dimensión (LDX,NRHS)
!>          En la entrada, la matriz solución X, calculada por SGETRS.
!>          En la salida, la matriz solución mejorada X.
!>

LDX

!>          LDX es INTEGER
!>          La dimensión principal del array X.  LDX >= max(1,N).
!>

FERR

!>          FERR es un array REAL, de dimensión (NRHS)
!>          La cota estimada del error hacia adelante para cada vector solución
!>          X(j) (la columna j de la matriz solución X).
!>          Si XTRUE es la solución verdadera correspondiente a X(j), FERR(j)
!>          es una cota superior estimada para la magnitud del elemento
!>          más grande en (X(j) - XTRUE) dividido por la magnitud del
!>          elemento más grande en X(j). La estimación es tan fiable como
!>          la estimación de RCOND, y es casi siempre una ligera
!>          sobrestimación del error real.
!>

BERR

!>          BERR es un array REAL, de dimensión (NRHS)
!>          El error relativo componente a componente hacia atrás de cada vector solución
!>          X(j) (es decir, el menor cambio relativo en
!>          cualquier elemento de A o B que haga que X(j) sea una solución exacta).
!>

WORK

!>          WORK es un array REAL, de dimensión (3*N)
!>

IWORK

!>          IWORK es un array INTEGER, de dimensión (N)
!>

INFO

!>          INFO es INTEGER
!>          = 0:  salida exitosa
!>          < 0:  si INFO = -i, el argumento i-ésimo tenía un valor ilegal
!>

Parámetros Internos:

!>  ITMAX es el número máximo de pasos de refinamiento iterativo.
!>
