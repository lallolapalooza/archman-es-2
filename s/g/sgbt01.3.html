DOCUMENTACIÓN DE FUNCIÓN/SUBRUTINA

subroutine sgbt01 (integer m, integer n, integer kl, integer ku, real, dimension( lda, * ) a, integer lda, real, dimension( ldafac, * ) afac, integer ldafac, integer, dimension( * ) ipiv, real, dimension( * ) work, real resid)

SGBT01

Propósito:

!>
!> SGBT01 reconstruye una matriz de banda A a partir de su factorización L*U y
!> calcula el residuo:
!>    norm(L*U - A) / ( N * norm(A) * EPS ),
!> donde EPS es la epsilon de la máquina.
!>
!> La expresión L*U - A se calcula una columna a la vez, por lo que A y
!> AFAC no se modifican.
!>

Parámetros

M 
!>          M es INTEGER
!>          El número de filas de la matriz A.  M >= 0.
!>

N

!>          N es INTEGER
!>          El número de columnas de la matriz A.  N >= 0.
!>

KL

!>          KL es INTEGER
!>          El número de subdiagonales dentro de la banda de A.  KL >= 0.
!>

KU

!>          KU es INTEGER
!>          El número de superdiagonales dentro de la banda de A.  KU >= 0.
!>

A

!>          A es array REAL, dimension (LDA,N)
!>          La matriz original A en almacenamiento de banda, almacenada en las
!>          filas 1 a KL+KU+1.
!>

LDA

!>          LDA es INTEGER.
!>          La dimensión principal del array A.  LDA >= max(1,KL+KU+1).
!>

AFAC

!>          AFAC es array REAL, dimension (LDAFAC,N)
!>          La forma factorizada de la matriz A.  AFAC contiene los factores
!>          de banda L y U de la factorización L*U, tal como se calculó con
!>          SGBTRF. U se almacena como una matriz de banda triangular superior
!>          con KL+KU superdiagonales en las filas 1 a KL+KU+1, y los
!>          multiplicadores utilizados durante la factorización se almacenan
!>          en las filas KL+KU+2 a 2*KL+KU+1. Ver SGBTRF para más detalles.
!>

LDAFAC

!>          LDAFAC es INTEGER
!>          La dimensión principal del array AFAC.
!>          LDAFAC >= max(1,2*KL*KU+1).
!>

IPIV

!>          IPIV es array INTEGER, dimension (min(M,N))
!>          Los índices de pivote de SGBTRF.
!>

WORK

!>          WORK es array REAL, dimension (2*KL+KU+1)
!>

RESID

!>          RESID es REAL
!>          norm(L*U - A) / ( N * norm(A) * EPS )
!>
