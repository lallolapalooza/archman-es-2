subroutine sgelqf (integer m, integer n, real, dimension( lda, * ) a, integer lda, real, dimension( * ) tau, real, dimension( * ) work, integer lwork, integer info)

SGELQF

Propósito:

!>
!> SGELQF calcula una factorización LQ de una matriz real M por N, A:
!>
!>    A = ( L 0 ) *  Q
!>
!> donde:
!>
!>    Q es una matriz ortogonal N por N;
!>    L es una matriz triangular inferior M por M;
!>    0 es una matriz cero M por (N-M), si M < N.
!>
!>

Parámetros

M 
!>          M es INTEGER
!>          El número de filas de la matriz A.  M >= 0.
!>

N

!>          N es INTEGER
!>          El número de columnas de la matriz A.  N >= 0.
!>

A

!>          A es un array REAL, de dimensión (LDA,N)
!>          En la entrada, la matriz M por N, A.
!>          En la salida, los elementos en y debajo de la diagonal del array
!>          contienen la matriz trapezoidal inferior M por min(M,N), L (L es
!>          triangular inferior si m <= n); los elementos por encima de la diagonal,
!>          junto con el array TAU, representan la matriz ortogonal Q como un
!>          producto de reflectores elementales (ver Detalles Adicionales).
!>

LDA

!>          LDA es INTEGER
!>          La dimensión principal del array A.  LDA >= max(1,M).
!>

TAU

!>          TAU es un array REAL, de dimensión (min(M,N))
!>          Los factores escalares de los reflectores elementales (ver Detalles
!>          Adicionales).
!>

WORK

!>          WORK es un array REAL, de dimensión (MAX(1,LWORK))
!>          En la salida, si INFO = 0, WORK(1) devuelve el LWORK óptimo.
!>

LWORK

!>          LWORK es INTEGER
!>          La dimensión del array WORK.
!>          LWORK >= 1, si MIN(M,N) = 0, y LWORK >= M en caso contrario.
!>          Para un rendimiento óptimo LWORK >= M*NB, donde NB es el
!>          tamaño de bloque óptimo.
!>
!>          Si LWORK = -1, entonces se asume una consulta de espacio de trabajo; la rutina
!>          solo calcula el tamaño óptimo del array WORK, devuelve
!>          este valor como la primera entrada del array WORK, y XERBLA no emite ningún mensaje de error
!>          relacionado con LWORK.
!>

INFO

!>          INFO es INTEGER
!>          = 0:  salida exitosa
!>          < 0:  si INFO = -i, el i-ésimo argumento tenía un valor ilegal
!>

Detalles Adicionales:

!>
!>  La matriz Q se representa como un producto de reflectores elementales
!>
!>     Q = H(k) . . . H(2) H(1), donde k = min(m,n).
!>
!>  Cada H(i) tiene la forma
!>
!>     H(i) = I - tau * v * v**T
!>
!>  donde tau es un escalar real, y v es un vector real con
!>  v(1:i-1) = 0 y v(i) = 1; v(i+1:n) se almacena en la salida en A(i,i+1:n),
!>  y tau en TAU(i).
!>
