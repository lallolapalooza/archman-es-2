subroutine sgetrf (integer m, integer n, real, dimension( lda, * ) a, integer lda, integer, dimension( * ) ipiv, integer info)

SGETRF

Propósito:

!>
!> SGETRF calcula una factorización LU de una matriz general M por N, A,
!> utilizando pivoteo parcial con intercambios de filas.
!>
!> La factorización tiene la forma
!>    A = P * L * U
!> donde P es una matriz de permutación, L es triangular inferior con elementos
!> unitarios en la diagonal (trapezoidal inferior si m > n) y U es triangular
!> superior (trapezoidal superior si m < n).
!>
!> Esta es la versión de algoritmo de BLAS de Nivel 3 de tipo "right-looking".
!>

Parámetros

M 
!>          M es INTEGER
!>          El número de filas de la matriz A.  M >= 0.
!>

N

!>          N es INTEGER
!>          El número de columnas de la matriz A.  N >= 0.
!>

A

!>          A es un array REAL de dimensión (LDA,N)
!>          En la entrada, la matriz M por N a factorizar.
!>          En la salida, los factores L y U de la factorización
!>          A = P*L*U; los elementos unitarios de la diagonal de L no se almacenan.
!>

LDA

!>          LDA es INTEGER
!>          La dimensión principal del array A.  LDA >= max(1,M).
!>

IPIV

!>          IPIV es un array INTEGER de dimensión (min(M,N))
!>          Los índices de pivote; para 1 <= i <= min(M,N), la fila i de la
!>          matriz fue intercambiada con la fila IPIV(i).
!>

INFO

!>          INFO es INTEGER
!>          = 0:  salida exitosa
!>          < 0:  si INFO = -i, el i-ésimo argumento tenía un valor ilegal
!>          > 0:  si INFO = i, U(i,i) es exactamente cero. La factorización
!>                se ha completado, pero el factor U es exactamente
!>                singular, y ocurrirá una división por cero si se utiliza
!>                para resolver un sistema de ecuaciones.
!>
