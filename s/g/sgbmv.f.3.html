DOCUMENTACIÓN DE FUNCIÓN/SUBRUTINA

subroutine sgbmv (character trans, integer m, integer n, integer kl, integer ku, real alpha, real, dimension(lda,*) a, integer lda, real, dimension(*) x, integer incx, real beta, real, dimension(*) y, integer incy)

SGBMV

Propósito:

!>
!> SGBMV  realiza una de las operaciones de matriz-vector
!>
!>    y := alpha*A*x + beta*y,   o   y := alpha*A**T*x + beta*y,
!>
!> donde alpha y beta son escalares, x e y son vectores y A es una
!> matriz de banda de m por n, con kl sub-diagonales y ku super-diagonales.
!>

Parámetros

TRANS 
!>          TRANS es CHARACTER*1
!>           Al entrar, TRANS especifica la operación a realizar como
!>           sigue:
!>
!>              TRANS = 'N' o 'n'   y := alpha*A*x + beta*y.
!>
!>              TRANS = 'T' o 't'   y := alpha*A**T*x + beta*y.
!>
!>              TRANS = 'C' o 'c'   y := alpha*A**T*x + beta*y.
!>

M

!>          M es INTEGER
!>           Al entrar, M especifica el número de filas de la matriz A.
!>           M debe ser al menos cero.
!>

N

!>          N es INTEGER
!>           Al entrar, N especifica el número de columnas de la matriz A.
!>           N debe ser al menos cero.
!>

KL

!>          KL es INTEGER
!>           Al entrar, KL especifica el número de sub-diagonales de la
!>           matriz A. KL debe satisfacer  0 .le. KL.
!>

KU

!>          KU es INTEGER
!>           Al entrar, KU especifica el número de super-diagonales de la
!>           matriz A. KU debe satisfacer  0 .le. KU.
!>

ALPHA

!>          ALPHA es REAL
!>           Al entrar, ALPHA especifica el escalar alpha.
!>

A

!>          A es array REAL, dimension ( LDA, N )
!>           Antes de entrar, la parte principal ( kl + ku + 1 ) por n
!>           del array A debe contener la matriz de coeficientes,
!>           suministrada columna por columna, con la diagonal principal
!>           de la matriz en la fila ( ku + 1 ) del array, la primera
!>           super-diagonal comenzando en la posición 2 en la fila ku,
!>           la primera sub-diagonal comenzando en la posición 1 en la fila
!>           ( ku + 2 ), y así sucesivamente.
!>           Los elementos del array A que no corresponden a elementos
!>           de la matriz de banda (como el triángulo superior izquierdo
!>           ku por ku) no se referencian.
!>           El siguiente segmento de programa transferirá una matriz de
!>           banda desde almacenamiento de matriz completa convencional
!>           a almacenamiento de banda:
!>
!>                 DO 20, J = 1, N
!>                    K = KU + 1 - J
!>                    DO 10, I = MAX( 1, J - KU ), MIN( M, J + KL )
!>                       A( K + I, J ) = matrix( I, J )
!>              10    CONTINUE
!>              20 CONTINUE
!>

LDA

!>          LDA es INTEGER
!>           Al entrar, LDA especifica la primera dimensión de A tal como
!>           se declaró en el programa (sub)llamador. LDA debe ser al menos
!>           ( kl + ku + 1 ).
!>

X

!>          X es array REAL, dimension al menos
!>           ( 1 + ( n - 1 )*abs( INCX ) ) cuando TRANS = 'N' o 'n'
!>           y al menos
!>           ( 1 + ( m - 1 )*abs( INCX ) ) en caso contrario.
!>           Antes de entrar, el array X incrementado debe contener el
!>           vector x.
!>

INCX

!>          INCX es INTEGER
!>           Al entrar, INCX especifica el incremento para los elementos de
!>           X. INCX no debe ser cero.
!>

BETA

!>          BETA es REAL
!>           Al entrar, BETA especifica el escalar beta. Cuando BETA se
!>           suministra como cero, entonces Y no necesita ser configurado
!>           en la entrada.
!>

Y

!>          Y es array REAL, dimension al menos
!>           ( 1 + ( m - 1 )*abs( INCY ) ) cuando TRANS = 'N' o 'n'
!>           y al menos
!>           ( 1 + ( n - 1 )*abs( INCY ) ) en caso contrario.
!>           Antes de entrar, el array Y incrementado debe contener el
!>           vector y. A la salida, Y se sobrescribe con el vector y
!>           actualizado.
!>           Si m o n es cero, entonces Y no se referencia y la función
!>           realiza un retorno rápido.
!>

INCY

!>          INCY es INTEGER
!>           Al entrar, INCY especifica el incremento para los elementos de
!>           Y. INCY no debe ser cero.
!>
