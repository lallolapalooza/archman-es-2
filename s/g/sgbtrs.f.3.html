subroutine sgbtrs (character trans, integer n, integer kl, integer ku, integer nrhs, real, dimension( ldab, * ) ab, integer ldab, integer, dimension( * ) ipiv, real, dimension( ldb, * ) b, integer ldb, integer info)

SGBTRS

Propósito:

!>
!> SGBTRS resuelve un sistema de ecuaciones lineales
!>    A * X = B  o  A**T * X = B
!> con una matriz de banda general A utilizando la factorización LU calculada
!> por SGBTRF.
!>

Parámetros

TRANS 
!>          TRANS es CHARACTER*1
!>          Especifica la forma del sistema de ecuaciones.
!>          = 'N':  A * X = B  (Sin transposición)
!>          = 'T':  A**T* X = B  (Transpuesta)
!>          = 'C':  A**T* X = B  (Transpuesta conjugada = Transpuesta)
!>

N

!>          N es INTEGER
!>          El orden de la matriz A.  N >= 0.
!>

KL

!>          KL es INTEGER
!>          El número de subdiagonales dentro de la banda de A.  KL >= 0.
!>

KU

!>          KU es INTEGER
!>          El número de superdiagonales dentro de la banda de A.  KU >= 0.
!>

NRHS

!>          NRHS es INTEGER
!>          El número de lados derechos, es decir, el número de columnas
!>          de la matriz B.  NRHS >= 0.
!>

AB

!>          AB es array REAL, dimension (LDAB,N)
!>          Detalles de la factorización LU de la matriz de banda A, tal como
!>          calculada por SGBTRF.  U se almacena como una matriz de banda triangular
!>          superior con KL+KU superdiagonales en las filas 1 a KL+KU+1, y
!>          los multiplicadores utilizados durante la factorización se almacenan en
!>          las filas KL+KU+2 a 2*KL+KU+1.
!>

LDAB

!>          LDAB es INTEGER
!>          La dimensión principal del array AB.  LDAB >= 2*KL+KU+1.
!>

IPIV

!>          IPIV es array INTEGER, dimension (N)
!>          Los índices de pivote; para 1 <= i <= N, la fila i de la matriz fue
!>          intercambiada con la fila IPIV(i).
!>

B

!>          B es array REAL, dimension (LDB,NRHS)
!>          En la entrada, la matriz del lado derecho B.
!>          En la salida, la matriz solución X.
!>

LDB

!>          LDB es INTEGER
!>          La dimensión principal del array B.  LDB >= max(1,N).
!>

INFO

!>          INFO es INTEGER
!>          = 0:  salida exitosa
!>          < 0: si INFO = -i, el i-ésimo argumento tenía un valor ilegal
!>
