Documentación de Función/Subrutina

subroutine sgrqts (integer m, integer p, integer n, real, dimension( lda, * ) a, real, dimension( lda, * ) af, real, dimension( lda, * ) q, real, dimension( lda, * ) r, integer lda, real, dimension( * ) taua, real, dimension( ldb, * ) b, real, dimension( ldb, * ) bf, real, dimension( ldb, * ) z, real, dimension( ldb, * ) t, real, dimension( ldb, * )bwk, integer ldb, real, dimension( * ) taub, real, dimension( lwork ) work, integer lwork, real, dimension( * ) rwork, real, dimension( 4 ) result)

SGRQTS

Propósito:

!>
!> SGRQTS prueba SGGRQF, que calcula la factorización GRQ de una
!> matriz A de M por N y una matriz B de P por N: A = R*Q y B = Z*T*Q.
!>

Parámetros

M 
!>          M es INTEGER
!>          El número de filas de la matriz A.  M >= 0.
!>

P

!>          P es INTEGER
!>          El número de filas de la matriz B.  P >= 0.
!>

N

!>          N es INTEGER
!>          El número de columnas de las matrices A y B.  N >= 0.
!>

A

!>          A es array REAL, dimension (LDA,N)
!>          La matriz A de M por N.
!>

AF

!>          AF es array REAL, dimension (LDA,N)
!>          Detalles de la factorización GRQ de A y B, como se devuelve
!>          por SGGRQF, ver SGGRQF para más detalles.
!>

Q

!>          Q es array REAL, dimension (LDA,N)
!>          La matriz ortogonal Q de N por N.
!>

R

!>          R es array REAL, dimension (LDA,MAX(M,N))
!>

LDA

!>          LDA es INTEGER
!>          La dimensión principal de los arrays A, AF, R y Q.
!>          LDA >= max(M,N).
!>

TAUA

!>          TAUA es array REAL, dimension (min(M,N))
!>          Los factores escalares de los reflectores elementales, como se devuelve
!>          por SGGQRC.
!>

B

!>          B es array REAL, dimension (LDB,N)
!>          En la entrada, la matriz A de P por N.
!>

BF

!>          BF es array REAL, dimension (LDB,N)
!>          Detalles de la factorización GQR de A y B, como se devuelve
!>          por SGGRQF, ver SGGRQF para más detalles.
!>

Z

!>          Z es array REAL, dimension (LDB,P)
!>          La matriz ortogonal Z de P por P.
!>

T

!>          T es array REAL, dimension (LDB,max(P,N))
!>

BWK

!>          BWK es array REAL, dimension (LDB,N)
!>

LDB

!>          LDB es INTEGER
!>          La dimensión principal de los arrays B, BF, Z y T.
!>          LDB >= max(P,N).
!>

TAUB

!>          TAUB es array REAL, dimension (min(P,N))
!>          Los factores escalares de los reflectores elementales, como se devuelve
!>          por SGGRQF.
!>

WORK

!>          WORK es array REAL, dimension (LWORK)
!>

LWORK

!>          LWORK es INTEGER
!>          La dimensión del array WORK, LWORK >= max(M,P,N)**2.
!>

RWORK

!>          RWORK es array REAL, dimension (M)
!>

RESULT

!>          RESULT es array REAL, dimension (4)
!>          Las relaciones de prueba:
!>            RESULT(1) = norm( R - A*Q' ) / ( MAX(M,N)*norm(A)*ULP)
!>            RESULT(2) = norm( T*Q - Z'*B ) / (MAX(P,N)*norm(B)*ULP)
!>            RESULT(3) = norm( I - Q'*Q ) / ( N*ULP )
!>            RESULT(4) = norm( I - Z'*Z ) / ( P*ULP )
!>
