subroutine sggbak (character job, character side, integer n, integer ilo, integer ihi, real, dimension( * ) lscale, real, dimension( * ) rscale, integer m, real, dimension( ldv, * ) v, integer ldv, integer info)

SGGBAK

Propósito:

!>
!> SGGBAK forma los vectores propios derechos o izquierdos de un problema
!> generalizado de valores propios real A*x = lambda*B*x, mediante
!> transformación inversa sobre los vectores propios calculados del par
!> balanceado de matrices devuelto por SGGBAL.
!>

Parámetros

JOB 
!>          JOB es CHARACTER*1
!>          Especifica el tipo de transformación inversa requerida:
!>          = 'N':  no hacer nada, retornar inmediatamente;
!>          = 'P':  hacer la transformación inversa solo para la permutación;
!>          = 'S':  hacer la transformación inversa solo para el escalado;
!>          = 'B':  hacer las transformaciones inversas tanto para la
!>                  permutación como para el escalado.
!>          JOB debe ser el mismo que el argumento JOB suministrado a SGGBAL.
!>

SIDE

!>          SIDE es CHARACTER*1
!>          = 'R':  V contiene vectores propios derechos;
!>          = 'L':  V contiene vectores propios izquierdos.
!>

N

!>          N es INTEGER
!>          El número de filas de la matriz V. N >= 0.
!>

ILO

!>          ILO es INTEGER
!>

IHI

!>          IHI es INTEGER
!>          Los enteros ILO e IHI determinados por SGGBAL.
!>          1 <= ILO <= IHI <= N, si N > 0; ILO=1 e IHI=0, si N=0.
!>

LSCALE

!>          LSCALE es un array REAL, dimension (N)
!>          Detalles de las permutaciones y/o factores de escala aplicados
!>          al lado izquierdo de A y B, como se devuelve por SGGBAL.
!>

RSCALE

!>          RSCALE es un array REAL, dimension (N)
!>          Detalles de las permutaciones y/o factores de escala aplicados
!>          al lado derecho de A y B, como se devuelve por SGGBAL.
!>

M

!>          M es INTEGER
!>          El número de columnas de la matriz V. M >= 0.
!>

V

!>          V es un array REAL, dimension (LDV,M)
!>          En la entrada, la matriz de vectores propios derechos o izquierdos
!>          a transformar, como se devuelve por STGEVC.
!>          En la salida, V se sobrescribe con los vectores propios
!>          transformados.
!>

LDV

!>          LDV es INTEGER
!>          La dimensión principal de la matriz V. LDV >= max(1,N).
!>

INFO

!>          INFO es INTEGER
!>          = 0:  salida exitosa.
!>          < 0:  si INFO = -i, el i-ésimo argumento tuvo un valor ilegal.
!>
