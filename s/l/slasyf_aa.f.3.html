subroutine slasyf_aa (character uplo, integer j1, integer m, integer nb, real, dimension( lda, * ) a, integer lda, integer, dimension( * ) ipiv, real, dimension( ldh, * ) h, integer ldh, real, dimension( * ) work)

SLASYF_AA

Propósito:

!>
!> DLATRF_AA factoriza un panel de una matriz simétrica real A utilizando
!> el algoritmo de Aasen. El panel consiste en un conjunto de NB filas de A
!> cuando UPLO es U, o un conjunto de NB columnas cuando UPLO es L.
!>
!> Para factorizar el panel, el algoritmo de Aasen requiere la
!> última fila, o columna, del panel anterior. La primera fila, o columna,
!> de A se establece como la primera fila, o columna, de una matriz identidad,
!> que se utiliza para factorizar el primer panel.
!>
!> La J-ésima fila resultante de U, o la J-ésima columna de L, se almacena en la
!> (J-1)-ésima fila, o columna, de A (sin las diagonales unitarias), mientras
!> que la diagonal y subdiagonal de A se sobrescriben con las de T.
!>
!>

Parámetros

UPLO 
!>          UPLO es CHARACTER*1
!>          = 'U':  El triángulo superior de A está almacenado;
!>          = 'L':  El triángulo inferior de A está almacenado.
!>

J1

!>          J1 es INTEGER
!>          La ubicación de la primera fila, o columna, del panel
!>          dentro de la submatriz de A, pasada a esta rutina, por ejemplo,
!>          cuando se llama desde SSYTRF_AA, para el primer panel, J1 es 1,
!>          mientras que para los paneles restantes, J1 es 2.
!>

M

!>          M es INTEGER
!>          La dimensión de la submatriz. M >= 0.
!>

NB

!>          NB es INTEGER
!>          La dimensión del panel a factorizar.
!>

A

!>          A es una matriz REAL, de dimensión (LDA,M) para
!>          el primer panel, mientras que de dimensión (LDA,M+1) para los
!>          paneles restantes.
!>
!>          A la entrada, A contiene la última fila, o columna, del
!>          panel anterior, y la submatriz final de A
!>          a factorizar, excepto para el primer panel, solo
!>          se pasa el panel.
!>
!>          A la salida, el panel principal está factorizado.
!>

LDA

!>          LDA es INTEGER
!>          La dimensión principal de la matriz A.  LDA >= max(1,M).
!>

IPIV

!>          IPIV es una matriz INTEGER, de dimensión (M)
!>          Detalles de los intercambios de filas y columnas,
!>          las filas y columnas k fueron intercambiadas con las filas y
!>          columnas IPIV(k).
!>

H

!>          H es un espacio de trabajo REAL, de dimensión (LDH,NB).
!>
!>

LDH

!>          LDH es INTEGER
!>          La dimensión principal del espacio de trabajo H. LDH >= max(1,M).
!>

WORK

!>          WORK es un espacio de trabajo REAL, de dimensión (M).
!>
