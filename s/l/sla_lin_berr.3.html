subroutine sla_lin_berr (integer n, integer nz, integer nrhs, real, dimension( n, nrhs ) res, real, dimension( n, nrhs ) ayb, real, dimension( nrhs ) berr)

SLA_LIN_BERR calcula un error residual relativo componente a componente.

Propósito:

!>
!>    SLA_LIN_BERR calcula el error residual relativo componente a componente
!>    a partir de la fórmula
!>        max(i) ( abs(R(i)) / ( abs(op(A_s))*abs(Y) + abs(B_s) )(i) )
!>    donde abs(Z) es el valor absoluto componente a componente de la matriz
!>    o vector Z.
!>

Parámetros

N 
!>          N is INTEGER
!>     El número de ecuaciones lineales, es decir, el orden de la
!>     matriz A.  N >= 0.
!>

NZ

!>          NZ is INTEGER
!>     A R(i) le añadimos (NZ+1)*SLAMCH( 'Safe minimum' ) en el numerador para
!>     evitar residuos espuriamente cero. El valor por defecto es N.
!>

NRHS

!>          NRHS is INTEGER
!>     El número de lados derechos, es decir, el número de columnas
!>     de las matrices AYB, RES y BERR.  NRHS >= 0.
!>

RES

!>          RES is REAL array, dimension (N,NRHS)
!>     La matriz residual, es decir, la matriz R en la fórmula del error
!>     residual relativo de arriba.
!>

AYB

!>          AYB is REAL array, dimension (N, NRHS)
!>     El denominador en la fórmula del error residual relativo de arriba,
!>     es decir, la matriz abs(op(A_s))*abs(Y) + abs(B_s). Las matrices A, Y y B
!>     provienen del refinamiento iterativo (ver sla_gerfsx_extended.f).
!>

BERR

!>          BERR is REAL array, dimension (NRHS)
!>     El error residual relativo componente a componente de la fórmula anterior.
!>
