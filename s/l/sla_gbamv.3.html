subroutine sla_gbamv (integer trans, integer m, integer n, integer kl, integer ku, real alpha, real, dimension( ldab, * ) ab, integer ldab, real, dimension( * ) x, integer incx, real beta, real, dimension( * ) y, integer incy)

SLA_GBAMV realiza una operación de vector de matriz para calcular límites de error.

Propósito:

!>
!> SLA_GBAMV  realiza una de las operaciones de vector de matriz
!>
!>         y := alpha*abs(A)*abs(x) + beta*abs(y),
!>    o    y := alpha*abs(A)**T*abs(x) + beta*abs(y),
!>
!> donde alpha y beta son escalares, x e y son vectores y A es una
!> matriz de m por n.
!>
!> Esta función se utiliza principalmente para calcular límites de error.
!> Para proteger contra el subdesbordamiento durante la evaluación, los
!> componentes del vector resultante se perturban lejos de cero en (N+1)
!> veces el umbral de subdesbordamiento. Para evitar errores
!> innecesariamente grandes para la estructura de bloque incrustada
!> en matrices generales, los componentes cero no se perturban.
!> Una entrada cero se considera si todas las multiplicaciones
!> involucradas en el cálculo de esa entrada tienen al menos un
!> multiplicando cero.
!>

Parámetros

TRANS 
!>          TRANS es INTEGER
!>           En la entrada, TRANS especifica la operación a realizar como
!>           sigue:
!>
!>             BLAS_NO_TRANS      y := alpha*abs(A)*abs(x) + beta*abs(y)
!>             BLAS_TRANS         y := alpha*abs(A**T)*abs(x) + beta*abs(y)
!>             BLAS_CONJ_TRANS    y := alpha*abs(A**T)*abs(x) + beta*abs(y)
!>
!>           Sin cambios a la salida.
!>

M

!>          M es INTEGER
!>           En la entrada, M especifica el número de filas de la matriz A.
!>           M debe ser mayor o igual a cero.
!>           Sin cambios a la salida.
!>

N

!>          N es INTEGER
!>           En la entrada, N especifica el número de columnas de la matriz A.
!>           N debe ser mayor o igual a cero.
!>           Sin cambios a la salida.
!>

KL

!>          KL es INTEGER
!>           El número de subdiagonales dentro de la banda de A.  KL >= 0.
!>

KU

!>          KU es INTEGER
!>           El número de superdiagonales dentro de la banda de A.  KU >= 0.
!>

ALPHA

!>          ALPHA es REAL
!>           En la entrada, ALPHA especifica el escalar alpha.
!>           Sin cambios a la salida.
!>

AB

!>          AB es array REAL, dimension ( LDAB, n )
!>           Antes de la entrada, la parte principal de m por n del array AB debe
!>           contener la matriz de coeficientes.
!>           Sin cambios a la salida.
!>

LDAB

!>          LDAB es INTEGER
!>           En la entrada, LDA especifica la primera dimensión de AB tal como se
!>           declara en el programa llamante (subrutina). LDAB debe ser al menos
!>           max( 1, m ).
!>           Sin cambios a la salida.
!>

X

!>          X es array REAL, dimension
!>           ( 1 + ( n - 1 )*abs( INCX ) ) cuando TRANS = 'N' o 'n'
!>           y al menos
!>           ( 1 + ( m - 1 )*abs( INCX ) ) en caso contrario.
!>           Antes de la entrada, el array X incrementado debe contener el
!>           vector x.
!>           Sin cambios a la salida.
!>

INCX

!>          INCX es INTEGER
!>           En la entrada, INCX especifica el incremento para los elementos de
!>           X. INCX no debe ser cero.
!>           Sin cambios a la salida.
!>

BETA

!>          BETA es REAL
!>           En la entrada, BETA especifica el escalar beta. Cuando BETA se
!>           proporciona como cero, entonces Y no necesita ser establecido en la
!>           entrada.
!>           Sin cambios a la salida.
!>

Y

!>          Y es array REAL, dimension
!>           ( 1 + ( m - 1 )*abs( INCY ) ) cuando TRANS = 'N' o 'n'
!>           y al menos
!>           ( 1 + ( n - 1 )*abs( INCY ) ) en caso contrario.
!>           Antes de la entrada con BETA distinto de cero, el array Y
!>           incrementado debe contener el vector y. A la salida, Y se
!>           sobrescribe con el vector y actualizado.
!>           Si m o n es cero, entonces Y no se referencia y la función
!>           realiza un retorno rápido.
!>

INCY

!>          INCY es INTEGER
!>           En la entrada, INCY especifica el incremento para los elementos de
!>           Y. INCY no debe ser cero.
!>           Sin cambios a la salida.
!>
!>  Rutina Blas de Nivel 2.
!>
