subroutine slaqge (integer m, integer n, real, dimension( lda, * ) a, integer lda, real, dimension( * ) r, real, dimension( * ) c, real rowcnd, real colcnd, real amax, character equed)

SLAQGE escala una matriz rectangular general, utilizando factores de escala de fila y columna calculados por sgeequ.

Propósito:

!>
!> SLAQGE equilibra una matriz general M por N, A, utilizando los factores
!> de escala de fila y columna en los vectores R y C.
!>

Parámetros

M 
!>          M es INTEGER
!>          El número de filas de la matriz A.  M >= 0.
!>

N

!>          N es INTEGER
!>          El número de columnas de la matriz A.  N >= 0.
!>

A

!>          A es array REAL, dimension (LDA,N)
!>          En la entrada, la matriz A de M por N.
!>          En la salida, la matriz equilibrada.  Ver EQUED para la forma de
!>          la matriz equilibrada.
!>

LDA

!>          LDA es INTEGER
!>          La dimensión principal del array A.  LDA >= max(M,1).
!>

R

!>          R es array REAL, dimension (M)
!>          Los factores de escala de fila para A.
!>

C

!>          C es array REAL, dimension (N)
!>          Los factores de escala de columna para A.
!>

ROWCND

!>          ROWCND es REAL
!>          Relación del R(i) más pequeño con el R(i) más grande.
!>

COLCND

!>          COLCND es REAL
!>          Relación del C(i) más pequeño con el C(i) más grande.
!>

AMAX

!>          AMAX es REAL
!>          Valor absoluto de la entrada matricial más grande.
!>

EQUED

!>          EQUED es CHARACTER*1
!>          Especifica la forma de la equilibración que se realizó.
!>          = 'N':  Sin equilibración
!>          = 'R':  Equilibración de fila, es decir, A se ha multiplicado previamente por
!>                  diag(R).
!>          = 'C':  Equilibración de columna, es decir, A se ha postmultiplicado
!>                  por diag(C).
!>          = 'B':  Equilibración tanto de fila como de columna, es decir, A ha sido
!>                  reemplazada por diag(R) * A * diag(C).
!>

Parámetros Internos:

!>  THRESH es un valor umbral utilizado para decidir si se debe hacer la escala
!>  de fila o columna basándose en la relación de los factores de escala de
!>  fila o columna.  Si ROWCND < THRESH, se hace la escala de fila, y si
!>  COLCND < THRESH, se hace la escala de columna.
!>
!>  LARGE y SMALL son valores umbral utilizados para decidir si se debe hacer
!>  la escala de fila basándose en el tamaño absoluto del elemento matricial
!>  más grande.  Si AMAX > LARGE o AMAX < SMALL, se hace la escala de fila.
!>
