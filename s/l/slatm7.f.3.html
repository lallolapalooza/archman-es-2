subroutine slatm7 (integer mode, real cond, integer irsign, integer idist, integer, dimension( 4 ) iseed, real, dimension( * ) d, integer n, integer rank, integer info)

SLATM7

Propósito:

!>
!>    SLATM7 calcula las entradas de D según lo especificado por MODE
!>    COND e IRSIGN. IDIST e ISEED determinan la generación
!>    de números aleatorios. SLATM7 es llamado por SLATMT para generar
!>    matrices de prueba aleatorias.
!> 
!>  MODE   - INTEGER
!>           A la entrada, describe cómo se debe calcular D:
!>           MODE = 0 significa no cambiar D.
!>
!>           MODE = 1 establece D(1)=1 y D(2:RANK)=1.0/COND
!>           MODE = 2 establece D(1:RANK-1)=1 y D(RANK)=1.0/COND
!>           MODE = 3 establece D(I)=COND**(-(I-1)/(RANK-1)) I=1:RANK
!>
!>           MODE = 4 establece D(i)=1 - (i-1)/(N-1)*(1 - 1/COND)
!>           MODE = 5 establece D a números aleatorios en el rango
!>                    ( 1/COND , 1 ) tales que sus logaritmos
!>                    estén distribuidos uniformemente.
!>           MODE = 6 establece D a números aleatorios de la misma distribución
!>                    que el resto de la matriz.
!>           MODE < 0 tiene el mismo significado que ABS(MODE), excepto que
!>              el orden de las entradas de D se invierte.
!>           Por lo tanto, si MODE es positivo, D tiene entradas que van de
!>              1 a 1/COND, si es negativo, de 1/COND a 1,
!>           No modificado.
!>
!>  COND   - REAL
!>           A la entrada, se usa como se describe en MODE arriba.
!>           Si se usa, debe ser >= 1. No modificado.
!>
!>  IRSIGN - INTEGER
!>           A la entrada, si MODE no es -6, 0 ni 6, determina el signo de
!>           las entradas de D
!>           0 => dejar las entradas de D sin cambios
!>           1 => multiplicar cada entrada de D por 1 o -1 con probabilidad 0.5
!>
!>  IDIST  - CHARACTER*1
!>           A la entrada, IDIST especifica el tipo de distribución a ser
!>           usada para generar una matriz aleatoria.
!>           1 => UNIFORME( 0, 1 )
!>           2 => UNIFORME( -1, 1 )
!>           3 => NORMAL( 0, 1 )
!>           No modificado.
!>
!>  ISEED  - Arreglo INTEGER, dimensión ( 4 )
!>           A la entrada, ISEED especifica la semilla del generador de números
!>           aleatorios. El generador de números aleatorios usa una
!>           secuencia congruencial lineal limitada a enteros pequeños,
!>           y por lo tanto debería producir números aleatorios independientes
!>           de la máquina. Los valores de ISEED se modifican a la salida,
!>           y pueden ser usados en la siguiente llamada a SLATM7
!>           para continuar la misma secuencia de números aleatorios.
!>           Modificado a la salida.
!>
!>  D      - Arreglo REAL, dimensión ( MIN( M , N ) )
!>           Arreglo a ser calculado de acuerdo con MODE, COND e IRSIGN.
!>           Puede ser modificado a la salida si MODE es distinto de cero.
!>
!>  N      - INTEGER
!>           Número de entradas de D. No modificado.
!>
!>  RANK   - INTEGER
!>           El rango de la matriz a ser generada solo para los modos 1,2,3.
!>           D( RANK+1:N ) = 0.
!>           No modificado.
!>
!>  INFO   - INTEGER
!>            0  => terminación normal
!>           -1  => si MODE no está en el rango -6 a 6
!>           -2  => si IRSIGN no es 0 ni 1 y MODE no es -6, 0 ni 6
!>           -3  => si MODE no es -6, 0 ni 6 y COND es menor que 1
!>           -4  => si MODE es 6 o -6 y IDIST no está en el rango 1 a 3
!>           -7  => si N es negativo
!>
