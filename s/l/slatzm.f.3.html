subroutine slatzm (character side, integer m, integer n, real, dimension( * ) v, integer incv, real tau, real, dimension( ldc, * ) c1, real, dimension( ldc, * ) c2, integer ldc, real, dimension( * ) work)

SLATZM

Propósito:

!>
!> Esta rutina está obsoleta y ha sido reemplazada por la rutina SORMRZ.
!>
!> SLATZM aplica una matriz de Householder generada por STZRQF a una matriz.
!>
!> Sea P = I - tau*u*u**T,   u = ( 1 ),
!>                               ( v )
!> donde v es un vector (m-1) si SIDE = 'L', o un vector (n-1) si
!> SIDE = 'R'.
!>
!> Si SIDE es igual a 'L', sea
!>        C = [ C1 ] 1
!>            [ C2 ] m-1
!>              n
!> Entonces C se sobrescribe con P*C.
!>
!> Si SIDE es igual a 'R', sea
!>        C = [ C1, C2 ] m
!>               1  n-1
!> Entonces C se sobrescribe con C*P.
!>

Parámetros

SIDE 
!>          SIDE es CHARACTER*1
!>          = 'L': forma P * C
!>          = 'R': forma C * P
!>

M

!>          M es INTEGER
!>          El número de filas de la matriz C.
!>

N

!>          N es INTEGER
!>          El número de columnas de la matriz C.
!>

V

!>          V es un array REAL, de dimensiones
!>                  (1 + (M-1)*abs(INCV)) si SIDE = 'L'
!>                  (1 + (N-1)*abs(INCV)) si SIDE = 'R'
!>          El vector v en la representación de P. V no se utiliza
!>          si TAU = 0.
!>

INCV

!>          INCV es INTEGER
!>          El incremento entre elementos de v. INCV <> 0
!>

TAU

!>          TAU es REAL
!>          El valor tau en la representación de P.
!>

C1

!>          C1 es un array REAL, de dimensiones
!>                         (LDC,N) si SIDE = 'L'
!>                         (M,1)   si SIDE = 'R'
!>          En la entrada, el vector C1 de n elementos si SIDE = 'L', o el
!>          vector C1 de m elementos si SIDE = 'R'.
!>
!>          En la salida, la primera fila de P*C si SIDE = 'L', o la primera
!>          columna de C*P si SIDE = 'R'.
!>

C2

!>          C2 es un array REAL, de dimensiones
!>                         (LDC, N)   si SIDE = 'L'
!>                         (LDC, N-1) si SIDE = 'R'
!>          En la entrada, la matriz C2 de (m - 1) x n si SIDE = 'L', o la
!>          matriz C2 de m x (n - 1) si SIDE = 'R'.
!>
!>          En la salida, las filas 2:m de P*C si SIDE = 'L', o las columnas 2:m de C*P
!>          si SIDE = 'R'.
!>

LDC

!>          LDC es INTEGER
!>          La dimensión principal de los arrays C1 y C2. LDC >= (1,M).
!>

WORK

!>          WORK es un array REAL, de dimensiones
!>                      (N) si SIDE = 'L'
!>                      (M) si SIDE = 'R'
!>
