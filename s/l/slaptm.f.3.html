Documentación de Función/Subrutina

subroutine slaptm (integer n, integer nrhs, real alpha, real, dimension( * ) d, real, dimension( * ) e, real, dimension( ldx, * ) x, integer ldx, real beta, real, dimension( ldb, * ) b, integer ldb)

SLAPTM

Propósito:

!>
!> SLAPTM multiplica una matriz N por NRHS X por una matriz tridiagonal
!> simétrica A y almacena el resultado en una matriz B. La operación tiene la
!> forma
!>
!>    B := alpha * A * X + beta * B
!>
!> donde alpha puede ser 1. o -1. y beta puede ser 0., 1. o -1.
!>

Parámetros

N 
!>          N es ENTERO
!>          El orden de la matriz A. N >= 0.
!>

NRHS

!>          NRHS es ENTERO
!>          El número de lados derechos, es decir, el número de columnas
!>          de las matrices X y B.
!>

ALPHA

!>          ALPHA es REAL
!>          El escalar alpha. ALPHA debe ser 1. o -1.; de lo contrario,
!>          se asume que es 0.
!>

D

!>          D es un array REAL, de dimensión (N)
!>          Los n elementos diagonales de la matriz tridiagonal A.
!>

E

!>          E es un array REAL, de dimensión (N-1)
!>          Los elementos subdiagonales o superdiagonales (n-1) de A.
!>

X

!>          X es un array REAL, de dimensión (LDX,NRHS)
!>          La matriz X de N por NRHS.
!>

LDX

!>          LDX es ENTERO
!>          La dimensión principal del array X. LDX >= max(N,1).
!>

BETA

!>          BETA es REAL
!>          El escalar beta. BETA debe ser 0., 1. o -1.; de lo contrario,
!>          se asume que es 1.
!>

B

!>          B es un array REAL, de dimensión (LDB,NRHS)
!>          En la entrada, la matriz B de N por NRHS.
!>          En la salida, B se sobrescribe con la expresión matricial
!>          B := alpha * A * X + beta * B.
!>

LDB

!>          LDB es ENTERO
!>          La dimensión principal del array B. LDB >= max(N,1).
!>
