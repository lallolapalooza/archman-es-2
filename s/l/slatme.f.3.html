subroutine slatme (integer n, character dist, integer, dimension( 4 ) iseed, real, dimension( * ) d, integer mode, real cond, real dmax, character, dimension( * ) ei, character rsign, character upper, character sim, real, dimension( * ) ds, integer modes, real conds, integer kl, integer ku, real anorm, real, dimension( lda, * ) a, integer lda, real, dimension( * ) work, integer info)

SLATME

Propósito:

!>
!>    SLATME genera matrices cuadradas no simétricas aleatorias con
!>    valores propios especificados para probar programas LAPACK.
!>
!>    SLATME opera aplicando la siguiente secuencia de
!>    operaciones:
!>
!>    1. Establece la diagonal a D, donde D puede ser de entrada o
!>         calculada según MODE, COND, DMAX y RSIGN
!>         como se describe a continuación.
!>
!>    2. Si se desean pares conjugados complejos (MODE=0 y EI(1)='R',
!>         o MODE=5), ciertos pares de elementos adyacentes de D se
!>         interpretan como las partes real e imaginaria de un par
!>         conjugado complejo; A se convierte así en diagonal por bloques, con
!>         bloques de 1x1 y 2x2.
!>
!>    3. Si UPPER='T', el triángulo superior de A se establece con valores
!>         aleatorios de la distribución DIST.
!>
!>    4. Si SIM='T', A se multiplica a la izquierda por una matriz aleatoria
!>         X, cuyos valores singulares se especifican por DS, MODES y
!>         CONDS, y a la derecha por X inversa.
!>
!>    5. Si KL < N-1, la banda inferior se reduce a KL usando
!>         transformaciones de Householder. Si KU < N-1, la banda superior
!>         se reduce a KU.
!>
!>    6. Si ANORM no es negativo, la matriz se escala para tener
!>         norma de elemento máximo ANORM.
!>
!>    (Nota: dado que la matriz no puede reducirse más allá de la forma de
!>     Hessenberg, no hay opciones de empaquetado disponibles).
!>

Parámetros

N 
!>          N es INTEGER
!>           El número de columnas (o filas) de A. No modificado.
!>

DIST

!>          DIST es CHARACTER*1
!>           A la entrada, DIST especifica el tipo de distribución a usar
!>           para generar los valores propios/singulares aleatorios, y para
!>           el triángulo superior (ver UPPER).
!>           'U' => UNIFORME( 0, 1 )  ( 'U' para uniforme )
!>           'S' => UNIFORME( -1, 1 ) ( 'S' para simétrico )
!>           'N' => NORMAL( 0, 1 )   ( 'N' para normal )
!>           No modificado.
!>

ISEED

!>          ISEED es un array INTEGER, dimensión ( 4 )
!>           A la entrada ISEED especifica la semilla del generador de números
!>           aleatorios. Deben estar entre 0 y 4095 inclusive,
!>           y ISEED(4) debe ser impar. El generador de números aleatorios
!>           usa una secuencia congruencial lineal limitada a enteros pequeños,
!>           y por lo tanto debe producir números aleatorios independientes de la máquina.
!>           Los valores de ISEED se cambian a la salida, y pueden usarse en
!>           la siguiente llamada a SLATME para continuar la misma secuencia
!>           de números aleatorios.
!>           Modificado a la salida.
!>

D

!>          D es un array REAL, dimensión ( N )
!>           Este array se utiliza para especificar los valores propios de A. Si
!>           MODE=0, entonces se asume que D contiene los valores propios (pero
!>           ver la descripción de EI), de lo contrario se calcularán
!>           según MODE, COND, DMAX y RSIGN y se colocarán en D.
!>           Modificado si MODE es distinto de cero.
!>

MODE

!>          MODE es INTEGER
!>           A la entrada esto describe cómo se van a especificar los valores propios:
!>           MODE = 0 significa usar D (con EI) como entrada
!>           MODE = 1 establece D(1)=1 y D(2:N)=1.0/COND
!>           MODE = 2 establece D(1:N-1)=1 y D(N)=1.0/COND
!>           MODE = 3 establece D(I)=COND**(-(I-1)/(N-1))
!>           MODE = 4 establece D(i)=1 - (i-1)/(N-1)*(1 - 1/COND)
!>           MODE = 5 establece D con números aleatorios en el rango
!>                    ( 1/COND , 1 ) tales que sus logaritmos
!>                    están distribuidos uniformemente. Cada par impar-par
!>                    se usará como dos valores propios reales o como la parte real
!>                    e imaginaria de un par de valores propios complejos conjugados;
!>                    la elección de cuál se hace es aleatoria, con
!>                    probabilidad del 50%, para cada par.
!>           MODE = 6 establece D con números aleatorios de la misma distribución
!>                    que el resto de la matriz.
!>           MODE < 0 tiene el mismo significado que ABS(MODE), excepto que
!>              el orden de los elementos de D se invierte.
!>           Así, si MODE está entre 1 y 4, D tiene entradas que varían
!>              de 1 a 1/COND, si está entre -1 y -4, D tiene entradas
!>              que varían de 1/COND a 1,
!>           No modificado.
!>

COND

!>          COND es REAL
!>           A la entrada, se utiliza como se describe en MODE arriba.
!>           Si se usa, debe ser >= 1. No modificado.
!>

DMAX

!>          DMAX es REAL
!>           Si MODE no es -6, 0 ni 6, los contenidos de D, calculados según
!>           MODE y COND, se escalarán por
!>           DMAX / max(abs(D(i))). Tenga en cuenta que DMAX no necesita
!>           ser positivo: si DMAX es negativo (o cero), D se escalará
!>           por un número negativo (o cero).
!>           No modificado.
!>

EI

!>          EI es un array CHARACTER*1, dimensión ( N )
!>           Si MODE es 0, y EI(1) no es ' ' (carácter de espacio),
!>           este array especifica qué elementos de D (en la entrada) son
!>           valores propios reales y cuáles son las partes real e imaginaria
!>           de un par de valores propios complejos conjugados. Los elementos
!>           de EI solo pueden tener los valores 'R' e 'I'. Si
!>           EI(j)='R' y EI(j+1)='I', entonces el j-ésimo valor propio es
!>           CMPLX( D(j) , D(j+1) ), y el (j+1)-ésimo es el conjugado complejo
!>           del mismo. Si EI(j)=EI(j+1)='R', entonces el j-ésimo
!>           valor propio es D(j) (es decir, real). EI(1) no puede ser 'I',
!>           ni dos elementos adyacentes de EI pueden tener el valor 'I'.
!>           Si MODE no es 0, entonces EI se ignora. Si MODE es 0 y
!>           EI(1)=' ', entonces todos los valores propios serán reales.
!>           No modificado.
!>

RSIGN

!>          RSIGN es CHARACTER*1
!>           Si MODE no es 0, 6, ni -6, y RSIGN='T', entonces los
!>           elementos de D, calculados según MODE y COND, se multiplicarán
!>           por un signo aleatorio (+1 o -1). Si RSIGN='F',
!>           no se multiplicarán. RSIGN solo puede tener los valores 'T' o
!>           'F'.
!>           No modificado.
!>

UPPER

!>          UPPER es CHARACTER*1
!>           Si UPPER='T', entonces los elementos de A por encima de la diagonal
!>           (y por encima de los bloques diagonales 2x2, si A tiene valores
!>           propios complejos) se establecerán con números aleatorios de DIST.
!>           Si UPPER='F', no se hará. UPPER solo puede tener los valores 'T' o
!>           'F'.
!>           No modificado.
!>

SIM

!>          SIM es CHARACTER*1
!>           Si SIM='T', entonces A se operará, es decir, se multiplicará a la
!>           izquierda por una matriz X y a la derecha por X inversa.
!>           X = U S V, donde U y V son matrices unitarias aleatorias y S es una matriz
!>           (diagonal) de valores singulares especificada por DS, MODES y

NAME

SLATME

constructs a general triangular matrix with specified singular values and/or eigenvalues.

Function/Subroutine Documentation

This subroutine constructs a general real matrix A which is factorized into a product of a lower triangular matrix L and an upper triangular matrix U.

The singular values and/or eigenvalues of A are specified by the contents of the arrays D, SD, E, and SE.

Other parameters control the distribution of the singular values and eigenvalues, the condition number of A, and the bandwidth of A.

A general triangular matrix A is generated subject to:

1. Its eigenvalues are D(I) if KEY = 'N' or 'T'.

2. Its singular values are SD(I) if KEY = 'N' or 'V'.

3. Its eigenvalues are D(I) and E(I) if KEY = 'N' or 'T'.

4. Its singular values are SD(I) and SE(I) if KEY = 'N' or 'V'.

Note that if KEY = 'T', the eigenvalues are D(I), and if KEY = 'V', the singular values are SD(I). If KEY = 'N', both are specified.

The diagonal elements of A are computed by SLATM1. The remaining elements are computed by SLATM2.

The singular values are computed by SLATM3.

The condition number of A is controlled by COND.

The bandwidth of A is controlled by KL and KU.

The maximum element norm of A is controlled by ANORM.

The subroutines SLATM1, SLATM2, SLATM3, and SLARGE are called by SLATME.

The input parameters DIST, MODE, PIVMIN, and PIVOTS specify the distribution of the eigenvalues and singular values.

The input parameters IL and IU specify the submatrix of A which is to be factored.

The input parameter KEY specifies whether eigenvalues or singular values are to be computed.

The input parameter N specifies the order of the matrix A.

The input parameter RSIGN specifies the signs of the eigenvalues.

The input parameter UPPER specifies whether A is upper or lower triangular.

The input parameter SIM specifies whether A is symmetric or non-symmetric.

The input parameter CONDS specifies the condition number of A.

The input parameters KL and KU specify the lower and upper bandwidths of A.

The input parameter ANORM specifies the maximum element norm of A.

The input parameter LDA specifies the leading dimension of A.

The input parameter WORK is a workspace array.

The output parameter NCONV specifies the number of eigenvalues which have been converged.

The output parameter A is the generated matrix.

The output parameter INFO indicates the exit status.

Parameters

N

!>          N is INTEGER
!>           The order of the matrix A.  N >= 0.
!>           Not modified.

DIST

!>          DIST is CHARACTER*1
!>           Specifies the storage format of the matrix A.
!>           Possible values for DIST are:
!>           'R' (random):   A is a general random matrix.
!>           'U' (upper):    A is upper triangular.
!>           'L' (lower):    A is lower triangular.
!>           'C' (codensed): A is a general random matrix, which is
!>                           factorized into a product of a lower
!>                           triangular matrix L and an upper
!>                           triangular matrix U.
!>           'A' (all):      A is a general random matrix.
!>                           The condition number is controlled by COND.
!>                           The bandwidth is controlled by KL and KU.
!>                           The maximum element norm is controlled by ANORM.
!>                           The signs of the eigenvalues are controlled by RSIGN.
!>           Not modified.

MODE

!>          MODE is INTEGER
!>           Specifies the form of the eigenvalues and/or singular values.
!>           Possible values for MODE are:
!>           0 : A is generated with random eigenvalues and singular
!>               values.
!>           1 : A is generated with real eigenvalues.
!>           2 : A is generated with real and random eigenvalues.
!>           3 : A is generated with strictly real eigenvalues.
!>           4 : A is generated with complex conjugate eigenvalues.
!>           5 : A is generated with complex conjugate and random
!>               eigenvalues.
!>           6 : A is generated with strictly complex conjugate eigenvalues.
!>           -1 : A is generated with real singular values.
!>           -2 : A is generated with real and random singular values.
!>           -3 : A is generated with strictly real singular values.
!>           -4 : A is generated with complex conjugate singular values.
!>           -5 : A is generated with complex conjugate and random
!>                singular values.
!>           -6 : A is generated with strictly complex conjugate singular
!>                values.
!>           Modified if MODE is nonzero.

IL

!>          IL is INTEGER
!>           Specifies the row and column indices of the submatrix,
!>           which is to be factored.
!>           1 <= IL <= IU <= N.
!>           Not modified.

IU

!>          IU is INTEGER
!>           Specifies the row and column indices of the submatrix,
!>           which is to be factored.
!>           1 <= IL <= IU <= N.
!>           Not modified.

KEY

!>          KEY is CHARACTER*1
!>           Specifies the type of factorization to be generated.
!>           Possible values for KEY are:
!>           'N' : A is factorized into a product of a lower triangular
!>                 matrix L and an upper triangular matrix U.
!>           'T' : A is factorized into a product of a lower triangular
!>                 matrix L and an upper triangular matrix U, where the
!>                 eigenvalues of A are specified by D.
!>           'V' : A is factorized into a product of a lower triangular
!>                 matrix L and an upper triangular matrix U, where the
!>                 singular values of A are specified by SD.
!>           Not modified.

PIVMIN

!>          PIVMIN is REAL
!>           The minimum value for the pivots to be generated.
!>           PIVMIN >= 0.
!>           Not modified.

NCONV

!>          NCONV is INTEGER
!>           The number of eigenvalues which have been converged.
!>           Modified.

A

!>          A is REAL array, dimension ( LDA, N )
!>           On exit A is the desired test matrix.
!>           Modified.

LDA

!>          LDA is INTEGER
!>           LDA specifies the first dimension of A as declared in the
!>           calling program.  LDA must be at least max(1,N).
!>           Not modified.

D

!>          D is REAL array, dimension ( N )
!>           If MODE is nonzero, the diagonal elements of A are set to
!>           the elements of D.
!>           Modified if MODE is nonzero.

SD

!>          SD is REAL array, dimension ( N )
!>           If MODE is nonzero, the singular values of A are set to
!>           the elements of SD.
!>           Modified if MODE is nonzero.

E

!>          E is REAL array, dimension ( N )
!>           If MODE is nonzero, the subdiagonal elements of A are set to
!>           the elements of E.
!>           Modified if MODE is nonzero.

SE

!>          SE is REAL array, dimension ( N )
!>           If MODE is nonzero, the sub-singular values of A are set to
!>           the elements of SE.
!>           Modified if MODE is nonzero.

WORK

!>          WORK is REAL array, dimension ( 3*N )
!>           Workspace.
!>           Modified.

INFO

!>          INFO is INTEGER
!>           Error code. On exit, INFO will be set to one of the
!>           following values:
!>             0 => normal return
!>            -1 => N negative
!>            -2 => DIST illegal string
!>            -3 => MODE not in range -6 to 6
!>            -4 => IL illegal value, or IL > IU
!>            -5 => IU illegal value, or IU > N
!>            -6 => KEY illegal string
!>            -7 => PIVMIN is negative
!>            -8 => LDA is less than max(1,N)
!>             1 => Error return from SLATM1
!>             2 => Error return from SLATM2
!>             3 => Error return from SLATM3
!>             4 => Error return from SLARGE

DS

!>           DS is REAL array, dimension ( N )
!>           This array is used to specify the singular values of X,
!>           in the same way that D specifies the eigenvalues of A.
!>           If MODE=0, the DS contains the singular values, which
!>           may not be zero.
!>           Modified if MODE is nonzero.

MODES

!>          MODES is INTEGER

CONDS

!>          CONDS is REAL
!>           Same as MODE and COND, but for specifying the diagonal
!>           of S.  MODES=-6 and +6 are not allowed (since they would
!>           result in randomly ill-conditioned eigenvalues.)

KL

!>          KL is INTEGER
!>           This specifies the lower bandwidth of the  matrix.  KL=1
!>           specifies upper Hessenberg form.  If KL is at least N-1,
!>           then A will have full lower bandwidth.  KL must be at
!>           least 1.
!>           Not modified.

KU

!>          KU is INTEGER
!>           This specifies the upper bandwidth of the  matrix.  KU=1
!>           specifies lower Hessenberg form.  If KU is at least N-1,
!>           then A will have full upper bandwidth; if KU and KL
!>           are both at least N-1, then A will be dense.  Only one of
!>           KU and KL may be less than N-1.  KU must be at least 1.
!>           Not modified.

ANORM

!>          ANORM is REAL
!>           If ANORM is not negative, then A will be scaled by a non-
!>           negative real number to make the maximum-element-norm of A
!>           to be ANORM.
!>           Not modified.

A

!>          A is REAL array, dimension ( LDA, N )
!>           On exit A is the desired test matrix.
!>           Modified.

LDA

!>          LDA is INTEGER
!>           LDA specifies the first dimension of A as declared in the
!>           calling program.  LDA must be at least N.
!>           Not modified.

WORK

!>          WORK is REAL array, dimension ( 3*N )
!>           Workspace.
!>           Modified.

INFO

!>          INFO is INTEGER
!>           Error code.  On exit, INFO will be set to one of the
!>           following values:
!>             0 => normal return
!>            -1 => N negative
!>            -2 => DIST illegal string
!>            -5 => MODE not in range -6 to 6
!>            -6 => COND less than 1.0, and MODE neither -6, 0 nor 6
!>            -8 => EI(1) is not ' ' or 'R', EI(j) is not 'R' or 'I', or
!>                  two adjacent elements of EI are 'I'.
!>            -9 => RSIGN is not 'T' or 'F'
!>           -10 => UPPER is not 'T' or 'F'
!>           -11 => SIM   is not 'T' or 'F'
!>           -12 => MODES=0 and DS has a zero singular value.
!>           -13 => MODES is not in the range -5 to 5.
!>           -14 => MODES is nonzero and CONDS is less than 1.
!>           -15 => KL is less than 1.
!>           -16 => KU is less than 1, or KL and KU are both less than
!>                  N-1.
!>           -19 => LDA is less than N.
!>            1  => Error return from SLATM1 (computing D)
!>            2  => Cannot scale to DMAX (max. eigenvalue is 0)
!>            3  => Error return from SLATM1 (computing DS)
!>            4  => Error return from SLARGE
!>            5  => Zero singular value from SLATM1.
