subrutina slakf2 (integer m, integer n, real, dimension( lda, * ) a, integer lda, real, dimension( lda, * ) b, real, dimension( lda, * ) d, real, dimension( lda, * ) e, real, dimension( ldz, * ) z, integer ldz)

SLAKF2

Propósito:

!>
!> Forma la matriz 2*M*N por 2*M*N
!>
!>        Z = [ kron(In, A)  -kron(B', Im) ]
!>            [ kron(In, D)  -kron(E', Im) ],
!>
!> donde In es la matriz identidad de tamaño n y X' es la transpuesta
!> de X. kron(X, Y) es el producto de Kronecker entre las matrices X
!> e Y.
!>

Parámetros

M 
!>          M es INTEGER
!>          Tamaño de la matriz, debe ser >= 1.
!>

N

!>          N es INTEGER
!>          Tamaño de la matriz, debe ser >= 1.
!>

A

!>          A es REAL, dimension ( LDA, M )
!>          La matriz A en la matriz de salida Z.
!>

LDA

!>          LDA es INTEGER
!>          La dimensión principal de A, B, D y E. ( LDA >= M+N )
!>

B

!>          B es REAL, dimension ( LDA, N )
!>

D

!>          D es REAL, dimension ( LDA, M )
!>

E

!>          E es REAL, dimension ( LDA, N )
!>
!>          Las matrices utilizadas en la formación de la matriz de salida Z.
!>

Z

!>          Z es REAL, dimension ( LDZ, 2*M*N )
!>          La matriz resultante de Kronecker M*N*2 por M*N*2 (ver arriba.)
!>

LDZ

!>          LDZ es INTEGER
!>          La dimensión principal de Z. ( LDZ >= 2*M*N )
!>
