subroutine slasy2 (logical ltranl, logical ltranr, integer isgn, integer n1, integer n2, real, dimension( ldtl, * ) tl, integer ldtl, real, dimension( ldtr, * ) tr, integer ldtr, real, dimension( ldb, * ) b, integer ldb, real scale, real, dimension( ldx, * ) x, integer ldx, real xnorm, integer info)

SLASY2 resuelve la ecuación matricial de Sylvester donde las matrices son de orden 1 o 2.

Propósito:

!>
!> SLASY2 resuelve para la matriz X de N1 por N2, 1 <= N1,N2 <= 2, en
!>
!>        op(TL)*X + ISGN*X*op(TR) = SCALE*B,
!>
!> donde TL es N1 por N1, TR es N2 por N2, B es N1 por N2, e ISGN = 1 o
!> -1.  op(T) = T o T**T, donde T**T denota la transpuesta de T.
!>

Parámetros

LTRANL 
!>          LTRANL es LOGICAL
!>          En la entrada, LTRANL especifica el op(TL):
!>             = .FALSE., op(TL) = TL,
!>             = .TRUE., op(TL) = TL**T.
!>

LTRANR

!>          LTRANR es LOGICAL
!>          En la entrada, LTRANR especifica el op(TR):
!>            = .FALSE., op(TR) = TR,
!>            = .TRUE., op(TR) = TR**T.
!>

ISGN

!>          ISGN es INTEGER
!>          En la entrada, ISGN especifica el signo de la ecuación
!>          como se describe antes. ISGN solo puede ser 1 o -1.
!>

N1

!>          N1 es INTEGER
!>          En la entrada, N1 especifica el orden de la matriz TL.
!>          N1 solo puede ser 0, 1 o 2.
!>

N2

!>          N2 es INTEGER
!>          En la entrada, N2 especifica el orden de la matriz TR.
!>          N2 solo puede ser 0, 1 o 2.
!>

TL

!>          TL es un array REAL, dimensión (LDTL,2)
!>          En la entrada, TL contiene una matriz de N1 por N1.
!>

LDTL

!>          LDTL es INTEGER
!>          La dimensión principal de la matriz TL. LDTL >= max(1,N1).
!>

TR

!>          TR es un array REAL, dimensión (LDTR,2)
!>          En la entrada, TR contiene una matriz de N2 por N2.
!>

LDTR

!>          LDTR es INTEGER
!>          La dimensión principal de la matriz TR. LDTR >= max(1,N2).
!>

B

!>          B es un array REAL, dimensión (LDB,2)
!>          En la entrada, la matriz B de N1 por N2 contiene el lado derecho
!>          de la ecuación.
!>

LDB

!>          LDB es INTEGER
!>          La dimensión principal de la matriz B. LDB >= max(1,N1).
!>

SCALE

!>          SCALE es REAL
!>          En la salida, SCALE contiene el factor de escala. SCALE se elige
!>          menor o igual a 1 para evitar el desbordamiento de la solución.
!>

X

!>          X es un array REAL, dimensión (LDX,2)
!>          En la salida, X contiene la solución de N1 por N2.
!>

LDX

!>          LDX es INTEGER
!>          La dimensión principal de la matriz X. LDX >= max(1,N1).
!>

XNORM

!>          XNORM es REAL
!>          En la salida, XNORM es la norma infinito de la solución.
!>

INFO

!>          INFO es INTEGER
!>          En la salida, INFO se establece en
!>             0: salida exitosa.
!>             1: TL y TR tienen valores propios demasiado cercanos, por lo que TL o
!>                TR se perturba para obtener una ecuación no singular.
!>          NOTA: En aras de la velocidad, esta rutina no
!>                verifica los errores de entrada.
!>
