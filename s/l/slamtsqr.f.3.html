subroutine slamtsqr (character side, character trans, integer m, integer n, integer k, integer mb, integer nb, real, dimension( lda, * ) a, integer lda, real, dimension( ldt, * ) t, integer ldt, real, dimension( ldc, * ) c, integer ldc, real, dimension( * ) work, integer lwork, integer info)

SLAMTSQR

Propósito:

!>
!>      SLAMTSQR sobrescribe la matriz general real M por N, C, con
!>
!>
!>                 SIDE = 'L'     SIDE = 'R'
!> TRANS = 'N':      Q * C          C * Q
!> TRANS = 'T':      Q**T * C       C * Q**T
!>      donde Q es una matriz ortogonal real definida como el producto
!>      de reflectores elementales bloqueados, calculados por la
!>      factorización QR delgada (SLATSQR)
!>

Parámetros

SIDE 
!>          SIDE es CHARACTER*1
!>          = 'L': aplicar Q o Q**T desde la Izquierda;
!>          = 'R': aplicar Q o Q**T desde la Derecha.
!>

TRANS

!>          TRANS es CHARACTER*1
!>          = 'N':  Sin transpuesta, aplicar Q;
!>          = 'T':  Transpuesta, aplicar Q**T.
!>

M

!>          M es INTEGER
!>          El número de filas de la matriz A.  M >=0.
!>

N

!>          N es INTEGER
!>          El número de columnas de la matriz C. N >= 0.
!>

K

!>          K es INTEGER
!>          El número de reflectores elementales cuyo producto define
!>          la matriz Q. M >= K >= 0;
!>
!>

MB

!>          MB es INTEGER
!>          El tamaño del bloque a utilizar en la QR bloqueada.
!>          MB > N. (debe ser el mismo que SLATSQR)
!>

NB

!>          NB es INTEGER
!>          El tamaño del bloque de columnas a utilizar en la QR bloqueada.
!>          N >= NB >= 1.
!>

A

!>          A es un array REAL, de dimensión (LDA,K)
!>          La i-ésima columna debe contener el vector que define el
!>          reflector elemental bloqueado H(i), para i = 1,2,...,k, tal como
!>          devuelto por SLATSQR en las primeras k columnas de
!>          su array de argumento A.
!>

LDA

!>          LDA es INTEGER
!>          La dimensión principal del array A.
!>          Si SIDE = 'L', LDA >= max(1,M);
!>          si SIDE = 'R', LDA >= max(1,N).
!>

T

!>          T es un array REAL, de dimensión
!>          ( N * Número de bloques(CEIL(M-K/MB-K)),
!>          Los reflectores de bloque triangulares superiores bloqueados, almacenados en forma compacta
!>          como una secuencia de bloques triangulares superiores. Véase abajo
!>          para más detalles.
!>

LDT

!>          LDT es INTEGER
!>          La dimensión principal del array T. LDT >= NB.
!>

C

!>          C es un array REAL, de dimensión (LDC,N)
!>          En la entrada, la matriz C de M por N.
!>          En la salida, C se sobrescribe con Q*C o Q**T*C o C*Q**T o C*Q.
!>

LDC

!>          LDC es INTEGER
!>          La dimensión principal del array C. LDC >= max(1,M).
!>

WORK

!>          (espacio de trabajo) array REAL, de dimensión (MAX(1,LWORK))
!>          En la salida, si INFO = 0, WORK(1) devuelve el LWORK mínimo.
!>

LWORK

!>          LWORK es INTEGER
!>          La dimensión del array WORK.
!>          Si MIN(M,N,K) = 0, LWORK >= 1.
!>          Si SIDE = 'L', LWORK >= max(1,N*NB).
!>          Si SIDE = 'R', LWORK >= max(1,MB*NB).
!>
!>          Si LWORK = -1, entonces se asume una consulta de espacio de trabajo; la rutina
!>          sólo calcula el tamaño mínimo del array WORK, devuelve
!>          este valor como la primera entrada del array WORK, y XERBLA no emite ningún mensaje de error
!>          relacionado con LWORK.
!>

INFO

!>          INFO es INTEGER
!>          = 0:  salida exitosa
!>          < 0:  si INFO = -i, el i-ésimo argumento tenía un valor ilegal
!>

Detalles Adicionales:

!> La Factorización QR delgada (TSQR) realiza la QR mediante una secuencia de transformaciones ortogonales,
!> representando Q como un producto de otras matrices ortogonales
!>   Q = Q(1) * Q(2) * . . . * Q(k)
!> donde cada Q(i) pone a cero las entradas subdiagonales de un bloque de MB filas de A:
!>   Q(1) pone a cero las entradas subdiagonales de las filas 1:MB de A
!>   Q(2) pone a cero las filas MB-N inferiores de las filas [1:N,MB+1:2*MB-N] de A
!>   Q(3) pone a cero las filas MB-N inferiores de las filas [1:N,2*MB-N+1:3*MB-2*N] de A
!>   . . .
!>
!> Q(1) se calcula con GEQRT, que representa Q(1) mediante vectores de Householder
!> almacenados debajo de la diagonal de las filas 1:MB de A, y mediante reflectores
!> de bloque triangulares superiores, almacenados en el array T(1:LDT,1:N).
!> Para más información, véase Detalles Adicionales en GEQRT.
!>
!> Q(i) para i>1 se calcula con TPQRT, que representa Q(i) mediante vectores de Householder
!> almacenados en las filas [(i-1)*(MB-N)+N+1:i*(MB-N)+N] de A, y mediante reflectores
!> de bloque triangulares superiores, almacenados en el array T(1:LDT,(i-1)*N+1:i*N).
!> La última Q(k) puede usar menos filas.
!> Para más información, véase Detalles Adicionales en TPQRT.
!>
!> Para más detalles del algoritmo general, véase la descripción de
!> TSQR Secuencial en la Sección 2.2 de [1].
!>
!> [1] “Communication-Optimal Parallel and Sequential QR and LU Factorizations,”
!>     J. Demmel, L. Grigori, M. Hoemmen, J. Langou,
!>     SIAM J. Sci. Comput, vol. 34, no. 1, 2012
!>
