subroutine slarfb_gett (character ident, integer m, integer n, integer k, real, dimension( ldt, * ) t, integer ldt, real, dimension( lda, * ) a, integer lda, real, dimension( ldb, * ) b, integer ldb, real, dimension( ldwork, * ) work, integer ldwork)

SLARFB_GETT

Propósito:

!>
!> SLARFB_GETT aplica un bloque reflector de Householder real H desde la
!> izquierda a una matriz real (K+M)-por-N
!> compuesta por dos matrices de bloque: una matriz A en forma trapezoidal superior K-por-N
!> almacenada en el array A, y una matriz rectangular M-por-(N-K) B, almacenada
!> en el array B. El bloque reflector H se almacena en una representación compacta
!> WY, donde los reflectores elementales están en los
!> arrays A, B y T. Ver la sección de Detalles Adicionales.
!>

Parámetros

IDENT 
!>          IDENT es CHARACTER*1
!>          Si IDENT = 'I' (o 'i'), entonces V1 es una matriz identidad
!>             y no se almacena.
!>          Si IDENT != 'I' (o 'i'), entonces V1 es una matriz
!>             unit lower-triangular y se almacena en el bloque K-por-K
!>             izquierdo de la matriz de entrada A.
!>          Ver la sección de Detalles Adicionales.
!>

M

!>          M es INTEGER
!>          El número de filas de la matriz B.
!>          M >= 0.
!>

N

!>          N es INTEGER
!>          El número de columnas de las matrices A y B.
!>          N >= 0.
!>

K

!>          K es INTEGER
!>          El número de filas de la matriz A.
!>          K es también el orden de la matriz T, es decir, el número de
!>          reflectores elementales cuyo producto define el bloque
!>          reflector. 0 <= K <= N.
!>

T

!>          T es un array REAL, de dimensión (LDT,K)
!>          La matriz T superior-triangular K-por-K en la representación
!>          del bloque reflector.
!>

LDT

!>          LDT es INTEGER
!>          La dimensión principal del array T. LDT >= K.
!>

A

!>          A es un array REAL, de dimensión (LDA,N)
!>
!>          En la entrada:
!>           a) En la parte superior trapezoidal K-por-N A: matriz de entrada A.
!>           b) En las columnas debajo de la diagonal: columnas de V1
!>              (los unos no se almacenan en la diagonal).
!>
!>          En la salida:
!>            A se sobrescribe con el producto rectangular K-por-N H*A.
!>
!>          Ver la sección de Detalles Adicionales.
!>

LDA

!>          LDA es INTEGER
!>          La dimensión principal del array A. LDA >= max(1,K).
!>

B

!>          B es un array REAL, de dimensión (LDB,N)
!>
!>          En la entrada:
!>            a) En el bloque derecho M-por-(N-K): matriz de entrada B.
!>            b) En el bloque izquierdo M-por-N: columnas de V2.
!>
!>          En la salida:
!>            B se sobrescribe con el producto rectangular M-por-N H*B.
!>
!>          Ver la sección de Detalles Adicionales.
!>

LDB

!>          LDB es INTEGER
!>          La dimensión principal del array B. LDB >= max(1,M).
!>

WORK

!>          WORK es un array REAL,
!>          de dimensión (LDWORK,max(K,N-K))
!>

LDWORK

!>          LDWORK es INTEGER
!>          La dimensión principal del array WORK. LDWORK>=max(1,K).
!>
!>

Detalles Adicionales:

!>
!>    (1) Descripción de la Operación Algebraica.
!>
!>    La matriz A es una matriz K-por-N compuesta por dos matrices de bloque
!>    de columna, A1, que es K-por-K, y A2, que es K-por-(N-K):
!>    A = ( A1, A2 ).
!>    La matriz B es una matriz M-por-N compuesta por dos matrices de bloque
!>    de columna, B1, que es M-por-K, y B2, que es M-por-(N-K):
!>    B = ( B1, B2 ).
!>
!>    Realiza la operación:
!>
!>       ( A_out ) := H * ( A_in ) = ( I - V * T * V**T ) * ( A_in ) =
!>       ( B_out )        ( B_in )                          ( B_in )
!>                  = ( I - ( V1 ) * T * ( V1**T, V2**T ) ) * ( A_in )
!>                          ( V2 )                            ( B_in )
!>     En la entrada:
!>
!>    a) ( A_in )  consiste en dos bloques de columna:
!>       ( B_in )
!>
!>       ( A_in ) = (( A1_in ) ( A2_in )) = (( A1_in ) ( A2_in ))
!>       ( B_in )   (( B1_in ) ( B2_in ))   ((     0 ) ( B2_in )),
!>
!>       donde los bloques de columna son:
!>
!>       (  A1_in )  es una matriz superior trapezoidal K-por-K almacenada en
!>                   la parte triangular superior del array A(1:K,1:K).
!>       (  B1_in )  es una matriz CERO rectangular M-por-K y no se almacena.
!>
!>       ( A2_in )  es una matriz rectangular K-por-(N-K) almacenada
!>                  en el array A(1:K,K+1:N).
!>       ( B2_in )  es una matriz rectangular M-por-(N-K) almacenada
!>                  en el array B(1:M,K+1:N).
!>
!>    b) V = ( V1 )
!>           ( V2 )
!>
!>       donde:
!>       1) si IDENT == 'I', V1 es una matriz identidad K-por-K, no se almacena;
!>       2) si IDENT != 'I', V1 es una matriz unit lower-triangular K-por-K,
!>          almacenada en la parte lower-triangular del array
!>          A(1:K,1:K) (los unos no se almacenan),
!>       y V2 es una matriz rectangular M-por-K almacenada en el array
!>          B(1:M,1:K) (porque en la entrada B1_in es una matriz cero
!>          rectangular que no se almacena y el espacio se usa para almacenar V2).
!>
!>    c) T es una matriz superior-triangular K-por-K almacenada
!>       en el array T(1:K,1:K).
!>
!>    En la salida:
!>
!>    a) ( A_out ) consiste en dos bloques de columna:
!>       ( B_out )
!>
!>       ( A_out ) = (( A1_out ) ( A2_out ))
!>       ( B_out )   (( B1_out ) ( B2_out )),
!>
!>       donde los bloques de columna son:
!>
!>       ( A1_out )  es una matriz K-por-K cuadrada, o una matriz
!>                   superior-triangular K-por-K, si V1 es una
!>                   matriz identidad. AiOut se almacena en
!>                   el array A(1:K,1:K).
!>       ( B1_out )  es una matriz rectangular M-por-K almacenada
!>                   en el array B(1:M,K:N).
!>
!>       ( A2_out )  es una matriz rectangular K-por-(N-K) almacenada
!>                   en el array A(1:K,K+1:N).
!>       ( B2_out )  es una matriz rectangular M-por-(N-K) almacenada
!>                   en el array B(1:M,K+1:N).
!>
!>
!>    La operación anterior se puede representar como la misma operación
!>    en cada bloque de columna:
!>
!>       ( A1_out ) := H * ( A1_in ) = ( I - V * T * V**T ) * ( A1_in )
!>       ( B1_out )        (     0 )                          (     0 )
!>
!>       ( A2_out ) := H * ( A2_in ) = ( I - V * T * V**T ) * ( A2_in )
!>       ( B2_out )        ( B2_in )                          ( B2_in )
!>
!>    Si IDENT != 'I':
!>
!>       El cálculo para el bloque de columna 1:
!>
!>       A1_out: = A1_in - V1*T*(V1**T)*A1_in
!>
!>       B1_out: = - V2*T*(V1**T)*A1_in
!>
!>       El cálculo para el bloque de columna 2, que existe si N > K:
!>
!>       A2_out: = A2_in - V1*T*( (V1**T)*A2_in + (V2**T)*B2_in )
!>
!>       B2_out: = B2_in - V2*T*( (V1**T)*A2_in + (V2**T)*B2_in )

!>    Si IDENT == 'I':

!>       B1_out: = - V2*T**A1_in

slarfb_gett(3) - LAPACK

NAME

slarfb_gett - generates the LAPACK implementation of the blocked Reflector update.

SYNOPSIS

      SUBROUTINE SLARFB_GETT( V, T, W, WSIDE, WTRANS )
!
!  -- LAPACK auxiliary routine (version 3.12.0) --
!  -- Contributed by Jessica L. Niephaus and David_Gonzalez (Oct 2020) --
!
!  Purpose
!  =======
!
!  SLARFB_GETT generates the LAPACK implementation of the blocked
!  Reflector update.
!
!  Arguments
!  =========
!
!  V       (input) REAL array, dimension (NV, K)
!          The matrix V with dimensions NV x K.
!
!  T       (input) REAL array, dimension (K, K)
!          The upper triangular matrix T with dimensions K x K.
!
!  W       (output) REAL array, dimension (NW, K)
!          The matrix W with dimensions NW x K.
!
!  WSIDE   (input) CHARACTER*1
!          Specifies whether W is a row or column block:
!          'L' or 'l': W is a row block, that is, W is of dimension
!                      K x NW, and the LAPACK implementation is of the
!                      form: op(A) := op(A) - V * T * W**T.
!          'R' or 'r': W is a column block, that is, W is of dimension
!                      NW x K, and the LAPACK implementation is of the
!                      form: op(A) := op(A) - V * T * W.
!
!  WTRANS  (input) CHARACTER*1
!          Specifies whether W is transposed or not:
!          'N' or 'n': No transpose, that is, W is not transposed in
!                      the update.
!          'T' or 't': Transpose, that is, W is transposed in the
!                      update.
!
!  =====================================================================
!
!  Local variables
!  ===============
!
!  NV = number of rows of V
!  NW = number of columns of W
!  K = number of columns of V and T and rows of T
!
!  We define A1, B1, A2, B2 as follows:
!  The matrix A is partitioned as
!     A = [ A1 A2 ]
!         [ B1 B2 ]
!  where A1 is K x K, A2 is K x (N-K), B1 is (M-K) x K, B2 is (M-K) x (N-K).
!
!  The matrix V is partitioned as
!     V = [ V1 ]
!         [ V2 ]
!  where V1 is K x K, V2 is (M-K) x K.
!
!  The matrix T is K x K.
!
!  The matrix W is partitioned as
!     W = [ W1 W2 ]
!  where W1 is K x K, W2 is K x (N-K).
!
!  The update is of the form
!     op(A) := op(A) - V * T * W**T
!  or
!     op(A) := op(A) - V * T * W
!
!  Case (A), when V1 is unit lower-triangular, i.e. IDENT != 'I',
!     more computations than in the Case (B).
!
!  if( IDENT != 'I' ) then
!   if ( N > K ) then
!     (First Step - column block 2)
!     col2_(1) W2: = A2
!     col2_(2) W2: = (V1**T) * W2 = (unit_lower_tr_of_(A1)**T) * W2
!     col2_(3) W2: = W2 + (V2**T) * B2 = W2 + (B1**T) * B2
!     col2_(4) W2: = T * W2
!     col2_(5) B2: = B2 - V2 * W2 = B2 - B1 * W2
!     col2_(6) W2: = V1 * W2 = unit_lower_tr_of_(A1) * W2
!     col2_(7) A2: = A2 - W2
!   else
!     (Second Step - column block 1)
!     col1_(1) W1: = A1
!     col1_(2) W1: = (V1**T) * W1 = (unit_lower_tr_of_(A1)**T) * W1
!     col1_(3) W1: = T * W1
!     col1_(4) B1: = - V2 * W1 = - B1 * W1
!     col1_(5) square W1: = V1 * W1 = unit_lower_tr_of_(A1) * W1
!     col1_(6) square A1: = A1 - W1
!   end if
!  end if
!
!  Case (B), when V1 is an identity matrix, i.e. IDENT == 'I',
!     less computations than in the Case (A)
!
!  if( IDENT == 'I' ) then
!   if ( N > K ) then
!     (First Step - column block 2)
!     col2_(1) W2: = A2
!     col2_(3) W2: = W2 + (V2**T) * B2 = W2 + (B1**T) * B2
!     col2_(4) W2: = T * W2
!     col2_(5) B2: = B2 - V2 * W2 = B2 - B1 * W2
!     col2_(7) A2: = A2 - W2
!   else
!     (Second Step - column block 1)
!     col1_(1) W1: = A1
!     col1_(3) W1: = T * W1
!     col1_(4) B1: = - V2 * W1 = - B1 * W1
!     col1_(6) upper-triangular_of_(A1): = A1 - W1
!   end if
!  end if
!
!  Combine these cases (A) and (B) together, this is the resulting
!  algorithm:
!
!  if ( N > K ) then
!
!    (First Step - column block 2)
!
!    col2_(1)  W2: = A2
!    if( IDENT != 'I' ) then
!      col2_(2)  W2: = (V1**T) * W2
!                    = (unit_lower_tr_of_(A1)**T) * W2
!    end if
!    col2_(3)  W2: = W2 + (V2**T) * B2 = W2 + (B1**T) * B2]
!    col2_(4)  W2: = T * W2
!    col2_(5)  B2: = B2 - V2 * W2 = B2 - B1 * W2
!    if( IDENT != 'I' ) then
!      col2_(6)    W2: = V1 * W2 = unit_lower_tr_of_(A1) * W2
!    end if
!    col2_(7) A2: = A2 - W2
!
!  else
!
!  (Second Step - column block 1)
!
!    col1_(1) W1: = A1
!    if( IDENT != 'I' ) then
!      col1_(2) W1: = (V1**T) * W1
!                  = (unit_lower_tr_of_(A1)**T) * W1
!    end if
!    col1_(3) W1: = T * W1
!    col1_(4) B1: = - V2 * W1 = - B1 * W1
!    if( IDENT != 'I' ) then
!      col1_(5) square W1: = V1 * W1 = unit_lower_tr_of_(A1) * W1
!      col1_(6_a) below_diag_of_(A1): =  - below_diag_of_(W1)
!    end if
!    col1_(6_b) up_tr_of_(A1): = up_tr_of_(A1) - up_tr_of_(W1)
!
!  end if
!

Description of the Algorithmic Computation.

In the first step, we compute column block 2, i.e. A2 and B2. Here, we need to use the K-by-(N-K) rectangular workspace matrix W2 that is of the same size as the matrix A2. W2 is stored in the array WORK(1:K,1:(N-K)).

In the second step, we compute column block 1, i.e. A1 and B1. Here, we need to use the K-by-K square workspace matrix W1 that is of the same size as the as the matrix A1. W1 is stored in the array WORK(1:K,1:K).

NOTE: Hence, in this routine, we need the workspace array WORK only of size WORK(1:K,1:max(K,N-K)) so it can hold both W2 from the first step and W1 from the second step.

Case (A), when V1 is unit lower-triangular, i.e. IDENT != 'I', more computations than in the Case (B).

    if( IDENT != 'I' ) then
     if ( N > K ) then
       (First Step - column block 2)
       col2_(1) W2: = A2
       col2_(2) W2: = (V1**T) * W2 = (unit_lower_tr_of_(A1)**T) * W2
       col2_(3) W2: = W2 + (V2**T) * B2 = W2 + (B1**T) * B2
       col2_(4) W2: = T * W2
       col2_(5) B2: = B2 - V2 * W2 = B2 - B1 * W2
       col2_(6) W2: = V1 * W2 = unit_lower_tr_of_(A1) * W2
       col2_(7) A2: = A2 - W2
     else
       (Second Step - column block 1)
       col1_(1) W1: = A1
       col1_(2) W1: = (V1**T) * W1 = (unit_lower_tr_of_(A1)**T) * W1
       col1_(3) W1: = T * W1
       col1_(4) B1: = - V2 * W1 = - B1 * W1
       col1_(5) square W1: = V1 * W1 = unit_lower_tr_of_(A1) * W1
       col1_(6) square A1: = A1 - W1
     end if
    end if

Case (B), when V1 is an identity matrix, i.e. IDENT == 'I', less computations than in the Case (A)

    if( IDENT == 'I' ) then
     if ( N > K ) then
       (First Step - column block 2)
       col2_(1) W2: = A2
       col2_(3) W2: = W2 + (V2**T) * B2 = W2 + (B1**T) * B2
       col2_(4) W2: = T * W2
       col2_(5) B2: = B2 - V2 * W2 = B2 - B1 * W2
       col2_(7) A2: = A2 - W2
     else
       (Second Step - column block 1)
       col1_(1) W1: = A1
       col1_(3) W1: = T * W1
       col1_(4) B1: = - V2 * W1 = - B1 * W1
       col1_(6) upper-triangular_of_(A1): = A1 - W1
     end if
    end if

Combine these cases (A) and (B) together, this is the resulting algorithm:

    if ( N > K ) then

      (First Step - column block 2)

      col2_(1)  W2: = A2
      if( IDENT != 'I' ) then
        col2_(2)  W2: = (V1**T) * W2
                      = (unit_lower_tr_of_(A1)**T) * W2
      end if
      col2_(3)  W2: = W2 + (V2**T) * B2 = W2 + (B1**T) * B2]
      col2_(4)  W2: = T * W2
      col2_(5)  B2: = B2 - V2 * W2 = B2 - B1 * W2
      if( IDENT != 'I' ) then
        col2_(6)    W2: = V1 * W2 = unit_lower_tr_of_(A1) * W2
      end if
      col2_(7) A2: = A2 - W2

    else

    (Second Step - column block 1)

      col1_(1) W1: = A1
      if( IDENT != 'I' ) then
        col1_(2) W1: = (V1**T) * W1
                    = (unit_lower_tr_of_(A1)**T) * W1
      end if
      col1_(3) W1: = T * W1
      col1_(4) B1: = - V2 * W1 = - B1 * W1
      if( IDENT != 'I' ) then
        col1_(5) square W1: = V1 * W1 = unit_lower_tr_of_(A1) * W1
        col1_(6_a) below_diag_of_(A1): =  - below_diag_of_(W1)
      end if
      col1_(6_b) up_tr_of_(A1): = up_tr_of_(A1) - up_tr_of_(W1)

    end if
