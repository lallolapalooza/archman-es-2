subroutine slarzt (character direct, character storev, integer n, integer k, real, dimension( ldv, * ) v, integer ldv, real, dimension( * ) tau, real, dimension( ldt, * ) t, integer ldt)

SLARZT forma el factor triangular T de un reflector de bloque H = I - vtvH.

Propósito:

!>
!> SLARZT forma el factor triangular T de un reflector de bloque real
!> H de orden > n, que se define como un producto de k reflectores elementales.
!>
!> Si DIRECT = 'F', H = H(1) H(2) . . . H(k) y T es triangular superior;
!>
!> Si DIRECT = 'B', H = H(k) . . . H(2) H(1) y T es triangular inferior.
!>
!> Si STOREV = 'C', el vector que define el reflector elemental
!> H(i) se almacena en la i-ésima columna del array V, y
!>
!>    H  =  I - V * T * V**T
!>
!> Si STOREV = 'R', el vector que define el reflector elemental
!> H(i) se almacena en la i-ésima fila del array V, y
!>
!>    H  =  I - V**T * T * V
!>
!> Actualmente, solo se admiten STOREV = 'R' y DIRECT = 'B'.
!>

Parámetros

DIRECT 
!>          DIRECT es CHARACTER*1
!>          Especifica el orden en que los reflectores elementales se
!>          multiplican para formar el reflector de bloque:
!>          = 'F': H = H(1) H(2) . . . H(k) (Adelante, todavía no soportado)
!>          = 'B': H = H(k) . . . H(2) H(1) (Atrás)
!>

STOREV

!>          STOREV es CHARACTER*1
!>          Especifica cómo se almacenan los vectores que definen los
!>          reflectores elementales (ver también Detalles adicionales):
!>          = 'C': en columnas (todavía no soportado)
!>          = 'R': en filas
!>

N

!>          N es INTEGER
!>          El orden del reflector de bloque H. N >= 0.
!>

K

!>          K es INTEGER
!>          El orden del factor triangular T (= el número de
!>          reflectores elementales). K >= 1.
!>

V

!>          V es un array REAL, de dimensiones
!>                               (LDV,K) si STOREV = 'C'
!>                               (LDV,N) si STOREV = 'R'
!>          La matriz V. Ver detalles adicionales.
!>

LDV

!>          LDV es INTEGER
!>          La dimensión principal del array V.
!>          Si STOREV = 'C', LDV >= max(1,N); si STOREV = 'R', LDV >= K.
!>

TAU

!>          TAU es un array REAL, de dimensión (K)
!>          TAU(i) debe contener el factor escalar del reflector
!>          elemental H(i).
!>

T

!>          T es un array REAL, de dimensiones (LDT,K)
!>          El factor triangular T de k por k del reflector de bloque.
!>          Si DIRECT = 'F', T es triangular superior; si DIRECT = 'B', T es
!>          triangular inferior. El resto del array no se utiliza.
!>

LDT

!>          LDT es INTEGER
!>          La dimensión principal del array T. LDT >= K.
!>
