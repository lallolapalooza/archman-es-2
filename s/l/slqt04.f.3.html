subroutine slqt04 (integer m, integer n, integer nb, real, dimension(6) result)

SLQT04

Propósito:

!>
!> SLQT04 prueba SGELQT y SGEMLQT.
!>

Parámetros

M 
!>          M es INTEGER
!>          Número de filas en la matriz de prueba.
!>

N

!>          N es INTEGER
!>          Número de columnas en la matriz de prueba.
!>

NB

!>          NB es INTEGER
!>          Tamaño de bloque de la matriz de prueba.  NB <= Min(M,N).
!>

RESULT

!>          RESULT es un array REAL, dimension (6)
!>          Resultados de cada una de las seis pruebas siguientes.
!>
!>          RESULT(1) = | A - L Q |
!>          RESULT(2) = | I - Q Q^H |
!>          RESULT(3) = | Q C - Q C |
!>          RESULT(4) = | Q^H C - Q^H C |
!>          RESULT(5) = | C Q - C Q |
!>          RESULT(6) = | C Q^H - C Q^H |
!>
