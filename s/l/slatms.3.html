subroutine slatms (integer m, integer n, character dist, integer, dimension( 4 ) iseed, character sym, real, dimension( * ) d, integer mode, real cond, real dmax, integer kl, integer ku, character pack, real, dimension( lda, * ) a, integer lda, real, dimension( * ) work, integer info)

SLATMS

Propósito:

!>
!>    SLATMS genera matrices aleatorias con valores singulares especificados
!>    (o simétricas/hermitianas con valores propios especificados)
!>    para probar programas LAPACK.
!>
!>    SLATMS opera aplicando la siguiente secuencia de
!>    operaciones:
!>
!>      Establece la diagonal en D, donde D puede ser de entrada o
!>         calculada según MODE, COND, DMAX y SYM
!>         como se describe a continuación.
!>
!>      Genera una matriz con la estructura de banda apropiada, mediante uno
!>         de dos métodos:
!>
!>      Método A:
!>          Genera una matriz densa M x N multiplicando D a la izquierda
!>              y a la derecha por matrices unitarias aleatorias, y luego:
!>
!>          Reduce el ancho de banda según KL y KU, usando
!>          transformaciones de Householder.
!>
!>      Método B:
!>          Convierte la matriz de ancho de banda 0 (es decir, diagonal) a una
!>              matriz de ancho de banda 1 usando rotaciones de Givens, 
!>              elementos fuera de banda de vuelta, de manera similar a QR; luego
!>              convierte el ancho de banda 1 a ancho de banda 2, etc.
!>              Tenga en cuenta que para anchos de banda razonablemente pequeños (relativos a
!>              M y N) esto requiere menos almacenamiento, ya que no se genera una matriz densa.
!>              Además, para matrices simétricas, solo se genera un triángulo.
!>
!>      Se elige el Método A si el ancho de banda es una fracción grande del
!>          orden de la matriz, y LDA es al menos M (por lo que se puede almacenar una matriz densa).
!>          Se elige el Método B si el ancho de banda
!>          es pequeño (< 1/2 N para simétrica, < .3 N+M para
!>          no simétrica), o LDA es menor que M y no menor que el
!>          ancho de banda.
!>
!>      Empaqueta la matriz si se desea. Las opciones especificadas por PACK son:
!>         sin empaquetar
!>         poner a cero la mitad superior (si es simétrica)
!>         poner a cero la mitad inferior (si es simétrica)
!>         almacenar la mitad superior en formato columna (si es simétrica o
!>               triangular superior)
!>         almacenar la mitad inferior en formato columna (si es simétrica o
!>               triangular inferior)
!>         almacenar el triángulo inferior en formato de banda (si es simétrica
!>               o triangular inferior)
!>         almacenar el triángulo superior en formato de banda (si es simétrica
!>               o triangular superior)
!>         almacenar toda la matriz en formato de banda
!>      Si se elige el Método B y se especifica el formato de banda, entonces la
!>         matriz se generará en formato de banda, por lo que no será necesario volver a empaquetar.
!>

Parámetros

M 
!>          M es INTEGER
!>           El número de filas de A. No modificado.
!>

N

!>          N es INTEGER
!>           El número de columnas de A. No modificado.
!>

DIST

!>          DIST es CHARACTER*1
!>           En la entrada, DIST especifica el tipo de distribución a utilizar
!>           para generar los valores propios/singulares aleatorios.
!>           'U' => UNIFORME( 0, 1 )  ('U' para uniforme)
!>           'S' => UNIFORME( -1, 1 ) ('S' para simétrica)
!>           'N' => NORMAL( 0, 1 )   ('N' para normal)
!>           No modificado.
!>

ISEED

!>          ISEED es array INTEGER, dimension ( 4 )
!>           En la entrada, ISEED especifica la semilla del generador de números
!>           aleatorios. Deben estar entre 0 y 4095 inclusive,
!>           y ISEED(4) debe ser impar. El generador de números aleatorios
!>           utiliza una secuencia congruencial lineal limitada a enteros pequeños,
!>           y por lo tanto debería producir números aleatorios independientes de la máquina.
!>           Los valores de ISEED se cambian al salir, y pueden usarse en la siguiente llamada
!>           a SLATMS para continuar la misma secuencia de números aleatorios.
!>           Modificado al salir.
!>

SYM

!>          SYM es CHARACTER*1
!>           Si SYM='S' o 'H', la matriz generada es simétrica, con
!>             valores propios especificados por D, COND, MODE y DMAX; pueden
!>             ser positivos, negativos o cero.
!>           Si SYM='P', la matriz generada es simétrica, con
!>             valores propios (= valores singulares) especificados por D, COND,
!>             MODE y DMAX; no serán negativos.
!>           Si SYM='N', la matriz generada es no simétrica, con
!>             valores singulares especificados por D, COND, MODE y DMAX;
!>             no serán negativos.
!>           No modificado.
!>

D

!>          D es array REAL, dimension ( MIN( M , N ) )
!>           Este array se utiliza para especificar los valores singulares o
!>           valores propios de A (ver SYM, arriba). Si MODE=0, entonces D se
!>           supone que contiene los valores singulares/propios, de lo contrario
!>           se calcularán según MODE, COND y DMAX,
!>           y se colocarán en D.
!>           Modificado si MODE es distinto de cero.
!>

MODE

!>          MODE es INTEGER
!>           En la entrada, esto describe cómo se especificarán los valores singulares/propios:
!>           MODE = 0 significa usar D como entrada
!>           MODE = 1 establece D(1)=1 y D(2:N)=1.0/COND
!>           MODE = 2 establece D(1:N-1)=1 y D(N)=1.0/COND
!>           MODE = 3 establece D(I)=COND**(-(I-1)/(N-1))
!>           MODE = 4 establece D(i)=1 - (i-1)/(N-1)*(1 - 1/COND)
!>           MODE = 5 establece D a números aleatorios en el rango
!>                    ( 1/COND , 1 ) tales que sus logaritmos
!>                    están distribuidos uniformemente.
!>           MODE = 6 establece D a números aleatorios de la misma distribución
!>                    que el resto de la matriz.
!>           MODE < 0 tiene el mismo significado que ABS(MODE), excepto que
!>              el orden de los elementos de D se invierte.
!>           Por lo tanto, si MODE es positivo, D tiene entradas que van de
!>              1 a 1/COND, si es negativo, de 1/COND a 1,
!>           Si SYM='S' o 'H', y MODE no es 0, 6 ni -6, entonces
!>              los elementos de D también se multiplicarán por un signo aleatorio
!>              (es decir, +1 o -1).
!>           No modificado.
!>

COND

!>          COND es REAL
!>           En la entrada, se usa como se describe en MODE arriba.
!>           Si se usa, debe ser >= 1. No modificado.
!>

DMAX

!>          DMAX es REAL
!>           Si MODE no es -6, 0 ni 6, el contenido de D, como se
!>           calcula según MODE y COND, se escalará por
!>           DMAX / max(abs(D(i))); por lo tanto, el valor propio o
!>           singular absoluto máximo (que es la norma) será abs(DMAX).
!>           Tenga en cuenta que DMAX no tiene por qué ser positivo: si DMAX es negativo
!>           (o cero), D se escalará por un número negativo (o cero).
!>           No modificado.
!>

KL

!>          KL es INTEGER
!>           Esto especifica el ancho de banda inferior de la matriz. Por
!>           ejemplo, KL=0 implica triangular superior, KL=1 implica Hessenberg
!>           superior, y KL al menos M-1 significa que la matriz
!>           tiene un ancho de banda inferior completo. KL debe ser igual a KU si la matriz
!>           es simétrica.
!>           No modificado.
!>

KU

!>          KU es INTEGER
!>           Esto especifica el ancho de banda superior de la matriz. Por
!>           ejemplo, KU=0 implica triangular inferior, KU=1 implica Hessenberg
!>           inferior, y KU al menos N-1 significa que la matriz
!>           tiene un ancho de banda superior completo. KL debe ser igual a KU



NAME


SLATMS


generates test matrices in the LAPACK machine-independent format

SUBROUTINE SLATMS(
  M, N, DIST, DTYPE, SUBTYP, PACK, A, LDA, WORK, INFO )

      CHARACTER*1          PACK
      REAL                 A( LDA, * )
      INTEGER              M, N, LDA, INFO
      REAL                 WORK( * )
      CHARACTER*1          DIST, DTYPE, SUBTYP








Function/Subroutine Documentation


This subroutine generates test matrices in the LAPACK machine-independent
    format. These matrices are used for testing LAPACK routines. The general
    form of the test matrices is:


      A = U * D * V**T



where U and V are orthogonal matrices, and D is a diagonal matrix with
    specified singular values. Various properties of the matrices can be controlled
    by the input parameters.


Parameters:


M


      M is INTEGER
       The number of rows of the matrix A.  M must be non-negative.
       Not modified.



N


      N is INTEGER
       The number of columns of the matrix A.  N must be non-negative.
       Not modified.



DIST


      DIST is CHARACTER*1
       Specifies the distribution of the singular values:
       'U' => uniformly distributed in [0,1]
       'S' => same as 'U'
       'D' => diagonally distributed with largest singular value 1 and
              smallest singular value given by COND
       'N' => normally distributed (mean 0, variance 1)
       'L' => large random values (mean 0, variance 100)
       'E' => exponentially distributed with mean 1
       'I' => distributed in a way that SLATM1 can generate the matrix
              with the desired condition number (e.g. largest singular
              value 1, smallest singular value 1/COND)
       Not modified.



DTYPE


      DTYPE is CHARACTER*1
       Specifies the type of matrix to be generated:
       'G' => general rectangular matrix
       'S' => symmetric matrix
       'H' => symmetric-hermitian matrix (complex only)
       'P' => positive definite matrix
       Not modified.



SUBTYP


      SUBTYP is CHARACTER*1
       Specifies the subtype of matrix to be generated:
       'A' => arbitrary matrix
       'R' => random orthogonal matrix
       'P' => permutation matrix
       'B' => block diagonal matrix
       'L' => lower triangular matrix
       'U' => upper triangular matrix
       'T' => tridiagonal matrix
       'P' => positive definite
       'C' => covariance matrix
       'B' => band matrix
       'Q' => quasi-triangular matrix
       Not modified.



PACK


      PACK is CHARACTER*1
       This specifies packing of matrix as follows:
       'N' => no packing
       'U' => zero out all subdiagonal entries (if symmetric)
       'L' => zero out all superdiagonal entries (if symmetric)
       'C' => store the upper triangle columnwise
              (only if the matrix is symmetric or upper triangular)
       'R' => store the lower triangle columnwise
              (only if the matrix is symmetric or lower triangular)
       'B' => store the lower triangle in band storage scheme
              (only if matrix symmetric or lower triangular)
       'Q' => store the upper triangle in band storage scheme
              (only if matrix symmetric or upper triangular)
       'Z' => store the entire matrix in band storage scheme
                  (pivoting can be provided for by using this
                  option to store A in the trailing rows of
                  the allocated storage)

       Using these options, the various LAPACK packed and banded
       storage schemes can be obtained:
       GB               - use 'Z'
       PB, SB or TB     - use 'B' or 'Q'
       PP, SP or TP     - use 'C' or 'R'

       If two calls to SLATMS differ only in the PACK parameter,
       they will generate mathematically equivalent matrices.
       Not modified.



A


      A is REAL array, dimension ( LDA, N )
       On exit A is the desired test matrix.  A is first generated
       in full (unpacked) form, and then packed, if so specified
       by PACK.  Thus, the first M elements of the first N
       columns will always be modified.  If PACK specifies a
       packed or banded storage scheme, all LDA elements of the
       first N columns will be modified; the elements of the
       array which do not correspond to elements of the generated
       matrix are set to zero.
       Modified.



LDA


      LDA is INTEGER
       LDA specifies the first dimension of A as declared in the
       calling program.  If PACK='N', 'U', 'L', 'C', or 'R', then
       LDA must be at least M.  If PACK='B' or 'Q', then LDA must
       be at least MIN( KL, M-1) (which is equal to MIN(KU,N-1)).
       If PACK='Z', LDA must be large enough to hold the packed
       array: MIN( KU, N-1) + MIN( KL, M-1) + 1.
       Not modified.



WORK


      WORK is REAL array, dimension ( 3*MAX( N , M ) )
       Workspace.
       Modified.



INFO


      INFO is INTEGER
       Error code.  On exit, INFO will be set to one of the
       following values:
         0 => normal return
        -1 => M negative or unequal to N and SYM='S', 'H', or 'P'
        -2 => N negative
        -3 => DIST illegal string
        -5 => SYM illegal string
        -7 => MODE not in range -6 to 6
        -8 => COND less than 1.0, and MODE neither -6, 0 nor 6
       -10 => KL negative
       -11 => KU negative, or SYM='S' or 'H' and KU not equal to KL
       -12 => PACK illegal string, or PACK='U' or 'L', and SYM='N';
              or PACK='C' or 'Q' and SYM='N' and KL is not zero;
              or PACK='R' or 'B' and SYM='N' and KU is not zero;
              or PACK='U', 'L', 'C', 'R', 'B', or 'Q', and M is not
              N.
       -14 => LDA is less than M, or PACK='Z' and LDA is less than
              MIN(KU,N-1) + MIN(KL,M-1) + 1.
         1  => Error return from SLATM1
         2  => Cannot scale to DMAX (max. sing. value is 0)
         3  => Error return from SLAGGE or SLAGSY
