subroutine slargv (integer n, real, dimension( * ) x, integer incx, real, dimension( * ) y, integer incy, real, dimension( * ) c, integer incc)

SLARG V genera un vector de rotaciones planas reales con cosenos reales y senos reales.

Propósito:

!>
!> SLARG V genera un vector de rotaciones planas reales, determinado por
!> elementos de los vectores reales x e y. Para i = 1,2,...,n
!>
!>    (  c(i)  s(i) ) ( x(i) ) = ( a(i) )
!>    ( -s(i)  c(i) ) ( y(i) ) = (   0  )
!>

Parámetros

N 
!>          N es ENTERO
!>          El número de rotaciones planas a generar.
!>

X

!>          X es un array REAL,
!>                         dimensión (1+(N-1)*INCX)
!>          En la entrada, el vector x.
!>          En la salida, x(i) se sobrescribe con a(i), para i = 1,...,n.
!>

INCX

!>          INCX es ENTERO
!>          El incremento entre elementos de X. INCX > 0.
!>

Y

!>          Y es un array REAL,
!>                         dimensión (1+(N-1)*INCY)
!>          En la entrada, el vector y.
!>          En la salida, los senos de las rotaciones planas.
!>

INCY

!>          INCY es ENTERO
!>          El incremento entre elementos de Y. INCY > 0.
!>

C

!>          C es un array REAL, dimensión (1+(N-1)*INCC)
!>          Los cosenos de las rotaciones planas.
!>

INCC

!>          INCC es ENTERO
!>          El incremento entre elementos de C. INCC > 0.
!>
