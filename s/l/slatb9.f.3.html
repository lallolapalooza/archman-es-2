subroutine slatb9 (character*3 path, integer imat, integer m, integer p, integer n, character type, integer kla, integer kua, integer klb, integer kub, real anorm, real bnorm, integer modea, integer modeb, real cndnma, real cndnmb, character dista, character distb)

SLATB9

Propósito:

!>
!> SLATB9 establece los parámetros para el generador de matrices basándose en el tipo
!> de matriz a generar.
!>

Parámetros

PATH 
!>          PATH es CHARACTER*3
!>          El nombre de la ruta LAPACK.
!>

IMAT

!>          IMAT es INTEGER
!>          Una clave entera que describe qué matriz generar para esta
!>          ruta.
!>          = 1:   A: diagonal, B: triangular superior
!>          = 2:   A: triangular superior, B: triangular superior
!>          = 3:   A: triangular inferior, B: triangular superior
!>          En otro caso:  A: general densa, B: general densa
!>

M

!>          M es INTEGER
!>          El número de filas de la matriz a generar.
!>

P

!>          P es INTEGER
!>

N

!>          N es INTEGER
!>          El número de columnas de la matriz a generar.
!>

TYPE

!>          TYPE es CHARACTER*1
!>          El tipo de matriz a generar:
!>          = 'S':  matriz simétrica;
!>          = 'P':  matriz simétrica positiva (semi)definida;
!>          = 'N':  matriz no simétrica.
!>

KLA

!>          KLA es INTEGER
!>          El ancho de banda inferior de la matriz a generar.
!>

KUA

!>          KUA es INTEGER
!>          El ancho de banda superior de la matriz a generar.
!>

KLB

!>          KLB es INTEGER
!>          El ancho de banda inferior de la matriz a generar.
!>

KUB

!>          KUA es INTEGER
!>          El ancho de banda superior de la matriz a generar.
!>

ANORM

!>          ANORM es REAL
!>          La norma deseada de la matriz a generar. La matriz diagonal
!>          de valores singulares o valores propios se escala por este
!>          valor.
!>

BNORM

!>          BNORM es REAL
!>          La norma deseada de la matriz a generar. La matriz diagonal
!>          de valores singulares o valores propios se escala por este
!>          valor.
!>

MODEA

!>          MODEA es INTEGER
!>          Una clave que indica cómo elegir el vector de valores propios.
!>

MODEB

!>          MODEB es INTEGER
!>          Una clave que indica cómo elegir el vector de valores propios.
!>

CNDNMA

!>          CNDNMA es REAL
!>          El número de condición deseado.
!>

CNDNMB

!>          CNDNMB es REAL
!>          El número de condición deseado.
!>

DISTA

!>          DISTA es CHARACTER*1
!>          El tipo de distribución a utilizar por el generador de números
!>          aleatorios.
!>

DISTB

!>          DISTB es CHARACTER*1
!>          El tipo de distribución a utilizar por el generador de números
!>          aleatorios.
!>
