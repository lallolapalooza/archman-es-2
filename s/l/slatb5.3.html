subroutine slatb5 (character*3 path, integer imat, integer n, character type, integer kl, integer ku, real anorm, integer mode, real cndnum, character dist)

SLATB5

Propósito:

!>
!> SLATB5 establece los parámetros para el generador de matrices basándose en el tipo
!> de matriz a generar.
!>

Parámetros

PATH 
!>          PATH es CHARACTER*3
!>          El nombre de la ruta LAPACK.
!>

IMAT

!>          IMAT es INTEGER
!>          Una clave entera que describe qué matriz generar para esta
!>          ruta.
!>

N

!>          N es INTEGER
!>          El número de filas y columnas en la matriz a generar.
!>

TYPE

!>          TYPE es CHARACTER*1
!>          El tipo de matriz a generar:
!>          = 'S':  matriz simétrica
!>          = 'P':  matriz definida positiva (semi)simétrica
!>          = 'N':  matriz no simétrica
!>

KL

!>          KL es INTEGER
!>          El ancho de banda inferior de la matriz a generar.
!>

KU

!>          KU es INTEGER
!>          El ancho de banda superior de la matriz a generar.
!>

ANORM

!>          ANORM es REAL
!>          La norma deseada de la matriz a generar. La matriz diagonal
!>          de valores singulares o valores propios se escala por este
!>          valor.
!>

MODE

!>          MODE es INTEGER
!>          Una clave que indica cómo elegir el vector de valores propios.
!>

CNDNUM

!>          CNDNUM es REAL
!>          El número de condición deseado.
!>

DIST

!>          DIST es CHARACTER*1
!>          El tipo de distribución a utilizar por el generador de números
!>          aleatorios.
!>
