subroutine slaqr1 (integer n, real, dimension( ldh, * ) h, integer ldh, real sr1, real si1, real sr2, real si2, real, dimension( * ) v)

SLAQR1 establece un múltiplo escalar de la primera columna del producto de una matriz H de 2x2 o 3x3 y desplazamientos especificados.

Propósito:

!>
!>      Dada una matriz H de 2x2 o 3x3, SLAQR1 establece v como un
!>      múltiplo escalar de la primera columna del producto
!>
!>      (*)  K = (H - (sr1 + i*si1)*I)*(H - (sr2 + i*si2)*I)
!>
!>      escalando para evitar desbordamientos y la mayoría de los desbordamientos
!>      inversos. Se asume que bien
!>
!>              1) sr1 = sr2 y si1 = -si2
!>          o
!>              2) si1 = si2 = 0.
!>
!>      Esto es útil para iniciar protuberancias de desplazamiento implícito doble
!>      en el algoritmo QR.
!>

Parámetros

N 
!>          N es INTEGER
!>              Orden de la matriz H. N debe ser 2 o 3.
!>

H

!>          H es un array REAL, de dimensión (LDH,N)
!>              La matriz H de 2x2 o 3x3 en (*).
!>

LDH

!>          LDH es INTEGER
!>              La dimensión principal de H según lo declarado en
!>              el procedimiento de llamada. LDH >= N
!>

SR1

!>          SR1 es REAL
!>

SI1

!>          SI1 es REAL
!>

SR2

!>          SR2 es REAL
!>

SI2

!>          SI2 es REAL
!>              Los desplazamientos en (*).
!>

V

!>          V es un array REAL, de dimensión (N)
!>              Un múltiplo escalar de la primera columna de la
!>              matriz K en (*).
!>
