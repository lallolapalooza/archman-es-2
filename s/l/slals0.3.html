subroutine slals0 (integer icompq, integer nl, integer nr, integer sqre, integer nrhs, real, dimension( ldb, * ) b, integer ldb, real, dimension( ldbx, * ) bx, integer ldbx, integer, dimension( * ) perm, integer givptr, integer, dimension( ldgcol, * ) givcol, integer ldgcol, real, dimension( ldgnum, * ) givnum, integer ldgnum, real, dimension( ldgnum, * ) poles, real, dimension( * ) difl, real, dimension( ldgnum, * ) difr, real, dimension( * ) z, integer k, real c, real s, real, dimension( * ) work, integer info)

SLALS0 aplica la multiplicación hacia atrás de los factores de la matriz de vectores singulares izquierda o derecha de una matriz diagonal añadida a la matriz de lado derecho B en la resolución del problema de mínimos cuadrados utilizando el enfoque SVD de división y conquista. Usado por sgelsd.

Propósito:

!>
!> SLALS0 aplica hacia atrás los factores multiplicadores de la matriz de
!> vectores singulares izquierda o derecha de una matriz diagonal añadida a la
!> matriz del lado derecho B en la resolución del problema de mínimos
!> cuadrados utilizando el enfoque SVD de división y conquista.
!>
!> Para la matriz de vectores singulares izquierdos, se involucran tres tipos de
!> matrices ortogonales:
!>
!> (1L) Rotaciones de Givens: el número de tales rotaciones es GIVPTR; los
!>      pares de columnas/filas a los que se aplicaron están almacenados en
!>      GIVCOL; y los valores C y S de estas rotaciones están almacenados en
!>      GIVNUM.
!>
!> (2L) Permutación. La fila (NL+1) de B se moverá a la primera fila, y para
!>      J=2:N, la fila PERM(J) de B se moverá a la fila J.
!>
!> (3L) La matriz de vectores singulares izquierdos de la matriz restante.
!>
!> Para la matriz de vectores singulares derechos, se involucran cuatro tipos de
!> matrices ortogonales:
!>
!> (1R) La matriz de vectores singulares derechos de la matriz restante.
!>
!> (2R) Si SQRE = 1, una rotación de Givens adicional para generar el espacio
!>      nulo derecho.
!>
!> (3R) La transformación inversa de (2L).
!>
!> (4R) La transformación inversa de (1L).
!>

Parámetros

ICOMPQ 
!>          ICOMPQ es INTEGER
!>         Especifica si los vectores singulares se calcularán en forma
!>         factorizada:
!>         = 0: Matriz de vectores singulares izquierdos.
!>         = 1: Matriz de vectores singulares derechos.
!>

NL

!>          NL es INTEGER
!>         La dimensión de fila del bloque superior. NL >= 1.
!>

NR

!>          NR es INTEGER
!>         La dimensión de fila del bloque inferior. NR >= 1.
!>

SQRE

!>          SQRE es INTEGER
!>         = 0: el bloque inferior es una matriz cuadrada de NRxNR.
!>         = 1: el bloque inferior es una matriz rectangular de NRx(NR+1).
!>
!>         La matriz bidiagonal tiene dimensión de fila N = NL + NR + 1,
!>         y dimensión de columna M = N + SQRE.
!>

NRHS

!>          NRHS es INTEGER
!>         El número de columnas de B y BX. NRHS debe ser al menos 1.
!>

B

!>          B es array REAL, dimensión ( LDB, NRHS )
!>         En la entrada, B contiene los lados derechos del problema de mínimos
!>         cuadrados en las filas 1 a M. En la salida, B contiene la solución
!>         X en las filas 1 a N.
!>

LDB

!>          LDB es INTEGER
!>         La dimensión principal de B. LDB debe ser al menos
!>         max(1, MAX( M, N ) ).
!>

BX

!>          BX es array REAL, dimensión ( LDBX, NRHS )
!>

LDBX

!>          LDBX es INTEGER
!>         La dimensión principal de BX.
!>

PERM

!>          PERM es array INTEGER, dimensión ( N )
!>         Las permutaciones (de deflación y ordenamiento) aplicadas
!>         a los dos bloques.
!>

GIVPTR

!>          GIVPTR es INTEGER
!>         El número de rotaciones de Givens que ocurrieron en esta
!>         subproblema.
!>

GIVCOL

!>          GIVCOL es array INTEGER, dimensión ( LDGCOL, 2 )
!>         Cada par de números indica un par de filas/columnas
!>         involucradas en una rotación de Givens.
!>

LDGCOL

!>          LDGCOL es INTEGER
!>         La dimensión principal de GIVCOL, debe ser al menos N.
!>

GIVNUM

!>          GIVNUM es array REAL, dimensión ( LDGNUM, 2 )
!>         Cada número indica el valor C o S utilizado en la
!>         rotación de Givens correspondiente.
!>

LDGNUM

!>          LDGNUM es INTEGER
!>         La dimensión principal de los arrays DIFR, POLES y
!>         GIVNUM, debe ser al menos K.
!>

POLES

!>          POLES es array REAL, dimensión ( LDGNUM, 2 )
!>         En la entrada, POLES(1:K, 1) contiene los nuevos valores
!>         singulares obtenidos al resolver la ecuación secular, y
!>         POLES(1:K, 2) es un array que contiene los polos en la
!>         ecuación secular.
!>

DIFL

!>          DIFL es array REAL, dimensión ( K ).
!>         En la entrada, DIFL(I) es la distancia entre el valor
!>         singular I-ésimo actualizado (no desdeflado) y el valor
!>         singular I-ésimo antiguo (no desdeflado).
!>

DIFR

!>          DIFR es array REAL, dimensión ( LDGNUM, 2 ).
!>         En la entrada, DIFR(I, 1) contiene las distancias entre
!>         el valor singular I-ésimo actualizado (no desdeflado) y el
!>         valor singular (I+1)-ésimo antiguo (no desdeflado). Y DIFR(I, 2)
!>         es el factor de normalización para el vector singular
!>         derecho I-ésimo.
!>

Z

!>          Z es array REAL, dimensión ( K )
!>         Contiene los componentes del vector de fila de actualización
!>         ajustado por deflación.
!>

K

!>          K es INTEGER
!>         Contiene la dimensión de la matriz no desdeflada,
!>         este es el orden de la ecuación secular relacionada. 1 <= K <= N.
!>

C

!>          C es REAL
!>         C contiene basura si SQRE = 0 y el valor C de una rotación
!>         de Givens relacionada con el espacio nulo derecho si SQRE = 1.
!>

S

!>          S es REAL
!>         S contiene basura si SQRE = 0 y el valor S de una rotación
!>         de Givens relacionada con el espacio nulo derecho si SQRE = 1.
!>

WORK

!>          WORK es array REAL, dimensión ( K )
!>

INFO

!>          INFO es INTEGER
!>          = 0:  salida exitosa.
!>          < 0:  si INFO = -i, el i-ésimo argumento tenía un valor ilegal.
!>

SUBROUTINE SLALS0(ICOMPV, NL, NR, SQRE, N, NCL, NCR, ISCALE, A, LDA, D, E, U, LDU, V, LDV, S, SL, SR, WORK, IWORK, INFO)

Function/Subroutine Documentation

SLALS0

Esta subroutine encuentra los valores singulares para una matriz diagonal en bloque dada la matriz diagonal en bloque dada por el usuario.

Los argumentos son:

ICOMPV
(input) INTEGER ICOMPV = .TRUE. siempre que se deseen los vectores singulares de la matriz diagonal en bloque. Si ICOMPV = .FALSE., los vectores singulares no son necesarios.
NL
(input) INTEGER El número de valores singulares en la parte izquierda de la matriz diagonal en bloque.
NR
(input) INTEGER El número de valores singulares en la parte derecha de la matriz diagonal en bloque.
SQRE
(input) INTEGER SQRE indica el número de columnas de la submatriz de A (N x NCL) en la primera columna. SQRE = 1 si NCL = N, y SQRE = 0 si NCL < N.
N
(input) INTEGER El orden de la matriz diagonal en bloque.
NCL
(input) INTEGER El número de columnas de la primera submatriz en la parte izquierda de la matriz diagonal en bloque.
NCR
(input) INTEGER El número de columnas de la última submatriz en la parte derecha de la matriz diagonal en bloque.
ISCALE
(input) INTEGER Valor de la escala para los bloques de la matriz.
A
(input) REAL de dimensiones (LDA, N) Los bloques de la matriz diagonal en bloque. Los bloques se almacenan en la matriz A de forma que el bloque (I, J) se almacena en A(I, J). Las submatrices de cada bloque se almacenan en la parte superior izquierda de A.
LDA
(input) INTEGER El ancho de la matriz A.
D
(output) REAL de dimensiones (N) Los valores singulares de la parte izquierda de la matriz diagonal en bloque. Ordenados en orden descendente.
E
(output) REAL de dimensiones (N) Los valores singulares de la parte derecha de la matriz diagonal en bloque. Ordenados en orden descendente.
U
(output) REAL de dimensiones (LDU, N) Si ICOMPV = .TRUE., U contiene los vectores singulares de la parte izquierda de la matriz diagonal en bloque.
LDU
(input) INTEGER El ancho de la matriz U.
V
(output) REAL de dimensiones (LDV, N) Si ICOMPV = .TRUE., V contiene los vectores singulares de la parte derecha de la matriz diagonal en bloque.
LDV
(input) INTEGER El ancho de la matriz V.
S
(output) REAL de dimensiones (N) Los valores singulares de la matriz diagonal en bloque. Ordenados en orden descendente.
SL
(output) REAL de dimensiones (NL) Los valores singulares de la parte izquierda de la matriz diagonal en bloque. Ordenados en orden descendente.
SR
(output) REAL de dimensiones (NR) Los valores singulares de la parte derecha de la matriz diagonal en bloque. Ordenados en orden descendente.
WORK
(workspace) REAL de dimensiones (2*N) Espacio de trabajo.
IWORK
(workspace) INTEGER de dimensiones (N) Espacio de trabajo.
INFO
(output) INTEGER = 0: éxito < 0: si el i-ésimo argumento es inválido.
