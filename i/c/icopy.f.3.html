subroutine icopy (integer n, integer, dimension( * ) sx, integer incx, integer, dimension( * ) sy, integer incy)

ICOPY

Propósito:

!>
!> ICOPY copia un vector entero x a un vector entero y.
!> Utiliza bucles desenrollados para incrementos iguales a 1.
!>

Parámetros

N 
!>          N es ENTERO
!>          La longitud de los vectores SX y SY.
!>

SX

!>          SX es un array de ENTEROS, dimension (1+(N-1)*abs(INCX))
!>          El vector X.
!>

INCX

!>          INCX es ENTERO
!>          El espaciado entre elementos consecutivos de SX.
!>

SY

!>          SY es un array de ENTEROS, dimension (1+(N-1)*abs(INCY))
!>          El vector Y.
!>

INCY

!>          INCY es ENTERO
!>          El espaciado entre elementos consecutivos de SY.
!>
