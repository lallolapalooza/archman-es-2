NAME

inetaddress - class implementing IP addresses

SYNOPSIS 

#include <bobcat/inetaddress/inetaddress.fwd.hpp>

inetaddress( const char *ipv4_address );
inetaddress( const std::string &ipv4_address );
inetaddress( const char *ipv6_address );
inetaddress( const std::string &ipv6_address );
inetaddress( const int address );
inetaddress( const unsigned int address );
inetaddress( const unsigned long address );
inetaddress( const struct in_addr &addr );
inetaddress( const struct in6_addr &addr );

class
    inetaddress
{
    public
        enum anon
        {
            any = INADDR_ANY,
            none = INADDR_NONE
        };

        explicit inetaddress( const char *addr );
        explicit inetaddress( const std::string &addr );
        inetaddress( int address );
        inetaddress( unsigned int address );
        inetaddress( unsigned long address );
        inetaddress( const struct in_addr &addr );
        inetaddress( const struct in6_addr &addr );
        inetaddress( const inetaddress &other );
        ~inetaddress();

        inetaddress &operator=( const inetaddress &other );

        void swap( inetaddress &other );

        bool isIpv4() const;
        bool isIpv6() const;

        std::string asString() const;
        unsigned int asLong() const;

        operator std::string() const;
        operator unsigned int() const;

    private
        struct anon;
        anon *d_ptr;

        static void throw_error();
};

void swap( inetaddress &lhs, inetaddress &rhs );

DESCRIPTION

The inetaddress class implements IP addresses.

The inetaddress class supports both IPv4 and IPv6 addresses.

NAMESPACE

None.

INHERITS FROM

None.

PROTECTED CONSTRUCTORS

None.

MEMBER FUNCTIONS

void swap(inetaddress &other)
Swap the contents of this inetaddress with those of other.
bool isIpv4() const
Return true if the IP address is an IPv4 address, false otherwise.
bool isIpv6() const
Return true if the IP address is an IPv6 address, false otherwise.
std::string asString() const
Return the IP address as a string.
unsigned int asLong() const
Return the IP address as an unsigned integer. This is only valid for IPv4 addresses.

PROTECTED MEMBER FUNCTIONS

None.

EXAMPLE 

#include <iostream>
#include <bobcat/inetaddress>

using namespace std;
using namespace bobcat;

int
main()
{
    inetaddress addr1("192.168.1.1");
    inetaddress addr2("fe80::1");
    inetaddress addr3(127.0.0.1);

    cout << "addr1: " << addr1.asString() << endl;
    cout << "addr2: " << addr2.asString() << endl;
    cout << "addr3: " << addr3.asString() << endl;

    cout << "addr1 is IPv4: " << boolalpha << addr1.isIpv4() << endl;
    cout << "addr2 is IPv6: " << boolalpha << addr2.isIpv6() << endl;

    return 0;
}
