iso(7) Administración de Sistemas Linux iso(7)

Nombre

iso - transporte ISO

SINOPSIS (desde Linux 6.0 [experimental])

#include <sys/socket.h>
#include <bluetooth/bluetooth.h>
#include <bluetooth/iso.h>
iso_socket = socket(PF_BLUETOOTH, SOCK_SEQPACKET, BTPROTO_ISO);

DESCRIPCIÓN

Los Canales de Bluetooth Sincrónicos (Isochronous Channels) son una característica introducida en Bluetooth 5.2 que permite la transmisión de múltiples flujos de audio sincronizados entre dispositivos.

Para audio unicast/multiflujo, se utilizan el grupo sincrónico conectado (CIG) y el flujo sincrónico conectado (CIS). Un CIG es creado por el central y puede incluir uno o más CIS. Un CIS es un flujo de transporte de datos punto a punto entre un central y un periférico determinado, y es un protocolo de comunicación bidireccional con reconocimiento (acknowledgment).

Para audio de difusión (broadcast), se utilizan el grupo sincrónico de difusión (BIG) y el flujo sincrónico de difusión (BIS). Hay dos tipos de dispositivos: el difusor sincrónico y el receptor sincronizado. Un BIG es creado por un difusor sincrónico y puede incluir uno o más BIS. Un BIS es un flujo de transporte de datos de uno a muchos.

DIRECCIÓN DEL SOCKET

struct sockaddr_iso_bc {
    bdaddr_t        bc_bdaddr;
    uint8_t         bc_bdaddr_type;
    uint8_t         bc_sid;
    uint8_t         bc_num_bis;
    uint8_t         bc_bis[ISO_MAX_NUM_BIS];
};
struct sockaddr_iso {
    sa_family_t     iso_family;
    bdaddr_t        iso_bdaddr;
    uint8_t         iso_bdaddr_type;
    struct sockaddr_iso_bc iso_bc[];
};

Ejemplo unicast:

struct sockaddr_iso addr;
memset(&addr, 0, sizeof(addr));
addr.iso_family = AF_BLUETOOTH;
bacpy(&addr.iso_bdaddr, bdaddr);
addr.iso_bdaddr_type = BDADDR_LE_PUBLIC;

Ejemplo de difusión (broadcast):

struct sockaddr_iso *addr;
size_t addr_len;
addr_len = sizeof(*addr) + sizeof(*addr->iso_bc);
memset(addr, 0, addr_len);
addr->iso_family = AF_BLUETOOTH;
bacpy(&addr->iso_bdaddr, bdaddr);
addr->iso_bdaddr_type = BDADDR_LE_PUBLIC;

Ejemplo de Fuente de Difusión (Difusor):

struct sockaddr_iso *addr;
size_t addr_len;
addr_len = sizeof(*addr) + sizeof(*addr->iso_bc);
memset(addr, 0, addr_len);
addr->iso_family = AF_BLUETOOTH;
/* Establecer la dirección en BDADR_ANY(00:00:00:00:00:00) */
bacpy(&addr->iso_bdaddr, BADDR_ANY);
addr->iso_bdaddr_type = BDADDR_LE_PUBLIC;
/* Conectar a la dirección de Difusión */
connect(iso_socket, (struct sockaddr *)addr, addr_len);

Ejemplo de Destino de Difusión (Receptor):

struct sockaddr_iso *addr;
size_t addr_len;
addr_len = sizeof(*addr) + sizeof(*addr->iso_bc);
memset(addr, 0, addr_len);
addr->iso_family = AF_BLUETOOTH;
/* Establecer el destino en la dirección del Difusor */
bacpy(&addr->iso_bdaddr, bdaddr);
addr->iso_bdaddr_type = BDADDR_LE_PUBLIC;
/* Vincular (bind) a la dirección del Difusor */
bind(iso_socket, (struct sockaddr *)addr, addr_len);

Fuente de Difusión (Difusor) o Destino de Difusión (Receptor) Transferencia de Sincronización de Publicidad Periódica (PAST):

struct sockaddr_iso *addr;
size_t addr_len;
addr_len = sizeof(*addr) + sizeof(*addr->iso_bc);
memset(addr, 0, addr_len);
addr->iso_family = AF_BLUETOOTH;
/* Establecer la dirección de destino en la dirección de destino de PAST */
bacpy(&addr->iso_bc->bc_bdaddr, (void *) bdaddr);
addr->iso_bc->bc_bdaddr_type = bdaddr_type;
/* Volver a vincular (rebind) el socket ya conectado a la dirección de PAST */
bind(iso_socket, (struct sockaddr *)addr, addr_len);

OPCIONES DEL SOCKET (SOL_BLUETOOTH)

Las opciones de socket listadas a continuación pueden establecerse usando setsockopt(2) y leerse con getsockopt(2) con el nivel del socket establecido en SOL_BLUETOOTH.

BT_SECURITY

Nivel de seguridad del canal, valores posibles:

Valor Nivel de Seguridad Tipo de Clave de Enlace Cifrado
BT_SECURITY_SDP 0 (Solo SDP) Ninguno No requerido
BT_SECURITY_LOW 1 (Bajo) No autenticado No requerido
BT_SECURITY_MEDIUM 2 (Medio - predeterminado) No autenticado Deseado
BT_SECURITY_HIGH 3 (Alto) Autenticado Requerido
BT_SECURITY_FIPS (desde Linux 3.15) 4 (Solo Seguro) Autenticado (Emparejamiento Simple Seguro y Autenticación Segura basados en P-256) Requerido

Ejemplo:

int level = BT_SECURITY_HIGH;
int err = setsockopt(iso_socket, SOL_BLUETOOTH, BT_SECURITY, &level,
                     sizeof(level));
if (err == -1) {
    perror("setsockopt");
    return 1;
}

BT_DEFER_SETUP

Diferir la configuración de la conexión del canal, esto controla si el procedimiento de conexión necesita ser autorizado por el espacio de usuario antes de responder, lo que permite la autorización a nivel de perfil. Valores posibles:

Valor Descripción Autorización
0 Deshabilitado (predeterminado) No requerida
1 Habilitado Requerida

Ejemplo:

int defer_setup = 1;
int err = setsockopt(iso_socket, SOL_BLUETOOTH, BT_DEFER_SETUP,
                     &defer_setup, sizeof(defer_setup));
if (err == -1) {
    perror("setsockopt");
    return err;
}
err = listen(iso_socket, 5);
if (err) {
    perror("listen");
    return err;
}
struct sockaddr_iso remote_addr = {0};
socklen_t addr_len = sizeof(remote_addr);
int new_socket = accept(iso_socket, (struct sockaddr*)&remote_addr,
                        &addr_len);
if (new_socket < 0) {
    perror("accept");
    return new_socket;
}
/* Para completar la configuración de la conexión de new_socket, leer 1 byte */
char c;
struct pollfd pfd;
memset(&pfd, 0, sizeof(pfd));
pfd.fd = new_socket;
pfd.events = POLLOUT;
err = poll(&pfd, 1, 0);
if (err) {
    perror("poll");
    return err;
}
if (!(pfd.revents & POLLOUT)) {
    err = read(sk, &c, 1);
    if (err < 0) {
        perror("read");
        return err;
    }
}

BT_PKT_STATUS

Habilitar el informe del estado de los paquetes a través de BT_SCM_PKT_STATUS CMSG en los paquetes recibidos. Valores posibles:

Valor Descripción
0 Deshabilitado (predeterminado)
1 Habilitadoble
BT_SCM_PKT_STATUS
Nivel SOL_BLUETOOTH CMSG con datos:
uint8_t pkt_status;

Los valores son iguales a "Packet_Status_Flag" definido en la Especificación del Núcleo v6.1, 5.4.5. Paquetes de datos HCI ISO:

https://www.bluetooth.com/wp-content/uploads/Files/Specification/HTML/Core-61/out/en/host-controller-interface/host-controller-interface-functional-specification.html#UUID-9b5fb085-278b-5084-ac33-bee2839abe6b

pkt_status Descripción
0x0 Datos válidos. Se recibió correctamente el SDU completo.
0x1 Datos posiblemente inválidos. El contenido del ISO_SDU_Fragment puede contener errores o faltar parte del SDU. Esto se informa como 'datos con posibles errores'.
0x2 Parte(s) del SDU no se recibieron correctamente. Esto se informa como 'datos perdidos'.

BT_PKT_SEQNUM (desde Linux 6.17-rc1)

Habilita el informe del número de secuencia ISO del paquete a través de BT_SCM_PKT_SEQNUM CMSG en los paquetes recibidos. Valores posibles:

Valor Descripción
0 Deshabilitado (predeterminado)
1 Habilitado
BT_SCM_PKT_SEQNUM
Nivel SOL_BLUETOOTH CMSG con datos:
uint16_t pkt_seqnum;

Los valores son iguales a "Packet_Sequence_Number" definido en la Especificación del Núcleo v6.1, 5.4.5. Paquetes de datos HCI ISO:

https://www.bluetooth.com/wp-content/uploads/Files/Specification/HTML/Core-61/out/en/host-controller-interface/host-controller-interface-functional-specification.html#UUID-9b5fb085-278b-5084-ac33-bee2839abe6b

Ejemplo (Habilitar números de secuencia):

uint32_t opt = 1;
if (setsockopt(fd, SOL_BLUETOOTH, BT_PKT_SEQNUM, &opt, sizeof(opt)) < 0)
    goto error;

Ejemplo (Leer paquete y su número de secuencia):

char data_buf[256];
uint16_t seqnum;
union {
    char buf[CMSG_SPACE(sizeof(uint16_t))];
    struct cmsghdr align;
} control;
struct iovec data = {
    .iov_base = data_buf,
    .iov_len = sizeof(data_buf),
};
struct msghdr msg = {
    .msg_iov = &data,
    .msg_iovlen = 1,
    .msg_control = control.buf,
    .msg_controllen = sizeof(control.buf),
};
struct scm_timestamping tss;
res = recvmsg(fd, &msg, 0);
if (res < 0)
    goto error;
for (cmsg = CMSG_FIRSTHDR(&msg); cmsg; cmsg = CMSG_NXTHDR(&msg, cmsg)) {
    if (cmsg->cmsg_level == SOL_BLUETOOTH && cmsg->cmsg_type == BT_PKT_SEQNUM)
        memcpy(&seqnum, CMSG_DATA(cmsg), sizeof(seqnum));
}

OPCIONES DE SOCKET (SOL_SOCKET)

SOL_SOCKET las opciones de socket de nivel que modifican características genéricas del socket (SO_SNDBUF, SO_RCVBUF, etc.) tienen su significado habitual, véase socket(7).

Las opciones de socket ISO de nivel SOL_SOCKET

que tienen un manejo específico de Bluetooth en el kernel se enumeran a continuación.

SO_TIMESTAMPING, SO_TIMESTAMP, SO_TIMESTAMPNS Véase

https://docs.kernel.org/networking/timestamping.html

Para los sockets ISO, se admiten marcas de tiempo RX por software. Las marcas de tiempo TX por software (SOF_TIMESTAMPING_TX_SOFTWARE) se admiten desde Linux 6.15.

La marca de tiempo RX por software es el momento en que el kernel recibió el paquete del controlador del driver. La marca de tiempo SCM_TSTAMP_SND

se emite cuando el paquete se envía al controlador del driver. La marca de tiempo SCM_TSTAMP_COMPLETION

se emite cuando el controlador informa que el paquete se ha completado. Las marcas de tiempo de finalización solo se admiten en controladores que tienen control de flujo ISO. No se admiten otros tipos de marcas de tiempo TX. Puede usar SIOCETHTOOL

para consultar los indicadores admitidos. Las marcas de tiempo están en el tiempo CLOCK_REALTIME

.

Ejemplo (Habilitar marcas de tiempo RX):

int flags = SOF_TIMESTAMPING_SOFTWARE | SOF_TIMESTAMPING_RX_SOFTWARE; setsockopt(fd, SOL_SOCKET, SO_TIMESTAMPING, &flags, sizeof(flags));

Ejemplo (Leer paquete y su marca de tiempo RX):

char data_buf[256]; union { char buf[CMSG_SPACE(sizeof(struct scm_timestamping))]; struct cmsghdr align; } control; struct iovec data = { .iov_base = data_buf, .iov_len = sizeof(data_buf), }; struct msghdr msg = { .msg_iov = &data, .msg_iovlen = 1, .msg_control = control.buf, .msg_controllen = sizeof(control.buf), }; struct scm_timestamping tss; res = recvmsg(fd, &msg, MSG_ERRQUEUE | MSG_DONTWAIT); if (res < 0) goto error; for (cmsg = CMSG_FIRSTHDR(&msg); cmsg; cmsg = CMSG_NXTHDR(&msg, cmsg)) { if (cmsg->cmsg_level == SOL_SOCKET && cmsg->cmsg_type == SCM_TIMESTAMPING) memcpy(&tss, CMSG_DATA(cmsg), sizeof(tss)); } tstamp_clock_realtime = tss.ts[0];

Ejemplo (Habilitar marcas de tiempo TX):

int flags = SOF_TIMESTAMPING_SOFTWARE | SOF_TIMESTAMPING_TX_SOFTWARE | SOF_TIMESTAMPING_OPT_ID; setsockopt(fd, SOL_SOCKET, SO_TIMESTAMPING, &flags, sizeof(flags));

Ejemplo (Leer marcas de tiempo TX):

union { char buf[CMSG_SPACE(sizeof(struct scm_timestamping))]; struct cmsghdr align; } control; struct iovec data = { .iov_base = NULL, .iov_len = 0 }; struct msghdr msg = { .msg_iov = &data, .msg_iovlen = 1, .msg_control = control.buf, .msg_controllen = sizeof(control.buf), }; struct cmsghdr *cmsg; struct scm_timestamping tss; struct sock_extended_err serr; int res; res = recvmsg(fd, &msg, MSG_ERRQUEUE | MSG_DONTWAIT); if (res < 0) goto error; for (cmsg = CMSG_FIRSTHDR(&msg); cmsg; cmsg = CMSG_NXTHDR(&msg, cmsg)) { if (cmsg->cmsg_level == SOL_SOCKET && cmsg->cmsg_type == SCM_TIMESTAMPING) memcpy(&tss, CMSG_DATA(cmsg), sizeof(tss)); else if (cmsg->cmsg_level == SOL_BLUETOOTH && cmsg->cmsg_type == BT_SCM_ERROR) memcpy(&serr, CMSG_DATA(cmsg), sizeof(serr)); } tstamp_clock_realtime = tss.ts[0]; tstamp_type = serr->ee_info; /* SCM_TSTAMP_SND or SCM_TSTAMP_COMPLETION */ tstamp_seqnum = serr->ee_data;

IOCTLS

Los siguientes ioctls con operaciones específicas para sockets ISO están disponibles.

SIOCETHTOOL (desde Linux 6.16-rc1) Admite solo el comandoETHTOOL_GET_TS_INFO , que puede usarse para consultar los indicadores SOF_TIMESTAMPING_* admitidos. Los indicadores SOF_TIMESTAMPING_OPT_*

siempre están disponibles según corresponda.

Ejemplo:

#include <linux/ethtool.h> #include <linux/sockios.h> #include <net/if.h> #include <sys/socket.h> #include <sys/ioctl.h> ... struct ifreq ifr = {}; struct ethtool_ts_info cmd = {}; int sk; snprintf(ifr.ifr_name, sizeof(ifr.ifr_name), "hci0"); ifr.ifr_data = (void *)& cmd.cmd = ETHTOOL_GET_TS_INFO; sk = socket(PF_BLUETOOTH, SOCK_SEQPACKET, BTPROTO_ISO); if (sk < 0) goto error; if (ioctl(sk, SIOCETHTOOL, &ifr)) goto error; sof_available = cmd.so_timestamping;
