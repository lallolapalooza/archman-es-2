Ejemplo: Adjuntar contadores estáticamente a un flujo

nuevo

Este ejemplo demuestra el uso de contadores que se adjuntan estáticamente con la creación de un flujo nuevo. Los contadores se leen del hardware periódicamente y, finalmente, se liberan todos los

recursos. 
/* crear objeto de contadores y definir sus puntos de contadores        */
/* crear flujo L2 simple con MAC fijo y una acción de conteo */
/* leer contadores periódicamente, cada 1seg, hasta que termine el bucle      */
/* asume que el usuario preparó un QP RAW_PACKET como entrada               */
/* solo se realiza una verificación limitada de errores en tiempo de ejecución por simplicidad del código  */
#include <inttypes.h>
#include <infiniband/verbs.h>
/* la siguiente MAC debe ser reemplazada por el usuario */
#define FLOW_SPEC_ETH_MAC_VAL {
    .dst_mac = { 0x00, 0x01, 0x02, 0x03, 0x04,0x05},
    .src_mac = { 0x00, 0x00, 0x00, 0x00, 0x00, 0x00},
    .ether_type = 0, .vlan_tag = 0, }
#define FLOW_SPEC_ETH_MAC_MASK {
    .dst_mac = { 0xFF, 0xFF, 0xFF, 0xFF, 0xFF, 0xFF},
    .src_mac = { 0xFF, 0xFF, 0xFF, 0xFF, 0xFF, 0xFF},
    .ether_type = 0, .vlan_tag = 0, }
void example_create_flow_with_counters_on_raw_qp(struct ibv_qp *qp) {
    int idx = 0;
    int loop = 10;
    struct ibv_flow *flow = NULL;
    struct ibv_counters *counters = NULL;
    struct ibv_counters_init_attr init_attr = {0};
    struct ibv_counter_attach_attr attach_attr = {0};
    /* crear un único manejador de contadores */
    counters = ibv_create_counters(qp->context, &init_attr);
    /* definir puntos de contadores */
    attach_attr.counter_desc = IBV_COUNTER_PACKETS;
    attach_attr.index = idx++;
    ret = ibv_attach_counters_point_flow(counters, &attach_attr, NULL);
    if (ret == ENOTSUP) {
        fprintf(stderr, "Adjuntar IBV_COUNTER_PACKETS al flujo no está \
soportado");
        exit(1);
    }
    attach_attr.counter_desc = IBV_COUNTER_BYTES;
    attach_attr.index = idx++;
    ibv_attach_counters_point_flow(counters, &attach_attr, NULL);
    if (ret == ENOTSUP) {
        fprintf(stderr, "Adjuntar IBV_COUNTER_BYTES al flujo no está \
soportado");
        exit(1);
    }
    /* definir un nuevo atributo de flujo que incluye el manejador de contadores */
    struct raw_eth_flow_attr {
        struct ibv_flow_attr              attr;
        struct ibv_flow_spec_eth          spec_eth;
        struct ibv_flow_spec_counter_action spec_count;
    } flow_attr = {
        .attr = {
                .comp_mask  = 0,
                .type       = IBV_FLOW_ATTR_NORMAL,
                .size       = sizeof(flow_attr),
                .priority   = 0,
                .num_of_specs = 2, /* ETH + COUNT */
                .port       = 1,
                .flags      = 0,
            },
        .spec_eth = {
                .type = IBV_EXP_FLOW_SPEC_ETH,
                .size = sizeof(struct ibv_flow_spec_eth),
                .val  = FLOW_SPEC_ETH_MAC_VAL,
                .mask = FLOW_SPEC_ETH_MAC_MASK,
            },
        .spec_count = {
                .type   = IBV_FLOW_SPEC_ACTION_COUNT,
                .size   = sizeof(struct ibv_flow_spec_counter_action),
                .counters = counters, /* se adjuntó este manejador de contadores
al ibv_flow recién creado */ } };
    /* crear el flujo */
    flow = ibv_create_flow(qp, &flow_attr.attr);
    /* asignar array para la lectura de valores de contadores */
    uint64_t *counters_value = malloc(sizeof(uint64_t) * idx);
    /* lectura e impresión periódica de los contadores del flujo */
    while (--loop) {
        sleep(1);
        /* leer valores de contadores del hardware */
        ibv_read_counters(counters, counters_value, idx,
                  IBV_READ_COUNTERS_ATTR_PREFER_CACHED);
        printf("PACKETS = %"PRIu64", BYTES = %"PRIu64 \n",
            counters_value[0], counters_value[1] );
    }
    /* todo listo, liberar todo */
    free(counters_value);
    /* destruir flujo y desadjuntar contadores */
    ibv_destroy_flow(flow);
    /* destruir manejador de contadores */
    ibv_destroy_counters(counters);
    ret

;
}
