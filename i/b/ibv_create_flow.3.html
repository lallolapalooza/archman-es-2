/* Las siguientes son capas opcionales según la solicitud del usuario

enum ibv_flow_attr_type {
IBV_FLOW_ATTR_NORMAL		= 0x0,		/* Enrutamiento según las especificaciones de la regla */
IBV_FLOW_ATTR_ALL_DEFAULT	= 0x1,		/* Regla unicast y multicast predeterminada - recibe todo el tráfico Ethernet que no esté enrutado a ningún QP */
IBV_FLOW_ATTR_MC_DEFAULT 	= 0x2,		/* Regla multicast predeterminada - recibe todo el tráfico multicast Ethernet que no esté enrutado a ningún QP */

enum ibv_flow_flags {

Estructura general de especificación de flujo:
struct ibv_flow_spec_xxx {
enum ibv_flow_spec_type  type;
uint16_t  size;                  /* Tamaño de la especificación de flujo = sizeof(struct ibv_flow_spec_xxx) */
struct ibv_flow_xxx_filter val;
struct ibv_flow_xxx_filter mask; /* Define qué bits del valor del filtro son aplicables al buscar una coincidencia en el paquete entrante */
}

Cada estructura de especificación contiene los parámetros de la capa de red relevantes para la coincidencia. Para hacer cumplir la coincidencia, el usuario establece una máscara para cada parámetro.
Los paquetes que provienen del cable se comparan con la especificación de flujo. Si se encuentra una coincidencia, se ejecutan las acciones de flujo asociadas en el paquete.
En los flujos de entrada, el parámetro QP se trata como otra acción de dispersión del paquete al QP correspondiente.
Si el bit está establecido en la máscara, el bit correspondiente en el valor debe coincidir.
Tenga en cuenta que la mayoría de los proveedores admiten una máscara completa (todos los "1") o una máscara cero (todos los "0").

Dominios y prioridad de

flujo

El enrutamiento de flujo define el concepto de dominio y prioridad. Cada dominio representa una aplicación que puede adjuntar un flujo. Los dominios tienen prioridad. Un dominio de mayor prioridad siempre prevalecerá sobre un dominio de menor prioridad cuando sus especificaciones de flujo se superpongan.
Los verbos IB tienen el dominio de mayor prioridad.
Además del dominio, existe una prioridad dentro de cada uno de los dominios. Un valor numérico de prioridad más bajo (mayor prioridad) tiene precedencia sobre las reglas coincidentes con un valor numérico de priorida

d más alto (menor prioridad). Es importante tener en cuenta que el valor de prioridad de una especificación de flujo se utiliza no solo para establecer la precedencia de las coincidencias de flujo en conflicto, sino t
ambién como una forma de abstraer el orden en el que se prueban las especificaciones de flujo para encontrar coincidencias. Los flujos con mayores prioridades se probarán antes que los flujos con menores prioridades.

Orden de definición de las

reglas

Una aplicación puede proporcionar las reglas ibv_flow_spec_xxx en un esquema no ordenado. En este caso, cada especificación debe estar bien definida y coincidir con una capa de cabecera de red específica. En algunos casos, cuando ciertos tipos de especificación de flujo están presentes en la lista de especificaciones, es necesario proporcionar la lista de manera ordenada para que la posición de ese tipo de especificación de flujo en el protocolo esté estrictamente definida. Cuando el cierto tipo de especificación, que requiere el ordenamiento, reside en la pila de protocolo de red interna (en protocolos de túnel), el ordenamiento debe aplicarse a las especificaciones de red internas y debe combinarse con la indicación de especificación interna usando la bandera IBV_FLOW_SPEC_INNER. Por ejemplo: Una especificación MPLS que intenta coincidir con una etiqueta MPLS en la red interna debe tener la bandera IBV_FLOW_SPEC_INNER establecida y así deben hacerlo el resto de las especificaciones de red internas. Además de eso, todas las especificaciones de red internas deben proporcionarse de manera ordenada. Esto es esencial para representar muchos de los protocolos de túnel de

encapsulación.

Los tipos de especificación de flujo que requieren este tipo de ordenamiento son:
1. IBV_FLOW_SPEC_MPLS -

esta especificación de acuerdo con su ubicación exacta en la pila de protocolos.

A continuación, flow_attr define una regla en la prioridad 0 para que coincida con una dirección MAC de destino y una dirección IPv4 de origen. Para ello, se utilizan las especificaciones L2 y L3.
Si hay una coincidencia en esta regla, significa que el paquete recibido tiene la dirección MAC de destino: 66:11:22:33:44:55 y la dirección IP de origen: 0x0B86C806, y el paquete se enruta a su qp

adjunta. 
struct raw_eth_flow_attr {
struct ibv_flow_attr            attr;
struct ibv_flow_spec_eth        spec_eth;
struct ibv_flow_spec_ipv4       spec_ipv4;
} __attribute__((packed));
struct raw_eth_flow_attr flow_attr = {
        .attr = {
                .comp_mask      = 0,
                .type           = IBV_FLOW_ATTR_NORMAL,
                .size           = sizeof(flow_attr),
                .priority       = 0,
                .num_of_specs   = 2,
                .port           = 1,
                .flags          = 0,
        },
        .spec_eth = {
                .type   = IBV_FLOW_SPEC_ETH,
                .size   = sizeof(struct ibv_flow_spec_eth),
                .val = {
                        .dst_mac = {0x66, 0x11, 0x22, 0x33, 0x44, 0x55},
                        .src_mac = { 0x00, 0x00, 0x00, 0x00, 0x00, 0x00},
                        .ether_type = 0,
                        .vlan_tag = 0,
                },
                .mask = {
                        .dst_mac = { 0xFF, 0xFF, 0xFF, 0xFF, 0xFF, 0xFF},
                        .src_mac = { 0xFF, 0xFF, 0xFF, 0xFF, 0xFF, 0xFF},
                        .ether_type = 0,
                        .vlan_tag = 0,
                }
        },
        .spec_ipv4 = {
                .type   = IBV_FLOW_SPEC_IPV4,
                .size   = sizeof(struct ibv_flow_spec_ipv4),
                .val = {
                        .src_ip = 0x0B86C806,
                        .dst_ip = 0,
                },
                .mask = {
                        .src_ip = 0xFFFFFFFF,
                        .dst_ip = 0,
                }
        }
};
