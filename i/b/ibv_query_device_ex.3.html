IBV_QUERY_DEVICE_EX(3) Manual del programador de Libibverbs IBV_QUERY_DEVICE_EX(3)

NOMBRE

ibv_query_device_ex - consulta los atributos de un dispositivo RDMA, incluidas las propiedades extendidas del dispositivo.

SINOPSIS

#include <infiniband/verbs.h>

int ibv_query_device_ex(struct ibv_context *context,
                        struct ibv_query_device_ex_input *input,
                        struct ibv_device_attr_ex *attr);

DESCRIPCIÓN

ibv_query_device_ex() devuelve los atributos del dispositivo con contexto context. El argumento input es un puntero a una estructura ibv_query_device_ex_input, utilizada para futuras extensiones. El argumento attr es un puntero a una estructura ibv_device_attr_ex, tal como se define en <infiniband/verbs.h>.

struct ibv_device_attr_ex {
struct ibv_device_attr orig_attr;
uint32_t               comp_mask;                  /* Máscara de compatibilidad que define qué de las siguientes variables son válidas */
struct ibv_odp_caps    odp_caps;                   /* Capacidades de paginación bajo demanda (On-Demand Paging) */
uint64_t               completion_timestamp_mask;  /* Máscara de marca de tiempo de finalización (0 = no soportado) */
uint64_t               hca_core_clock;             /* La frecuencia (en kHZ) de la HCA (0 = no soportado) */
uint64_t               device_cap_flags_ex;        /* Banderas de capacidad extendida del dispositivo */
struct ibv_tso_caps    tso_caps;                   /* Capacidades de descarga de segmentación TCP (TCP Segmentation Offload) */
struct ibv_rss_caps    rss_caps;                   /* Capacidades RSS (Receive Side Scaling) */
uint32_t               max_wq_type_rq;             /* Máxima Cola de Trabajo (Work Queue) de tipo RQ */
struct ibv_packet_pacing_caps packet_pacing_caps; /* Capacidades de control de ritmo de paquetes */
uint32_t               raw_packet_caps;            /* Capacidades de paquetes crudos, usar enum ibv_raw_packet_caps */
struct ibv_tm_caps     tm_caps;                    /* Capacidades de coincidencia de etiquetas (Tag Matching) */
struct ibv_cq_moderation_caps  cq_mod_caps;        /* Capacidades máximas de moderación de CQ (Completion Queue) */
uint64_t     	       max_dm_size;		   /* Tamaño máximo de Memoria del Dispositivo (Device Memory) (en bytes) disponible para asignación */
struct ibv_pci_atomic_caps atomic_caps;            /* Capacidades de operaciones atómicas PCI, usar enum ibv_pci_atomic_op_size */
uint32_t               xrc_odp_caps;               /* Máscara con enum ibv_odp_transport_cap_bits para saber qué operaciones son compatibles. */
uint32_t	       phys_port_cnt_ex		   /* Número extendido de puertos físicos, permite exponer más de 255 puertos en el dispositivo */
};
struct ibv_odp_caps {
        uint64_t general_odp_caps;    /* Máscara con enum ibv_odp_general_cap_bits */
        struct {
                uint32_t rc_odp_caps; /* Máscara con enum ibv_odp_tranport_cap_bits para saber qué operaciones son compatibles. */
                uint32_t uc_odp_caps; /* Máscara con enum ibv_odp_tranport_cap_bits para saber qué operaciones son compatibles. */
                uint32_t ud_odp_caps; /* Máscara con enum ibv_odp_tranport_cap_bits para saber qué operaciones son compatibles. */
        } per_transport_caps;
};
enum ibv_odp_general_cap_bits {
        IBV_ODP_SUPPORT = 1 << 0, /* Paginación bajo demanda es compatible */
        IBV_ODP_SUPPORT_IMPLICIT = 1 << 1, /* Paginación bajo demanda implícita es compatible */
};
enum ibv_odp_transport_cap_bits {
        IBV_ODP_SUPPORT_SEND     = 1 << 0, /* Operaciones de envío admiten paginación bajo demanda */
        IBV_ODP_SUPPORT_RECV     = 1 << 1, /* Operaciones de recepción admiten paginación bajo demanda */
        IBV_ODP_SUPPORT_WRITE    = 1 << 2, /* Operaciones RDMA-Write admiten paginación bajo demanda */
        IBV_ODP_SUPPORT_READ     = 1 << 3, /* Operaciones RDMA-Read admiten paginación bajo demanda */
        IBV_ODP_SUPPORT_ATOMIC   = 1 << 4, /* Operaciones RDMA-Atomic admiten paginación bajo demanda */
        IBV_ODP_SUPPORT_SRQ_RECV = 1 << 5, /* Operaciones de recepción SRQ admiten paginación bajo demanda */
        IBV_ODP_SUPPORT_FLUSH    = 1 << 6, /* Operaciones RDMA-Flush admiten paginación bajo demanda */
        IBV_ODP_SUPPORT_ATOMIC_WRITE     = 1 << 7, /* Operaciones RDMA-Atomic-write admiten paginación bajo demanda */
};
struct ibv_tso_caps {
        uint32_t max_tso;        /* Tamaño máximo de carga útil en bytes admitido para segmentación por el motor TSO. */
        uint32_t supported_qpts; /* Mapa de bits que muestra qué tipos de QP son compatibles con la operación TSO. */
};
struct ibv_rss_caps {
        uint32_t supported_qpts;                   /* Mapa de bits que muestra qué tipos de QP son compatibles con RSS */
        uint32_t max_rwq_indirection_tables;       /* Máximas tablas de indirección de cola de trabajo de recepción */
        uint32_t max_rwq_indirection_table_size;   /* Tamaño máximo de tabla de indirección de cola de trabajo de recepción */
        uint64_t rx_hash_fields_mask;              /* Máscara con enum ibv_rx_hash_fields para saber qué campo del paquete entrante puede participar en el hash RX */
        uint8_t  rx_hash_function;                 /* Máscara con enum ibv_rx_hash_function_flags para saber qué funciones hash son compatibles */
};
struct ibv_packet_pacing_caps {
       uint32_t qp_rate_limit_min; /* Límite de tasa mínimo en kbps */
       uint32_t qp_rate_limit_max; /* Límite de tasa máximo en kbps */
       uint32_t supported_qpts;    /* Mapa de bits que muestra qué tipos de QP son compatibles. */
};
enum ibv_raw_packet_caps {
IBV_RAW_PACKET_CAP_CVLAN_STRIPPING	= 1 << 0, /* Se admite la eliminación de CVLAN */
IBV_RAW_PACKET_CAP_SCATTER_FCS		= 1 << 1, /* Se admite la dispersión de FCS */
IBV_RAW_PACKET_CAP_IP_CSUM		= 1 << 2, /* Se admite la descarga de suma de verificación IP (IP CSUM) */
};
enum ibv_tm_cap_flags {
IBV_TM_CAP_RC   = 1 << 0,            /* Compatibilidad con coincidencia de etiquetas en transporte RC */
};
struct ibv_tm_caps {
uint32_t        max_rndv_hdr_size;   /* Tamaño máximo de cabecera de solicitud de entrega de rendezvous */
uint32_t        max_num_tags;        /* Número máximo de búferes etiquetados en una lista de coincidencia TM-SRQ */
uint32_t        flags;    	     /* De enum ibv_tm_cap_flags */
uint32_t        max_ops;             /* Número máximo de operaciones de lista pendientes */
uint32_t        max_sge;             /* Número máximo de SGEs en un búfer etiquetado */
};
struct ibv_cq_moderation_caps {
	uint16_t max_cq_count;
	uint16_t max_cq_period;
};
enum ibv_pci_atomic_op_size {
IBV_PCI_ATOMIC_OPERATION_4_BYTE_SIZE_SUP = 1 << 0,
IBV_PCI_ATOMIC_OPERATION_8_BYTE_SIZE_SUP = 1 << 1,
IBV_PCI_ATOMIC_OPERATION_16_BYTE_SIZE_SUP = 1 << 2,
};
struct ibv_pci_atomic_caps {
uint16_t fetch_add;	/* Tamaños compatibles para una operación atómica de obtener y sumar, usar enum ibv_pci_atomic_op_size */
uint16_t swap;		/* Tamaños compatibles para una operación atómica incondicional de intercambio, usar enum ibv_pci_atomic_op_size */
uint16_t compare_swap;	/* Tamaños compatibles para una operación atómica de comparar e intercambiar, usar enum ibv_pci_atomic_op_size */
};

Banderas de capacidad extendida del dispositivo (device_cap_flags_ex):

IBV_DEVICE_PCI_WRITE_END_PADDING

Indica que el dispositivo tiene soporte para el relleno de escrituras PCI a una línea de caché completa.

Rellenar paquetes a líneas de caché completas reduce la cantidad de tráfico requerido en el controlador de memoria a expensas de crear más tráfico en el puerto PCI-E.

Las cargas de trabajo con alta carga de memoria de CPU y baja utilización de PCI-E se beneficiarán de esta característica, mientras que las cargas de trabajo con alta utilización de PCI-E y paquetes pequeños se verán perjudicadas.

Por ejemplo, con un tamaño de línea de caché de 128 bytes, la transferencia de cualquier paquete inferior a 128 bytes requerirá una transferencia completa de 128 bytes en PCI, lo que podría duplicar el ancho de banda PCI-E requerido.

Esta característica se puede habilitar por QP o WQ mediante las banderas IBV_QP_CREATE_PCI_WRITE_END_PADDING o IBV_WQ_FLAGS_PCI_WRITE_END_PADDING.

VALOR DEVUELTO

ibv_query_device_ex() devuelve 0 si tiene éxito, o el valor de errno en caso de error (que indica el motivo del fallo).

NOTAS

Los valores máximos devueltos por esta función son los límites superiores de los recursos soportados por el dispositivo. Sin embargo, puede que no sea posible utilizar estos valores máximos, ya que el número real de cualquier recurso que se pueda crear puede estar limitado por la configuración de la máquina, la cantidad de memoria del host, los permisos del usuario y la cantidad de recursos ya en uso por otros usuarios/procesos.
