DESCRIPCIÓ

N

ibv_create_qp_ex() crea un par de colas (QP) asociado con el dominio de protección pd. El argumento qp_init_attr_ex es una estructura ibv_qp_init_attr_ex, definida en <infiniband/verbs.h>

. 
struct ibv_qp_init_attr_ex {
void                   *qp_context;     /* Contexto asociado del QP */
struct ibv_cq          *send_cq;        /* CQ a asociar con la Cola de Envío (SQ) */
struct ibv_cq          *recv_cq;        /* CQ a asociar con la Cola de Recepción (RQ) */
struct ibv_srq         *srq;            /* Manejador de SRQ si el QP va a asociarse con un SRQ, de lo contrario NULL */
struct ibv_qp_cap       cap;            /* Capacidades del QP */
enum ibv_qp_type        qp_type;        /* Tipo de Servicio de Transporte del QP: IBV_QPT_RC, IBV_QPT_UC, IBV_QPT_UD, IBV_QPT_RAW_PACKET o IBV_QPT_DRIVER */
int                     sq_sig_all;     /* Si está establecido, cada Solicitud de Trabajo (WR) enviada a la SQ genera una entrada de finalización */
uint32_t                comp_mask;	/* Identifica campos válidos */
struct ibv_pd          *pd;		/* PD a asociar con el QP */
struct ibv_xrcd        *xrcd;		/* Dominio XRC a asociar con el QP de destino */
enum ibv_qp_create_flags create_flags;	/* Banderas de creación para este QP */
uint16_t                max_tso_header; /* Tamaño máximo de cabecera TSO */
struct ibv_rwq_ind_table *rwq_ind_tbl;  /* Tabla de indirección a asociar con el QP */
struct ibv_rx_hash_conf  rx_hash_conf;  /* Configuración de hash RX a utilizar */
uint32_t                source_qpn;     /* Número de QP de origen, la bandera de creación IBV_QP_CREATE_SOURCE_QPN debe establecerse, ver algunas NOTAS más abajo */
uint64_t                send_ops_flags; /* Selecciona qué operaciones de envío del QP se definirán en struct ibv_qp_ex. Usar enum ibv_qp_create_send_ops_flags */
};
struct ibv_qp_cap {
uint32_t                max_send_wr;    /* Número máximo solicitado de WRs pendientes en la SQ */
uint32_t                max_recv_wr;    /* Número máximo solicitado de WRs pendientes en la RQ */
uint32_t                max_send_sge;   /* Número máximo solicitado de elementos de dispersión/recogida (s/g) en un WR en la SQ */
uint32_t                max_recv_sge;   /* Número máximo solicitado de elementos s/g en un WR en la RQ */
uint32_t                max_inline_data;/* Número máximo solicitado de datos (bytes) que pueden publicarse en línea en la SQ, de lo contrario 0 */
};
enum ibv_qp_create_flags {
IBV_QP_CREATE_BLOCK_SELF_MCAST_LB       = 1 << 1, /* Previene el retorno de multicast a uno mismo */
IBV_QP_CREATE_SCATTER_FCS               = 1 << 8, /* El campo FCS se dispersará a la memoria del host */
IBV_QP_CREATE_CVLAN_STRIPPING           = 1 << 9, /* El campo CVLAN se eliminará de los paquetes entrantes */
IBV_QP_CREATE_SOURCE_QPN                = 1 << 10, /* El QP creado usará el source_qpn como su número de QP en el cable */
IBV_QP_CREATE_PCI_WRITE_END_PADDING     = 1 << 11, /* Los paquetes entrantes se rellenarán al tamaño de la línea de caché */
}
; 
struct ibv_rx_hash_conf {
uint8_t                rx_hash_function;       /* Función de hash RX, usar enum ibv_rx_hash_function_flags */
uint8_t                rx_hash_key_len;        /* Longitud de la clave de hash RX */
uint8_t                *rx_hash_key;           /* Datos de la clave de hash RX */
uint64_t               rx_hash_fields_mask;    /* Campos RX que deben participar en el hashing, usar enum ibv_rx_hash_fields */
}
; 
enum ibv_rx_hash_fields {
IBV_RX_HASH_SRC_IPV4            = 1 << 0,
IBV_RX_HASH_DST_IPV4            = 1 << 1,
IBV_RX_HASH_SRC_IPV6            = 1 << 2,
IBV_RX_HASH_DST_IPV6            = 1 << 3,
IBV_RX_HASH_SRC_PORT_TCP        = 1 << 4,
IBV_RX_HASH_DST_PORT_TCP        = 1 << 5,
IBV_RX_HASH_SRC_PORT_UDP        = 1 << 6,
IBV_RX_HASH_DST_PORT_UDP        = 1 << 7,
IBV_RX_HASH_IPSEC_SPI           = 1 << 8,
/* Al usar un protocolo de túnel, p. ej. VXLAN, tenemos un paquete interior (encapsulado) y uno exterior.
 * Para aplicar RSS al paquete interior, el siguiente campo debe establecerse con uno de los campos L3/L4.
*/
IBV_RX_HASH_INNER		= (1UL << 31),
}
; 
struct ibv_qp_create_send_ops_flags {
IBV_QP_EX_WITH_RDMA_WRITE		= 1 << 0,
IBV_QP_EX_WITH_RDMA_WRITE_WITH_IMM	= 1 << 1,
IBV_QP_EX_WITH_SEND			= 1 << 2,
IBV_QP_EX_WITH_SEND_WITH_IMM		= 1 << 3,
IBV_QP_EX_WITH_RDMA_READ		= 1 << 4,
IBV_QP_EX_WITH_ATOMIC_CMP_AND_SWP	= 1 << 5,
IBV_QP_EX_WITH_ATOMIC_FETCH_AND_ADD	= 1 << 6,
IBV_QP_EX_WITH_LOCAL_INV		= 1 << 7,
IBV_QP_EX_WITH_BIND_MW			= 1 << 8,
IBV_QP_EX_WITH_SEND_WITH_INV		= 1 << 9,
IBV_QP_EX_WITH_TSO			= 1 << 10,
}
;

La función ibv_create_qp_ex() actualizará la estructura qp_init_attr_ex->cap con los valores reales del QP del QP que fue creado; los valores serán mayores o iguales a los valores

solicitados.

ibv_destroy_qp() destruye el QP q

p
.

VALOR DE

RETORNO

ibv_create_qp_ex() devuelve un puntero al QP creado, o NULL si la solicitud falla. Verifique el número de QP (qp_num) en el QP

devuelto.

ibv_destroy_qp() devuelve 0 en caso de éxito, o el valor de

errno en caso de fallo (lo c
ual indica la razón del fallo).

NOTA

S

Los atributos max_recv_wr y max_recv_sge son ignorados por ibv_create_qp_ex() si el QP va a asociarse con un

SRQ.

El atributo source_qpn solo está soportado en QP UD, sin direccionamiento de flujo la recepción (RX) no debería ser

posible.

Use ibv_qp_to_qp_ex() para obtener el ibv_qp_ex para acceder a la interfaz del iterador de operaciones de envío, cuando se usa el atributo de creación del QP IBV_QP_INIT_ATTR_SEND_OPS_FLAGS

.

ibv_destroy_qp() falla si el QP está unido a un grupo de

multidifusión.

IBV_QPT_DRIVER no representa un servicio específico y

se usa para lógica de Q
P específica del proveedor.
