io(3)

Paquete: libaio

io(3) Linux AIO io(3)

NOMBRE

io - Entrada/Salida Asíncrona

SINOPSIS

#include <errno.h>

#include <libaio.h>

DESCRIPCIÓN

La biblioteca libaio define un nuevo conjunto de operaciones de E/S que pueden reducir significativamente el tiempo que una aplicación pasa esperando E/S. Las nuevas funciones permiten a un programa iniciar una o más operaciones de E/S y luego reanudar inmediatamente el trabajo normal mientras las operaciones de E/S se ejecutan en paralelo.

Estas funciones forman parte de la biblioteca con funciones de tiempo real llamadas libaio. En realidad, no forman parte del binario libc. La implementación de estas funciones se puede realizar utilizando el soporte del kernel.

Todas las operaciones de E/S operan sobre archivos que se abrieron previamente. Puede haber un número arbitrario de operaciones en ejecución para un solo archivo. Las operaciones de E/S asíncrona se controlan mediante una estructura de datos llamada struct iocb. Se define en libaio.h de la siguiente manera.

typedef struct io_context *io_context_t;
typedef enum io_iocb_cmd {
        IO_CMD_PREAD = 0,
        IO_CMD_PWRITE = 1,
        IO_CMD_FSYNC = 2,
        IO_CMD_FDSYNC = 3,
        IO_CMD_POLL = 5,
        IO_CMD_NOOP = 6,
} io_iocb_cmd_t;
struct io_iocb_common {
        void            *buf;
        unsigned        __pad1;
        long            nbytes;
        unsigned        __pad2;
        long long       offset;
        long long       __pad3, __pad4;
};      /* el código de resultado es la cantidad leída o el errno negativo */
struct iocb {
        void            *data;
        unsigned        key;
        short           aio_lio_opcode;
        short           aio_reqprio;
        int             aio_fildes;
        union {
                struct io_iocb_common           c;
                struct io_iocb_vector           v;
                struct io_iocb_poll             poll;
                struct io_iocb_sockaddr saddr;
        } u;
};
int aio_fildes
Este elemento especifica el descriptor de archivo que se utilizará para la operación. Debe ser un descriptor válido, de lo contrario, la operación fallará.

El dispositivo en el que se abre el archivo debe permitir la operación de búsqueda. Es decir, no es posible utilizar ninguna de las operaciones de E/S en dispositivos como terminales donde una llamada a lseek(2) resultaría en un error.

long u.c.offset
Este elemento especifica el desplazamiento en el archivo en el que se realiza la operación (entrada o salida). Dado que las operaciones se realizan en un orden arbitrario y se puede iniciar más de una operación para un descriptor de archivo, no se puede esperar una posición de lectura/escritura actual del descriptor de archivo.
void *buf
Este es un puntero al búfer con los datos que se escribirán o al lugar donde se almacenarán los datos leídos.
long u.c.nbytes
Este elemento especifica la longitud del búfer al que apunta io_buf.
int aio_reqprio
Actualmente no se utiliza.
IO_CMD_PREAD
Inicia una operación de lectura. Lee del archivo en la posición u.c.offset y almacena los siguientes u.c.nbytes bytes en el búfer apuntado por buf.
IO_CMD_PWRITE
Inicia una operación de escritura. Escribe u.c.nbytes bytes a partir de buf en el archivo a partir de la posición u.c.offset.
IO_CMD_NOP
No hace nada para este bloque de control. Este valor es útil a veces cuando un arreglo de valores struct iocb contiene huecos, es decir, algunos de los valores no deben ser manejados aunque el arreglo completo se presente a la función io_submit(3).
IO_CMD_FSYNC
IO_CMD_POLL
Esto es experimental.

EJEMPLO

/*
 * Versión simplista del comando copy usando E/S asíncrona
 *
 * De:	Stephen Hemminger <shemminger@osdl.org>
 * Copia un archivo utilizando una máquina de estados de E/S asíncrona.
 * 1. Inicia la solicitud de lectura
 * 2. Cuando la lectura se completa, la convierte en una solicitud de escritura
 * 3. Cuando la escritura se completa, decrementa el contador y libera recursos
 *
 *
 * Uso: aiocp archivo(s) destino
 */
#include <unistd.h>
#include <stdio.h>
#include <sys/types.h>
#include <sys/stat.h>
#include <sys/param.h>
#include <fcntl.h>
#include <errno.h>
#include <libaio.h>
#define AIO_BLKSIZE	(64*1024)
#define AIO_MAXIO	32
static int busy = 0;		// # de E/S en curso
static int tocopy = 0;		// # de bloques restantes por copiar
static int dstfd = -1;		// descriptor de archivo de destino
static const char *dstname = NULL;
static const char *srcname = NULL;
/* Manejador de errores fatales */
static void io_error(const char *func, int rc)
{
    if (rc == -ENOSYS)
	fprintf(stderr, "AIO no está en este kernel\n");
    else if (rc < 0 && -rc < sys_nerr)
	fprintf(stderr, "%s: %s\n", func, sys_errlist[-rc]);
    else
	fprintf(stderr, "%s: error %d\n", func, rc);
    if (dstfd > 0)
	close(dstfd);
    if (dstname)
	unlink(dstname);
    exit(1);
}
/*
 * Callback de escritura completa.
 * Ajusta los contadores y libera recursos.
 */
static void wr_done(io_context_t ctx, struct iocb *iocb, long res, long res2)
{
    if (res2 != 0) {
	io_error("aio write", res2);
    }
    if (res != iocb->u.c.nbytes) {
	fprintf(stderr, "la escritura omitió bytes esperados %d obtenidos %d\n", iocb->u.c.nbytes, res2);
	exit(1);
    }
    --tocopy;
    --busy;
    free(iocb->u.c.buf);
    memset(iocb, 0xff, sizeof(iocb));	// paranoia
    free(iocb);
    write(2, "w", 1);
}
/*
 * Callback de lectura completa.
 * Convierte el iocb de lectura en un iocb de escritura y lo inicia.
 */
static void rd_done(io_context_t ctx, struct iocb *iocb, long res, long res2)
{
    /* ¿la biblioteca necesita accesores para ver iocb? */
    int iosize = iocb->u.c.nbytes;
    char *buf = iocb->u.c.buf;
    off_t offset = iocb->u.c.offset;
    if (res2 != 0)
	io_error("aio read", res2);
    if (res != iosize) {
	fprintf(stderr, "la lectura omitió bytes esperados %d obtenidos %d\n", iocb->u.c.nbytes, res);
	exit(1);
    }
    /* convierte lectura en escritura */
    io_prep_pwrite(iocb, dstfd, buf, iosize, offset);
    io_set_callback(iocb, wr_done);
    if (1 != (res = io_submit(ctx, 1, &iocb)))
	io_error("io_submit write", res);
    write(2, "r", 1);
}
int main(int argc, char *const *argv)
{
    int srcfd;
    struct stat st;
    off_t length = 0, offset = 0;
    io_context_t myctx;
    if (argc != 3 || argv[1][0] == '-') {
	fprintf(stderr, "Uso: aiocp FUENTE DESTINO");
	exit(1);
    }
    if ((srcfd = open(srcname = argv[1], O_RDONLY)) < 0) {
	perror(srcname);
	exit(1);
    }
    if (fstat(srcfd, &st) < 0) {
	perror("fstat");
	exit(1);
    }
    length = st.st_size;
    if ((dstfd = open(dstname = argv[2], O_WRONLY | O_CREAT, 0666)) < 0) {
	close(srcfd);
	perror(dstname);
	exit(1);
    }
    /* inicializa la máquina de estados */
    memset(&myctx, 0, sizeof(myctx));
    io_queue_init(AIO_MAXIO, &myctx);
    tocopy = howmany(length, AIO_BLKSIZE);
    while (tocopy > 0) {
	int i, rc;
	/* Envía tantas lecturas como sea posible a la vez hasta AIO_MAXIO */
	int n = MIN(MIN(AIO_MAXIO - busy, AIO_MAXIO / 2),
		    howmany(length - offset, AIO_BLKSIZE));
	if (n > 0) {
	    struct iocb *ioq[n];
	    for (i = 0; i < n; i++) {
		struct iocb *io = (struct iocb *) malloc(sizeof(struct iocb));
		int iosize = MIN(length - offset, AIO_BLKSIZE);
		char *buf = (char *) malloc(iosize);
		if (NULL == buf || NULL == io) {
		    fprintf(stderr, "sin memoria\n");
		    exit(1);
		}
		io_prep_pread(io, srcfd, buf, iosize, offset);
		io_set_callback(io, rd_done);
		ioq[i] = io;
		offset += iosize;
	    }
	    rc = io_submit(myctx, n, ioq);
	    if (rc < 0)
		io_error("io_submit", rc);
	    busy += n;
	}
	// Maneja las E/S que se han completado
	rc = io_queue_run(myctx);
	if (rc < 0)
	    io_error("io_queue_run", rc);
	// si tenemos el número máximo de E/S en curso
	// entonces espera a que una se complete
	if (busy == AIO_MAXIO) {
	    rc = io_queue_wait(myctx, NULL);
	    if (rc < 0)
		io_error("io_queue_wait", rc);
	}
    }
    close(srcfd);
    close(dstfd);
    exit(0);
}
