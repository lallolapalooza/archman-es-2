nbd_aio_pread(3)

Paquete: libnbd

Licencias: LGPL-2.1-or-later

nbd_aio_pread(3) LIBNBD nbd_aio_pread(3)

NOMBRE

nbd_aio_pread - leer del servidor NBD

SINOPSIS

#include <libnbd.h>
typedef struct {
  int (*callback) (void *user_data, int *error);
  void *user_data;
  void (*free) (void *user_data);
} nbd_completion_callback;
int64_t nbd_aio_pread (
          struct nbd_handle *h, void *buf, size_t count,
          uint64_t offset,
          nbd_completion_callback completion_callback,
          uint32_t flags
        );

DESCRIPCIÓN

Emite un comando de lectura al servidor NBD.

Para comprobar si el comando se completó, llame a nbd_aio_command_completed(3). O proporcione el "completion_callback" opcional que se invocará como se describe en "Callbacks de finalización" en libnbd(3).

Tenga en cuenta que debe asegurarse de que "buf" sea válido hasta que el comando se haya completado. Además, si el parámetro "error" de "completion_callback" está establecido o si nbd_aio_command_completed(3) informa de un fallo, y si nbd_get_pread_initialize(3) devuelve verdadero, entonces libnbd sanitizó "buf", pero no se especifica si los contenidos de "buf" se leerán como cero o como resultados parciales del servidor. Si nbd_get_pread_initialize(3) devuelve falso, entonces libnbd no sanitizó "buf", y los contenidos son indefinidos en caso de fallo.

Otros parámetros se comportan como se documenta en nbd_pread(3).

Por defecto, libnbd rechazará los intentos de usar esta función con parámetros que probablemente resulten en un fallo del servidor, como solicitar una bandera de comando desconocida. La función nbd_set_strict_mode(3) puede usarse para alterar qué escenarios deben esperar una respuesta del servidor en lugar de fallar rápidamente.

VALOR DE RETORNO

Esta llamada devuelve la cookie de 64 bits del comando. La cookie es ≥ 1. Las cookies son únicas (por manejador libnbd, no globalmente).

ERRORES

En caso de error, se devuelve -1.

Consulte "MANEJO DE ERRORES" en libnbd(3) para obtener más detalles sobre el error.

Los siguientes parámetros no deben ser NULL: "h", "buf". Para más información, consulte "Parámetros no NULL" en libnbd(3).

ESTADO DEL MANEJADOR

nbd_aio_pread puede llamarse cuando el manejador está en el siguiente estado:

┌─────────────────────────────────────┬─────────────────────────┐
│ Manejador creado, antes de conectar   │ ❌ error                │
│ Conectando                          │ ❌ error                │
│ Conectando y haciendo handshake (modo_opt) │ ❌ error                │
│ Conectado al servidor             │ ✅ permitido              │
│ Conexión cerrada                   │ ❌ error                │
│ Manejador muerto                         │ ❌ error                │
└─────────────────────────────────────┴─────────────────────────┘

VERSIÓN

Esta función apareció por primera vez en libnbd 1.0.

Si necesita probar si esta función está disponible en tiempo de compilación, compruebe si el siguiente macro está definido:

#define LIBNBD_HAVE_NBD_AIO_PREAD 1

EJEMPLO

Este ejemplo también está disponible como examples/aio-connect-read.c en el código fuente de libnbd.

/* Este ejemplo muestra cómo usar la API de bajo nivel AIO (asíncrona)
 * para conectarse a un servidor y leer el disco.
 *
 * Aquí hay algunas formas de probar este ejemplo:
 *
 * nbdkit -U - linuxdisk . \
 *   --run './aio-connect-read $unixsocket'
 *
 * nbdkit -U - floppy . \
 *   --run './aio-connect-read $unixsocket'
 *
 * nbdkit -U - pattern size=1M \
 *   --run './aio-connect-read $unixsocket'
 */
#include <stdio.h>
#include <stdlib.h>
#include <stdint.h>
#include <inttypes.h>
#include <errno.h>
#include <assert.h>
#include <libnbd.h>
#define NR_SECTORS 32
#define SECTOR_SIZE 512
struct data {
  uint64_t offset;
  char sector[SECTOR_SIZE];
};
static int
hexdump (void *user_data, int *error)
{
  struct data *data = user_data;
  FILE *pp;
  if (*error) {
    errno = *error;
    perror ("fallo al leer");
    exit (EXIT_FAILURE);
  }
  printf ("sector en offset 0x%" PRIx64 ":\n",
          data->offset);
  pp = popen ("hexdump -C", "w");
  if (pp == NULL) {
    perror ("popen: hexdump");
    exit (EXIT_FAILURE);
  }
  fwrite (data->sector, SECTOR_SIZE, 1, pp);
  pclose (pp);
  printf ("\n");
  /* Devolver 1 desde el callback retira automáticamente
   * el comando.
   */
  return 1;
}
static struct data data[NR_SECTORS];
int
main (int argc, char *argv[])
{
  struct nbd_handle *nbd;
  size_t i;
  if (argc != 2) {
    fprintf (stderr, "%s socket\n", argv[0]);
    exit (EXIT_FAILURE);
  }
  /* Crear el manejador libnbd. */
  nbd = nbd_create ();
  if (nbd == NULL) {
    fprintf (stderr, "%s\n", nbd_get_error ());
    exit (EXIT_FAILURE);
  }
  /* Conectarse al servidor NBD a través de un socket de dominio Unix.
   * Esto solo inicia la conexión.
   */
  if (nbd_aio_connect_unix (nbd, argv[1]) == -1) {
    fprintf (stderr, "%s\n", nbd_get_error ());
    exit (EXIT_FAILURE);
  }
  /* Esperar a que la conexión se complete. El uso de
   * nbd_poll aquí es solo un ejemplo. También podría
   * integrarse con poll(2), glib u otro bucle principal.
   * Lea libnbd(3) y el archivo fuente lib/poll.c.
   */
  while (!nbd_aio_is_ready (nbd)) {
    if (nbd_poll (nbd, -1) == -1) {
      fprintf (stderr, "%s\n", nbd_get_error ());
      exit (EXIT_FAILURE);
    }
  }
  assert (nbd_get_size (nbd) >= NR_SECTORS * SECTOR_SIZE);
  /* Emitir comandos de lectura para los primeros NR sectores. */
  for (i = 0; i < NR_SECTORS; ++i) {
    data[i].offset = i * SECTOR_SIZE;
    /* El callback (hexdump) se llama cuando el comando
     * se completa. El buffer debe seguir existiendo mientras
     * el comando está en ejecución.
     */
    if (nbd_aio_pread (nbd, data[i].sector, SECTOR_SIZE,
                       data[i].offset,
                       (nbd_completion_callback) {
                         .callback = hexdump,
                         .user_data = &data[i],
                       }, 0) == -1) {
      fprintf (stderr, "%s\n", nbd_get_error ());
      exit (EXIT_FAILURE);
    }
  }
  /* Ejecutar el bucle principal hasta que todos los comandos se hayan
   * completado y retirado. Nuevamente, el uso de nbd_poll
   * aquí es solo un ejemplo.
   */
  while (nbd_aio_in_flight (nbd) > 0) {
    if (nbd_poll (nbd, -1) == -1) {
      fprintf (stderr, "%s\n", nbd_get_error ());
      exit (EXIT_FAILURE);
    }
  }
  /* Cerrar el manejador libnbd. */
  nbd_close (nbd);
  exit (EXIT_SUCCESS);
}
