nbd_set_opt_mode(3)

Paquete: libnbd

Licencias: LGPL-2.1-or-later

nbd_set_opt_mode(3) LIBNBD nbd_set_opt_mode(3)

NOMBRE

nbd_set_opt_mode - controla el modo de opciones, para pausar durante la negociación de opciones

SINOPSIS

#include <libnbd.h>
int nbd_set_opt_mode (
      struct nbd_handle *h, bool enable
    );

DESCRIPCIÓN

Establezca este indicador a verdadero para solicitar que un comando de conexión "nbd_connect_*" se detenga para negociar opciones en lugar de proceder hasta el estado listo, al comunicarse con un servidor de nuevo estilo. Esta configuración no tiene ningún efecto al conectarse a un servidor de estilo antiguo.

Tenga en cuenta que libnbd intenta por defecto "NBD_OPT_STARTTLS", "NBD_OPT_EXTENDED_HEADERS" y "NBD_OPT_STRUCTURED_REPLY" antes de permitirle controlar los pasos de negociación restantes; si necesita control sobre estos pasos también, primero establezca nbd_set_tls(3) a "LIBNBD_TLS_DISABLE", y nbd_set_request_extended_headers(3) o nbd_set_request_structured_replies(3) a falso, antes de iniciar el intento de conexión.

Cuando el modo de opciones está habilitado, tiene un control detallado sobre qué opciones se negocian, en comparación con el valor predeterminado de que el servidor negocie todo en su nombre utilizando la configuración realizada antes de iniciar la conexión. Para salir del modo y proceder al estado listo, debe usar nbd_opt_go(3) con éxito; un nbd_opt_go(3) fallido regresa al estado de negociación para permitir un cambio de nombre de exportación antes de intentarlo de nuevo. También puede usar nbd_opt_abort(3) o nbd_shutdown(3) para finalizar la conexión sin completar la negociación.

VALOR DE RETORNO

Si la llamada es exitosa, la función devuelve 0.

ERRORES

En caso de error, se devuelve -1.

Consulte "MANEJO DE ERRORES" en libnbd(3) para obtener más detalles sobre el error.

Los siguientes parámetros no deben ser NULL: "h". Para más información, consulte "Parámetros no nulos" en libnbd(3).

ESTADO DEL MANEJADOR

nbd_set_opt_mode puede llamarse cuando el manejador se encuentra en el siguiente estado:

┌─────────────────────────────────────┬─────────────────────────┐
│ Manejador creado, antes de conectar │ ✅ permitido              │
│ Conectando                          │ ❌ error                │
│ Conectando y handshaking (opt_mode) │ ❌ error                │
│ Conectado al servidor             │ ❌ error                │
│ Conexión cerrada                 │ ❌ error                │
│ Manejador muerto                  │ ❌ error                │
└─────────────────────────────────────┴─────────────────────────┘

VERSIÓN

Esta función apareció por primera vez en libnbd 1.4.

Si necesita probar si esta función está disponible en tiempo de compilación, compruebe si el siguiente macro está definido:

#define LIBNBD_HAVE_NBD_SET_OPT_MODE 1

EJEMPLO

Este ejemplo también está disponible como examples/list-exports.c en el código fuente de libnbd.

/* Este ejemplo muestra cómo listar exportaciones NBD.
 *
 * Para probar esto con qemu-nbd:
 *   $ qemu-nbd -x "hello" -t -k /tmp/sock disk.img
 *   $ ./run examples/list-exports /tmp/sock
 *   [0] hello
 *   ¿Qué exportación conectar? (-1 para salir) 0
 *   Conectando a hello ...
 *   /tmp/sock: hello: tamaño = 2048 bytes
 *
 * Para probar esto con nbdkit (requiere 1.22):
 *   $ nbdkit -U /tmp/sock sh - <<\EOF
 *   case $1 in
 *     list_exports) echo NAMES; echo foo; echo foobar ;;
 *     open) echo "$3" ;;
 *     get_size) echo "$2" | wc -c ;;
 *     pread) echo "$2" | dd bs=1 skip=$4 count=$3 ;;
 *     *) exit 2 ;;
 *   esac
 *   EOF
 *   $ ./run examples/list-exports /tmp/sock
 *   [0] foo
 *   [1] foobar
 *   ¿Qué exportación conectar? (-1 para salir) 1
 *   Conectando a foobar ...
 *   /tmp/sock: foobar: tamaño = 7 bytes
 */
#include <stdio.h>
#include <stdlib.h>
#include <stdbool.h>
#include <stdint.h>
#include <string.h>
#include <inttypes.h>
#include <errno.h>
#include <libnbd.h>
struct export_list {
  int i;
  char **names;
};
/* Función de devolución de llamada para nbd_opt_list */
static int
list_one (void *opaque, const char *name,
          const char *description)
{
  struct export_list *l = opaque;
  char **names;
  printf ("[%d] %s\n", l->i, name);
  if (*description)
    printf ("  (%s)\n", description);
  names = realloc (l->names,
                   (l->i + 1) * sizeof *names);
  if (!names) {
    perror ("realloc");
    exit (EXIT_FAILURE);
  }
  names[l->i] = strdup (name);
  if (!names[l->i]) {
    perror ("strdup");
    exit (EXIT_FAILURE);
  }
  l->names = names;
  l->i++;
  return 0;
}
int
main (int argc, char *argv[])
{
  struct nbd_handle *nbd;
  int i;
  const char *name;
  int64_t size;
  struct export_list list = { 0 };
  if (argc != 2) {
    fprintf (stderr, "%s socket\n", argv[0]);
    exit (EXIT_FAILURE);
  }
  /* Crear el manejador libnbd. */
  nbd = nbd_create ();
  if (nbd == NULL) {
    fprintf (stderr, "%s\n", nbd_get_error ());
    exit (EXIT_FAILURE);
  }
  /* Establecer el modo de opciones. */
  nbd_set_opt_mode (nbd, true);
  /* Conectarse al servidor NBD a través de un
   * socket de dominio Unix. Si no
   * entramos en modo de opciones, entonces
   * una lista no es posible.
   */
  if (nbd_connect_unix (nbd, argv[1]) == -1) {
    fprintf (stderr, "%s\n", nbd_get_error ());
    exit (EXIT_FAILURE);
  }
  if (!nbd_aio_is_negotiating (nbd)) {
    fprintf (stderr, "El servidor no soporta "
             "listar exportaciones.\n");
    exit (EXIT_FAILURE);
  }
  /* Imprimir la lista de exportaciones. */
  if (nbd_opt_list (nbd,
                    (nbd_list_callback) {
                      .callback = list_one,
                      .user_data = &list, }) == -1) {
    fprintf (stderr, "%s\n", nbd_get_error ());
    exit (EXIT_FAILURE);
  }
  /* Mostrar la lista de exportaciones. */
  printf ("¿Qué exportación conectar? ");
  if (scanf ("%d", &i) != 1) exit (EXIT_FAILURE);
  if (i == -1) {
    if (nbd_opt_abort (nbd) == -1) {
      fprintf (stderr, "%s\n", nbd_get_error ());
      exit (EXIT_FAILURE);
    }
    nbd_close (nbd);
    exit (EXIT_SUCCESS);
  }
  if (i < 0 || i >= list.i) {
    fprintf (stderr, "índice %d fuera de rango", i);
    exit (EXIT_FAILURE);
  }
  name = list.names[i];
  printf ("Conectando a %s ...\n", name);
  /* Reanudar la conexión a la exportación elegida. */
  if (nbd_set_export_name (nbd, name) == -1 ||
      nbd_opt_go (nbd) == -1) {
    fprintf (stderr, "%s\n", nbd_get_error ());
    exit (EXIT_FAILURE);
  }
  if (!nbd_aio_is_ready (nbd)) {
    fprintf (stderr, "el servidor cerró prematuramente\n");
    exit (EXIT_FAILURE);
  }
  /* Leer el tamaño en bytes e imprimirlo. */
  size = nbd_get_size (nbd);
  if (size == -1) {
    fprintf (stderr, "%s\n", nbd_get_error ());
    exit (EXIT_FAILURE);
  }
  printf ("%s: %s: tamaño = %" PRIi64 " bytes\n",
          argv[1], name, size);
  /* Cerrar el manejador libnbd. */
  nbd_close (nbd);
  for (i = 0; i < list.i; i++)
    free (list.names[i]);
  free (list.names);
  exit (EXIT_SUCCESS);
}
