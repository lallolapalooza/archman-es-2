nbd_can_cache(3)

Paquete: libnbd

Licencias: LGPL-2.1-or-later

nbd_can_cache(3) LIBNBD nbd_can_cache(3)

NOMBRE

nbd_can_cache - ¿admite el servidor el comando de caché?

SINOPSIS

#include <libnbd.h>
int nbd_can_cache (
      struct nbd_handle *h
    );

DESCRIPCIÓN

Devuelve verdadero si el servidor admite el comando de caché (ver nbd_cache(3), nbd_aio_cache(3)). Devuelve falso si el servidor no lo hace.

Esta llamada no bloquea, ya que devuelve datos que se guardan en el manejador del protocolo de enlace NBD.

VALOR DE RETORNO

Esta llamada devuelve un valor booleano.

ERRORES

En caso de error, se devuelve -1.

Consulte "MANEJO DE ERRORES" en libnbd(3) para obtener más detalles sobre el error.

Los siguientes parámetros no deben ser NULL: "h". Para más información, consulte "Parámetros no nulos" en libnbd(3).

ESTADO DEL MANEJADOR

nbd_can_cache se puede llamar cuando el manejador está en los siguientes estados:

┌─────────────────────────────────────┬─────────────────────────┐
│ Manejador creado, antes de conectar │ ❌ error                │
│ Conectando                          │ ❌ error                │
│ Conectando y handshake (opt_mode) │ ✅ permitido             │
│ Conectado al servidor             │ ✅ permitido             │
│ Conexión cerrada                   │ ✅ permitido             │
│ Manejador muerto                    │ ❌ error                │
└─────────────────────────────────────┴─────────────────────────┘

VERSIÓN

Esta función apareció por primera vez en libnbd 1.0.

Si necesita probar si esta función está disponible en tiempo de compilación, compruebe si el siguiente macro está definido:

#define LIBNBD_HAVE_NBD_CAN_CACHE 1

EJEMPLO

Este ejemplo también está disponible como examples/server-flags.c en el código fuente de libnbd.

/* Este ejemplo muestra cómo conectarse a un servidor NBD
 * e imprimir las banderas de exportación.
 *
 * Puedes probarlo con nbdkit así:
 *
 * nbdkit -U - memory 1M \
 *   --run './server-flags $unixsocket'
 */
#include <stdio.h>
#include <stdlib.h>
#include <stdbool.h>
#include <libnbd.h>
int
main (int argc, char *argv[])
{
  struct nbd_handle *nbd;
  char *str;
  int flag;
  if (argc != 2) {
    fprintf (stderr, "%s socket\n", argv[0]);
    exit (EXIT_FAILURE);
  }
  /* Crear el manejador de libnbd. */
  nbd = nbd_create ();
  if (nbd == NULL) {
    fprintf (stderr, "%s\n", nbd_get_error ());
    exit (EXIT_FAILURE);
  }
  /* Solicitar información completa. */
#if LIBNBD_HAVE_NBD_SET_FULL_INFO /* Añadido en 1.4 */
  if (nbd_set_full_info (nbd, true) == -1) {
    fprintf (stderr, "%s\n", nbd_get_error ());
    exit (EXIT_FAILURE);
  }
#endif
  /* Conectarse al servidor NBD a través de
   * un socket de dominio Unix.
   */
  if (nbd_connect_unix (nbd, argv[1]) == -1) {
    fprintf (stderr, "%s\n", nbd_get_error ());
    exit (EXIT_FAILURE);
  }
  /* Ver si el servidor proporcionó detalles adicionales,
   * usando funciones añadidas en 1.4
   */
#if LIBNBD_HAVE_NBD_GET_EXPORT_DESCRIPTION
  str = nbd_get_canonical_export_name (nbd);
  if (str)
    printf ("canonical_name = %s\n", str);
  free (str);
  str = nbd_get_export_description (nbd);
  if (str)
    printf ("description = %s\n", str);
  free (str);
#endif
  /* Leer e imprimir las banderas. */
#define PRINT_FLAG(flag_fn)                     \
  flag = flag_fn (nbd);                         \
  if (flag == -1) {                             \
    fprintf (stderr, "%s\n", nbd_get_error ()); \
    exit (EXIT_FAILURE);                        \
  }                                             \
  printf (#flag_fn " = %s\n",                   \
          flag ? "true" : "false");
  PRINT_FLAG (nbd_can_cache);
  PRINT_FLAG (nbd_can_df);
  PRINT_FLAG (nbd_can_flush);
  PRINT_FLAG (nbd_can_fua);
  PRINT_FLAG (nbd_can_multi_conn);
  PRINT_FLAG (nbd_can_trim);
  PRINT_FLAG (nbd_can_zero);
#if LIBNBD_HAVE_NBD_CAN_FAST_ZERO
  /* Añadido en 1.2 */
  PRINT_FLAG (nbd_can_fast_zero);
#endif
#if LIBNBD_HAVE_NBD_CAN_BLOCK_STATUS_PAYLOAD
  /* Añadido en 1.18 */
  PRINT_FLAG (nbd_can_block_status_payload);
#endif
  PRINT_FLAG (nbd_is_read_only);
  PRINT_FLAG (nbd_is_rotational);
  /* Cerrar el manejador de libnbd. */
  nbd_close (nbd);
  exit (EXIT_SUCCESS);
}

nbd_can_cache(3)

NAME

nbd_can_cache - determines if the NBD export supports caching

SYNOPSIS

#include <libnbd-1/nbd.h>

int nbd_can_cache(NbdHandle *handle);

DESCRIPTION

nbd_can_cache checks if the NBD export that handle is connected to supports the NBD_OPT_CACHE caching option.

This option can be used to request that the server cache data for reads and writes. The exact caching behaviour is server-defined.

If the server supports this option, the function will return 1. Otherwise, it will return 0.

ARGUMENTS

  • handle: An NbdHandle obtained from a previous call to nbd_connect_uri or nbd_connect.

RETURN VALUE

Returns 1 if the server supports caching, 0 otherwise.

EXAMPLE

#include <stdio.h>
#include <stdlib.h>
#include <libnbd-1/nbd.h>

int main(int argc, char **argv) {
    NbdHandle *handle;
    int ret;

    if (argc < 2) {
        fprintf(stderr, "Usage: %s uri\n", argv[0]);
        return 1;
    }

    handle = nbd_connect_uri(argv[1]);
    if (!handle) {
        fprintf(stderr, "Failed to connect to %s: %s\n", argv[1], nbd_get_error(NULL));
        return 1;
    }

    ret = nbd_can_cache(handle);
    if (ret == -1) {
        fprintf(stderr, "Failed to check if server supports caching: %s\n", nbd_get_error(NULL));
        nbd_shutdown(handle);
        nbd_close(handle);
        return 1;
    }

    if (ret) {
        printf("The NBD export supports caching.\n");
    } else {
        printf("The NBD export does not support caching.\n");
    }

    nbd_shutdown(handle);
    nbd_close(handle);
    return 0;
}
