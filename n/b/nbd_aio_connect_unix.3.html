nbd_aio_connect_unix(3) LIBNBD nbd_aio_connect_unix(3)

NOMBRE

nbd_aio_connect_unix - conectar al servidor NBD a través de un socket de dominio Unix

SINOPSIS

#include <libnbd.h>
int nbd_aio_connect_unix (
      struct nbd_handle *h, const char *unixsocket
    );

DESCRIPCIÓN

Comenzar la conexión al servidor NBD a través de un socket de dominio Unix ("unixsocket"). Los parámetros se comportan como se documenta en nbd_connect_unix(3).

Puede verificar si el intento de conexión todavía está en curso llamando a nbd_aio_is_connecting(3). Si nbd_set_opt_mode(3) está habilitado, la conexión está lista para la negociación manual de opciones una vez que nbd_aio_is_negotiating(3) devuelve verdadero; de lo contrario, el intento de conexión incluirá el handshake NBD, y estará listo para usarse una vez que nbd_aio_is_ready(3) devuelva verdadero.

VALOR DE RETORNO

Si la llamada tiene éxito, la función devuelve 0.

ERRORES

En caso de error, se devuelve -1.

Consulte "MANEJO DE ERRORES" en libnbd(3) para obtener más detalles sobre el error.

Los siguientes parámetros no deben ser NULL: "h", "unixsocket". Para más información, consulte "Parámetros no NULL" en libnbd(3).

ESTADO DEL MANEJADOR

nbd_aio_connect_unix puede llamarse cuando el manejador está en el siguiente estado:

┌─────────────────────────────────────┬─────────────────────────┐
│ Manejador creado, antes de conectar   │ ✅ permitido              │
│ Conectando                          │ ❌ error                │
│ Conectando & handshake (opt_mode) │ ❌ error                │
│ Conectado al servidor             │ ❌ error                │
│ Conexión cerrada                   │ ❌ error                │
│ Manejador muerto                    │ ❌ error                │
└─────────────────────────────────────┴─────────────────────────┘

VERSIÓN

Esta función apareció por primera vez en libnbd 1.0.

Si necesita probar si esta función está disponible en tiempo de compilación, verifique si el siguiente macro está definido:

#define LIBNBD_HAVE_NBD_AIO_CONNECT_UNIX 1

EJEMPLO

Este ejemplo también está disponible como examples/aio-connect-read.c en el código fuente de libnbd.

/* Este ejemplo muestra cómo usar la API de bajo
 * nivel AIO (asíncrona) para conectarse a un servidor y leer el disco.
 *
 * Aquí hay algunas formas de probar este ejemplo:
 *
 * nbdkit -U - linuxdisk . \
 *   --run './aio-connect-read $unixsocket'
 *
 * nbdkit -U - floppy . \
 *   --run './aio-connect-read $unixsocket'
 *
 * nbdkit -U - pattern size=1M \
 *   --run './aio-connect-read $unixsocket'
 */
#include <stdio.h>
#include <stdlib.h>
#include <stdint.h>
#include <inttypes.h>
#include <errno.h>
#include <assert.h>
#include <libnbd.h>
#define NR_SECTORS 32
#define SECTOR_SIZE 512
struct data {
  uint64_t offset;
  char sector[SECTOR_SIZE];
};
static int
hexdump (void *user_data, int *error)
{
  struct data *data = user_data;
  FILE *pp;
  if (*error) {
    errno = *error;
    perror ("falló la lectura");
    exit (EXIT_FAILURE);
  }
  printf ("sector en el offset 0x%" PRIx64 ":\n",
          data->offset);
  pp = popen ("hexdump -C", "w");
  if (pp == NULL) {
    perror ("popen: hexdump");
    exit (EXIT_FAILURE);
  }
  fwrite (data->sector, SECTOR_SIZE, 1, pp);
  pclose (pp);
  printf ("\n");
  /* Devolver 1 desde la devolución de llamada retira automáticamente
   * el comando.
   */
  return 1;
}
static struct data data[NR_SECTORS];
int
main (int argc, char *argv[])
{
  struct nbd_handle *nbd;
  size_t i;
  if (argc != 2) {
    fprintf (stderr, "%s socket\n", argv[0]);
    exit (EXIT_FAILURE);
  }
  /* Crear el manejador libnbd. */
  nbd = nbd_create ();
  if (nbd == NULL) {
    fprintf (stderr, "%s\n", nbd_get_error ());
    exit (EXIT_FAILURE);
  }
  /* Conectar al servidor NBD a través de un socket de dominio Unix.
   * Esto solo inicia la conexión.
   */
  if (nbd_aio_connect_unix (nbd, argv[1]) == -1) {
    fprintf (stderr, "%s\n", nbd_get_error ());
    exit (EXIT_FAILURE);
  }
  /* Esperar a que se complete la conexión. El uso de
   * nbd_poll aquí es solo un ejemplo. También podrías
   * integrarlo con poll(2), glib u otro bucle principal.
   * Lee libnbd(3) y el archivo fuente lib/poll.c.
   */
  while (!nbd_aio_is_ready (nbd)) {
    if (nbd_poll (nbd, -1) == -1) {
      fprintf (stderr, "%s\n", nbd_get_error ());
      exit (EXIT_FAILURE);
    }
  }
  assert (nbd_get_size (nbd) >= NR_SECTORS * SECTOR_SIZE);
  /* Emitir comandos de lectura para los primeros NR sectores. */
  for (i = 0; i < NR_SECTORS; ++i) {
    data[i].offset = i * SECTOR_SIZE;
    /* La devolución de llamada (hexdump) se llama cuando el comando
     * se completa. El búfer debe seguir existiendo mientras
     * el comando se ejecuta.
     */
    if (nbd_aio_pread (nbd, data[i].sector, SECTOR_SIZE,
                       data[i].offset,
                       (nbd_completion_callback) {
                         .callback = hexdump,
                         .user_data = &data[i],
                       }, 0) == -1) {
      fprintf (stderr, "%s\n", nbd_get_error ());
      exit (EXIT_FAILURE);
    }
  }
  /* Ejecutar el bucle principal hasta que todos los comandos se
   * hayan completado y retirado. Nuevamente, el uso de nbd_poll
   * aquí es solo un ejemplo.
   */
  while (nbd_aio_in_flight (nbd) > 0) {
    if (nbd_poll (nbd, -1) == -1) {
      fprintf (stderr, "%s\n", nbd_get_error ());
      exit (EXIT_FAILURE);
    }
  }
  /* Cerrar el manejador libnbd. */
  nbd_close (nbd);
  exit (EXIT_SUCCESS);
}

NAME

nbd_aio_connect_unix - asynchronously connect to an NBD server via a Unix domain socket

SYNOPSIS

#include <libnbd.h>

void nbd_aio_connect_unix (
    struct nbd_handle *nbd,
    const char *path,
    nbd_completion_func callback,
    void *callback_data,
    int *error
);

DESCRIPTION

nbd_aio_connect_unix is the asynchronous version of nbd_connect_unix(3). It attempts to connect to an NBD server specified by the Unix domain socket path.

The connection attempt is non-blocking. When the connection attempt is complete (either successful or failed), the callback function will be called with callback_data as its argument.

The callback function should have the signature:

void callback (struct nbd_handle *nbd, void *data, int error);

If error is 0, the connection was successful. Otherwise, it contains an error code. See ERRORS.

The error parameter of nbd_aio_connect_unix itself is not used for reporting connection errors. It is currently unused and is reserved for future use. You should pass a pointer to an integer, but it will always be set to 0 on successful return from the function itself.

This function is used in conjunction with nbd_aio_context_run(3) or your own GLib main loop to handle the asynchronous I/O.

RETURN VALUE

On success, nbd_aio_connect_unix returns immediately. The result of the connection attempt is reported via the callback function.

On failure, nbd_aio_connect_unix returns immediately and sets the integer pointed to by error to an error code. However, this only indicates that the connection attempt could not even be started. It does not mean the connection itself failed.

ERRORS

The error parameter passed to the callback function can be set to:

0
Success.
-EACCES
Permission denied to access the socket file.
-ECONNREFUSED
Connection refused by the server.
-ENOENT
The socket file does not exist.
-ENOTSOCK
The path is not a socket.
-ETIMEDOUT
The connection timed out.
-EPROTO
The server did not speak the NBD protocol.
-ENOMEM
Out of memory.
-EHOSTDOWN
The host is down.
-EAGAIN
Temporary unavailability of resources. Try again later.
-EINTR
The operation was interrupted by a signal.
-EPIPE
The connection was broken.
-EPROTO
The server sent an invalid reply.
-EBADFD
The handle is invalid.
-EINVAL
Invalid arguments.
-ENOTSUP
The requested operation is not supported.

You can use strerror(3) to convert error codes to human-readable strings.

HANDLE STATE

After calling this function, the handle is in the NBD_STATE_CONNECTING state until the callback is invoked.

VERSION

This function is available since libnbd 1.2.

EXAMPLE

Here is a simple example of how to use nbd_aio_connect_unix:

#include <libnbd.h>
#include <stdio.h>
#include <stdlib.h>
#include <unistd.h>
#include <errno.h>

static void connect_callback (struct nbd_handle *nbd, void *data, int error) {
    if (error == 0) {
        printf ("Successfully connected.\n");
        // Now you can use the handle for NBD operations.
        // For example, get the size:
        long long size;
        error = nbd_aio_get_size (nbd, &size, NULL, NULL, NULL);
        if (error == 0) {
            printf ("Device size: %lld bytes\n", size);
        } else {
            fprintf (stderr, "Failed to get size: %s\n", strerror (error));
        }
    } else {
        fprintf (stderr, "Connection failed: %s\n", strerror (error));
    }

    // Clean up and exit the loop
    nbd_shutdown (nbd);
    nbd_aio_context_unref (nbd_aio_context (nbd));
}

int main (int argc, char **argv) {
    if (argc != 2) {
        fprintf (stderr, "Usage: %s <socket_path>\n", argv[0]);
        return 1;
    }

    const char *path = argv[1];
    struct nbd_handle *nbd;
    int error;

    // Create a new NBD handle
    nbd = nbd_create ();
    if (nbd == NULL) {
        fprintf (stderr, "Failed to create NBD handle\n");
        return 1;
    }

    // Create an async context
    struct nbd_aio_context *ctx = nbd_aio_context_create ();
    if (ctx == NULL) {
        fprintf (stderr, "Failed to create NBD async context\n");
        nbd_destroy (nbd);
        return 1;
    }
    nbd_set_aio_context (nbd, ctx);

    // Start the asynchronous connection
    nbd_aio_connect_unix (nbd, path, connect_callback, NULL, &error);
    if (error != 0) {
        fprintf (stderr, "Failed to start connection: %s\n", strerror (error));
        nbd_destroy (nbd);
        nbd_aio_context_unref (ctx);
        return 1;
    }

    // Run the async loop until shutdown
    nbd_aio_context_run (ctx);

    // Clean up
    nbd_destroy (nbd);
    nbd_aio_context_unref (ctx);

    return 0;
}
