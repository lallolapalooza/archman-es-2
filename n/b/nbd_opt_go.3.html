nbd_opt_go(3)

Paquete: libnbd

Licencias: LGPL-2.1-or-later

nbd_opt_go(3) LIBNBD nbd_opt_go(3)

NOMBRE

nbd_opt_go - finalizar la negociación y pasar a usar una exportación

SINOPSIS

#include <libnbd.h>
int nbd_opt_go (
      struct nbd_handle *h
    );

DESCRIPCIÓN

Solicita que el servidor finalice la negociación y pase a servir la exportación especificada previamente por la llamada más reciente a nbd_set_export_name(3) o nbd_connect_uri(3). Esto solo se puede usar si nbd_set_opt_mode(3) habilitó el modo de opciones.

Por defecto, libnbd solicitará automáticamente todos los contextos meta registrados por nbd_add_meta_context(3) como parte de esta llamada; pero esto se puede suprimir con nbd_set_request_meta_context(3), especialmente si se utilizó nbd_opt_set_meta_context(3) anteriormente en la secuencia de negociación.

Si esto falla, el servidor aún podría estar en negociación, donde es posible intentar otra opción, como un nombre de exportación diferente; aunque los servidores más antiguos habrán cerrado la conexión.

VALOR DE RETORNO

Si la llamada es exitosa, la función devuelve 0.

ERRORES

En caso de error, se devuelve -1.

Consulte "MANEJO DE ERRORES" en libnbd(3) para obtener más detalles sobre el error.

Los siguientes parámetros no deben ser NULL: "h". Para más información, consulte "Parámetros no nulos" en libnbd(3).

ESTADO DEL MANEJADOR

nbd_opt_go se puede llamar cuando el manejador se encuentra en el siguiente estado:

┌─────────────────────────────────────┬─────────────────────────┐
│ Man. creado, antes de conectar      │ ❌ error                │
│ Conectando                          │ ❌ error                │
│ Conectando y handshake (opt_mode) │ ✅ permitido             │
│ Conectado al servidor             │ ❌ error                │
│ Conexión cerrada                   │ ❌ error                │
│ Man. muerto                         │ ❌ error                │
└─────────────────────────────────────┴─────────────────────────┘

VERSIÓN

Esta función apareció por primera vez en libnbd 1.4.

Si necesita probar si esta función está disponible en tiempo de compilación, verifique si el siguiente macro está definido:

#define LIBNBD_HAVE_NBD_OPT_GO 1

EJEMPLO

Este ejemplo también está disponible como examples/list-exports.c en el código fuente de libnbd.

/* Este ejemplo muestra cómo listar exportaciones NBD.
 *
 * Para probar esto con qemu-nbd:
 *   $ qemu-nbd -x "hello" -t -k /tmp/sock disk.img
 *   $ ./run examples/list-exports /tmp/sock
 *   [0] hello
 *   ¿Qué exportación conectar? (-1 para salir) 0
 *   Conectando a hello ...
 *   /tmp/sock: hello: tamaño = 2048 bytes
 *
 * Para probar esto con nbdkit (requiere 1.22):
 *   $ nbdkit -U /tmp/sock sh - <<\EOF
 *   case $1 in
 *     list_exports) echo NAMES; echo foo; echo foobar ;;
 *     open) echo "$3" ;;
 *     get_size) echo "$2" | wc -c ;;
 *     pread) echo "$2" | dd bs=1 skip=$4 count=$3 ;;
 *     *) exit 2 ;;
 *   esac
 *   EOF
 *   $ ./run examples/list-exports /tmp/sock
 *   [0] foo
 *   [1] foobar
 *   ¿Qué exportación conectar? (-1 para salir) 1
 *   Conectando a foobar ...
 *   /tmp/sock: foobar: tamaño = 7 bytes
 */
#include <stdio.h>
#include <stdlib.h>
#include <stdbool.h>
#include <stdint.h>
#include <string.h>
#include <inttypes.h>
#include <errno.h>
#include <libnbd.h>
struct export_list {
  int i;
  char **names;
};
/* Función de devolución de llamada para nbd_opt_list */
static int
list_one (void *opaque, const char *name,
          const char *description)
{
  struct export_list *l = opaque;
  char **names;
  printf ("[%d] %s\n", l->i, name);
  if (*description)
    printf ("  (%s)\n", description);
  names = realloc (l->names,
                   (l->i + 1) * sizeof *names);
  if (!names) {
    perror ("realloc");
    exit (EXIT_FAILURE);
  }
  names[l->i] = strdup (name);
  if (!names[l->i]) {
    perror ("strdup");
    exit (EXIT_FAILURE);
  }
  l->names = names;
  l->i++;
  return 0;
}
int
main (int argc, char *argv[])
{
  struct nbd_handle *nbd;
  int i;
  const char *name;
  int64_t size;
  struct export_list list = { 0 };
  if (argc != 2) {
    fprintf (stderr, "%s socket\n", argv[0]);
    exit (EXIT_FAILURE);
  }
  /* Crear el manejador libnbd. */
  nbd = nbd_create ();
  if (nbd == NULL) {
    fprintf (stderr, "%s\n", nbd_get_error ());
    exit (EXIT_FAILURE);
  }
  /* Establecer el modo de opciones. */
  nbd_set_opt_mode (nbd, true);
  /* Conectar al servidor NBD a través de un
   * socket de dominio Unix. Si no
   * entramos en modo de opciones, entonces
   * no es posible una lista.
   */
  if (nbd_connect_unix (nbd, argv[1]) == -1) {
    fprintf (stderr, "%s\n", nbd_get_error ());
    exit (EXIT_FAILURE);
  }
  if (!nbd_aio_is_negotiating (nbd)) {
    fprintf (stderr, "El servidor no soporta "
             "listar exportaciones.\n");
    exit (EXIT_FAILURE);
  }
  /* Imprimir la lista de exportaciones. */
  if (nbd_opt_list (nbd,
                    (nbd_list_callback) {
                      .callback = list_one,
                      .user_data = &list, }) == -1) {
    fprintf (stderr, "%s\n", nbd_get_error ());
    exit (EXIT_FAILURE);
  }
  /* Mostrar la lista de exportaciones. */
  printf ("¿Qué exportación conectar? ");
  if (scanf ("%d", &i) != 1) exit (EXIT_FAILURE);
  if (i == -1) {
    if (nbd_opt_abort (nbd) == -1) {
      fprintf (stderr, "%s\n", nbd_get_error ());
      exit (EXIT_FAILURE);
    }
    nbd_close (nbd);
    exit (EXIT_SUCCESS);
  }
  if (i < 0 || i >= list.i) {
    fprintf (stderr, "índice %d fuera de rango", i);
    exit (EXIT_FAILURE);
  }
  name = list.names[i];
  printf ("Conectando a %s ...\n", name);
  /* Reanudar la conexión a la exportación elegida. */
  if (nbd_set_export_name (nbd, name) == -1 ||
      nbd_opt_go (nbd) == -1) {
    fprintf (stderr, "%s\n", nbd_get_error ());
    exit (EXIT_FAILURE);
  }
  if (!nbd_aio_is_ready (nbd)) {
    fprintf (stderr, "el servidor se cerró temprano\n");
    exit (EXIT_FAILURE);
  }
  /* Leer el tamaño en bytes e imprimirlo. */
  size = nbd_get_size (nbd);
  if (size == -1) {
    fprintf (stderr, "%s\n", nbd_get_error ());
    exit (EXIT_FAILURE);
  }
  printf ("%s: %s: tamaño = %" PRIi64 " bytes\n",
          argv[1], name, size);
  /* Cerrar el manejador libnbd. */
  nbd_close (nbd);
  for (i = 0; i < list.i; i++)
    free (list.names[i]);
  free (list.names);
  exit (EXIT_SUCCESS);
}
