nbd_poll(3)

Paquete: libnbd

Licencias: LGPL-2.1-or-later

nbd_poll(3) LIBNBD nbd_poll(3)

NOMBRE

nbd_poll - sondea el manejador una vez

SINOPSIS

#include <libnbd.h>
int nbd_poll (
      struct nbd_handle *h, int timeout
    );

DESCRIPCIÓN

Esta es una implementación simple de poll(2) que se utiliza internamente en las llamadas a la API síncrona. En caso de éxito, devuelve 0 si ocurre el "timeout" (en milisegundos), o 1 si el sondeo se completó y la máquina de estados progresó. Establezca "timeout" a -1 para bloquear indefinidamente (pero tenga cuidado de que la acción eventual sea realmente esperada; por ejemplo, si la conexión está establecida pero no hay comandos en curso, usar un tiempo de espera infinito bloqueará permanentemente).

Esta función es útil principalmente como ejemplo de cómo podría integrar libnbd con su propio bucle principal, en lugar de ser algo que se pretende que utilice.

VALOR DE RETORNO

Esta llamada devuelve un entero ≥ 0.

ERRORES

En caso de error, se devuelve -1.

Consulte "MANEJO DE ERRORES" en libnbd(3) para obtener más detalles sobre el error.

Los siguientes parámetros no deben ser NULL: "h". Para más información, consulte "Parámetros no nulos" en libnbd(3).

VERSIÓN

Esta función apareció por primera vez en libnbd 1.0.

Si necesita comprobar si esta función está disponible en tiempo de compilación, verifique si el siguiente macro está definido:

#define LIBNBD_HAVE_NBD_POLL 1

EJEMPLO

Este ejemplo también está disponible como examples/aio-connect-read.c en el código fuente de libnbd.

/* Este ejemplo muestra cómo utilizar la API de bajo nivel AIO (asíncrona)
 * para conectarse a un servidor y leer el disco.
 *
 * Aquí hay algunas formas de probar este ejemplo:
 *
 * nbdkit -U - linuxdisk . \
 *   --run './aio-connect-read $unixsocket'
 *
 * nbdkit -U - floppy . \
 *   --run './aio-connect-read $unixsocket'
 *
 * nbdkit -U - pattern size=1M \
 *   --run './aio-connect-read $unixsocket'
 */
#include <stdio.h>
#include <stdlib.h>
#include <stdint.h>
#include <inttypes.h>
#include <errno.h>
#include <assert.h>
#include <libnbd.h>
#define NR_SECTORS 32
#define SECTOR_SIZE 512
struct data {
  uint64_t offset;
  char sector[SECTOR_SIZE];
};
static int
hexdump (void *user_data, int *error)
{
  struct data *data = user_data;
  FILE *pp;
  if (*error) {
    errno = *error;
    perror ("fallo al leer");
    exit (EXIT_FAILURE);
  }
  printf ("sector en el desplazamiento 0x%" PRIx64 ":\n",
          data->offset);
  pp = popen ("hexdump -C", "w");
  if (pp == NULL) {
    perror ("popen: hexdump");
    exit (EXIT_FAILURE);
  }
  fwrite (data->sector, SECTOR_SIZE, 1, pp);
  pclose (pp);
  printf ("\n");
  /* Devolver 1 desde la devolución de llamada retira
   * el comando automáticamente.
   */
  return 1;
}
static struct data data[NR_SECTORS];
int
main (int argc, char *argv[])
{
  struct nbd_handle *nbd;
  size_t i;
  if (argc != 2) {
    fprintf (stderr, "%s socket\n", argv[0]);
    exit (EXIT_FAILURE);
  }
  /* Crear el manejador libnbd. */
  nbd = nbd_create ();
  if (nbd == NULL) {
    fprintf (stderr, "%s\n", nbd_get_error ());
    exit (EXIT_FAILURE);
  }
  /* Conectarse al servidor NBD a través de un socket de dominio Unix.
   * Esto solo inicia la conexión.
   */
  if (nbd_aio_connect_unix (nbd, argv[1]) == -1) {
    fprintf (stderr, "%s\n", nbd_get_error ());
    exit (EXIT_FAILURE);
  }
  /* Esperar a que la conexión se complete. El uso de
   * nbd_poll aquí es solo un ejemplo. También podría
   * integrarlo con poll(2), glib u otro bucle
   * principal. Lea libnbd(3) y el archivo fuente lib/poll.c.
   */
  while (!nbd_aio_is_ready (nbd)) {
    if (nbd_poll (nbd, -1) == -1) {
      fprintf (stderr, "%s\n", nbd_get_error ());
      exit (EXIT_FAILURE);
    }
  }
  assert (nbd_get_size (nbd) >= NR_SECTORS * SECTOR_SIZE);
  /* Emitir comandos de lectura para los primeros NR sectores. */
  for (i = 0; i < NR_SECTORS; ++i) {
    data[i].offset = i * SECTOR_SIZE;
    /* La devolución de llamada (hexdump) se llama cuando el comando
     * se completa. El búfer debe seguir existiendo mientras
     * el comando se ejecuta.
     */
    if (nbd_aio_pread (nbd, data[i].sector, SECTOR_SIZE,
                       data[i].offset,
                       (nbd_completion_callback) {
                         .callback = hexdump,
                         .user_data = &data[i],
                       }, 0) == -1) {
      fprintf (stderr, "%s\n", nbd_get_error ());
      exit (EXIT_FAILURE);
    }
  }
  /* Ejecutar el bucle principal hasta que todos los comandos se hayan
   * completado y retirado. Nuevamente, el uso de nbd_poll
   * aquí es solo un ejemplo.
   */
  while (nbd_aio_in_flight (nbd) > 0) {
    if (nbd_poll (nbd, -1) == -1) {
      fprintf (stderr, "%s\n", nbd_get_error ());
      exit (EXIT_FAILURE);
    }
  }
  /* Cerrar el manejador libnbd. */
  nbd_close (nbd);
  exit (EXIT_SUCCESS);
}
