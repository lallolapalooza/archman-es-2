MLX5DV_QUERY_DEVICE(3) Manual de funciones de la biblioteca MLX5DV_QUERY_DEVICE(3)

NOMBRE

mlx5dv_query_device - Consulta las capacidades del dispositivo específicas de mlx5

SINOPSIS

#include <infiniband/mlx5dv.h>

int mlx5dv_query_device(struct ibv_context *ctx_in,
                        struct mlx5dv_context *attrs_out);

DESCRIPCIÓN

mlx5dv_query_device() consulta información específica del hardware del dispositivo que es importante para la ruta de datos, pero que no es proporcionada por ibv_query_device(3).

Esta función devuelve la versión, las banderas y la máscara de compatibilidad. La versión representa el formato de las estructuras internas del hardware que mlx5dv.h representa. Las adiciones de nuevos campos a las estructuras existentes se manejan a través del campo comp_mask.

struct mlx5dv_sw_parsing_caps {
uint32_t sw_parsing_offloads; /* Usar enum mlx5dv_sw_parsing_offloads */
uint32_t supported_qpts;
};
struct mlx5dv_striding_rq_caps {
uint32_t min_single_stride_log_num_of_bytes; /* log mínimo de tamaño de cada paso */
uint32_t max_single_stride_log_num_of_bytes; /* log máximo de tamaño de cada paso */
uint32_t min_single_wqe_log_num_of_strides; /* log mínimo de número de pasos por WQE */
uint32_t max_single_wqe_log_num_of_strides; /* log máximo de número de pasos por WQE */
uint32_t supported_qpts;
};
struct mlx5dv_dci_streams_caps {
uint8_t max_log_num_concurent; /* log máximo de transmisiones concurrentes diferentes que el HW podría manejar */
uint8_t max_log_num_errored; /* canales máximos de transmisiones de error DCI soportados por DCI antes de que un DCI pase a un estado de error */
};
struct mlx5dv_context {
uint8_t         version;
uint64_t        flags;
uint64_t        comp_mask; /* Usar enum mlx5dv_context_comp_mask */
struct mlx5dv_cqe_comp_caps     cqe_comp_caps;
struct mlx5dv_sw_parsing_caps sw_parsing_caps;
uint32_t	tunnel_offloads_caps;
uint32_t        max_dynamic_bfregs /* registros blue-flame máximos que se pueden asignar dinámicamente */
uint64_t        max_clock_info_update_nsec;
uint32_t        flow_action_flags; /* usar enum mlx5dv_flow_action_cap_flags */
uint32_t        dc_odp_caps; /* usar enum ibv_odp_transport_cap_bits */
void		*hca_core_clock; /* apunta a una ubicación de memoria mapeada al reloj central del HCA */
struct mlx5dv_sig_caps sig_caps;
size_t max_wr_memcpy_length; /* longitud máxima soportada por el WR DMA memcpy */
struct mlx5dv_crypto_caps crypto_caps;
uint64_t max_dc_rd_atom; /* Número máximo de lecturas/atómicas RDMA pendientes por DC QP como solicitante */
uint64_t max_dc_init_rd_atom; /* Número máximo de lecturas/atómicas RDMA pendientes por DC QP como respondedor */
struct mlx5dv_reg reg_c0; /* valor y máscara para hacer coincidir el tráfico de egreso del vport local en FDB */
struct mlx5dv_ooo_recv_wrs_caps ooo_recv_wrs_caps; /* Número máximo de WRs pendientes por tipo de QP fuera de orden */
};
enum mlx5dv_context_flags {
/*
 * Esta bandera indica si se necesita la versión 0 o 1 de CQE.
 */
 MLX5DV_CONTEXT_FLAGS_CQE_V1 = (1 << 0),
 MLX5DV_CONTEXT_FLAGS_OBSOLETE    =  (1 << 1), /* Obsoleto, no usar */
 MLX5DV_CONTEXT_FLAGS_MPW_ALLOWED  = (1 << 2), /* Se permite el WQE de múltiples paquetes */
 MLX5DV_CONTEXT_FLAGS_ENHANCED_MPW = (1 << 3), /* Se soporta o no el WQE de múltiples paquetes mejorado */
 MLX5DV_CONTEXT_FLAGS_CQE_128B_COMP = (1 << 4), /* Soporta compresión CQE de 128B */
 MLX5DV_CONTEXT_FLAGS_CQE_128B_PAD = (1 << 5), /* Soporta relleno CQE de 128B */
 MLX5DV_CONTEXT_FLAGS_PACKET_BASED_CREDIT_MODE = (1 << 6), /* Soporta modo de crédito basado en paquetes en RC QP */
/*
 * Si CQ se creó con IBV_WC_EX_WITH_COMPLETION_TIMESTAMP_WALLCLOCK, los timestamps de CQE estarán en formato de tiempo real.
 */
 MLX5DV_CONTEXT_FLAGS_REAL_TIME_TS = (1 << 7),
 MLX5DV_CONTEXT_FLAGS_BLUEFLAME = (1 << 8), /* Indica si BlueFlame es compatible con el dispositivo */
};
enum mlx5dv_context_comp_mask {
MLX5DV_CONTEXT_MASK_CQE_COMPRESION      = 1 << 0,
MLX5DV_CONTEXT_MASK_SWP                 = 1 << 1,
MLX5DV_CONTEXT_MASK_STRIDING_RQ         = 1 << 2,
MLX5DV_CONTEXT_MASK_TUNNEL_OFFLOADS     = 1 << 3,
MLX5DV_CONTEXT_MASK_DYN_BFREGS          = 1 << 4,
MLX5DV_CONTEXT_MASK_CLOCK_INFO_UPDATE   = 1 << 5,
MLX5DV_CONTEXT_MASK_FLOW_ACTION_FLAGS   = 1 << 6,
MLX5DV_CONTEXT_MASK_DC_ODP_CAPS         = 1 << 7,
MLX5DV_CONTEXT_MASK_HCA_CORE_CLOCK      = 1 << 8,
MLX5DV_CONTEXT_MASK_NUM_LAG_PORTS       = 1 << 9,
MLX5DV_CONTEXT_MASK_SIGNATURE_OFFLOAD   = 1 << 10,
MLX5DV_CONTEXT_MASK_DCI_STREAMS         = 1 << 11,
MLX5DV_CONTEXT_MASK_WR_MEMCPY_LENGTH    = 1 << 12,
MLX5DV_CONTEXT_MASK_CRYPTO_OFFLOAD      = 1 << 13,
MLX5DV_CONTEXT_MASK_MAX_DC_RD_ATOM      = 1 << 14,
MLX5DV_CONTEXT_MASK_REG_C0              = 1 << 15,
MLX5DV_CONTEXT_MASK_OOO_RECV_WRS        = 1 << 16,
};
enum enum mlx5dv_sw_parsing_offloads {
MLX5DV_SW_PARSING         = 1 << 0,
MLX5DV_SW_PARSING_CSUM    = 1 << 1,
MLX5DV_SW_PARSING_LSO     = 1 << 2,
};
enum mlx5dv_tunnel_offloads {
MLX5DV_RAW_PACKET_CAP_TUNNELED_OFFLOAD_VXLAN  = 1 << 0,
MLX5DV_RAW_PACKET_CAP_TUNNELED_OFFLOAD_GRE    = 1 << 1,
MLX5DV_RAW_PACKET_CAP_TUNNELED_OFFLOAD_GENEVE = 1 << 2,
};
enum mlx5dv_flow_action_cap_flags {
MLX5DV_FLOW_ACTION_FLAGS_ESP_AES_GCM                = 1 << 0, /* La acción de flujo ESP (con clave AES_GCM) es compatible */
MLX5DV_FLOW_ACTION_FLAGS_ESP_AES_GCM_REQ_METADATA   = 1 << 1, /* La acción de flujo ESP siempre devuelve metadatos en la carga útil */
MLX5DV_FLOW_ACTION_FLAGS_ESP_AES_GCM_SPI_STEERING   = 1 << 2, /* ESP (con clave AESGCM) Soporta la coincidencia por SPI (en lugar de hash contra SPI) */
MLX5DV_FLOW_ACTION_FLAGS_ESP_AES_GCM_FULL_OFFLOAD   = 1 << 3, /* La acción de flujo ESP soporta la descarga completa (con clave AES_GCM) */
MLX5DV_FLOW_ACTION_FLAGS_ESP_AES_GCM_TX_IV_IS_ESN   = 1 << 4, /* Acción de flujo ESP (con clave AES_GCM), ESN viene implícitamente del IV. */
};
struct mlx5dv_sig_caps {
uint64_t block_size; /* usar enum mlx5dv_block_size_caps */
uint32_t block_prot; /* usar enum mlx5dv_sig_prot_caps */
uint16_t t10dif_bg; /* usar enum mlx5dv_sig_t10dif_bg_caps */
uint16_t crc_type; /* usar enum mlx5dv_sig_crc_type_caps */
};
enum mlx5dv_sig_prot_caps {
MLX5DV_SIG_PROT_CAP_T10DIF = 1 << MLX5DV_SIG_TYPE_T10DIF,
MLX5DV_SIG_PROT_CAP_CRC = 1 << MLX5DV_SIG_TYPE_CRC,
};
enum mlx5dv_sig_t10dif_bg_caps {
MLX5DV_SIG_T10DIF_BG_CAP_CRC = 1 << MLX5DV_SIG_T10DIF_CRC,
MLX5DV_SIG_T10DIF_BG_CAP_CSUM = 1 << MLX5DV_SIG_T10DIF_CSUM,
};
enum mlx5dv_sig_crc_type_caps {
MLX5DV_SIG_CRC_TYPE_CAP_CRC32 = 1 << MLX5DV_SIG_CRC_TYPE_CRC32,
MLX5DV_SIG_CRC_TYPE_CAP_CRC32C = 1 << MLX5DV_SIG_CRC_TYPE_CRC32C,
MLX5DV_SIG_CRC_TYPE_CAP_CRC64_XP10 = 1 << MLX5DV_SIG_CRC_TYPE_CRC64_XP10,
};
enum mlx5dv_block_size_caps {
MLX5DV_BLOCK_SIZE_CAP_512 = 1 << MLX5DV_BLOCK_SIZE_512,
MLX5DV_BLOCK_SIZE_CAP_520 = 1 << MLX5DV_BLOCK_SIZE_520,
MLX5DV_BLOCK_SIZE_CAP_4048 = 1 << MLX5DV_BLOCK_SIZE_4048,
MLX5DV_BLOCK_SIZE_CAP_4096 = 1 << MLX5DV_BLOCK_SIZE_4096,
MLX5DV_BLOCK_SIZE_CAP_4160 = 1 << MLX5DV_BLOCK_SIZE_4160,
};
struct mlx5dv_crypto_caps {
/*
 * si failed_selftests != 0, significa que hay algunos errores en las autocomprobaciones
 * que pueden dejar inutilizables ciertos motores criptográficos. El significado exacto del código
 * debe consultarse con NVIDIA.
 */
uint16_t failed_selftests;
uint8_t crypto_engines; /* usar enum mlx5dv_crypto_engines_caps */
uint8_t wrapped_import_method; /* usar enum mlx5dv_crypto_wrapped_import_method_caps */
uint8_t log_max_num_deks;
uint32_t flags; /* usar enum mlx5dv_crypto_caps_flags */
};
/* Este mapa de bits indica qué motores criptográficos son compatibles con el dispositivo. */
enum mlx5dv_crypto_engines_caps {
/* Obsoleto, reemplazado por MLX5DV_CRYPTO_ENGINES_CAP_AES_XTS_SINGLE_BLOCK */
 MLX5DV_CRYPTO_ENGINES_CAP_AES_XTS = 1 << 0,
/*
 * Indica que AES-XTS solo soporta la encriptación de un solo bloque
 * a la vez.
 */
 MLX5DV_CRYPTO_ENGINES_CAP_AES_XTS_SINGLE_BLOCK = 1 << 1,
/* Indica que AES-XTS soporta la encriptación de múltiples bloques. */
 MLX5DV_CRYPTO_ENGINES_CAP_AES_XTS_MULTI_BLOCK = 1 << 2,
};
/*
 * Este mapa de bits indica el método de importación de cada motor criptográfico.
 *
 * Si un bit no está activado, el motor criptográfico correspondiente utiliza el método de importación en texto plano y la
 * aplicación debe usar DEKs en texto plano para este motor criptográfico.
 *
 * De lo contrario, el motor criptográfico correspondiente utiliza el método de importación envuelto y la aplicación
 * debe usar DEKs envueltos para este motor criptográfico. Para cargar DEKs envueltos, la aplicación debe realizar
 * un inicio de sesión criptográfico, lo que a su vez requiere que MLX5DV_CRYPTO_CAPS_WRAPPED_CRYPTO_OPERATIONAL, a continuación, esté configurado.
 */
enum mlx5dv_crypto_wrapped_import_method_caps {
 MLX5DV_CRYPTO_WRAPPED_IMPORT_METHOD_CAP_AES_XTS = 1 << 0,
};
enum mlx5dv_crypto_caps_flags {
/* Indica si las capacidades criptográficas están habilitadas en el dispositivo. */
 MLX5DV_CRYPTO_CAPS_CRYPTO = 1 << 0,
/* Indica si los motores criptográficos que utilizan el método de importación envuelto están operativos. */
 MLX5DV_CRYPTO_CAPS_WRAPPED_CRYPTO_OPERATIONAL = 1 << 1,
/*
 * Si está configurado, indica que después del próximo reinicio de FW, el dispositivo volverá
 * al modo de puesta en marcha, lo que significa que MLX5DV_CRYPTO_CAPS_WRAPPED_CRYPTO_OPERATIONAL
 * se configurará como 0.
 */
 MLX5DV_CRYPTO_CAPS_WRAPPED_CRYPTO_GOING_TO_COMMISSIONING = 1 << 2,
};
struct mlx5dv_ooo_recv_wrs_caps {
uint32_t max_rc;
uint32_t max_xrc;
uint32_t max_dct;
uint32_t max_ud;
uint32_t max_uc;
};

VALOR DE RETORNO

0 en caso de éxito o el valor de errno en caso de fallo (que indica el motivo del fallo).

NOTAS


* La máscara de compatibilidad (comp_mask) es un campo de entrada/salida.
