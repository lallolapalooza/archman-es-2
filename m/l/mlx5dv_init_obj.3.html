MLX5DV_INIT_OBJ(3) Manual de funciones de biblioteca MLX5DV_INIT_OBJ(3)

NOMBRE

mlx5dv_init_obj - Inicializa el objeto mlx5 direct verbs a partir de estructuras ibv_xxx o mlx5dv_xxx

SINOPSIS

#include <infiniband/mlx5dv.h>

int mlx5dv_init_obj(struct mlx5dv_obj *obj, uint64_t obj_type);

DESCRIPCIÓN

mlx5dv_init_obj() Esta función inicializará las estructuras mlx5dv_xxx basándose en el tipo proporcionado. La información para la inicialización se toma de las estructuras ibv_xx o mlx5dv_xxx que se suministran como parte de la entrada.

La solicitud de información de CQ marca su propiedad por parte de direct verbs para todas las acciones relacionadas con el índice del consumidor. El tipo de inicialización puede ser una combinación de varios tipos.

struct mlx5dv_qp {
uint32_t                *dbrec;
struct {
void            *buf;
uint32_t        wqe_cnt;
uint32_t        stride;
} sq;
struct {
void            *buf;
uint32_t        wqe_cnt;
uint32_t        stride;
} rq;
struct {
void            *reg;
uint32_t        size;
} bf;
uint64_t                comp_mask;
off_t			uar_mmap_offset;
uint32_t		tirn;
uint32_t		tisn;
uint32_t		rqn;
uint32_t		sqn;
uint64_t 		tir_icm_address;
};
struct mlx5dv_cq {
void                    *buf;
uint32_t                *dbrec;
uint32_t                cqe_cnt;
uint32_t                cqe_size;
void                    *cq_uar;
uint32_t                cqn;
uint64_t                comp_mask;
};
struct mlx5dv_srq {
void                    *buf;
uint32_t                *dbrec;
uint32_t                stride;
uint32_t                head;
uint32_t                tail;
uint64_t                comp_mask;
uint32_t                srqn;
};
struct mlx5dv_rwq {
void            *buf;
uint32_t        *dbrec;
uint32_t        wqe_cnt;
uint32_t        stride;
uint64_t        comp_mask;
};
struct mlx5dv_dm {
void		*buf;
uint64_t	length;
uint64_t	comp_mask;
uint64_t	remote_va;
};
struct mlx5dv_ah {
struct mlx5_wqe_av    *av;
uint64_t              comp_mask;
};
struct mlx5dv_pd {
uint32_t              pdn;
uint64_t              comp_mask;
};
struct mlx5dv_devx {
uint32_t              handle; /* El handle del kernel, se puede usar en la destrucción directa de ioctl */
};
struct mlx5dv_obj {
struct {
struct ibv_qp           *in;
struct mlx5dv_qp        *out;
} qp;
struct {
struct ibv_cq           *in;
struct mlx5dv_cq        *out;
} cq;
struct {
struct ibv_srq          *in;
struct mlx5dv_srq       *out;
} srq;
struct {
struct ibv_wq           *in;
struct mlx5dv_rwq       *out;
} rwq;
struct {
struct ibv_dm		*in;
struct mlx5dv_dm	*out;
} dm;
struct {
struct ibv_ah		*in;
struct mlx5dv_ah	*out;
} ah;
struct {
struct ibv_pd           *in;
struct mlx5dv_pd        *out;
} pd;
struct {
struct mlx5dv_devx_obj  *in;
struct mlx5dv_devx      *out;
} devx;
};
enum mlx5dv_obj_type {
MLX5DV_OBJ_QP   = 1 << 0,
MLX5DV_OBJ_CQ   = 1 << 1,
MLX5DV_OBJ_SRQ  = 1 << 2,
MLX5DV_OBJ_RWQ  = 1 << 3,
MLX5DV_OBJ_DM   = 1 << 4,
MLX5DV_OBJ_AH   = 1 << 5,
MLX5DV_OBJ_PD   = 1 << 6,
MLX5DV_OBJ_DEVX = 1 << 7,
};

VALOR DE RETORNO

0 si tiene éxito o el valor de errno en caso de fallo (que indica el motivo del fallo).

NOTAS


* La información sobre si el timbre es "blueflame" se basa en mlx5dv_qp->bf->size, en caso de 0 no es un BF.
* Las máscaras de compatibilidad (comp_mask) son campos de entrada/salida.
