Si '+' no es el primer carácter de optstring, se trata como una opción normal. Si se requiere el comportamiento POSIXLY_CORRECT en este caso optstring contendrá dos símbolos '+'. Si el primer carácter de optstring es '-', entonces cada elemento de argv que no es una opción se maneja como si fuera el argumento de una opción con código de carácter 1. HISTORIA

getopt()
POSIX.1-2001 y POSIX.2.

En algunas implementaciones antiguas, getopt() se declaraba en <stdio.h>. SUSv1 permitía que la declaración apareciera en <unistd.h> o <stdio.h>. POSIX.1-1996 marcó el uso de <stdio.h> para este propósito como LEGACY. POSIX.1-2001 no requiere que la declaración aparezca en <stdio.h>.

Versiones muy antiguas de glibc se vieron afectadas por una variable de entorno _PID_GNU_nonoption_argv_flags_.

NOTAS

Un programa que escanea múltiples vectores de argumentos, o que vuelve a escanear el mismo vector más de una vez, y desea utilizar las extensiones de GNU como '+' y '-' al principio de optstring, o cambia el valor de POSIXLY_CORRECT entre escaneos, debe reinicializar getopt() restableciendo optind a 0, en lugar del valor tradicional de 1. (Restablecer a 0 fuerza la invocación de una rutina de inicialización interna que vuelve a verificar POSIXLY_CORRECT y verifica las extensiones de GNU en optstring).

Los argumentos de línea de comandos se analizan en orden estricto, lo que significa que una opción que requiere un argumento consumirá el siguiente argumento, independientemente de si ese argumento es el argumento de opción especificado correctamente o simplemente la siguiente opción (en el escenario en que el usuario especifica incorrectamente la línea de comandos). Por ejemplo, si optstring se especifica como "1n:" y el usuario especifica los argumentos de línea de comandos incorrectamente como prog -n -1, la opción -n recibirá el valor optarg de "-1", y la opción -1 se considerará no especificada.

EJEMPLOS

getopt()

El siguiente ejemplo trivial de programa utiliza getopt() para manejar dos opciones de programa: -n, sin valor asociado; y -t val, que espera un valor asociado.

#include <stdio.h>
#include <stdlib.h>
#include <unistd.h>
int
main(int argc, char *argv[])
{
    int flags, opt;
    int nsecs, tfnd;
    nsecs = 0;
    tfnd = 0;
    flags = 0;
    while ((opt = getopt(argc, argv, "nt:")) != -1) {
        switch (opt) {
        case 'n':
            flags = 1;
            break;
        case 't':
            nsecs = atoi(optarg);
            tfnd = 1;
            break;
        default: /* '?' */
            fprintf(stderr, "Uso: %s [-t nsecs] [-n] nombre\n",
                    argv[0]);
            exit(EXIT_FAILURE);
        }
    }
    printf("flags=%d; tfnd=%d; nsecs=%d; optind=%d\n",
           flags, tfnd, nsecs, optind);
    if (optind >= argc) {
        fprintf(stderr, "Se esperaba argumento después de las opciones\n");
        exit(EXIT_FAILURE);
    }
    printf("argumento de nombre = %s\n", argv[optind]);
    /* Otro código omitido */
    exit(EXIT_SUCCESS);
}

getopt_long()

El siguiente ejemplo de programa ilustra el uso de getopt_long() con la mayoría de sus características.

#include <getopt.h>
#include <stdio.h>     /* para printf */
#include <stdlib.h>    /* para exit */
int
main(int argc, char *argv[])
{
    int c;
    int digit_optind = 0;
    while (1) {
        int this_option_optind = optind ? optind : 1;
        int option_index = 0;
        static struct option long_options[] = {
            {"add",     required_argument, 0,  0 },
            {"append",  no_argument,       0,  0 },
            {"delete",  required_argument, 0,  0 },
            {"verbose", no_argument,       0,  0 },
            {"create",  required_argument, 0, 'c'},
            {"file",    required_argument, 0,  0 },
            {0,         0,                 0,  0 }
        };
        c = getopt_long(argc, argv, "abc:d:012",
                        long_options, &option_index);
        if (c == -1)
            break;
        switch (c) {
        case 0:
            printf("opción %s", long_options[option_index].name);
            if (optarg)
                printf(" con argumento %s", optarg);
            printf("\n");
            break;
        case '0':
        case '1':
        case '2':
            if (digit_optind != 0 && digit_optind != this_option_optind)
              printf("dígitos ocurren en dos elementos argv diferentes.\n");
            digit_optind = this_option_optind;
            printf("opción %c\n", c);
            break;
        case 'a':
            printf("opción a\n");
            break;
        case 'b':
            printf("opción b\n");
            break;
        case 'c':
            printf("opción c con valor '%s'\n", optarg);
            break;
        case 'd':
            printf("opción d con valor '%s'\n", optarg);
            break;
        case '?':
            break;
        default:
            printf("?? getopt devolvió el código de carácter 0%o ??\n", c);
        }
    }
    if (optind < argc) {
        printf("elementos ARGV no opciones: ");
        while (optind < argc)
            printf("%s ", argv[optind++]);
        printf("\n");
    }
    exit(EXIT_SUCCESS);
}
