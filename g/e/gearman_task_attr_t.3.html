#include <libgearman/gearman.h>

tipo gearman_work_t
tipo gearman_job_priority_t
gearman_work_t gearman_work(gearman_job_priority_t priority)
gearman_work_t gearman_work_epoch(time_t epoch, gearman_job_priority_t priority)
gearman_work_t gearman_work_background(gearman_job_priority_t priority)

Compilar y enlazar con -lgearman

DESCRIPCIÓN

gearman_work_t describe el trabajo para gearman_execute().

gearman_work() crea un gearman_work_t con una prioridad.

gearman_work_epoch() crea un gearman_work_t que indica a gearman_execute() que ejecute la carga de trabajo en el momento especificado por epoch.

gearman_work_background() crea un gearman_work_t que indica a gearman_execute() que ejecute la carga de trabajo como un trabajo en segundo plano.
