void *gearman_client_do_low(gearman_client_st *client, const char *function_name, const char *unique, const void *workload, size_t workload_size, size_t *result_size, gearman_return_t *ret_ptr)

DESCRIPCIÓN

gearman_client_do() ejecuta una solicitud única al servidor gearmand y espera una respuesta.

gearman_client_do_high() y gearman_client_do_low() son idénticas a gearman_client_do(), solo que establecen la prioridad a alta o baja.

Todas las funciones se bloquearán hasta que se devuelva una respuesta o un error.

VALOR DEVUELTO

gearman_client_do() devuelve un puntero a un valor que el llamador debe liberar. Si se proporciona ret_ptr, los errores que hayan ocurrido se almacenarán en él. Dado que un valor NULL/cero es un valor válido, siempre deberá verificar ret_ptr si le preocupan los errores.

#include <string.h>
#include <stdlib.h>
#include <libgearman/gearman.h>
int main(void)
{
  gearman_client_st *client= gearman_client_create(NULL);
  gearman_return_t ret= gearman_client_add_server(client, "localhost", 0);
  if (gearman_failed(ret))
  {
    return EXIT_FAILURE;
  }
  size_t result_size;
  gearman_return_t rc;
  void *value= gearman_client_do(client, "reverse_function", "unique_value", 
                                 "my string to reverse", strlen("my string to reverse"), 
                                 &result_size, &rc);
  if (gearman_success(rc))
  {
    // Hacer uso de value
  }
  free(value);
  gearman_client_free(client);
  return 0;
}
