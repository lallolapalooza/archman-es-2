LIBGVC(3) Manual de funciones de biblioteca LIBGVC(3)

NOMBRE

libgvc - Biblioteca de contexto Graphviz

SINOPSIS

#include <graphviz/gvc.h>
/* configura un contexto graphviz */
extern GVC_t *gvNEWcontext(const lt_symlist_t *builtins, int demand_loading);
/* configura un contexto graphviz - alternativo */
/* (envuelve las dos funciones anteriores usando la información integrada en libgvc) */
extern GVC_t *gvContext(void);
/* analiza los argumentos de la línea de comandos - mínimamente argv[0] establece el motor de diseño */
extern int gvParseArgs(GVC_t *gvc, int argc, char **argv);
extern graph_t *gvNextInputGraph(GVC_t *gvc);
/* calcula un diseño usando un motor especificado */
extern int gvLayout(GVC_t *gvc, graph_t *g, char *engine);
/* calcula un diseño usando el motor de diseño de los argumentos de la línea de comandos */
extern int gvLayoutJobs(GVC_t *gvc, graph_t *g);
/* comprueba si se ha realizado un diseño */
extern bool gvLayoutDone(graph_t *g);
/* renderiza el diseño en atributos de cadena del gráfico */
extern void attach_attrs(graph_t *g);
/* analiza una cadena html */
extern char *agstrdup_html(char *s);
extern int aghtmlstr(char *s);
/* renderiza el diseño en un formato especificado a un FILE abierto */
extern int gvRender(GVC_t *gvc, graph_t *g, char *format, FILE *out);
/* renderiza el diseño en un formato especificado a un FILE abierto */
extern int gvRenderFilename(GVC_t *gvc, graph_t *g, char *format, char *filename);
/* renderiza el diseño según las opciones -T y -o encontradas por gvParseArgs */
extern int gvRenderJobs(GVC_t *gvc, graph_t *g);
/* limpia las estructuras de datos de diseño - los diseños no son anidables (todavía) */
extern int gvFreeLayout(GVC_t *gvc, graph_t *g);
/* limpia el contexto graphviz */
extern int gvFreeContext(GVC_t *gvc);
/* consulta los plugins disponibles */
/* Ver comentario en gvc.h            */
extern char** gvPluginList(GVC_t *gvc, char* kind, int* cnt, char*);

DESCRIPCIÓN

libgvc proporciona un contexto para aplicaciones que deseen manipular y renderizar gráficos. Proporciona análisis de línea de comandos, código de renderizado común y un mecanismo de plugins para los renderizadores.

VER TAMBIÉN

dot(1), neato(1), cdt(3), cgraph(3)
