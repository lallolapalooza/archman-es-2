EJEMPL

O 
# Buscar pull requests que coincidan con el conjunto de palabras clave "fix" y "bug"
$ gh search prs fix bug
# Buscar pull requests en borrador en el repositorio cli
$ gh search prs --repo=cli/cli --draft
# Buscar pull requests abiertos que solicitan tu revisión
$ gh search prs --review-requested=@me --state=open
# Buscar pull requests fusionados asignados a ti mismo
$ gh search prs --assignee=@me --merged
# Buscar pull requests con numerosas reacciones
$ gh search prs --reactions=">100"
# Buscar pull requests sin la etiqueta "bug"
$ gh search prs -- -label:bug
# Buscar pull requests solo de repositorios no archivados (el valor predeterminado es todos los repositorios)
$ gh search prs --owner github
--archived=fals
