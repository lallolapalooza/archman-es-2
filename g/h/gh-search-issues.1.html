EJEMPL

O 
# Buscar issues que coincidan con el conjunto de palabras clave "readme" y "typo"
$ gh search issues readme typo
# Buscar issues que coincidan con la frase "broken feature"
$ gh search issues "broken feature"
# Buscar issues y pull requests en la organización cli
$ gh search issues --include-prs --owner=cli
# Buscar issues abiertos asignados a ti mismo
$ gh search issues --assignee=@me --state=open
# Buscar issues con numerosos comentarios
$ gh search issues --comments=">100"
# Buscar issues sin la etiqueta "bug"
$ gh search issues -- -label:bug
# Buscar issues solo de repositorios no archivados (por defecto son todos los repositorios)
$ gh search issues --owner github
--archived=fals
