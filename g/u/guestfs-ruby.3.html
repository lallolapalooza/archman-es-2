NOMBRE

guestfs-ruby - Cómo usar libguestfs desde Ruby

SINOPSIS

require 'guestfs'
g = Guestfs::Guestfs.new()
g.add_drive_opts("disk.img",
                 :readonly => true, :format => "raw")
g.launch()

DESCRIPCIÓN

Esta página del manual documenta cómo llamar a libguestfs desde el lenguaje de programación Ruby. Esta página solo documenta las diferencias con la API de C y proporciona algunos ejemplos. Si no estás familiarizado con el uso de libguestfs, también debes leer guestfs(3).

EXCEPCIONES

Los errores de las funciones de libguestfs se mapean a la excepción "Error". Esta tiene un solo parámetro que es el mensaje de error (una cadena).

EJEMPLO 1: CREAR UNA IMAGEN DE DISCO

# Ejemplo que muestra cómo crear una imagen de disco.

require 'guestfs'

output = "disk.img"

g = Guestfs::Guestfs.new()

# Crea una imagen de disco dispersa en formato raw, de 512 MB de tamaño.
g.disk_create (output, "raw", 512 * 1024 * 1024)

# Establece la bandera de rastreo para que podamos ver cada llamada a libguestfs.
g.set_trace(1)

# Adjunta la imagen de disco a libguestfs.
g.add_drive_opts(output, :format => "raw")

# Ejecuta el backend de libguestfs.
g.launch();

# Obtiene la lista de dispositivos. Como solo adjuntamos una unidad
# arriba, esperamos que esta lista contenga un solo
# elemento.
devices = g.list_devices()
if devices.length != 1 then
  raise "error: se esperaba un solo dispositivo de list-devices"
end

# Particiona el disco como una única partición MBR.
g.part_disk(devices[0], "mbr")

# Obtiene la lista de particiones. Esperamos un solo elemento, que
# es la partición que acabamos de crear.
partitions = g.list_partitions()
if partitions.length != 1 then
  raise "error: se esperaba una sola partición de list-partitions"
end

# Crea un sistema de archivos en la partición.
g.mkfs("ext4", partitions[0])

# Ahora monta el sistema de archivos para que podamos añadir archivos.
g.mount(partitions[0], "/")

# Crea algunos archivos y directorios.
g.touch("/empty")
message = "Hola, mundo\n"
g.write("/hello", message)
g.mkdir("/foo")

# Esto sube el archivo local /etc/resolv.conf a
# la imagen de disco.
g.upload("/etc/resolv.conf", "/foo/resolv.conf")

# Como escribimos en el disco y queremos detectar errores de escritura, llamamos a g.shutdown.
# No necesitas hacer esto: g.close lo hará implícitamente.
g.shutdown()

# Ten en cuenta también que los manejadores se cierran automáticamente si son
# recolectados por el recolector de basura. Solo necesitas llamar a close
# si quieres cerrar el manejador de inmediato.
g.close()

EJEMPLO 2: INSPECCIONAR UNA IMAGEN DE DISCO DE MÁQUINA VIRTUAL

# Ejemplo que muestra cómo inspeccionar un disco de máquina virtual.

require 'guestfs'

if ARGV.length == 0
  puts "uso: inspect_vm disk.img"
  exit 1
end
disk = ARGV[0]

g = Guestfs::Guestfs.new()

# Adjunta la imagen de disco en modo de solo lectura a libguestfs.
g.add_drive_opts(disk, :readonly => true)

# Ejecuta el backend de libguestfs.
g.launch()

# Pide a libguestfs que inspeccione los sistemas operativos.
roots = g.inspect_os()
if roots.length == 0
  puts "inspect_vm: no se encontraron sistemas operativos"
  exit 1
end

for root in roots do
  printf("Dispositivo raíz: %s\n", root)

  # Imprime información básica sobre el sistema operativo.
  printf("  Nombre del producto: %s\n", g.inspect_get_product_name(root))
  printf("  Versión:      %d.%d\n",
         g.inspect_get_major_version(root),
         g.inspect_get_minor_version(root))
  printf("  Tipo:         %s\n", g.inspect_get_type(root))
  printf("  Distribución:       %s\n", g.inspect_get_distro(root))

  # Monta los discos, como guestfish -i.
  #
  # Ordena las claves por longitud, la más corta primero, para que terminemos
  # montando los sistemas de archivos en el orden correcto.
  mps = g.inspect_get_mountpoints(root)
  mps = mps.sort {|a,b| a[0].length <=> b[0].length}
  for mp in mps do
    begin
      g.mount_ro(mp[1], mp[0])
    rescue Guestfs::Error => msg
      printf("%s (ignorado)\n", msg)
    end
  end

  # Si el archivo /etc/issue.net existe, imprime hasta 3 líneas.
  filename = "/etc/issue.net"
  if g.is_file filename then
    printf("--- %s ---\n", filename)
    lines = g.head_n(3, filename)
    for line in lines do
      puts line
    end
  end

  # Desmonta todo.
  g.umount_all()
end

Para obtener una lista de errores contra libguestfs, utiliza este enlace: https://bugzilla.redhat.com/buglist.cgi?component=libguestfs&product=Virtualization+Tools

Para reportar un nuevo error contra libguestfs, utiliza este enlace: https://bugzilla.redhat.com/enter_bug.cgi?component=libguestfs&product=Virtualization+Tools

Al reportar un error, por favor proporciona:

  • La versión de libguestfs.
  • De dónde obtuviste libguestfs (por ejemplo, qué distribución de Linux, compilado desde el código fuente, etc.)
  • Describe el error con precisión y proporciona una forma de reproducirlo.
  • Ejecuta libguestfs-test-tool(1) y pega la salida completa y sin editar en el informe de error.
