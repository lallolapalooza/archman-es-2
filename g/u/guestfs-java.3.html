// Ejemplo que muestra

cómo crear una imagen de

disco. 
import java.io.*;
import java.util.Map;
import java.util.HashMap;
import com.redhat.et.libguestfs.*;

public class CreateDisk
{
    static String output = "disk.img";

    public static void main (String[] argv)
    {
        try {
            GuestFS g = new GuestFS ();

            // Crear una imagen de disco dispersa en formato raw, de 512 MB de tamaño.
            RandomAccessFile f = new RandomAccessFile (output, "rw");
            f.setLength (512 * 1024 * 1024);
            f.close ();

            // Establecer la bandera de seguimiento para poder ver cada llamada a libguestfs.
            g.set_trace (true);

            // Adjuntar la imagen de disco a libguestfs.
            @SuppressWarnings("serial") Map<String, Object> optargs =
                new HashMap<String, Object>() {
                {
                    put ("format", "raw");
                    put ("readonly", Boolean.FALSE);
                }
            };
            g.add_drive_opts (output, optargs);

            // Ejecutar el backend de libguestfs.
            g.launch ();

            // Obtener la lista de dispositivos. Como solo añadimos un disco arriba,
            // esperamos que esta lista contenga un solo elemento.
            String[] devices = g.list_devices ();
            if (devices.length != 1)
                throw new Error ("expected a single device from list-devices");

            // Particionar el disco como una sola partición MBR.
            g.part_disk (devices[0], "mbr");

            // Obtener la lista de particiones. Esperamos un solo elemento, que es
            // la partición que acabamos de crear.
            String[] partitions = g.list_partitions ();
            if (partitions.length != 1)
                throw new Error ("expected a single partition from list-partitions");

            // Crear un sistema de archivos en la partición.
            g.mkfs ("ext4", partitions[0]);

            // Ahora montar el sistema de archivos para poder añadir archivos.
            g.mount (partitions[0], "/");

            // Crear algunos archivos y directorios.
            g.touch ("/empty");
            String message = "Hello, world\n";
            g.write ("/hello", message.getBytes());
            g.mkdir ("/foo");

            // Este sube el archivo local /etc/resolv.conf a la imagen de disco.
            g.upload ("/etc/resolv.conf", "/foo/resolv.conf");

            // Como escribimos en el disco y queremos detectar errores de escritura,
            // llamar a g.shutdown. No necesita hacer esto:
            // g.close lo hará implícitamente.
            g.shutdown ();

            // Tenga en cuenta también que los manejadores se cierran automáticamente si son
            // recogidos por el recolector de basura. Solo necesita llamar a close
            // si quiere cerrar el manejador inmediatamente.
            g.close ();
        }
        catch (Exception exn) {
            System.err.println (exn);
            System.exit (1);
        }
    }
}

EJEMPLO 2: INSPECCIONAR UNA IMAGEN DE DISCO DE MÁQUINA

VIRTUAL 
// Ejemplo que muestra cómo inspeccionar un disco de máquina virtual.

import java.util.ArrayList;
import java.util.Collections;
import java.util.Comparator;
import java.util.HashMap;
import java.util.List;
import java.util.Map;
import com.redhat.et.libguestfs.*;

public class InspectVM
{
    static final Comparator<String> COMPARE_KEYS_LEN =
        new Comparator<String>() {
        public int compare (String k1, String k2) {
            return k1.length() - k2.length();
        }
    };

    public static void main (String[] argv)
    {
        try {
            if (argv.length != 1)
                throw new Error ("usage: InspectVM disk.img");

            String disk = argv[0];

            GuestFS g = new GuestFS ();

            // Adjuntar la imagen de disco de solo lectura a libguestfs.
            @SuppressWarnings("serial") Map<String, Object> optargs =
                new HashMap<String, Object>() {
                {
                    //put ("format", "raw");
                    put ("readonly", Boolean.TRUE);
                }
            };

            g.add_drive_opts (disk, optargs);

            // Ejecutar el backend de libguestfs.
            g.launch ();

            // Pedir a libguestfs que inspeccione los sistemas operativos.
            String roots[] = g.inspect_os ();
            if (roots.length == 0)
                throw new Error ("inspect_vm: no operating systems found");

            for (String root : roots) {
                System.out.println ("Root device: " + root);

                // Imprimir información básica sobre el sistema operativo.
                System.out.println ("  Product name: " +
                                    g.inspect_get_product_name (root));
                System.out.println ("  Version:      " +
                                    g.inspect_get_major_version (root) +
                                    "." +
                                    g.inspect_get_minor_version (root));
                System.out.println ("  Type:         " +
                                    g.inspect_get_type (root));
                System.out.println ("  Distro:       " +
                                    g.inspect_get_distro (root));

                // Montar los discos, como guestfish -i.
                //
                // Ordenar las claves por longitud, de la más corta a la más
                // larga, para que montemos los sistemas de archivos en el
                // orden correcto.
                Map<String,String> mps = g.inspect_get_mountpoints (root);
                List<String> mps_keys = new ArrayList<String> (mps.keySet ());
                Collections.sort (mps_keys, COMPARE_KEYS_LEN);

                for (String mp : mps_keys) {
                    String dev = mps.get (mp);
                    try {
                        g.mount_ro (dev, mp);
                    }
                    catch (Exception exn) {
                        System.err.println (exn + " (ignored)");
                    }
                }

                // Si el archivo /etc/issue.net existe, imprimir hasta 3 líneas.
                String filename = "/etc/issue.net";
                if (g.is_file (filename)) {
                    System.out.println ("--- " + filename + " ---");
                    String[] lines = g.head_n (3, filename);
                    for (String line : lines)
                        System.out.println (line);
                }

                // Desmontar todo.
                g.umount_all ();
            }
        }
        catch (Exception exn) {
            System.err.println (exn);
            System.exit (1);
        }
    }
}
