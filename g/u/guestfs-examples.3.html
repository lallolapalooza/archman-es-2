EJEMPLO: CREAR UNA IMAGEN DE

DISCO 
/* Ejemplo que muestra cómo crear una imagen de disco. */

#include <stdio.h>
#include <stdlib.h>
#include <string.h>
#include <fcntl.h>
#include <unistd.h>
#include <guestfs.h>

int
main (int argc, char *argv[])
{
  guestfs_h *g;
  size_t i;

  g = guestfs_create ();
  if (g == NULL) {
    perror ("failed to create libguestfs handle");
    exit (EXIT_FAILURE);
  }

  /* Establece la bandera de traza para que podamos ver cada llamada a libguestfs. */
  guestfs_set_trace (g, 1);

  /* Crea una imagen de disco dispersa en formato raw, de 512 MB de tamaño. */
  if (guestfs_disk_create (g, "disk.img", "raw", UINT64_C(512)*1024*1024,
                           -1) == -1)
    exit (EXIT_FAILURE);

  /* Añade la imagen de disco a libguestfs. */
  if (guestfs_add_drive_opts (g, "disk.img",
                             GUESTFS_ADD_DRIVE_OPTS_FORMAT, "raw", /* formato raw */
                             GUESTFS_ADD_DRIVE_OPTS_READONLY, 0, /* para escritura */
                             -1) /* esto marca el final de los argumentos opcionales */
      == -1)
    exit (EXIT_FAILURE);

  /* Ejecuta el backend de libguestfs. */
  if (guestfs_launch (g) == -1)
    exit (EXIT_FAILURE);

  /* Obtiene la lista de dispositivos.  Como solo añadimos un disco
   * arriba, esperamos que esta lista contenga un solo
   * elemento.
   */
  char **devices = guestfs_list_devices (g);
  if (devices == NULL)
    exit (EXIT_FAILURE);
  if (devices[0] == NULL || devices[1] != NULL) {
    fprintf (stderr, "error: expected a single device from list-devices\n");
    exit (EXIT_FAILURE);
  }

  /* Particiona el disco como una sola partición MBR. */
  if (guestfs_part_disk (g, devices[0], "mbr") == -1)
    exit (EXIT_FAILURE);

  /* Obtiene la lista de particiones.  Esperamos un solo elemento, que
   * es la partición que acabamos de crear.
   */
  char **partitions = guestfs_list_partitions (g);
  if (partitions == NULL)
    exit (EXIT_FAILURE);
  if (partitions[0] == NULL || partitions[1] != NULL) {
    fprintf (stderr, "error: expected a single partition from list-partitions\n");
    exit (EXIT_FAILURE);
  }

  /* Crea un sistema de archivos en la partición. */
  if (guestfs_mkfs (g, "ext4", partitions[0]) == -1)
    exit (EXIT_FAILURE);

  /* Ahora monta el sistema de archivos para que podamos añadir archivos. */
  if (guestfs_mount (g, partitions[0], "/") == -1)
    exit (EXIT_FAILURE);

  /* Crea algunos archivos y directorios. */
  if (guestfs_touch (g, "/empty") == -1)
    exit (EXIT_FAILURE);
  const char *message = "Hello, world\n";
  if (guestfs_write (g, "/hello", message, strlen (message)) == -1)
    exit (EXIT_FAILURE);
  if (guestfs_mkdir (g, "/foo") == -1)
    exit (EXIT_FAILURE);

  /* Este sube el archivo local /etc/resolv.conf a
   * la imagen de disco.
   */
  if (guestfs_upload (g, "/etc/resolv.conf", "/foo/resolv.conf") == -1)
    exit (EXIT_FAILURE);

  /* Como escribimos en el disco y queremos detectar errores de
   * escritura, llama a guestfs_shutdown.  No necesita hacer esto:
   * guestfs_close lo hará implícitamente.
   */
  if (guestfs_shutdown (g) == -1)
    exit (EXIT_FAILURE);

  guestfs_close (g);

  /* Libera las listas. */
  for (i = 0; devices[i] != NULL; ++i)
    free (devices[i]);
  free (devices);
  for (i = 0; partitions[i] != NULL; ++i)
    free (partitions[i]);
  free (partitions);

  exit (EXIT_SUCCESS);
}

EJEMPLO: INSPECCIONAR UNA IMAGEN DE DISCO DE MÁQUINA

VIRTUAL 
/* Inspecciona una imagen de disco y muestra los sistemas operativos que puede contener. */

#include <stdio.h>
#include <stdlib.h>
#include <string.h>
#include <guestfs.h>

static int
compare_keys_len (const void *p1, const void *p2)
{
  const char *key1 = * (char * const *) p1;
  const char *key2 = * (char * const *) p2;
  return strlen (key1) - strlen (key2);
}

static size_t
count_strings (char *const *argv)
{
  size_t c;

  for (c = 0; argv[c]; ++c)
    ;
  return c;
}

int
main (int argc, char *argv[])
{
  guestfs_h *g;
  const char *disk;
  char **roots, *root, *str, **mountpoints, **lines;
  size_t i, j;

  if (argc != 2) {
    fprintf (stderr, "usage: inspect_vm disk.img\n");
    exit (EXIT_FAILURE);
  }
  disk = argv[1];

  g = guestfs_create ();
  if (g == NULL) {
    perror ("failed to create libguestfs handle");
    exit (EXIT_FAILURE);
  }

  /* Adjunta la imagen de disco de solo lectura a libguestfs. */
  if (guestfs_add_drive_opts (g, disk,
                             /* GUESTFS_ADD_DRIVE_OPTS_FORMAT, "raw", */
                             GUESTFS_ADD_DRIVE_OPTS_READONLY, 1,
                             -1) /* esto marca el final de los argumentos opcionales */
      == -1)
    exit (EXIT_FAILURE);

  /* Ejecuta el backend de libguestfs. */
  if (guestfs_launch (g) == -1)
    exit (EXIT_FAILURE);

  /* Pide a libguestfs que inspeccione los sistemas operativos. */
  roots = guestfs_inspect_os (g);
  if (roots == NULL)
    exit (EXIT_FAILURE);
  if (roots[0] == NULL) {
    fprintf (stderr, "inspect_vm: no operating systems found\n");
    exit (EXIT_FAILURE);
  }

  for (j = 0; roots[j] != NULL; ++j) {
    root = roots[j];

    printf ("Root device: %s\n", root);

    /* Imprime información básica sobre el sistema operativo. */
    str = guestfs_inspect_get_product_name (g, root);
    if (str)
      printf ("  Product name: %s\n", str);
    free (str);

    printf ("  Version:      %d.%d\n",
            guestfs_inspect_get_major_version (g, root),
            guestfs_inspect_get_minor_version (g, root));

    str = guestfs_inspect_get_type (g, root);
    if (str)
      printf ("  Type:         %s\n", str);
    free (str);
    str = guestfs_inspect_get_distro (g, root);
    if (str)
      printf ("  Distro:       %s\n", str);
    free (str);

    /* Monta los discos, como guestfish -i.
     *
     * Ordena las claves por longitud, la más corta primero, para que
     * terminemos montando los sistemas de archivos en el orden
     * correcto.
     */
    mountpoints = guestfs_inspect_get_mountpoints (g, root);
    if (mountpoints == NULL)
      exit (EXIT_FAILURE);

    qsort (mountpoints, count_strings (mountpoints) / 2, 2 * sizeof (char *),
           compare_keys_len);
    for (i = 0; mountpoints[i] != NULL; i += 2) {
      /* Ignora los fallos de esta llamada, ya que pueden aparecer
       * entradas falsas en el /etc/fstab del huésped.
       */
      guestfs_mount_ro (g, mountpoints[i+1], mountpoints[i]);
      free (mountpoints[i]);
      free (mountpoints[i+1]);
    }
    free (mountpoints);

    /* Si el archivo /etc/issue.net existe, imprime hasta 3 líneas. */
    if (guestfs_is_file (g, "/etc/issue.net") > 0) {
      printf ("--- /etc/issue.net ---\n");
      lines = guestfs_head_n (g, 3, "/etc/issue.net");
      if (lines == NULL)
        exit (EXIT_FAILURE);
      for (i = 0; lines[i] != NULL; ++i) {
        printf ("%s\n", lines[i]);
        free (lines[i]);
      }
      free (lines);
    }

    /* Desmonta todo. */
    if (guestfs_umount_all (g) == -1)
      exit (EXIT_FAILURE);

    free (root);
  }
  free (roots);

  guestfs_close (g);

  exit (EXIT_SUCCESS);
}

EJEMPLO: HABILITAR LA DEPURACIÓN Y EL

REGISTRO 
/* Ejemplo que muestra cómo habilitar la depuración y capturarla en cualquier
 * sistema de registro personalizado (syslog en este ejemplo, pero se podría usar
 * cualquiera).  Note que esto usa la API de eventos, que también está disponible en
 * enlaces de lenguajes no C.
 */

#include <stdio.h>
#include <stdlib.h>
#include <string.h>
#include <unistd.h>
#include <syslog.h>
#include <guestfs.h>

static void message_callback (guestfs_h *g, void *opaque, uint64_t event, int event_handle, int flags, const char *buf, size_t buf_len, const uint64_t *array, size_t array_len);

/* Eventos en los que estamos interesados.  Esta máscara de bits cubre todos los
 * mensajes de traza y depuración.
 */
static const uint64_t event_bitmask =
  GUESTFS_EVENT_LIBRARY |
  GUESTFS_EVENT_WARNING |
  GUESTFS_EVENT_APPLIANCE |
  GUESTFS_EVENT_TRACE;

int
main (int argc, char *argv[])
{
  guestfs_h *g;

  g = guestfs_create ();
  if (g == NULL) {
    perror ("failed to create libguestfs handle");
    exit (EXIT_FAILURE);
  }

  /* Por defecto, la información de depuración se envnted en stderr. Para
   * capturarlo en otro lugar, debe configurar un manejador de eventos
   * que se llame de vuelta cuando se generen mensajes de depuración. Para hacer
   * esto, use la API de eventos.
   *
   * Para más información, vea EVENTS en guestfs(3).
   */
  if (guestfs_set_event_callback (g, message_callback,
                                  event_bitmask, 0, NULL) == -1)
    exit (EXIT_FAILURE);

  /* Así se habilita la depuración:
   *
   * Establecer la bandera 'trace' en el manejador significa que cada llamada a
   * libguestfs se registra (nombre, parámetros, retorno). Esta bandera es útil
   * para ver cómo un programa está usando libguestfs.
   *
   * Establecer la bandera 'verbose' habilita una gran cantidad de depuración
   * adicional en todo el sistema. Esto es útil si hay un error de libguestfs
   * que no entiende.
   *
   * Note que debe establecer las banderas temprano después de crear el
   * manejador. En particular, si establece la bandera verbose después de launch
   * entonces no verá todos los mensajes.
   *
   * Para más información vea:
   * http://libguestfs.org/guestfs-faq.1.html#debugging-libguestfs
   *
   * Los mensajes de error lanzados por las APIs *no* son información de depuración,
   * y no se ven afectados por nada de esto. Puede que tenga que registrarlos
   * por separado.
   */
  guestfs_set_trace (g, 1);
  guestfs_set_verbose (g, 1);

  /* Hacer algunas operaciones que generarán muchos mensajes de traza y depuración.
   */
  if (guestfs_add_drive (g, "/dev/null") == -1)
    exit (EXIT_FAILURE);

  printf ("Este programa no tiene salida. "
          "Eche un vistazo en su archivo de registro del sistema,\n"
          "p. ej. /var/log/messages.\n");

  if (guestfs_launch (g) == -1)
    exit (EXIT_FAILURE);

  guestfs_close (g);

  exit (EXIT_SUCCESS);
}

/* Esta función es llamada de vuelta por libguestfs siempre que se genera un
 * mensaje de traza o depuración.
 *
 * Para las clases de eventos que hemos registrado arriba, 'array' y
 * 'array_len' no serán significativos. Solo 'buf' y 'buf_len' serán
 * interesantes y estos contendrán el mensaje de traza o depuración.
 *
 * Este ejemplo simplemente redirige estos mensajes a syslog, pero
 * obviamente podría hacer algo más avanzado aquí.
 */
static void
message_callback (guestfs_h *g, void *opaque,
                  uint64_t event, int event_handle,
                  int flags,
                  const char *buf, size_t buf_len,
                  const uint64_t *array, size_t array_len)
{
  const int priority = LOG_USER|LOG_INFO;
  char *event_name, *msg;

  if (buf_len > 0) {
    event_name = guestfs_event_to_string (event);
    msg = strndup (buf, buf_len);
    syslog (priority, "[%s] %s", event_name, msg);
    free (msg);
    free (event_name);
  }
}

EJEMPLO: MOSTRAR EL ICONO DEL SISTEMA OPERATIVO DE UN INVITADO

/* Este ejemplo inspecciona un invitado usando la inspección de libguestfs (vea
 * "INSPECTION" en guestfs(3)), y si es posible muestra un
 * icono o logotipo representativo para el sistema operativo del invitado.
 */

#include <stdio.h>
#include <stdlib.h>
#include <string.h>
#include <guestfs.h>

static int
compare_keys_len (const void *p1, const void *p2)
{
  const char *key1 = * (char * const *) p1;
  const char *key2 = * (char * const *) p2;
  return strlen (key1) - strlen (key2);
}

static size_t
count_strings (char *const *argv)
{
  size_t c;

  for (c = 0; argv[c]; ++c)
    ;
  return c;
}

int
main (int argc, char *argv[])
{
  guestfs_h *g;
  const char *disk;
  char **roots, *root, **mountpoints, *icon;
  size_t i, j, icon_size;
  FILE *fp;

  if (argc != 2) {
    fprintf (stderr, "uso: display-icon disk.img\n");
    exit (EXIT_FAILURE);
  }
  disk = argv[1];

  g = guestfs_create ();
  if (g == NULL) {
    perror ("falló al crear el manejador de libguestfs");
    exit (EXIT_FAILURE);
  }

  /* Adjuntar la imagen de disco de solo lectura a libguestfs. */
  if (guestfs_add_drive_opts (g, disk,
                             /* GUESTFS_ADD_DRIVE_OPTS_FORMAT, "raw", */
                             GUESTFS_ADD_DRIVE_OPTS_READONLY, 1,
                             -1) /* esto marca el final de los argumentos opcionales */
      == -1)
    exit (EXIT_FAILURE);

  /* Ejecutar el backend de libguestfs. */
  if (guestfs_launch (g) == -1)
    exit (EXIT_FAILURE);

  /* Pedir a libguestfs que inspeccione los sistemas operativos. */
  roots = guestfs_inspect_os (g);
  if (roots == NULL)
    exit (EXIT_FAILURE);
  if (roots[0] == NULL) {
    fprintf (stderr, "display-icon: no se encontraron sistemas operativos\n");
    exit (EXIT_FAILURE);
  }

  for (j = 0; roots[j] != NULL; ++j) {
    root = roots[j];

    /* Montar los discos, como guestfish -i.
     *
     * Ordenar las claves por longitud, la más corta primero, para que
     * terminemos montando los sistemas de archivos en el orden correcto.
     */
    mountpoints = guestfs_inspect_get_mountpoints (g, root);
    if (mountpoints == NULL)
      exit (EXIT_FAILURE);

    qsort (mountpoints, count_strings (mountpoints) / 2, 2 * sizeof (char *),
           compare_keys_len);
    for (i = 0; mountpoints[i] != NULL; i += 2) {
      /* Ignorar los fallos de esta llamada, ya que pueden aparecer
       * entradas falsas en el /etc/fstab del invitado.
       */
      guestfs_mount_ro (g, mountpoints[i+1], mountpoints[i]);
      free (mountpoints[i]);
      free (mountpoints[i+1]);
    }
    free (mountpoints);

    /* Obtener el icono.
     * Esta función devuelve un búfer ('icon'). Normalmente es un archivo
     * png, devuelto como una cadena, pero también puede ser un búfer de
     * longitud cero que tiene un significado especial, o NULL que significa
     * que hubo un error.
     */
    icon = guestfs_inspect_get_icon (g, root, &icon_size, -1);
    if (!icon)                  /* error real de libguestfs */
      exit (EXIT_FAILURE);
    if (icon_size == 0)         /* no hay icono disponible */
      fprintf (stderr, "%s: %s: no hay icono disponible para este sistema operativo\n",
               disk, root);
    else {
      /* Mostrar el icono. */
      fp = popen ("display -", "w");
      if (fp == NULL) {
        perror ("display");
        exit (EXIT_FAILURE);
      }
      if (fwrite (icon, 1, icon_size, fp) != icon_size) {
        perror ("write");
        exit (EXIT_FAILURE);
      }
      if (pclose (fp) == -1) {
        perror ("pclose");
        exit (EXIT_FAILURE);
      }
    }
    free (icon);

    /* Desmontar todo. */
    if (guestfs_umount_all (g) == -1)
      exit (EXIT_FAILURE);

    free (root);
  }
  free (roots);

  guestfs_close (g);

  exit (EXIT_SUCCESS);
}

EJEMPLO: LA API DE AUTENTICACIÓN DE LIBVIRT

/* Ejemplo de uso de la API de autenticación basada en eventos de libvirt.
 *
 * Vea "LIBVIRT AUTHENTICATION" en guestfs(3).
 */

#include <stdio.h>
#include <stdlib.h>
#include <string.h>
#include <unistd.h>

#include <guestfs.h>

static void
usage (void)
{
  fprintf (stderr,
          "Uso:\n"
          "\n"
          "  libvirt-auth URI domain\n"
          "\n"
          "donde:\n"
          "\n"
          "  URI     es la URI de libvirt, p. ej. qemu+libssh2://USER@localhost/system\n"
          "  domain  es el nombre del invitado\n"
          "\n"
          "Ejemplo:\n"
          "\n"
          "  libvirt-auth 'qemu+libssh2://USER@localhost/system' 'foo'\n"
          "\n"
          "se conectará (solo lectura) a la URI de libvirt dada y abrirá el invitado\n"
          "llamado 'foo' y listará alguna información sobre sus sistemas de archivos.\n"
          "\n"
          "El punto importante de este ejemplo es que cualquier autenticación de libvirt\n"
          "necesaria para conectarse al servidor debe realizarse.\n"
          "\n");
}

static void auth_callback (guestfs_h *g, void *opaque, uint64_t event, int event_handle, int flags, const char *buf, size_t buf_len, const uint64_t *array, size_t array_len);

int
main (int argc, char *argv[])
{
  const char *uri, *dom;
  guestfs_h *g;
  const char *creds[] = { "authname", "passphrase",
                          "echoprompt", "noechoprompt", NULL };
  int r, eh;
  char **filesystems;
  size_t i;

  if (argc != 3) {
    usage ();
    exit (EXIT_FAILURE);
  }
  uri = argv[1];
  dom = argv[2];

  g = guestfs_create ();
  if (!g)
    exit (EXIT_FAILURE);

  r = guestfs_set_libvirt_supported_credentials (g, (char **) creds);
  if (r == -1)
    exit (EXIT_FAILURE);

  /* Configurar el manejador de eventos. */
  eh = guestfs_set_event_callback (g, auth_callback,
                                   GUESTFS_EVENT_LIBVIRT_AUTH, 0, NULL);
  if (eh == -1)
    exit (EXIT_FAILURE);

  /* Añadir el dominio con nombre. */
  r = guestfs_add_domain (g, dom,
                          GUESTFS_ADD_DOMAIN_LIBVIRTURI, uri,
                          -1);
  if (r == -1)
    exit (EXIT_FAILURE);

  /* Lanzar y hacer una inspección simple. */
  r = guestfs_launch (g);
  if (r == -1)
    exit (EXIT_FAILURE);

  filesystems = guestfs_list_filesystems (g);
  if (filesystems == NULL)
    exit (EXIT_FAILURE);

  for (i = 0; filesystems[i] != NULL; i += 2) {
    printf ("%s:%s es un sistema de archivos %s\n",
            dom, filesystems[i], filesystems[i+1]);
    free (filesystems[i]);
    free (filesystems[i+1]);
  }
  free (filesystems);

  exit (EXIT_SUCCESS);
}

static void
auth_callback (guestfs_h *g,
               void *opaque,
               uint64_t event,
               int event_handle,
               int flags,
               const char *buf, size_t buf_len,
               const uint64_t *array, size_t array_len)
{
  char **creds;
  size_t i;
  char *prompt;
  char *reply = NULL;
  size_t allocsize = 0;
  char *pass;
  ssize_t len;
  int r;

  printf ("libvirt-auth.c: se requiere autenticación para la URI de libvirt '%s'\n\n",
          buf);

  /* Preguntar a libguestfs qué credenciales está exigiendo libvirt. */
  creds = guestfs_get_libvirt_requested_credentials (g);
  if (creds == NULL)
    exit (EXIT_FAILURE);

  /* Ahora preguntar al usuario por las respuestas. */
  for (i = 0; creds[i] != NULL; ++i)
  {
    printf ("libvirt-auth.c: credencial '%s'\n", creds[i]);

    if (strcmp (creds[i], "authname") == 0 ||
        strcmp (creds[i], "echoprompt") == 0) {
      prompt = guestfs_get_libvirt_requested_credential_prompt (g, i);
      if (prompt && strcmp (prompt, "") != 0)
        printf ("%s: ", prompt);
      free (prompt);

      len = getline (&reply, &allocsize, stdin);
      if (len == -1) {
        perror ("getline");
        exit (EXIT_FAILURE);
      }
      if (len > 0 && reply[len-1] == '\n')
        reply[--len] = '\0';

      r = guestfs_set_libvirt_requested_credential (g, i, reply, len);
      if (r == -1)
        exit (EXIT_FAILURE);
    } else if (strcmp (creds[i], "passphrase") == 0 ||
               strcmp (creds[i], "noechoprompt") == 0) {
      prompt = guestfs_get_libvirt_requested_credential_prompt (g, i);
      if (prompt && strcmp (prompt, "") != 0)
        printf ("%s: ", prompt);
      free (prompt);

      pass = getpass ("");
      if (pass == NULL) {
        perror ("getpass");
        exit (EXIT_FAILURE);
      }
      len = strlen (pass);

      r = guestfs_set_libvirt_requested_credential (g, i, pass, len);
      if (r == -1)
        exit (EXIT_FAILURE);
    }

    free (creds[i]);
  }

  free (reply);
  free (creds);
}

EJEMPLO: LA API MOUNT-LOCAL

/* Demuestra el uso de la API 'mount-local'.
 *
 * Ejecuta este programa como (por ejemplo) mount-local /tmp/test.img. Nota que
 * '/tmp/test.img' se crea o se sobrescribe. Sigue las instrucciones
 * en pantalla.
 *
 * Ver "MOUNT LOCAL" en guestfs(3).
 */

#include <stdio.h>
#include <stdlib.h>
#include <string.h>
#include <unistd.h>
#include <fcntl.h>
#include <sys/wait.h>

#include <guestfs.h>

/* Define una lista de opciones de montaje del sistema de archivos (utilizadas en el lado de libguestfs,
 * nada que ver con FUSE). Se puede usar una cadena vacía aquí en su lugar.
 */
#define MOUNT_OPTIONS "acl,user_xattr"

/* Tamaño del disco (megabytes). */
#define SIZE_MB 512

static void
usage (void)
{
  fprintf (stderr,
           "Uso: mount-local disco.img\n"
           "\n"
           "NOTA: disco.img se creará o se sobrescribirá.\n"
           "\n");
}

int
main (int argc, char *argv[])
{
  guestfs_h *g;
  int r;
  char tempdir[] = "/tmp/mlXXXXXX";
  pid_t pid;
  char *shell, *p;

  if (argc != 2) {
    usage ();
    exit (EXIT_FAILURE);
  }

  if (argv[1][0] == '-') {
    usage ();
    exit (EXIT_FAILURE);
  }

  printf ("\n"
          "Este es el programa de demostración 'mount-local'. Sigue las\n"
          "instrucciones en pantalla.\n"
          "\n"
          "Creando y formateando la imagen de disco, espera un momento ...\n");
  fflush (stdout);

  /* Handle de Guestfs. */
  g = guestfs_create ();
  if (g == NULL) {
    perror ("no se pudo crear el handle de libguestfs");
    exit (EXIT_FAILURE);
  }

  /* Crea la imagen de disco de salida: raw dispersa. */
  if (guestfs_disk_create (g, argv[1], "raw", SIZE_MB * 1024 * 1024, -1) == -1)
    exit (EXIT_FAILURE);

  /* Crea la imagen de disco y la formatea con una partición y un sistema de archivos. */
  if (guestfs_add_drive_opts (g, argv[1],
                              GUESTFS_ADD_DRIVE_OPTS_FORMAT, "raw",
                              -1) == -1)
    exit (EXIT_FAILURE);

  if (guestfs_launch (g) == -1)
    exit (EXIT_FAILURE);

  if (guestfs_part_disk (g, "/dev/sda", "mbr") == -1)
    exit (EXIT_FAILURE);

  if (guestfs_mkfs (g, "ext2", "/dev/sda1") == -1)
    exit (EXIT_FAILURE);

  /* Monta el sistema de archivos vacío. */
  if (guestfs_mount_options (g, MOUNT_OPTIONS, "/dev/sda1", "/") == -1)
    exit (EXIT_FAILURE);

  /* Crea un archivo en el nuevo sistema de archivos. */
  if (guestfs_touch (g, "/PON_AQUI_ARCHIVOS_Y_DIRECTORIOS") == -1)
    exit (EXIT_FAILURE);

  /* Crea un directorio de montaje temporal. */
  if (mkdtemp (tempdir) == NULL) {
    perror ("mkdtemp");
    exit (EXIT_FAILURE);
  }

  /* Monta el sistema de archivos. */
  if (guestfs_mount_local (g, tempdir, -1) == -1)
    exit (EXIT_FAILURE);

  /* Crea el shell para el usuario. */
  pid = fork ();
  if (pid == -1) {
    perror ("fork");
    exit (EXIT_FAILURE);
  }

  if (pid == 0) {               /* Hijo. */
    if (chdir (tempdir) == -1) {
      perror (tempdir);
      _exit (EXIT_FAILURE);
    }

    printf ("\n"
            "El *directorio actual* es un sistema de archivos FUSE respaldado por el disco\n"
            "imagen que es administrada por libguestfs. Cualquier archivo o directorio\n"
            "que copies aquí (hasta %d MB) se guardará en el disco\n"
            "imagen. También puedes eliminar archivos, crear ciertos archivos especiales\n"
            "y así sucesivamente.\n"
            "\n"
            "Cuando hayas terminado de agregar archivos, presiona ^D o escribe 'exit' para\n"
            "salir del shell y regresar al programa mount-local.\n"
            "\n",
            SIZE_MB);

    shell = getenv ("SHELL");
    if (!shell)
      r = system ("/bin/sh");
    else {
      /* Establece un prompt mágico. Solo sabemos cómo hacerlo para bash. */
      p = strrchr (shell, '/');
      if (p && strcmp (p+1, "bash") == 0) {
        const size_t len = 64 + strlen (shell);
        char *buf;

        buf = malloc (len);
        if (buf == NULL) {
          perror ("malloc");
          _exit (EXIT_FAILURE);
        }
        snprintf (buf, len, "PS1='mount-local-shell> ' %s --norc -i", shell);
        r = system (buf);
        free (buf);
      } else
        r = system (shell);
    }
    if (r == -1) {
      fprintf (stderr, "error: no se pudo ejecutar el sub-shell (%s) "
               "(¿está $SHELL configurado correctamente?)\n",
               shell);
      // FALLTHROUGH
    }

    if (chdir ("/") == -1)
      perror ("chdir: /");
    guestfs_umount_local (g, GUESTFS_UMOUNT_LOCAL_RETRY, 1, -1);
    _exit (EXIT_SUCCESS);
  }

  /* Nota que *no* estamos esperando al hijo todavía. Queremos
   * ejecutar el código FUSE en paralelo con el subshell.
   */

  /* Vamos a ocultar los errores de libguestfs aquí, pero en un programa real
   * probablemente querrías registrarlos en algún lugar.
   */
  guestfs_push_error_handler (g, NULL, NULL);

  /* Ahora ejecuta el hilo FUSE. */
  if (guestfs_mount_local_run (g) == -1)
    exit (EXIT_FAILURE);

  guestfs_pop_error_handler (g);

  waitpid (pid, NULL, 0);

  /* Cierra el handle explícitamente para que los errores de escritura puedan ser detectados. */
  if (guestfs_shutdown (g) == -1)
    exit (EXIT_FAILURE);

  guestfs_close (g);

  printf ("\n"
          "Cualquier archivo o directorio que hayas copiado se ha guardado en\n"
          "la imagen de disco llamada '%s'.\n"
          "\n"
          "Intenta abrir la imagen de disco con guestfish para ver esos archivos:\n"
          "\n"
          "  guestfish -a %s -m /dev/sda1\n"
          "\n",
          argv[1], argv[1]);

  exit (EXIT_SUCCESS);
}

/* Copia un directorio de una máquina virtual libvirt a otra.
 *
 * Este es un ejemplo más sustancial de uso de la API de libguestfs,
 * que demuestra entre otras cosas:
 *
 * - uso de múltiples manejadores con hilos
 * - carga y descarga (usando una tubería entre manejadores)
 * - inspección
 */

#include <stdio.h>
#include <stdlib.h>
#include <stdint.h>
#include <inttypes.h>
#include <string.h>
#include <unistd.h>
#include <fcntl.h>
#include <errno.h>
#include <sys/time.h>

#include <pthread.h>

#include <guestfs.h>

struct threaddata {
  const char *src;
  const char *srcdir;
  int fd;
  pthread_t mainthread;
};

static void *start_srcthread (void *);
static int open_guest (guestfs_h *g, const char *dom, int readonly);
static int64_t timeval_diff (const struct timeval *x, const struct timeval *y);
static int compare_keys_len (const void *p1, const void *p2);
static size_t count_strings (char *const *argv);

static void
usage (void)
{
  fprintf (stderr,
          "Uso: copy-over origen srcdir destino destdir\n"
          "\n"
          "  origen  : el dominio de origen (un nombre de máquina virtual libvirt)\n"
          "  srcdir  : el directorio para copiar desde la máquina virtual de origen\n"
          "  destino : el dominio de destino (un nombre de máquina virtual libvirt)\n"
          "  destdir : el directorio de destino (debe existir en el destino)\n"
          "\n"
          "ej: copy-over Src /home/rjones Dest /tmp/dir\n"
          "copiaría /home/rjones de Src a /tmp/dir en Dest\n"
          "\n"
          "La máquina virtual de destino no puede estar en ejecución.\n");
}

int
main (int argc, char *argv[])
{
  const char *src, *srcdir, *dest, *destdir;
  guestfs_h *destg;
  int fd[2];
  pthread_t srcthread;
  struct threaddata threaddata;
  int err;
  char fdname[128];
  struct timeval start_t, end_t;
  int64_t ms;

  if (argc != 5) {
    usage ();
    exit (EXIT_FAILURE);
  }

  src = argv[1];
  srcdir = argv[2];
  dest = argv[3];
  destdir = argv[4];

  /* En lugar de descargar a disco local y cargar, vamos a
   * conectar la descarga de origen y la carga de destino usando una
   * tubería. Crear esa tubería.
   */
  if (pipe (fd) == -1) {
    perror ("pipe");
    exit (EXIT_FAILURE);
  }

  /* No queremos que la tubería se pase a subprocesos. */
  if (fcntl (fd[0], F_SETFD, FD_CLOEXEC) == -1 ||
      fcntl (fd[1], F_SETFD, FD_CLOEXEC) == -1) {
    perror ("fcntl");
    exit (EXIT_FAILURE);
  }

  /* La API de libguestfs es síncrona, por lo que si queremos usar dos
   * manejadores concurrentemente, entonces debemos tener dos hilos. En este
   * caso, el hilo principal (este) está manejando el dominio de destino
   * (cargando), y creamos un hilo más para manejar el dominio de origen
   * (descargando).
   */
  threaddata.src = src;
  threaddata.srcdir = srcdir;
  threaddata.fd = fd[1];
  threaddata.mainthread = pthread_self ();
  err = pthread_create (&srcthread, NULL, start_srcthread, &threaddata);
  if (err != 0) {
    fprintf (stderr, "pthread_create: %s\n", strerror (err));
    exit (EXIT_FAILURE);
  }

  /* Abre el dominio de destino. */
  destg = guestfs_create ();
  if (!destg) {
    perror ("fallo al crear el handle de libguestfs");
    pthread_cancel (srcthread);
    exit (EXIT_FAILURE);
  }
  if (open_guest (destg, dest, 0) == -1) {
    pthread_cancel (srcthread);
    exit (EXIT_FAILURE);
  }

  gettimeofday (&start_t, NULL);

  /* Comienza la carga. */
  snprintf (fdname, sizeof fdname, "/dev/fd/%d", fd[0]);
  if (guestfs_tar_in (destg, fdname, destdir) == -1) {
    pthread_cancel (srcthread);
    exit (EXIT_FAILURE);
  }

  /* Cierra nuestro extremo de la tubería. El otro hilo cerrará
   * el otro extremo de la tubería.
   */
  close (fd[0]);

  /* Espera a que el otro hilo termine. */
  err = pthread_join (srcthread, NULL);
  if (err != 0) {
    fprintf (stderr, "pthread_join: %s\n", strerror (err));
    exit (EXIT_FAILURE);
  }

  /* Limpia. */
  if (guestfs_shutdown (destg) == -1)
    exit (EXIT_FAILURE);
  guestfs_close (destg);

  gettimeofday (&end_t, NULL);

  /* Imprime el tiempo transcurrido. */
  ms = timeval_diff (&start_t, &end_t);
  printf ("copia finalizada, tiempo transcurrido (excluyendo inicio) fue "
          "%" PRIi64 ".%03" PRIi64 " s\n",
          ms / 1000, ms % 1000);

  exit (EXIT_SUCCESS);
}

static void *
start_srcthread (void *arg)
{
  struct threaddata *threaddata = arg;
  guestfs_h *srcg;
  char fdname[128];

  /* Abre el dominio de origen. */
  srcg = guestfs_create ();
  if (!srcg) {
    perror ("fallo al crear el handle de libguestfs");
    pthread_cancel (threaddata->mainthread);
    exit (EXIT_FAILURE);
  }
  if (open_guest (srcg, threaddata->src, 1) == -1) {
    pthread_cancel (threaddata->mainthread);
    exit (EXIT_FAILURE);
  }

  /* Comienza la descarga. */
  snprintf (fdname, sizeof fdname, "/dev/fd/%d", threaddata->fd);
  if (guestfs_tar_out (srcg, threaddata->srcdir, fdname) == -1) {
    pthread_cancel (threaddata->mainthread);
    exit (EXIT_FAILURE);
  }

  /* Cierra la tubería; esto hará que el receptor finalice la carga. */
  if (close (threaddata->fd) == -1) {
    pthread_cancel (threadd
ata->mainthread);
    exit (EXIT_FAILURE);
  }

  /* Limpieza. */
  guestfs_close (srcg);

  return NULL;
}

/* Esta función se encarga de la complejidad de añadir el dominio,
 * lanzar el manejador y montar los sistemas de archivos. Consulta
 * 'examples/inspect-vm.c' para entender cómo funciona esto.
 */
static int
open_guest (guestfs_h *g, const char *dom, int readonly)
{
  char **roots, *root, **mountpoints;
  size_t i;

  /* Usa libvirt para encontrar los discos del huésped y añadirlos al manejador. */
  if (guestfs_add_domain (g, dom,
                          GUESTFS_ADD_DOMAIN_READONLY, readonly,
                          -1) == -1)
    return -1;

  if (guestfs_launch (g) == -1)
    return -1;

  /* Inspecciona el huésped, buscando sistemas operativos. */
  roots = guestfs_inspect_os (g);
  if (roots == NULL)
    return -1;

  if (roots[0] == NULL || roots[1] != NULL) {
    fprintf (stderr, "copy-over: %s: no se encontraron sistemas operativos o se encontraron múltiples sistemas operativos\n", dom);
    return -1;
  }

  root = roots[0];

  /* Monta los sistemas de archivos (como 'guestfish -i'). */
  mountpoints = guestfs_inspect_get_mountpoints (g, root);
  if (mountpoints == NULL)
    return -1;

  qsort (mountpoints, count_strings (mountpoints) / 2, 2 * sizeof (char *),
         compare_keys_len);
  for (i = 0; mountpoints[i] != NULL; i += 2) {
    /* Ignora los fallos de esta llamada, ya que pueden aparecer
     * entradas falsas en el /etc/fstab del huésped.
     */
    (readonly ? guestfs_mount_ro : guestfs_mount)
      (g, mountpoints[i+1], mountpoints[i]);
    free (mountpoints[i]);
    free (mountpoints[i+1]);
  }

  free (mountpoints);

  free (root);
  free (roots);

  /* Todo listo, sin errores. */
  return 0;
}

/* Calcula Y - X y devuelve el resultado en milisegundos.
 * Aproximadamente igual que este código:
 * http://www.mpp.mpg.de/~huber/util/timevaldiff.c
 */
static int64_t
timeval_diff (const struct timeval *x, const struct timeval *y)
{
  int64_t msec;

  msec = (y->tv_sec - x->tv_sec) * 1000;
  msec += (y->tv_usec - x->tv_usec) / 1000;
  return msec;
}

static int
compare_keys_len (const void *p1, const void *p2)
{
  const char *key1 = * (char * const *) p1;
  const char *key2 = * (char * const *) p2;
  return strlen (key1) - strlen (key2);
}

static size_t
count_strings (char *const *argv)
{
  size_t c;

  for (c = 0; argv[c]; ++c)
    ;
  return c;
}

HUÉSPED 
/* Este es un ejemplo más significativo de una herramienta que puede obtener la
 * dirección DHCP de algunos tipos de máquina virtual. Dado que hay
 * tantas formas posibles de hacer esto, sin claridad sobre cuál es la
 * mejor manera, no quiero convertir esto en una herramienta virtual oficial.
 *
 * Para más información, consulta:
 *
 * https://rwmj.wordpress.com/2010/10/26/tip-find-the-ip-address-of-a-virtual-machine/
 * https://rwmj.wordpress.com/2011/03/30/tip-another-way-to-get-the-ip-address-of-a-virtual-machine/
 */

#include <stdio.h>
#include <stdlib.h>
#include <string.h>
#include <errno.h>
#include <unistd.h>
#include <assert.h>

#include <guestfs.h>

static int compare_keys_len (const void *p1, const void *p2);
static size_t count_strings (char *const *argv);
static void free_strings (char **argv);
static void mount_disks (guestfs_h *g, char *root);
static void print_dhcp_address (guestfs_h *g, char *root);
static void print_dhcp_address_linux (guestfs_h *g, char *root, const char *logfile);
static void print_dhcp_address_windows (guestfs_h *g, char *root);

int
main (int argc, char *argv[])
{
  guestfs_h *g;
  size_t i;
  char **roots, *root;

  if (argc < 2) {
    fprintf (stderr,
             "Uso: virt-dhcp-address disk.img [disk.img [...]]\n"
             "Ten en cuenta que todos los discos deben provenir de una sola máquina virtual.\n");
    exit (EXIT_FAILURE);
  }

  g = guestfs_create ();
  if (g == NULL) {
    perror ("failed to create libguestfs handle");
    exit (EXIT_FAILURE);
  }

  for (i = 1; i < (size_t) argc; ++i) {
    /* Adjunta la(s) imagen(es) de disco en modo de solo lectura a libguestfs. */
    if (guestfs_add_drive_opts (g, argv[i],
                                /* GUESTFS_ADD_DRIVE_OPTS_FORMAT, "raw", */
                                GUESTFS_ADD_DRIVE_OPTS_READONLY, 1,
                                -1) /* esto marca el final de los argumentos opcionales */
        == -1)
      exit (EXIT_FAILURE);
  }

  /* Ejecuta el backend de libguestfs. */
  if (guestfs_launch (g) == -1)
    exit (EXIT_FAILURE);

  /* Pide a libguestfs que inspeccione los sistemas operativos. */
  roots = guestfs_inspect_os (g);
  if (roots == NULL)
    exit (EXIT_FAILURE);
  if (roots[0] == NULL) {
    fprintf (stderr, "virt-dhcp-address: no se encontraron sistemas operativos\n");
    exit (EXIT_FAILURE);
  }
  if (count_strings (roots) > 1) {
    fprintf (stderr, "virt-dhcp-address: sistema operativo de arranque múltiple\n");
    exit (EXIT_FAILURE);
  }

  root = roots[0];

  /* Monta los discos del huésped. */
  mount_disks (g, root);

  /* Imprime la dirección DHCP. */
  print_dhcp_address (g, root);

  /* Cierra el manejador y sale. */
  guestfs_close (g);
  free_strings (roots);

  exit (EXIT_SUCCESS);
}

static void
mount_disks (guestfs_h *g, char *root)
{
  char **mountpoints;
  size_t i;

  /* Monta los discos, como guestfish -i.
   *
   * Ordena las claves por longitud, la más corta primero, para que
   * terminemos montando los sistemas de archivos en el orden correcto.
   */
  mountpoints = guestfs_inspect_get_mountpoints (g, root);
  if (mountpoints == NULL)
    exit (EXIT_FAILURE);

  qsort (mountpoints, count_strings (mountpoints) / 2, 2 * sizeof (char *),
         compare_keys_len);

  for (i = 0; mountpoints[i] != NULL; i += 2) {
    /* Ignora los fallos de esta llamada, ya que pueden aparecer
     * entradas falsas en el /etc/fstab del huésped.
     */
    guestfs_mount_ro (g, mountpoints[i+1], mountpoints[i]);
  }

  free_strings (mountpoints);
}

static void
print_dhcp_address (guestfs_h *g, char *root)
{
  char *guest_type, *guest_distro;

  /* Dependiendo del tipo de huésped, intenta obtener la dirección DHCP. */
  guest_type = guestfs_inspect_get_type (g, root);
  if (guest_type == NULL)
    exit (EXIT_FAILURE);

  if (strcmp (guest_type, "linux") == 0) {
    guest_distro = guestfs_inspect_get_distro (g, root);
    if (guest_distro == NULL)
      exit (EXIT_FAILURE);

    if (strcmp (guest_distro, "fedora") == 0 ||
        strcmp (guest_distro, "rhel") == 0 ||
        strcmp (guest_distro, "redhat-based") == 0) {
      print_dhcp_address_linux (g, root, "/var/log/messages");
    }
    else if (strcmp (guest_distro, "debian") == 0 ||
             strcmp (guest_distro, "ubuntu") == 0) {
      print_dhcp_address_linux (g, root, "/var/log/syslog");
    }
    else {
      fprintf (stderr, "virt-dhcp-address: no sé cómo obtener la dirección DHCP de '%s'\n",
               guest_distro);
      exit (EXIT_FAILURE);
    }

    free (guest_distro);
  }
  else if (strcmp (guest_type, "windows") == 0) {
    print_dhcp_address_windows (g, root);
  }
  else {
    fprintf (stderr, "virt-dhcp-address: no sé cómo obtener la dirección DHCP de '%s'\n",
             guest_type);
    exit (EXIT_FAILURE);
  }

  free (guest_type);
}

/* Busca mensajes de dhclient en el archivo de registro.
 */
static void
print_dhcp_address_linux (guestfs_h *g, char *root, const char *logfile)
{
  char **lines, *p;
  size_t len;

  lines = guestfs_grep_opts (g, "dhclient.*: bound to ", logfile,
                             GUESTFS_GREP_OPTS_EXTENDED, 1,
                             -1);
  if (lines == NULL)
    exit (EXIT_FAILURE);

  len = count_strings (lines);
  if (len == 0) {
    fprintf (stderr, "virt-dhcp-address: no se puede encontrar la dirección DHCP para este huésped.\n");
    exit (EXIT_FAILURE);
  }

  /* Solo queremos el último mensaje. */
  p = strstr (lines[len-1], "bound to ");
  assert (p);
  p += 9;
  len = strcspn (p, " ");
  p[len] = '\0';

  printf ("%s\n", p);

  free_strings (lines);
}

/* Descarga el colmena SYSTEM de Windows y busca la configuración DHCP ahí dentro. */
static void
print_dhcp_address_windows (guestfs_h *g, char *root_fs)
{
  char *system_path;
  int64_t root, node, value;
  struct guestfs_hivex_node_list *nodes;
  char *controlset;
  size_t i;
  char *p;

  /* Localiza la colmena SYSTEM. */
  system_path = guestfs_inspect_get_windows_system_hive (g, root_fs);
  if (!system_path)
    exit (EXIT_FAILURE);

  /* Abre la colmena para analizarla. Ten en cuenta que antes de libguestfs 1.19.35
   * tenías que descargar el archivo y analizarlo usando hivex(3). Desde
   * libguestfs 1.19.35, partes de la API de hivex(3) ahora se exponen
   * a través de libguestfs, y eso es lo que usaremos aquí porque es
   * más conveniente y evita tener que descargar la colmena.
   */
  if (guestfs_hivex_open (g, system_path, -1) == -1)
    exit (EXIT_FAILURE);

  free (system_path);

  root = guestfs_hivex_root (g);
  if (root == -1)
    exit (EXIT_FAILURE);

  /* Obtiene ControlSetXXX\Services\Tcpip\Parameters\Interfaces. */
  controlset = guestfs_inspect_get_windows_current_control_set (g, root_fs);
  if (controlset == NULL)
    exit (EXIT_FAILURE);
  const char *path[] = { controlset, "Services", "Tcpip", "Parameters",
                         "Interfaces" };
  node = root;
  for (i = 0; node > 0 && i < sizeof path / sizeof path[0]; ++i)
    node = guestfs_hivex_node_get_child (g, node, path[i]);

  if (node == -1)
    exit (EXIT_FAILURE);

  if (node == 0) {
    fprintf (stderr, "virt-dhcp-address: HKLM\\System\\%s\\Services\\Tcpip\\Parameters\\Interfaces no encontrado.", controlset);
    exit (EXIT_FAILURE);
  }

  free (controlset);

  /* Busca un nodo debajo de aquí que tenga una entrada "DhcpIPAddress". */
  nodes = guestfs_hivex_node_children (g, node);
  if (nodes == NULL)
    exit (EXIT_FAILURE);

  value = 0;
  for (i = 0; value == 0 && i < nodes->len; ++i) {
    value = guestfs_hivex_node_get_value (g, nodes->val[i].hivex_node_h,
                                          "DhcpIPAddress");
    if (value == -1)
      exit (EXIT_FAILURE);
  }

  if (value == 0) {
    fprintf (stderr, "virt-dhcp-address: no se puede encontrar la dirección DHCP para este huésped.\n");
    exit (EXIT_FAILURE);
  }

  guestfs_free_hivex_node_list (nodes);

  /* Obtiene la cadena y usa la autoconversión de libguestfs para convertirla
   * a UTF-8 para la salida.
   */
  p = guestfs_hivex_value_string (g, value);
  if (!p)
    exit (EXIT_FAILURE);

  printf ("%s\n", p);

  free (p);

  /* Cierra el manejador de la colmena. */
  guestfs_hivex_close (g);
}

static int
compare_keys_len (const void *p1, const void *p2)
{
  const char *key1 = * (char * const *) p1;
  const char *key2 = * (char * const *) p2;
  return strlen (key1) - strlen (key2);
}

static size_t
count_strings (char *const *argv)
{
  size_t c;

  for (c = 0; argv[c]; ++c)
    ;
  return c;
}

static void
free_strings (char **argv)
{
  size_t i;

  for (i = 0; argv[i]; ++i)
    free (argv[i]);
  free (argv);
}
