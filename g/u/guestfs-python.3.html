EJEMPLO 1: CREAR UNA IMAGEN DE DISCO # Ejemplo que muestra cómo crear una imagen de disco. import guestfs output = "disk.img" # Todo el código Python nuevo debe pasar python_return_dict=True # al constructor. Indica que su programa desea recibir # diccionarios de Python para los métodos de la API que devuelven # tablas hash. g = guestfs.GuestFS(python_return_dict=True) # Crear una imagen de disco dispersa en formato raw, de 512 MB de tamaño. g.disk_create(output, "raw", 512 * 1024 * 1024) # Establecer el indicador de traza para poder ver cada llamada a libguestfs. g.set_trace(1) # Adjuntar la imagen de disco a libguestfs. g.add_drive_opts(output, format="raw", readonly=0) # Ejecutar el backend de libguestfs. g.launch() # Obtener la lista de dispositivos. Como solo añadimos una unidad # arriba, esperamos que esta lista contenga un solo # elemento. devices = g.list_devices() assert (len(devices) == 1) # Particionar el disco como una única partición MBR. g.part_disk(devices[0], "mbr") # Obtener la lista de particiones. Esperamos un solo elemento, que es # la partición que acabamos de crear. partitions = g.list_partitions() assert (len(partitions) == 1) # Crear un sistema de archivos en la partición. g.mkfs("ext4", partitions[0]) # Ahora montar el sistema de archivos para poder añadir archivos. g.mount(partitions[0], "/") # Crear algunos archivos y directorios. g.touch("/empty") message = "Hola, mundo\n" g.write("/hello", message) g.mkdir("/foo") # Este comando sube el archivo local /etc/resolv.conf a la # imagen de disco. g.upload("/etc/resolv.conf", "/foo/resolv.conf") # Como escribimos en el disco y queremos detectar errores de # escritura, llame a g.shutdown. No necesita hacer esto: # g.close lo hará implícitamente. g.shutdown() # Tenga en cuenta que los manejadores se cierran automáticamente si son # reciclados por el conteo de referencias. Solo necesita llamar a close # si desea cerrar el manejador inmediatamente. g.close() EJEMPLO 2: INSPECCIONAR UNA IMAGEN DE DISCO DE UNA MÁQUINA

# Ejemplo que muestra cómo inspeccionar un disco de una máquina virtual. import sys import guestfs if len(sys.argv) < 2: print("inspect_vm: falta la imagen de disco para inspeccionar", file=sys.stderr) sys.exit(1) disk = sys.argv[1] # Todo el código Python nuevo debe pasar python_return_dict=True # al constructor. Indica que su programa desea recibir # diccionarios de Python para los métodos de la API que devuelven # tablas hash. g = guestfs.GuestFS(python_return_dict=True) # Adjuntar la imagen de disco a libguestfs en modo solo lectura. g.add_drive_opts(disk, readonly=1) # Ejecutar el backend de libguestfs. g.launch() # Pedir a libguestfs que inspeccione los sistemas operativos. roots = g.inspect_os() if len(roots) == 0: print("inspect_vm: no se encontraron sistemas operativos", file=sys.stderr) sys.exit(1) for root in roots: print("Dispositivo raíz: %s" % root) # Imprimir información básica sobre el sistema operativo. print(" Nombre del producto: %s" % (g.inspect_get_product_name(root))) print(" Versión: %d.%d" % (g.inspect_get_major_version(root), g.inspect_get_minor_version(root))) print(" Tipo: %s" % (g.inspect_get_type(root))) print(" Distribución: %s" % (g.inspect_get_distro(root))) # Montar los discos, como guestfish -i. # # Ordenar las claves por longitud, de la más corta a la más larga, para # que terminemos montando los sistemas de archivos en el orden correcto. mps = g.inspect_get_mountpoints(root) for device, mp in sorted(mps.items(), key=lambda k: len(k[0])): try: g.mount_ro(mp, device) except RuntimeError as msg: print("%s (ignorado)" % msg) # Si el archivo /etc/issue.net existe, imprimir hasta 3 líneas. filename = "/etc/issue.net" if g.is_file(filename): print("--- %s ---" % filename) lines = g.head_n(3, filename) for line in lines: print(line) # Desmontar todo. g.umount_all() VÉASE TAMBIÉN g
