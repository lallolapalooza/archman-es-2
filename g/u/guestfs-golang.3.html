NOMBRE

guestfs-golang - Cómo usar libguestfs desde Go

SINOPSIS

import "libguestfs.org/guestfs"

g, errno := guestfs.Create ()
if errno != nil {
    panic (fmt.Sprintf ("no se pudo crear el manejador: %s", errno))
}
defer g.Close ()
if err := g.Add_drive ("test.img"); err != nil {
    panic (err)
}
if err := g.Launch (); err != nil {
    panic (err)
}
if err := g.Shutdown (); err != nil {
    panic (err)
}

DESCRIPCIÓN

Esta página del manual documenta cómo llamar a libguestfs desde el lenguaje de programación Go. Esta página solo documenta las diferencias con la API de C y proporciona algunos ejemplos. Si no estás familiarizado con el uso de libguestfs, también debes leer guestfs(3).

IMPORTANDO EL MÓDULO

El módulo se llama "guestfs". El nombre completo del paquete para importar es "libguestfs.org/guestfs".

CREANDO Y CERRANDO EL MANEJADOR

Usa "guestfs.Create" o "guestfs.Create_flags" para crear el manejador. El manejador se cierra implícitamente si es recolectado por el recolector de basura. Sin embargo, probablemente sea una buena idea cerrarlo explícitamente, ya sea llamando a "g.Close ()" o difiriendo lo mismo.

ERRORES

"guestfs.Create" y "guestfs.Create_flags" devuelven un simple *error, que en realidad es solo un C "errno" envuelto en la estructura golang apropiada.

Todas las demás llamadas devuelven un *GuestfsError que (si no es nulo) es una estructura más rica que contiene la cadena de error de libguestfs, el errno (si está disponible) y la operación que falló. Esto también se puede convertir en una cadena para su visualización.

LIMITACIONES

  • No hay soporte para eventos (ver "EVENTOS" en guestfs(3)).
  • Los UUID no se devuelven en las estructuras.

EJEMPLO 1: CREAR UNA IMAGEN DE DISCO

/* Ejemplo que muestra cómo crear una imagen de disco. */

package main

import (
       "fmt"
       "libguestfs.org/guestfs"
)

func main() {
       output := "disk.img"

       g, errno := guestfs.Create ()
       if errno != nil {
               panic (errno)
       }
       defer g.Close ()

       /* Crear una imagen de disco dispersa en formato raw, de 512 MB. */
       if err := g.Disk_create (output, "raw", 512 * 1024 * 1024); err != nil {
               panic (err)
       }

       /* Establecer la bandera de rastreo para que podamos ver cada llamada a libguestfs. */
       g.Set_trace (true)

       /* Adjuntar la imagen de disco a libguestfs. */
       optargs := guestfs.OptargsAdd_drive{
               Format_is_set: true,
               Format: "raw",
               Readonly_is_set: true,
               Readonly: false,
       }
       if err := g.Add_drive (output, &optargs); err != nil {
               panic (err)
       }

       /* Ejecutar el backend de libguestfs. */
       if err := g.Launch (); err != nil {
               panic (err)
       }

       /* Obtener la lista de dispositivos. Como solo adjuntamos una unidad
        * arriba, esperamos que esta lista contenga un solo
        * elemento.
        */
       devices, err := g.List_devices ()
       if err != nil {
               panic (err)
       }
       if len(devices) != 1 {
               panic ("se esperaba un solo dispositivo de list-devices")
       }

       /* Particionar el disco como una única partición MBR. */
       err = g.Part_disk (devices[0], "mbr")
       if err != nil {
               panic (err)
       }

       /* Obtener la lista de particiones. Esperamos un solo elemento, que
        * es la partición que acabamos de crear.
        */
       partitions, err := g.List_partitions ()
       if err != nil {
               panic (err)
       }
       if len(partitions) != 1 {
               panic ("se esperaba una sola partición de list-partitions")
       }

       /* Crear un sistema de archivos en la partición. */
       err = g.Mkfs ("ext4", partitions[0], nil)
       if err != nil {
               panic (err)
       }

       /* Ahora montar el sistema de archivos para que podamos agregar archivos. */
       err = g.Mount (partitions[0], "/")
       if err != nil {
               panic (err)
       }

       /* Crear algunos archivos y directorios. */
       err = g.Touch ("/empty")
       if err != nil {
               panic (err)
       }
       message := []byte("Hola, mundo\n")
       err = g.Write ("/hello", message)
       if err != nil {
               panic (err)
       }
       err = g.Mkdir ("/foo")
       if err != nil {
               panic (err)
       }

       /* Esto carga el archivo local /etc/resolv.conf en
        * la imagen del disco.
        */
       err = g.Upload ("/etc/resolv.conf", "/foo/resolv.conf")
       if err != nil {
               panic (err)
       }

       /* Dado que escribimos en el disco y queremos detectar errores de escritura,
        * llamamos a g:shutdown. No necesitas hacer esto:
        * g.Close lo hará implícitamente.
        */
       if err = g.Shutdown (); err != nil {
               panic (fmt.Sprintf ("la escritura en el disco falló: %s", err))
       }
}

EJEMPLO 2: INSPECCIONAR UNA IMAGEN DE DISCO DE UNA MÁQUINA VIRTUAL

/* Ejemplo que muestra cómo inspeccionar un disco de máquina virtual. */

package main

import (
       "fmt"
       "os"
       "libguestfs.org/guestfs"
)

func main() {
       if len(os.Args) < 2 {
               panic ("uso: inspect-vm disk.img")
       }
       disk := os.Args[1]

       g, errno := guestfs.Create ()
       if errno != nil {
               panic (fmt.Sprintf ("no se pudo crear el manejador: %s", errno))
       }

       /* Adjuntar la imagen de disco en modo de solo lectura a libguestfs. */
       optargs := guestfs.OptargsAdd_drive{
               Format_is_set: true,
               Format: "raw",
               Readonly_is_set: true,
               Readonly: true,
       }
       if err := g.Add_drive (disk, &optargs); err != nil {
               panic (err)
       }

       /* Ejecutar el backend de libguestfs. */
       if err := g.Launch (); err != nil {
               panic (err)
       }

       /* Pedir a libguestfs que inspeccione los sistemas operativos. */
       roots, err := g.Inspect_os ()
       if err != nil {
               panic (err)
       }
       if len(roots) == 0 {
               panic ("inspect-vm: no se encontraron sistemas operativos")
       }

       for _, root := range roots {
               fmt.Printf ("Dispositivo raíz: %s\n", root)

               /* Imprimir información básica sobre el sistema operativo. */
               s, _ := g.Inspect_get_product_name (root)
               fmt.Printf ("  Nombre del producto: %s\n", s)
               major, _ := g.Inspect_get_major_version (root)
               minor, _ := g.Inspect_get_minor_version (root)
               fmt.Printf ("  Versión:      %d.%d\n", major, minor)
               s, _ = g.Inspect_get_type (root)
               fmt.Printf ("  Tipo:         %s\n", s)
               s, _ = g.Inspect_get_distro (root)
               fmt.Printf ("  Distribución: %s\n", s)

               /* XXX Ejemplo incompleto. Ordenar las claves por longitud
                * es innecesariamente difícil en golang.
                */
       }
}
