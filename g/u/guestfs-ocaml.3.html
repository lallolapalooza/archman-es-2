NOMBRE

guestfs-ocaml - Cómo usar libguestfs desde OCaml

SINOPSIS

Estilo de módulo:

let g = Guestfs.create () in
Guestfs.add_drive_opts g ~format:"raw" ~readonly:true "disk.img";
Guestfs.launch g;

Estilo orientado a objetos:

 let g = new Guestfs.guestfs () in
 g#add_drive_opts ~format:"raw" ~readonly:true "disk.img";
 g#launch ();
 ocamlfind opt prog.ml -package guestfs -linkpkg -o prog
o bien:
 ocamlopt -I +guestfs mlguestfs.cmxa prog.ml -o prog

DESCRIPCIÓN

Esta página del manual documenta cómo llamar a libguestfs desde el lenguaje de programación OCaml. Esta página solo documenta las diferencias con la API de C y proporciona algunos ejemplos. Si no está familiarizado con el uso de libguestfs, también necesita leer guestfs(3).

ESTILOS DE PROGRAMACIÓN

Las vinculaciones de OCaml admiten dos estilos de programación diferentes. Puede usar un estilo de módulo, con cada función de C mapeada a una función de OCaml:

int guestfs_set_verbose (guestfs_h *g, int flag);

se convierte en:

val Guestfs.set_verbose : Guestfs.t -> bool -> unit

Alternativamente, puede usar un estilo orientado a objetos, llamando a métodos en la clase "Guestfs.guestfs":

method set_verbose : bool -> unit

El estilo orientado a objetos suele ser más breve, y la pequeña penalización de rendimiento no es perceptible en la sobrecarga general de realizar funciones de libguestfs.

CERRAR EL MANEJADOR

El manejador se cierra cuando es recogido por el recolector de basura. Debido a que los manejadores de libguestfs incluyen mucho estado, también es posible cerrarlos (y por lo tanto liberarlos) explícitamente llamando a "Guestfs.close" o al método "#close" .

EXCEPCIONES

Los errores de las funciones de libguestfs se mapean en la excepción "Guestfs.Error" . Esta tiene un único parámetro que es el mensaje de error (una cadena de texto).

Llamar a cualquier función/método en un manejador cerrado lanza "Guestfs.Handle_closed". El único parámetro es el nombre de la función que usted llamó.

EJEMPLO: CREAR UNA IMAGEN DE DISCO

(* Ejemplo que muestra cómo crear una imagen de disco. *)

open Unix
open Printf

let output = "disk.img"

let () =
  let g = new Guestfs.guestfs () in

  (* Crear una imagen de disco dispersa de formato raw, de 512 MB de
   * tamaño. *)
  g#disk_create output "raw" (Int64.of_int (512 * 1024 * 1024));

  (* Establecer el indicador de traza para que podamos ver cada llamada
   * a libguestfs. *)
  g#set_trace true;

  (* Adjuntar la imagen de disco a libguestfs. *)
  g#add_drive_opts ~format:"raw" ~readonly:false output;

  (* Ejecutar el backend de libguestfs. *)
  g#launch ();

  (* Obtener la lista de dispositivos. Como solo añadimos una unidad
   * arriba, esperamos que esta lista contenga un solo elemento. *)
  let devices = g#list_devices () in
  if Array.length devices <> 1 then
    failwith "error: se esperaba un solo dispositivo de list-devices";

  (* Particionar el disco como una sola partición MBR. *)
  g#part_disk devices.(0) "mbr";

  (* Obtener la lista de particiones. Esperamos un solo elemento, que
   * es la partición que acabamos de crear. *)
  let partitions = g#list_partitions () in
  if Array.length partitions <> 1 then
    failwith "error: se esperaba una sola partición de list-partitions";

  (* Crear un sistema de archivos en la partición. *)
  g#mkfs "ext4" partitions.(0);

  (* Ahora montar el sistema de archivos para poder añadir archivos. *)
  g#mount partitions.(0) "/";

  (* Crear algunos archivos y directorios. *)
  g#touch "/empty";
  let message = "Hola, mundo\n" in
  g#write "/hello" message;
  g#mkdir "/foo";

  (* Este sube el archivo local /etc/resolv.conf a la imagen de disco. *)
  g#upload "/etc/resolv.conf" "/foo/resolv.conf";

  (* Como escribimos en el disco y queremos detectar errores de
   * escritura, llamamos a g#shutdown. No necesita hacer esto:
   * g#close lo hará implícitamente. *)
  g#shutdown ();

  (* Tenga en cuenta también que los manejadores se cierran
   * automáticamente si son recogidos por el recolector de basura. Solo
   * necesita llamar a close si quiere cerrar el manejador de inmediato. *)
  g#close ()

EJEMPLO: INSPECCIONAR UNA IMAGEN DE DISCO DE UNA MÁQUINA VIRTUAL

(* Ejemplo que muestra cómo inspeccionar un disco de máquina virtual. *)

open Printf

let disk =
  if Array.length Sys.argv = 2 then
    Sys.argv.(1)
  else
    failwith "uso: inspect_vm disk.img"

let () =
  let g = new Guestfs.guestfs () in

  (* Adjuntar la imagen de disco de solo lectura a libguestfs. *)
  g#add_drive_opts (*~format:"raw"*) ~readonly:true disk;

  (* Ejecutar el backend de libguestfs. *)
  g#launch ();

  (* Pedir a libguestfs que inspeccione los sistemas operativos. *)
  let roots = g#inspect_os () in
  if Array.length roots = 0 then
    failwith "inspect_vm: no se encontraron sistemas operativos";

  Array.iter (
    fun root ->
      printf "Dispositivo raíz: %s\n" root;

      (* Imprimir información básica sobre el sistema operativo. *)
      printf "  Nombre del producto: %s\n" (g#inspect_get_product_name root);
      printf "  Versión:            %d.%d\n"
        (g#inspect_get_major_version root)
        (g#inspect_get_minor_version root);
      printf "  Tipo:               %s\n" (g#inspect_get_type root);
      printf "  Distribución:       %s\n" (g#inspect_get_distro root);

      (* Montar los discos, como guestfish -i.
       *
       * Ordenar las claves por longitud, la más corta primero, para
       * que terminemos montando los sistemas de archivos en el orden
       * correcto. *)
      let mps = g#inspect_get_mountpoints root in
      let cmp (a,_) (b,_) =
        compare (String.length a) (String.length b) in
      let mps = List.sort cmp mps in
      List.iter (
        fun (mp, dev) ->
          try g#mount_ro dev mp
          with Guestfs.Error msg -> eprintf "%s (ignorado)\n" msg
      ) mps;

      (* Si el archivo /etc/issue.net existe, imprimir hasta 3 líneas. *)
      let filename = "/etc/issue.net" in
      if g#is_file filename then (
        printf "--- %s ---\n" filename;
        let lines = g#head_n 3 filename in
        Array.iter print_endline lines
      );

      (* Desmontar todo. *)
      g#umount_all ()
  ) roots

EJEMPLO: HABILITAR LA DEPURACIÓN Y EL REGISTRO

(* Ejemplo que muestra cómo habilitar la depuración y capturarla en
 * cualquier sistema de registro personalizado.
 *)

(* Eventos en los que estamos interesados. Esta máscara de bits cubre
 * todos los mensajes de traza y depuración. *)
let event_bitmask = [
  Guestfs.EVENT_LIBRARY;
  Guestfs.EVENT_WARNING;
  Guestfs.EVENT_APPLIANCE;
  Guestfs.EVENT_TRACE
]

let rec main () =
  let g = new Guestfs.guestfs () in

  (* Por defecto, la información de depuración se imprime en stderr.
   * Para capturarla en otro lugar, debe configurar un manejador de
   * eventos que será llamado de vuelta a medida que se generan los
   * mensajes de depuración. Para hacer esto, use la API de eventos.
   *
   * Para más información, vea EVENTS en guestfs(3). *)
  ignore (g#set_event_callback message_callback event_bitmask);

  (* Así es como se habilita la depuración:
   *
   * Establecer el indicador 'trace' en el manejador significa que cada
   * llamada a libguestfs se registra (nombre, parámetros, retorno).
   * Este indicador es útil para ver cómo un programa está usando
   * libguestfs.
   *
   * Establecer el indicador 'verbose' habilita una gran cantidad de
   * depuración adicional en todo el sistema. Esto es útil si hay un
   * error de libguestfs que no entiende.
   *
   * Tenga en cuenta que debe establecer los indicadores pronto después
   * de crear el manejador. En particular, si establece el indicador
   * verbose después del lanzamiento, no verá todos los mensajes.
   *
   * Para más información, vea:
   * http://libguestfs.org/guestfs-faq.1.html#debugging-libguestfs
   *
   * Los mensajes de error lanzados por las APIs *no* son información de
   * depuración, y no se ven afectados por nada de esto. Puede que
   * tenga que registrarlos por separado. *)
  g#set_trace true;
  g#set_verbose true;

  (* Realizar algunas operaciones que generarán muchos mensajes de
   * traza y depuración. *)
  g#add_drive "/dev/null";
  g#launch ();
  g#close ()

(* Esta función es llamada de vuelta por libguestfs siempre que se
 * genera un mensaje de traza o depuración.
 *
 * Para la classes de eventos que hemos registrado anteriormente, 'array' y
 * 'array_len' no serán significativos. Solo 'buf' y 'buf_len' serán
 * interesantes y contendrán el rastro o el mensaje de depuración.
 *
 * Este ejemplo simplemente redirige estos mensajes a syslog, pero
 * obviamente podrías hacer algo más avanzado aquí.
 *)
and message_callback event event_handle buf array =
  if String.length buf > 0 then (
    let event_name = Guestfs.event_to_string [event] in
    Printf.printf "[%s] %S\n%!" event_name buf
  )

let () = main ()
