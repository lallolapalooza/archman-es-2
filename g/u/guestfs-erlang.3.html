DESCRIPCIÓ

N

Esta página manual documenta cómo llamar a libguestfs desde el lenguaje de programación Erlang. Esta página solo documenta las diferencias con la API de C y proporciona algunos ejemplos. Si no está familiarizado con el uso de libguestfs, también necesita leer guestfs

(3).

APERTURA Y CIERRE DEL

MANEJADOR

Los enlaces de Erlang se implementan utilizando un programa externo llamado "erl-guestfs". Este programa debe estar en el PATH actual, o de lo contrario debe especificar la ruta completa al

programa: 
{ok, G} = guestfs:create().
{ok, G} = guestfs:create("/path/to/erl-guestfs")
.

"G" es el manejador de libguestfs que debe pasar a otras

funciones.

Para cerrar el

manejador:

OPCIONALES

VALORES DE RETORNO Y

ERRORES

En caso de éxito, la mayoría de las funciones devuelven un término "Result" (que puede ser una lista, cadena, tupla, etc.). Si la función no tiene nada que devolver, entonces se devuelve el átomo "ok"

.

En caso de error, verá una de las siguientes

tuplas:
"{error, Msg, Errno}
"
Esto indica un error ordinario de la función.

"Msg" es el mensaje de error (cadena de texto) y "Errno" es el error de Unix

(entero).

EJEMPLO 1: CREAR UNA IMAGEN DE

DISCO 
#!/usr/bin/env escript
%%! -smp enable -sname create_disk debug verbose
% Ejemplo que muestra cómo crear una imagen de disco.

main(_) ->
    Output = "disk.img",

    {ok, G} = guestfs:create(),

    % Crear una imagen de disco dispersa en formato raw, de 512 MB de tamaño.
    {ok, File} = file:open(Output, [raw, write, binary]),
    {ok, _} = file:position(File, 512 * 1024 * 1024 - 1),
    ok = file:write(File, " "),
    ok = file:close(File),

    % Establecer la bandera de rastreo para que podamos ver cada llamada a libguestfs.
    ok = guestfs:set_trace(G, true),

    % Adjuntar la imagen de disco a libguestfs.
    ok = guestfs:add_drive_opts(G, Output,
                                [{format, "raw"}, {readonly, false}]),

    % Ejecutar el
backend de libguestfs. ok = guestfs:launch(G), % Obtener la lista de dispositivos. Como solo añadimos una unidad % arriba, esperamos que esta lista contenga un solo % elemento. [Device] = guestfs:list_devices(G), % Particionar el disco como una sola partición MBR. ok = guestfs:part_disk(G, Device, "mbr"), % Obtener la lista de particiones. Esperamos un solo elemento, que % es la partición que acabamos de crear. [Partition] = guestfs:list_partitions(G), % Crear un sistema de archivos en la partición. ok = guestfs:mkfs(G, "ext4", Partition), % Ahora montar el sistema de archivos para que podamos añadir archivos. *) ok = guestfs:mount(G, Partition, "/"), % Crear algunos archivos y dir
ectorios. *) ok = guestfs:touch(G, "/empty"), Message = "Hello, world\n", ok = guestfs:write(G, "/hello", Message), ok = guestfs:mkdir(G, "/foo"), % Este sube el archivo local /etc/resolv.conf dentro de % la imagen de disco. ok = guestfs:upload(G, "/etc/resolv.conf", "/foo/resolv.conf"), % Como escribimos en el disco y queremos detectar errores de % escritura, llamamos a guestfs:shutdown. No necesita hacer esto: % guestfs:close lo hará implícitamente. ok = guestfs:shutdown(G), % Tenga en cuenta que los manejadores se cierran automáticamente si son % recogidos por el recolector de basura. Solo necesita llamar a close % si quiere cerrar el manejador inmediatamente. ok = guestfs:close(G).

EJEMPLO 2: INSPECCIONAR UNA IMAGEN DE DISCO DE MÁQUINA

VIRTUAL 
#!/usr/bin/env escript
%%! -smp enable -sname inspect_vm debug verbose
% Ejemplo que muestra cómo inspeccionar un disco de máquina virtual.

main([Disk]) ->
    {ok, G} = guestfs:create(),

    % Adjuntar la imagen de disco de solo lectura a libguestfs.
    ok = guestfs:add_drive_opts(G, Disk, [{readonly, true}]),

    % Ejecutar el backend de libguestfs.
    ok = guestfs:launch(G),

    % Pedir a libguestfs que inspeccione los sistemas operativos.
    case guestfs:inspect_os(G) of
        [] ->
            io:fwrite("inspect_vm: no operating systems found~n"),
            exit(no_operating_system);
        Roots ->
            list_os(G, Roots)
    end.

list_os(_, []) ->
    ok;
list_os(G, [Root|Roots]) ->
    io:fwrite("Root device: ~s~n", [Root]),

    % Imprimir información básica sobre el sistem
a operativo. Product_name = guestfs:inspect_get_product_name(G, Root), io:fwrite(" Product name: ~s~n", [Product_name]), Major = guestfs:inspect_get_major_version(G, Root), Minor = guestfs:inspect_get_minor_version(G, Root), io:fwrite(" Version: ~w.~w~n", [Major, Minor]), Type = guestfs:inspect_get_type(G, Root), io:fwrite(" Type: ~s~n", [Type]), Distro = guestfs:inspect_get_distro(G, Root), io:fwrite(" Distro: ~s~n", [Distro]), % Montar los discos, como guestfish -i. Mps = sort_mps(guestfs:inspect_get_mountpoints(G, Root)), mount_mps(G, Mps), % Si el archivo /etc/issue.net existe, imprimir hasta 3 líneas. *) Filename = "/etc/issue.net", Is_file = guestfs:is_file(G, Filename), if Is_file -> io:fwrite("-
-- ~s ---~n", [Filename]), Lines = guestfs:head_n(G, 3, Filename), write_lines(Lines); true -> ok end, % Desmontar todo. ok = guestfs:umount_all(G), list_os(G, Roots). % Ordenar las claves por longitud, la más corta primero, para que acabemos % montando los sistemas de archivos en el orden correcto. sort_mps(Mps) -> Cmp = fun ({A,_}, {B,_}) -> length(A) =< length(B) end, lists:sort(Cmp, Mps). mount_mps(_, []) -> ok; mount_mps(G, [{Mp, Dev}|Mps]) -> case guestfs:mount_ro(G, Dev, Mp) of ok -> ok; { error, Msg, _ } -> io:fwrite("~s (ignored)~n", [Msg]) end, mount_mps(G, Mps). write_lines([]) -> ok; write_lines([Line|Lines]) -> io:fwrite("~s~n", [Line]), write_lines(Lines).
