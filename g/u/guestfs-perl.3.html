DESCRIPCIÓ

N

Esta página manual documenta cómo llamar a libguestfs desde el lenguaje de programación Perl. Esta página solo documenta las diferencias con la API de C y proporciona algunos ejemplos. Si no está familiarizado con el uso de libguestfs, también necesita leer guestfs(3). Para leer la API completa de Perl, consulte Sys::Guestf

s(3).

EJEMPLO 1: CREAR UNA IMAGEN DE

DISCO 
#!/usr/bin/env perl

# Ejemplo que muestra cómo crear una imagen de disco.

use strict;
use warnings;
use Sys::Guestfs;

my $output = "disk.img";

my $g = new Sys::Guestfs ();

# Crear una imagen de disco dispersa en formato raw, de 512 MB de tamaño.
$g->disk_create ($output, "raw", 512 * 1024 * 1024);

# Establecer la bandera de seguimiento para poder ver cada llamada a libguestfs.
$g->set_trace (1);

# Adjuntar la imagen de disco a libguestfs.
$g->add_drive_opts ($output, format => "raw", readonly => 0);

# Ejecutar el backend de libguestfs.
$g->launch ();

# Obtener la lista de dispositivos. Como solo añadimos un disco
# arriba, esperamos que esta lista contenga un único
# elemento.
my @devices = $g->list_devices ();
if (@devices != 1) {
    die "error: expected a single device from list-devices";
}

# Particionar el disco como una única partición MBR.
$g->part_disk ($devices[0], "mbr");

# Obtener la lista de particiones. Esperamos un único elemento, que es
# la partición que acabamos de crear.
my @partitions = $g->list_partitions ();
if (@partitions != 1) {
    die "error: expected a single partition from list-partitions";
}

# Crear un sistema de archivos en la partición.
$g->mkfs ("ext4", $partitions[0]);

# Ahora montar el sistema de archivos para poder añadir archivos.
$g->mount ($partitions[0], "/");

# Crear algunos archivos y directorios.
$g->touch ("/empty");
my $message = "Hello, world\n";
$g->write ("/hello", $message);
$g->mkdir ("/foo");

# Este sube el archivo local /etc/resolv.conf dentro de
# la imagen de disco.
$g->upload ("/etc/resolv.conf", "/foo/resolv.conf");

# Como escribimos en el disco y queremos detectar errores de
# escritura, llamar a $g->shutdown. No necesita hacer esto:
# $g->close lo hará implícitamente.
$g->shutdown ();

# Tenga en cuenta también que los manejadores se cierran automáticamente si son
# recolectados por conteo de referencias. Solo necesita llamar a close
# si quiere cerrar el manejador de inmediato.
$g->close
()
;

EJEMPLO 2: INSPECCIONAR UNA IMAGEN DE DISCO DE MÁQUINA

VIRTUAL 
#!/usr/bin/env perl

# Ejemplo que muestra cómo inspeccionar un disco de máquina virtual.

use strict;
use warnings;
use Sys::Guestfs;

if (@ARGV < 1) {
    die "usage: inspect_vm disk.img"
}

my $disk = $ARGV[0];

my $g = new Sys::Guestfs ();

# Adjuntar la imagen de disco de solo lectura a libguestfs.
# También podría añadir aquí un argumento opcional format => ... Esto es
# aconsejable ya que la detección automática de formato no es segura.
$g->add_drive_opts ($disk, readonly => 1);

# Ejecutar el backend de libguestfs.
$g->launch ();

# Pedir a libguestfs que inspeccione los sistemas operativos.
my @roots = $g->inspect_os ();
if (@roots == 0) {
    die "inspect_vm: no operating systems found";
}

for my $root (@roots) {
    printf "Root device: %s\n", $root;

    # Imprimir información básica sobre el sistema operativo.
    printf "  Product name: %s\n", $g->inspect_get_product_name ($root);
    printf "  Version:      %d.%d\n",
        $g->inspect_get_major_version ($root),
        $g->inspect_get_minor_version ($root);
    printf "  Type:         %s\n", $g->inspect_get_type ($root);
    printf "  Distro:       %s\n", $g->inspect_get_distro ($root);

    # Montar los discos, como guestfish -i.
    #
    # Ordenar las claves por longitud, la más corta primero, para que
    # terminemos montando los sistemas de archivos en el orden
    # correcto.
    my %mps = $g->inspect_get_mountpoints ($root);
    my @mps = sort { length $a <=> length $b } (keys %mps);
    for my $mp (@mps) {
        eval { $g->mount_ro ($mps{$mp}, $mp) };
        if ($@) {
            print "$@ (ignored)\n"
        }
    }

    # Si el archivo /etc/issue.net existe, imprimir hasta 3 líneas.
    my $filename = "/etc/issue.net";
    if ($g->is_file ($filename)) {
        printf "--- %s ---\n", $filename;
        my @lines = $g->head_n (3, $filename);
        print "$_\n" foreach @lines;
    }

    # Desmontar todo.
    $g->umount_all ()
}
