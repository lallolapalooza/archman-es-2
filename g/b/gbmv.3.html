subrutina cgbmv (character trans, integer m, integer n, integer kl, integer ku, complex alpha, complex, dimension(lda,*) a, integer lda, complex, dimension(*) x, integer incx, complex beta, complex, dimension(*) y, integer

incy)

CGBM

V

Propósito

: 
!>
!> CGBMV realiza una de las operaciones matriz-vector
!>
!>    y := alpha*A*x + beta*y,   o   y := alpha*A**T*x + beta*y,   o
!>
!>    y := alpha*A**H*x + beta*y,
!>
!> donde alpha y beta son escalares, x y y son vectores y A es una
!> matriz banda de m por n, con kl subdiagonales y ku superdiagonales.
!
>

Parámetro

s
TRANS 
!>          TRANS es CHARACTER*1
!>           A la entrada, TRANS especifica la operación a realizar como
!>           sigue:
!>
!>              TRANS = 'N' o 'n'   y := alpha*A*x + beta*y.
!>
!>              TRANS = 'T' o 't'   y := alpha*A**T*x + beta*y.
!>
!>              TRANS = 'C' o 'c'   y := alpha*A**H*x + beta*y.
!
>

M 
!>          M es INTEGER
!>           A la entrada, M especifica el número de filas de la matriz A.
!>           M debe ser al menos cero.
!
>

N 
!>          N es INTEGER
!>           A la entrada, N especifica el número de columnas de la matriz A.
!>           N debe ser al menos cero.
!
>

K

L 
!>          KL es INTEGER
!>           A la entrada, KL especifica el número de subdiagonales de la
!>           matriz A. KL debe satisfacer  0 .le. KL.
!
>

K

U 
!>          KU es INTEGER
!>           A la entrada, KU especifica el número de superdiagonales de la
!>           matriz A. KU debe satisfacer  0 .le. KU.
!
>

ALPH

A 
!>          ALPHA es COMPLEX
!>           A la entrada, ALPHA especifica el escalar alpha.
!
>

A 
!>          A es un array COMPLEX, dimensión ( LDA, N )
!>           Antes de la entrada, la parte principal del array A de
!>           dimensión ( kl + ku + 1 ) por n debe contener la matriz de
!>           coeficientes, suministrada columna por columna, con la
!>           diagonal principal de la matriz en la fila ( ku + 1 ) del
!>           array, la primera superdiagonal comenzando en la posición 2
!>           en la fila ku, la primera subdiagonal comenzando en la
!>           posición 1 en la fila ( ku + 2 ), y así sucesivamente.
!>           Los elementos del array A que no corresponden a elementos
!>           en la matriz banda (como el triángulo superior izquierdo
!>           ku por ku) no son referenciados.
!>           El siguiente segmento de programa transferirá una matriz
!>           banda desde el almacenamiento convencional de matriz
!>           completa al almacenamiento de banda:
!>
!>                 DO 20, J = 1, N
!>                    K = KU + 1 - J
!>                    DO 10, I = MAX( 1, J - KU ), MIN( M, J + KL )
!>                       A( K + I, J ) = matrix( I, J )
!>              10    CONTINUE
!>              20 CONTINUE
!
>

LD

A 
!>          LDA es INTEGER
!>           A la entrada, LDA especifica la primera dimensión de A
!>           declarada en el (sub)programa que llama. LDA debe ser al
!>           menos ( kl + ku + 1 ).
!
>

X 
!>          X es un array COMPLEX, dimensión al menos
!>           ( 1 + ( n - 1 )*abs( INCX ) ) cuando TRANS = 'N' o 'n'
!>           y al menos
!>           ( 1 + ( m - 1 )*abs( INCX ) ) en otro caso.
!>           Antes de la entrada, el array X incrementado debe contener
!>           el vector x.
!
>

INC

X 
!>          INCX es INTEGER
!>           A la entrada, INCX especifica el incremento para los
!>           elementos de X. INCX no debe ser cero.
!
>

BET

A 
!>          BETA es COMPLEX
!>           A la entrada, BETA especifica el escalar beta. Cuando BETA
!>           se suministra como cero, entonces Y no necesita estar
!>           establecido en la entrada.
!
>

Y 
!>          Y es un array COMPLEX, dimensión al menos
!>           ( 1 + ( m - 1 )*abs( INCY ) ) cuando TRANS = 'N' o 'n'
!>           y al menos
!>           ( 1 + ( n - 1 )*abs( INCY ) ) en otro caso.
!>           Antes de la entrada, el array Y incrementado debe contener
!>           el vector y. A la salida, Y es sobrescrito por el vector y
!>           actualizado.
!>           Si m o n es cero, entonces Y no es referenciado y la
!>           función realiza una salida rápida.
!
>

INC

Y 
!>          INCY es INTEGER
!>           A la entrada, INCY especifica el incremento para los
!>           elementos de Y. INCY no debe ser cero.
!
>

subroutine   cgbmv (character trans, integer m, integer n, integer kl, integer ku,   complex alpha, complex, dimension(lda,*) a, integer lda, complex,   dimension(*) x, integer incx, complex beta, complex, dimension(*) y,   integer incy)


CGBMV


Purpose:




!>
!> CGBMV  performs one of the matrix-vector operations
!>
!>    y := alpha*A*x + beta*y,   or   y := alpha*A**T*x + beta*y,   or
!>
!>    y := alpha*A**H*x + beta*y,
!>
!> where alpha and beta are scalars, x and y are vectors and A is an
!> m by n band matrix, with kl sub-diagonals and ku super-diagonals.
!> 




Parameters


TRANS

!>          TRANS is CHARACTER*1
!>           On entry, TRANS specifies the operation to be performed as
!>           follows:
!>
!>              TRANS = 'N' or 'n'   y := alpha*A*x + beta*y.
!>
!>              TRANS = 'T' or 't'   y := alpha*A**T*x + beta*y.
!>
!>              TRANS = 'C' or 'c'   y := alpha*A**H*x + beta*y.
!> 


M


!>          M is INTEGER
!>           On entry, M specifies the number of rows of the matrix A.
!>           M must be at least zero.
!> 


N


!>          N is INTEGER
!>           On entry, N specifies the number of columns of the matrix A.
!>           N must be at least zero.
!> 


KL


!>          KL is INTEGER
!>           On entry, KL specifies the number of sub-diagonals of the
!>           matrix A. KL must satisfy  0 .le. KL.
!> 


KU


!>          KU is INTEGER
!>           On entry, KU specifies the number of super-diagonals of the
!>           matrix A. KU must satisfy  0 .le. KU.
!> 


ALPHA


!>          ALPHA is COMPLEX
!>           On entry, ALPHA specifies the scalar alpha.
!> 


A


!>          A is COMPLEX array, dimension ( LDA, N )
!>           Before entry, the leading ( kl + ku + 1 ) by n part of the
!>           array A must contain the matrix of coefficients, supplied
!>           column by column, with the leading diagonal of the matrix in
!>           row ( ku + 1 ) of the array, the first super-diagonal
!>           starting at position 2 in row ku, the first sub-diagonal
!>           starting at position 1 in row ( ku + 2 ), and so on.
!>           Elements in the array A that do not correspond to elements
!>           in the band matrix (such as the top left ku by ku triangle)
!>           are not referenced.
!>           The following program segment will transfer a band matrix
!>           from conventional full matrix storage to band storage:
!>
!>                 DO 20, J = 1, N
!>                    K = KU + 1 - J
!>                    DO 10, I = MAX( 1, J - KU ), MIN( M, J + KL )
!>                       A( K + I, J ) = matrix( I, J )
!>              10    CONTINUE
!>              20 CONTINUE
!> 


LDA


!>          LDA is INTEGER
!>           On entry, LDA specifies the first dimension of A as declared
!>           in the calling (sub) program. LDA must be at least
!>           ( kl + ku + 1 ).
!> 


X


!>          X is COMPLEX array, dimension at least
!>           ( 1 + ( n - 1 )*abs( INCX ) ) when TRANS = 'N' or 'n'
!>           and at least
!>           ( 1 + ( m - 1 )*abs( INCX ) ) otherwise.
!>           Before entry, the incremented array X must contain the
!>           vector x.
!> 


INCX


!>          INCX is INTEGER
!>           On entry, INCX specifies the increment for the elements of
!>           X. INCX must not be zero.
!> 


BETA


!>          BETA is COMPLEX
!>           On entry, BETA specifies the scalar beta. When BETA is
!>           supplied as zero then Y need not be set on input.
!> 


Y


!>          Y is COMPLEX array, dimension at least
!>           ( 1 + ( m - 1 )*abs( INCY ) ) when TRANS = 'N' or 'n'
!>           and at least
!>           ( 1 + ( n - 1 )*abs( INCY ) ) otherwise.
!>           Before entry, the incremented array Y must contain the
!>           vector y. On exit, Y is overwritten by the updated vector y.
!>           If either m or n is zero, then Y not referenced and the function
!>           performs a quick return.
!> 


INCY


!>          INCY is INTEGER
!>           On entry, INCY specifies the increment for the elements of
!>           Y. INCY must not be zero.
!>
