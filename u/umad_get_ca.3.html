UMAD_GET_CA(3) Manual del programador de OpenIB UMAD_GET_CA(3)

NOMBRE

umad_get_ca, umad_release_ca - obtener y liberar atributos de puerto de dispositivo InfiniBand

SINOPSIS

#include <infiniband/umad.h>

int umad_get_ca(char *ca_name, umad_ca_t *ca);

int umad_release_ca(umad_ca_t *ca);

DESCRIPCIÓN

umad_get_ca() obtiene los atributos del dispositivo InfiniBand ca_name. Llena la estructura ca con los atributos del dispositivo especificados por ca_name o con los atributos del dispositivo por defecto si ca_name es NULL. umad_release_ca() debe llamarse antes de que la estructura ca sea desasignada. El argumento ca es una estructura umad_ca_t, como se especifica en <infiniband/umad.h>.

typedef struct umad_ca {
char ca_name[UMAD_CA_NAME_LEN];                 /* Nombre del dispositivo */
uint node_type;                                 /* Tipo del dispositivo */
int numports;                                   /* Número de puertos físicos */
char fw_ver[20];                                /* Versión del firmware */
char ca_type[40];                               /* Tipo de CA (ej. MT23108, etc.) */
char hw_ver[20];                                /* Versión de hardware */
uint64_t node_guid;                             /* GUID del nodo */
uint64_t system_guid;                           /* GUID de la imagen del sistema */
umad_port_t *ports[UMAD_CA_MAX_PORTS];          /* Array de propiedades de puertos del dispositivo */
} umad_ca_t;

umad_release_ca() libera los recursos que fueron asignados en la función umad_get_ca().

VALOR DE RETORNO

umad_get_ca() y umad_release_ca() devuelven 0 en caso de éxito y un valor negativo en caso de error.
