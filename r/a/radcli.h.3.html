struct send_data {
  uint8_t      code;          // Código del paquete RADIUS. 
  uint8_t      seq_nbr;       // Número de secuencia del paquete. 
  char        *server;        // Nombre/dirección del servidor RADIUS. 
  int          svc_port;      // Puerto de destino del protocolo RADIUS. 
  char        *secret;        // Secreto compartido del servidor RADIUS. 
  int          timeout;       // Tiempo de espera de la sesión en segundos. 
  int          retries;
  VALUE_PAIR  *send_pairs;    // Más pares a/v para enviar. 
  VALUE_PAIR  *receive_pairs; // Dónde colocar los pares a/v recibidos. 
};

struct rc_value_pair {
  char                   name[RC_NAME_LENGTH+1];      // nombre del atributo si se conoce. 
  uint64_t               attribute; // valor numérico del atributo de tipo rc_attr_id, incluido el proveedor; use VENDOR() y ATTRID() para separar. 
  rc_attr_type           type;      // tipo de atributo. 
  uint32_t               lvalue;    // valor del atributo si el tipo es PW_TYPE_INTEGER, PW_TYPE_DATE o PW_TYPE_IPADDR. 
  char                   strvalue[AUTH_STRING_LEN+1];  // contiene el valor del atributo en otros casos. 
  struct rc_value_pair  *next;
  char                   pad[32];       // pad no usado 
};
