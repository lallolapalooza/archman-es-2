NOMBRE

RPCSEC_GSS — autenticación basada en GSS-API para RPC

SINOPSIS

#include <rpc/rpcsec_gss.h>

DESCRIPCIÓN

RPCSEC_GSS es un mecanismo de seguridad para el protocolo RPC. Utiliza la Generic Security Service API (GSS-API) para establecer un contexto de seguridad entre un cliente y un servidor y para asegurar que toda la comunicación subsiguiente entre cliente y servidor sea autenticada correctamente. Opcionalmente, se puede aplicar protección adicional a la conexión. El servicio de integridad utiliza sumas de verificación para asegurar que todos los datos enviados por un par se reciban sin modificación. El servicio de privacidad utiliza cifrado para asegurar que ningún tercero pueda acceder a los datos de una conexión.

Para utilizar este sistema, una aplicación debe primero usar rpc_gss_seccreate() para establecer un contexto de seguridad.

ESTRUCTURAS DE DATOS

Las estructuras de datos utilizadas por RPCSEC_GSS se muestran a continuación.

rpc_gss_service_t
Este tipo define los tipos de servicio de seguridad requeridos para rpc_gss_seccreate(). 
typedef enum {
	rpc_gss_svc_default	= 0,
	rpc_gss_svc_none	= 1,
	rpc_gss_svc_integrity	= 2,
	rpc_gss_svc_privacy	= 3
} rpc_gss_service_t;
rpc_gss_options_ret_t
Esta estructura contiene varios valores opcionales que se utilizan al crear un contexto de seguridad. 
typedef struct {
	int		req_flags;	/* Banderas de solicitud GSS */
	int		time_req;	/* vida útil solicitada */
	gss_cred_id_t	my_cred;	/* credencial GSS */
	gss_channel_bindings_t input_channel_bindings;
} rpc_gss_options_req_t;
rpc_gss_options_ret_t
Varios detalles del contexto de seguridad creado se devuelven utilizando esta estructura. 
typedef struct {
	int		major_status;
	int		minor_status;
	u_int		rpcsec_version;
	int		ret_flags;
	int		time_req;
	gss_ctx_id_t	gss_context;
	char		actual_mechanism[MAX_GSS_MECH];
} rpc_gss_options_ret_t;
rpc_gss_principal_t
Este tipo se utiliza para referirse a un principal de cliente representado en formato de nombre exportado de GSS-API (ver gss_export_name(3) para más detalles). Los nombres en este formato pueden almacenarse en listas de control de acceso o compararse con otros nombres en formato de nombre exportado. Esta estructura es devuelta por rpc_gss_get_principal_name() y también se hace referencia a ella en la estructura rpc_gss_rawcred_t. 
typedef struct {
	int		len;
	char		name[1];
} *rpc_gss_principal_t;
rpc_gss_rawcred_t
Esta estructura se utiliza para acceder a las credenciales brutas asociadas con un contexto de seguridad. 
typedef struct {
	u_int		version;	/* Número de versión RPC */
	const char	*mechanism;	/* mecanismo de seguridad */
	const char	*qop;		/* calidad de protección */
	rpc_gss_principal_t client_principal; /* nombre del cliente */
	const char	*svc_principal;	/* nombre del servidor */
	rpc_gss_service_t service;	/* tipo de servicio */
} rpc_gss_rawcred_t;
rpc_gss_ucred_t
Credenciales de Unix que se derivan de las credenciales brutas, accedidas a través de rpc_gss_getcred(). 
typedef struct {
	uid_t		uid;		/* ID de usuario */
	gid_t		gid;		/* ID de grupo */
	short		gidlen;
	gid_t		*gidlist;	/* lista de grupos */
} rpc_gss_ucred_t;
rpc_gss_lock_t
Estructura utilizada para forzar una QOP y servicio particulares. 
typedef struct {
	bool_t		locked;
	rpc_gss_rawcred_t *raw_cred;
} rpc_gss_lock_t;
rpc_gss_callback_t
Estructura de devolución de llamada utilizada por rpc_gss_set_callback(). 
typedef struct {
	u_int		program;	/* número de programa RPC */
	u_int		version;	/* número de versión RPC */
					/* devolución de llamada definida por el usuario */
	bool_t		(*callback)(struct svc_req *req,
				    gss_cred_id_t deleg,
				    gss_ctx_id_t gss_context,
				    rpc_gss_lock_t *lock,
				    void **cookie);
} rpc_gss_callback_t;
rpc_gss_error_t
Estructura utilizada para devolver información de error por rpc_gss_get_error(). 
typedef struct {
	int		rpc_gss_error;
	int		system_error;	/* igual que errno */
} rpc_gss_error_t;

/*
 * Valores para rpc_gss_error
 */
#define RPC_GSS_ER_SUCCESS	0	/* sin error */
#define RPC_GSS_ER_SYSTEMERROR	1	/* error del sistema */
