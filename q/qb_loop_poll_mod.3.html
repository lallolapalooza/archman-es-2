NOMBRE

qb_loop_poll_mod - Modificar un trabajo de sondeo.

SINOPSIS

#include <qb/qbloop.h>

int32_t qb_loop_poll_mod(
    qb_loop_t                *l,            /* puntero a la instancia del bucle */
    enum qb_loop_priority     p,            /* la prioridad */
    int32_t                   fd,           /* descriptor de archivo. */
    int32_t                   events,       /* (POLLIN|POLLOUT) etc .... */
    void                     *data,         /* datos de usuario pasados a la función de despacho */
    qb_loop_poll_dispatch_fn  dispatch_fn   /* función de devolución de llamada */
);

PARÁMETROS

l puntero a la instancia del bucle

p la prioridad

fd descriptor de archivo.

events (POLLIN|POLLOUT) etc ....

data datos de usuario pasados a la función de despacho

dispatch_fn función de devolución de llamada

DESCRIPCIÓN

ESTRUCTURAS


enum qb_loop_priority {


      QB_LOOP_LOW;

      QB_LOOP_MED;

      QB_LOOP_HIGH;
};

VALOR DE RETORNO

estado (0 == ok, -errno == fallo)
