NOMBRE

qb_loop_signal_del - Eliminar el trabajo de señal.

SINOPSIS

#include <qb/qbloop.h>

int32_t qb_loop_signal_del(
    qb_loop_t             *l,       /* puntero a la instancia del bucle */
    qb_loop_signal_handle  handle   /* (entrada) una referencia al trabajo de señal */
);

PARÁMETROS

l puntero a la instancia del bucle

handle (entrada) una referencia al trabajo de señal

DESCRIPCIÓN

VALOR DE RETORNO

estado (0 == ok, -errno == fallo)
