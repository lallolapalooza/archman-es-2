qb_log_filter_ctl(3)

Paquete: libqb

Licencias: LGPL-2.1-or-later

QB_LOG_FILTER_CTL(3) Manual del programador de libqb QB_LOG_FILTER_CTL(3)

NOMBRE

qb_log_filter_ctl - Esto le permite modificar los campos de llamada 'tags' y 'targets' en tiempo de ejecución.

SINOPSIS

#include <qb/qblog.h>

int32_t qb_log_filter_ctl(
    int32_t                  value,
    enum qb_log_filter_conf  c,
    enum qb_log_filter_type  type,
    const char              *text,
    uint8_t                  low_priority
);

ESTRUCTURAS


enum qb_log_filter_conf {


      QB_LOG_FILTER_ADD;

      QB_LOG_FILTER_REMOVE;

      QB_LOG_FILTER_CLEAR_ALL;

      QB_LOG_TAG_SET;

      QB_LOG_TAG_CLEAR;

      QB_LOG_TAG_CLEAR_ALL;
};


enum qb_log_filter_type {


      QB_LOG_FILTER_FILE;

      QB_LOG_FILTER_FUNCTION;

      QB_LOG_FILTER_FORMAT;

      QB_LOG_FILTER_FILE_REGEX;

      QB_LOG_FILTER_FUNCTION_REGEX;

      QB_LOG_FILTER_FORMAT_REGEX;
};
