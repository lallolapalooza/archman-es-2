NOMBRE

qb_ipcs_connection_stats_get - Obtiene las estadísticas de la conexión.

SINOPSIS

#include <qb/qbipcs.h>

int32_t qb_ipcs_connection_stats_get(
    qb_ipcs_connection_t              *c,                 /* instancia de conexión */
    struct qb_ipcs_connection_stats   *stats,             /* (salida) la estructura de estadísticas */
    int32_t                            clear_after_read   /* limpia las estadísticas después de copiarlas en stats */
);

PARÁMETROS

c instancia de conexión

stats (salida) la estructura de estadísticas

clear_after_read limpia las estadísticas después de copiarlas en stats

DESCRIPCIÓN

ESTRUCTURAS


struct qb_ipcs_connection_stats {


    int32_t   client_pid;

    uint64_t  requests;

    uint64_t  responses;

    uint64_t  events;

    uint64_t  send_retries;

    uint64_t  recv_retries;

    int32_t   flow_control_state;

    uint64_t  flow_control_count;
};

VALOR DE RETORNO

0 == ok; -errno para indicar un fallo
