DESCRIPCIÓ

N

La función kbuffer_timestamp() devuelve la marca de tiempo del evento actual de kbuf

.

La función kbuffer_subbuf_timestamp() devuelve la marca de tiempo del sub-búfer que fue cargado en kbuf. Esto generalmente es (pero no está garantizado que sea) la marca de tiempo del primer evento en el

sub-búfer.

La función kbuffer_start_of_data() devuelve el

desplazamiento de
donde el delta

VALOR

DEVUELTO

kbuffer_read_event() devuelve el evento al que el descriptor kbuf está apuntando actualmente, o NULL si el último evento ya fue pasado (por kbuffer_next_event())

.

kbuffer_next_event() devuelve el siguiente evento después del evento actual o NULL si no hay más

eventos.

kbuffer_read_at_offset() devuelve el evento en un desplazamiento dado desde el inicio del sub-búfer almacenado en kbuf, o NULL si no existe ningún evento. Nótese, offset solo necesita ser un desplazamiento que caiga en el registro, o esté en su inicio. No necesita estar exactamente en el comienzo del

registro.

kbuffer_missed_events() devuelve 0 si no hubo eventos perdidos antes del sub-búfer cargado. Devuelve -1 si hubo un número desconocido de eventos perdidos, o si el número de eventos perdidos es conocido, ese número será

devuelto.

kbuffer_event_size() devuelve el tamaño de la carga útil de datos del evento actual de kbuf

.

kbuffer_curr_size() devuelve el tamaño del registro completo del evento actual de kbuf. Esto incluye el tamaño de los metadatos de ese

registro.

kbuf_curr_offset() devuelve el desplazamiento del registro actual desde el inicio del sub-búfer

kbuf.

kbuf_curr_index() devuelve el índice del registro actual desde el inicio de la sección de

datos
kbuf.

EJEMPL

O 
#include <stdio.h>
#include <stdlib.h>
#include <fcntl.h>
#include <unistd.h>
#include <sys/stat.h>
#include <kbuffer.h>
int main (int argc, char **argv)
{
        unsigned long long ts;
        struct kbuffer *kbuf;
        struct stat st;
        char *buf;
        void *event;
        int save_offset = -1;
        int record_size;
        int offset;
        int index;
        int size;
        int ret;
        int fd;
        int i = 0;
        if (argc < 2) {
                printf("usage: %s raw-subbuffer-page\n", argv[0]);
                printf(" Try: dd count=1 bs=4096 if=/sys/kernel/tracing/per_cpu/cpu0/trace_pipe_raw of=/tmp/file\n");
                exit(0);
        }
        if (stat(argv[1], &st) < 0) {
                perror("stat");
                exit(-1);
        }
        buf = malloc(st.st_size);
        if (!buf) {
                perror("Allocating buffer");
                exit(-1);
        }
        fd = open(argv[1], O_RDONLY);
        if (fd < 0) {
                perror(argv[1]);
                exit(-1);
        }
        ret = read(fd, buf, st.st_size);
        if (ret < 0) {
                perror("Reading buffer");
                exit(-1);
        }
        close(fd);
        kbuf = kbuffer_alloc(KBUFFER_ENDIAN_SAME_AS_HOST,
                             KBUFFER_LSIZE_SAME_AS_HOST);
        if (!kbuf) {
                perror("Creating kbuffer");
                exit(-1);
        }
        ret = kbuffer_load_subbuffer(kbuf, buf);
        if (ret < 0) {
                perror("Loading sub bufer");
                exit(-1);
        }
        if (kbuffer_subbuffer_size(kbuf) > st.st_size) {
                fprintf(stderr, "kbuffer is bigger than raw size %d > %ld\n",
                        kbuffer_subbuffer_size(kbuf), st.st_size);
                exit(-1);
        }
        ret = kbuffer_missed_events(kbuf);
        if (ret) {
                if (ret > 0)
                        printf("Missed %d events before this buffer\n", ret);
                else
                        printf("Missed unknown number of events before this buffer\n");
        }
        do {
                event = kbuffer_read_event(kbuf, &ts);
                if (event) {
                        record_size = kbuffer_curr_size(kbuf);
                        offset = kbuffer_curr_offset(kbuf);
                        index = kbuffer_curr_index(kbuf);
                        size = kbuffer_event_size(kbuf);
                        if (i == 20)
                                save_offset = offset;
                        printf(" event %3d ts:%lld\trecord_size:%d size:%d\tindex:%d offset:%d\n",
                               i++, ts, record_size, size, index, offset);
                        event = kbuffer_next_event(kbuf, NULL);
                }
        } while (event);
        if (!event)
                printf("Finished sub buffer\n");
        if (save_offset > 0) {
                event = kbuffer_read_at_offset(kbuf, save_offset, &ts);
                if (!event) {
                        fprintf(stderr, "Funny, can't find event 20 at offset %d\n", save_offset);
                        exit(-1);
                }
                record_size = kbuffer_curr_size(kbuf);
                offset = kbuffer_curr_offset(kbuf);
                index = kbuffer_curr_index(kbuf);
                size = kbuffer_event_size(kbuf);
                printf("\n saved event 20 ts:%lld\trecord_size:%d size:%d\tindex:%d offset:%d\n\n",
                       ts, record_size, size, index, offset);
        }
        kbuffer_free(kbuf);
        return
0;
}
