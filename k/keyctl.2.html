DESCRIPCIÓ

N

keyctl() permite a los programas de espacio de usuario realizar manipulaciones de

claves.

La operación realizada por keyctl() está determinada por el valor del argumento op. Cada una de estas operaciones está envuelta por la biblioteca libkeyutils (proporcionada por el paquete keyutils) en funciones individuales (véase keyctl(3)) para permitir que el compilador verifique los

tipos.

EJEMPLO

S

El programa siguiente proporciona un subconjunto de la funcionalidad del programa request-key(8) proporcionado por el paquete keyutils. Para fines informativos, el programa registra varios datos en un archivo de

registro.

Como se describe en request_key(2), el programa request-key(8) se invoca con argumentos de línea de comandos que describen una clave que va a ser instanciada. El programa de ejemplo recupera y registra estos argumentos. El programa asume la autoridad para instanciar la clave solicitada, y luego instancia esa

clave.

La siguiente sesión de shell demuestra el uso de este programa. En la sesión, compilamos el programa y luego lo usamos para reemplazar temporalmente el programa estándar request-key(8). (Tenga en cuenta que deshabilitar temporalmente el programa estándar request-key(8) puede no ser seguro en algunos sistemas.) Mientras nuestro programa de ejemplo está instalado, usamos el programa de ejemplo mostrado en request_key(2) para solicitar una

clave. 


$ cc -o key_instantiate key_instantiate.c -lkeyutils;
$ sudo mv /sbin/request-key /sbin/request-key.backup;
$ sudo cp key_instantiate /sbin/request-key;
$ ./t_request_key user mykey somepayloaddata;
Key ID is 20d035bf
$ sudo mv /sbin/request-key.backup /sbin/request-key
;

Programa

fuente 
/* key_instantiate.c */
#include <errno.h>
#include <keyutils.h>
#include <stdint.h>
#include <stdio.h>
#include <stdlib.h>
#include <string.h>
#include <sys/types.h>
#include <time.h>
#ifndef KEY_SPEC_REQUESTOR_KEYRING
#define KEY_SPEC_REQUESTOR_KEYRING      (-8)
#endif
int
main(int argc, char *argv[])
{
    int           akp_size;       /* Tamaño de auth_key_payload */
    int           auth_key;
    char          dbuf[256];
    char          auth_key_payload[256];
    char          *operation;
    FILE          *fp;
    gid_t         gid;
    uid_t         uid;
    time_t        t;
    key_serial_t  key_to_instantiate, dest_keyring;
    key_serial_t  thread_keyring, process_keyring, session_keyring;
    if (argc != 8) {
        fprintf(stderr, "Usage: %s op key uid gid thread_keyring "
                        "process_keyring session_keyring\n", argv[0]);
        exit(EXIT_FAILURE);
    }
    fp = fopen("/tmp/key_instantiate.log", "w");
    if (fp == NULL)
        exit(EXIT_FAILURE);
    setbuf(fp, NULL);
    t = time(NULL);
    fprintf(fp, "Time: %s\n", ctime(&t));
    /*
     * El kernel pasa un conjunto fijo de argumentos al programa
     * que ejecuta; obtenerlos.
     */
    operation = argv[1];
    key_to_instantiate = atoi(argv[2]);
    uid = atoi(argv[3]);
    gid = atoi(argv[4]);
    thread_keyring = atoi(argv[5]);
    process_keyring = atoi(argv[6]);
    session_keyring = atoi(argv[7]);
    fprintf(fp, "Command line arguments:\n");
    fprintf(fp, "  argv[0]:            %s\n", argv[0]);
    fprintf(fp, "  operation:          %s\n", operation);
    fprintf(fp, "  key_to_instantiate: %jx\n",
            (uintmax_t) key_to_instantiate);
    fprintf(fp, "  UID:                %jd\n", (intmax_t) uid);
    fprintf(fp, "  GID:                %jd\n", (intmax_t) gid);
    fprintf(fp, "  thread_keyring:     %jx\n",
            (uintmax_t) thread_keyring);
    fprintf(fp, "  process_keyring:    %jx\n",
            (uintmax_t) process_keyring);
    fprintf(fp, "  session_keyring:    %jx\n",
            (uintmax_t) session_keyring);
    fprintf(fp, "\n");
    /*
     * Asumir la autoridad para instanciar la clave nombrada en argv[2].
     */
    if (keyctl(KEYCTL_ASSUME_AUTHORITY, key_to_instantiate) == -1) {
        fprintf(fp, "KEYCTL_ASSUME_AUTHORITY failed: %s\n",
                strerror(errno));
        exit(EXIT_FAILURE);
    }
    /*
     * Obtener la descripción de la clave que se va a instanciar.
     */
    if (keyctl(KEYCTL_DESCRIBE, key_to_instantiate,
               dbuf, sizeof(dbuf)) == -1) {
        fprintf(fp, "KEYCTL_DESCRIBE failed: %s\n", strerror(errno));
        exit(EXIT_FAILURE);
    }
    fprintf(fp, "Key description:      %s\n", dbuf);
    /*
     * Obtener la carga útil de la clave de autorización, que es
     * en realidad los datos de llamada dados a request_key().
     */
    akp_size = keyctl(KEYCTL_READ, KEY_SPEC_REQKEY_AUTH_KEY,
                      auth_key_payload, sizeof(auth_key_payload));
    if (akp_size == -1) {
        fprintf(fp, "KEYCTL_READ failed: %s\n", strerror(errno));
        exit(EXIT_FAILURE);
    }
    auth_key_payload[akp_size] = '\0';
    fprintf(fp, "Auth key payload:     %s\n", auth_key_payload);
    /*
     * Por interés, obtener el ID de la clave de autorización y
     * mostrarlo.
     */
    auth_key = keyctl(KEYCTL_GET_KEYRING_ID,
                      KEY_SPEC_REQKEY_AUTH_KEY);
    if (auth_key == -1) {
        fprintf(fp, "KEYCTL_GET_KEYRING_ID failed: %s\n",
                strerror(errno));
        exit(EXIT_FAILURE);
    }
    fprintf(fp, "Auth key ID:          %jx\n", (uintmax_t) auth_key);
    /*
     * Obtener el ID de la clave para el anillo de claves de destino de request_key(2).
     */
    dest_keyring = keyctl(KEYCTL_GET_KEYRING_ID,
                          KEY_SPEC_REQUESTOR_KEYRING);
    if (dest_keyring == -1) {
        fprintf(fp, "KEYCTL_GET_KEYRING_ID failed: %s\n",
                strerror(errno));
        exit(EXIT_FAILURE);
    }
    fprintf(fp, "Destination keyring:  %jx\n", (uintmax_t) dest_keyring);
    /*
     * Obtener la descripción de la clave de autorización. Esto
     * nos permite ver el tipo de clave, UID, GID, permisos
     * y descripción (nombre) de la clave. Entre otras cosas,
     * veremos que el nombre de la clave es una cadena hexadecimal
     * que representa el ID de la clave a instanciar.
     */
    if (keyctl(KEYCTL_DESCRIBE, KEY_SPEC_REQKEY_AUTH_KEY,
               dbuf, sizeof(dbuf)) == -1)
    {
        fprintf(fp, "KEYCTL_DESCRIBE failed: %s\n", strerror(errno));
        exit(EXIT_FAILURE);
    }
    fprintf(fp, "Auth key description: %s\n", dbuf);
    /*
     * Instanciar la clave usando los datos de llamada que se proporcionaron
     * en la carga útil de la clave de autorización.
     */
    if (keyctl(KEYCTL_INSTANTIATE, key_to_instantiate,
               auth_key_payload, akp_size + 1, dest_keyring) == -1)
    {
        fprintf(fp, "KEYCTL_INSTANTIATE failed: %s\n",
                strerror(errno));
        exit(EXIT_FAILURE);
    }
    exit(EXIT_SUCCESS)
;
}
