NOMBRE

libcaca-canvas - El formato de lienzo de libcaca (versión 1)

Todos los tipos son big endian.

struct
{
magic:
   uint8_t caca_header[2];    // ""\xCA\xCA""
   uint8_t caca_file_type[2]; // ""CV""
canvas_header:
   uint32_t control_size;     // Tamaño del control (datos del lienzo - encabezado del lienzo)
   uint32_t data_size;        // Tamaño de los datos (EOF - datos del lienzo)
   uint16_t version;          // Versión del formato del lienzo
                              //  bit 0: establecido a 1 si el lienzo es compatible
                              //         con la versión 1 del formato
                              //  bits 1-15: aún no utilizados, deben ser 0
   uint32_t frames;           // Número de fotogramas
   uint16_t flags;            // Banderas de características
                              //  bits 0-15: aún no utilizados, deben ser 0
frame_info:
   struct
   {
      uint32_t width;         // Ancho del fotograma
      uint32_t height;        // Alto del fotograma
      uint32_t duration;      // Duración del fotograma en milisegundos, 0 para
                              // no especificar una duración
      uint32_t attr;          // Atributo del contexto gráfico
      int32_t cursor_x;       // Coordenada X del cursor
      int32_t cursor_y;       // Coordenada Y del cursor
      int32_t handle_x;       // Coordenada X del manejador
      int32_t handle_y;       // Coordenada Y del manejador
   }
   frame_list[frames];
control_extension_1:
control_extension_2:
   ...
control_extension_N:
   ...                        // reservado para uso futuro
canvas_data:
   uint8_t data[data_size];   // datos del lienzo
data_extension_1:
data_extension_2:
   ...
data_extension_N:
   ...                        // reservado para uso futuro
};
