NOMBRE

biblioteca “libcrypto”
#include <openssl/evp.h>
#include <openssl/kdf.h>

int
EVP_PKEY_CTX_hkdf_mode(EVP_PKEY_CTX *pctx, int mode);

int
EVP_PKEY_CTX_set_hkdf_md(EVP_PKEY_CTX *pctx, const EVP_MD *md);

int
EVP_PKEY_CTX_set1_hkdf_salt(EVP_PKEY_CTX *pctx, unsigned char *salt, int saltlen);

int
EVP_PKEY_CTX_set1_hkdf_key(EVP_PKEY_CTX *pctx, unsigned char *key, int keylen);

int
EVP_PKEY_CTX_add1_hkdf_info(EVP_PKEY_CTX *pctx, unsigned char *info, int infolen);

DESCRIPCIÓN

El algoritmo EVP_PKEY_HKDF implementa la función de derivación de claves HKDF. HKDF sigue el paradigma de "extraer-luego-expandir", donde la KDF consta lógicamente de dos módulos. La primera etapa toma el material de clave de entrada y "extrae" de él una clave pseudorandom de longitud fija K. La segunda etapa "expande" la clave K en varias claves pseudorandom adicionales (la salida de la KDF).

EVP_PKEY_CTX_hkdf_mode() establece el modo para la operación HKDF. Actualmente hay tres modos definidos:

EVP_PKEY_HKDEF_MODE_EXTRACT_AND_EXPAND
Este es el modo predeterminado. Llamar a EVP_PKEY_derive(3) en un EVP_PKEY_CTX configurado para HKDF realizará una operación de extracción seguida de una operación de expansión en un solo paso. La clave derivada devuelta será el resultado después de la operación de expansión. La clave pseudorandom intermedia de longitud fija K no se devuelve.

En este modo, el digest, la clave, la sal y la información deben establecerse antes de derivar una clave o se producirá un error.

EVP_PKEY_HKDEF_MODE_EXTRACT_ONLY
En este modo, llamar a EVP_PKEY_derive(3) solo realizará la operación de extracción. El valor devuelto será la clave pseudorandom intermedia de longitud fija K.

El digest, la clave y la sal deben establecerse antes de derivar una clave o se producirá un error.

EVP_PKEY_HKDEF_MODE_EXPAND_ONLY
En este modo, llamar a EVP_PKEY_derive(3) solo realizará la operación de expansión. La clave de entrada debe establecerse en la clave pseudorandom intermedia de longitud fija K devuelta por una operación de extracción anterior.

El digest, la clave y la información deben establecerse antes de derivar una clave o se producirá un error.

EVP_PKEY_CTX_set_hkdf_md() establece el digest de mensaje asociado con el HKDF.

EVP_PKEY_CTX_set1_hkdf_salt() establece la sal en saltlen bytes del búfer salt. Se reemplaza cualquier valor existente.

EVP_PKEY_CTX_set1_hkdf_key() establece la clave en keylen bytes del búfer key. Se reemplaza cualquier valor existente.

EVP_PKEY_CTX_add1_hkdf_info() establece el valor de información en infolen bytes del búfer info. Si ya hay un valor establecido, se añade al valor existente.

CONTROLES DE CADENA

HKDF también admite operaciones de control basadas en cadenas a través de EVP_PKEY_CTX_ctrl_str(3). El parámetro type "md" utiliza el value proporcionado como el nombre del algoritmo de digest a utilizar. El parámetro type "mode" acepta "EXTRACT_AND_EXPAND", "EXTRACT_ONLY" y "EXPAND_ONLY" como value para determinar el modo a utilizar. Los parámetros type "salt", "key" e "info" utilizan el parámetro value proporcionado como semilla, clave o información. Los nombres "hexsalt", "hexkey" y "hexinfo" son similares, excepto que toman una cadena hexadecimal que se convierte a binario.

NOTAS

Todas estas funciones se implementan como macros.

Se puede obtener un contexto para HKDF llamando a:

EVP_PKEY_CTX *pctx = EVP_PKEY_CTX_new_id(EVP_PKEY_HKDF, NULL);

La longitud total del búfer de información no puede exceder los 1024 bytes: esto debería ser más que suficiente para cualquier uso normal de HKDF.

La longitud de salida de una operación de expansión HKDF se especifica a través del parámetro de longitud de la función EVP_PKEY_derive(3). Dado que la longitud de salida de HKDF es variable, pasar un búfer NULL como medio para obtener la longitud requerida no es significativo con HKDF en ningún modo que realice una operación de expansión. En su lugar, el llamador debe asignar un búfer de la longitud deseada y pasarlo a EVP_PKEY_derive(3) junto con (un puntero inicializado a) la longitud deseada. Pasar un búfer NULL para obtener la longitud está permitido cuando se usa EVP_PKEY_HKDEF_MODE_EXTRACT_ONLY.

VALORES DE RETORNO

Todas estas funciones devuelven 1 en caso de éxito y 0 o un valor negativo en caso de error. En particular, un valor de retorno de -2 indica que la operación no es compatible con el algoritmo de clave pública.

EJEMPLOS

Este ejemplo deriva 10 bytes usando SHA-256 con la clave secreta "secret", el valor de sal "salt" y el valor de información "label":

EVP_PKEY_CTX *pctx;
unsigned char out[10];
size_t outlen = sizeof(out);

if ((pctx = EVP_PKEY_CTX_new_id(EVP_PKEY_HKDF, NULL)) == NULL)
	/* Error */

if (EVP_PKEY_derive_init(pctx) <= 0)
	/* Error */
if (EVP_PKEY_CTX_set_hkdf_md(pctx, EVP_sha256()) <= 0)
	/* Error */
if (EVP_PKEY_CTX_set1_hkdf_salt(pctx, "salt", 4) <= 0)
	/* Error */
if (EVP_PKEY_CTX_set1_hkdf_key(pctx, "secret", 6) <= 0)
	/* Error */
if (EVP_PKEY_CTX_add1_hkdf_info(pctx, "label", 5) <= 0)
	/* Error */
if (EVP_PKEY_derive(pctx, out, &outlen) <= 0)
	/* Error */

ESTÁNDARES

RFC 5869

libressl-EVP_PKEY_CTX_set_hkdf_md.3

NAME

EVP_PKEY_CTX_set_hkdf_md, EVP_PKEY_CTX_set1_hkdf_salt, EVP_PKEY_CTX_set1_hkdf_key, EVP_PKEY_CTX_add1_hkdf_info, EVP_PKEY_CTX_hkdf_mode - HKDF (HMAC-based Extract-and-Expand Key Derivation Function) key generation context control

SYNOPSIS

#include <openssl/evp.h>

int EVP_PKEY_CTX_set_hkdf_md(EVP_PKEY_CTX *ctx, const EVP_MD *md);
int EVP_PKEY_CTX_set1_hkdf_salt(EVP_PKEY_CTX *ctx, const unsigned char *salt,
                              int len);
int EVP_PKEY_CTX_set1_hkdf_key(EVP_PKEY_CTX *ctx, const unsigned char *key,
                             int len);
int EVP_PKEY_CTX_add1_hkdf_info(EVP_PKEY_CTX *ctx, const unsigned char *info,
                              int len);
int EVP_PKEY_CTX_hkdf_mode(EVP_PKEY_CTX *ctx, int mode);

DESCRIPTION

The HKDF algorithm is a Key Derivation Function (KDF) specified in RFC 5869. It uses an underlying HMAC cryptographic primitive. HKDF has two distinct phases: an extraction phase and an expansion phase.

The extraction phase takes a possibly variable-length input keying material (IKM) and a salt, and uses the HMAC to derive a fixed-length pseudorandom key (PRK).

The expansion phase takes the PRK and expands it into a specified length of output keying material (OKM). It can also take optional application-specific information (info) as input.

The functions EVP_PKEY_CTX_set_hkdf_md(), EVP_PKEY_CTX_set1_hkdf_salt(), EVP_PKEY_CTX_set1_hkdf_key(), EVP_PKEY_CTX_add1_hkdf_info(), and EVP_PKEY_CTX_hkdf_mode() are used to configure an EVP_PKEY_CTX for use with HKDF.

EVP_PKEY_CTX_set_hkdf_md() sets the digest algorithm to be used for the underlying HMAC. The digest algorithm must be registered with OpenSSL (e.g., via EVP_add_md()). If md is NULL, the default digest (SHA1) is used.

EVP_PKEY_CTX_set1_hkdf_salt() sets the salt to be used for the extraction phase. The salt is a byte array of length len. If salt is NULL or len is 0, then a salt of all zeros is used (as per RFC 5869 section 3).

EVP_PKEY_CTX_set1_hkdf_key() sets the input keying material (IKM) to be used for the extraction phase. The IKM is a byte array of length len. This is the key that is being derived from. If key is NULL or len is 0, then a key of all zeros is used (as per RFC 5869 section 3).

EVP_PKEY_CTX_add1_hkdf_info() adds application-specific information (info) to be used in the expansion phase. This function can be called multiple times to concatenate different info components. The info parameter is a byte array of length len.

EVP_PKEY_CTX_hkdf_mode() sets the mode of operation. The only currently supported mode is EVP_PKEY_HKDF_MODE_EXTRACT_THEN_EXPAND, which performs both the extraction and expansion phases. Other modes may be added in the future.

STRING CTRLS

The following string controls are also available:

EVP_PKEY_CTRL_HKDF_MD

Sets the digest algorithm. The value is a pointer to an EVP_MD struct.

EVP_PKEY_CTRL_HKDF_SALT

Sets the salt. The value is a pointer to a structure containing the salt data and its length. The exact structure definition is implementation-dependent.

EVP_PKEY_CTRL_HKDF_KEY

Sets the input keying material (IKM). The value is a pointer to a structure containing the IKM data and its length. The exact structure definition is implementation-dependent.

EVP_PKEY_CTRL_HKDF_INFO

Adds application-specific information. The value is a pointer to a structure containing the info data and its length. This can be called multiple times.

EVP_PKEY_CTRL_HKDF_MODE

Sets the HKDF mode. The value is an integer representing the mode.

NOTES

The functions described here are intended to be used with EVP_PKEY_derive_init(), EVP_PKEY_derive_set_param(), and EVP_PKEY_derive().

The underlying digest algorithm set by EVP_PKEY_CTX_set_hkdf_md() determines the length of the pseudorandom key (PRK) generated in the extraction phase. For example, SHA256 produces a 32-byte PRK.

The length of the output keying material (OKM) is determined by the EVP_PKEY_CTX_set_derive_param() function, which should be called after setting the HKDF parameters.

RFC 5869 recommends that the salt should be at least 16 bytes long. If a salt is not provided, a salt of all zeros is used, which is generally not recommended for security reasons.

The info parameter is optional but recommended. It should be unique for each derived key and should reflect the context in which the key will be used.

RETURN VALUES

All the functions described in this page return 1 on success, and 0 or a negative value on failure.

EXAMPLES

/*
 * Example demonstrating the use of HKDF to derive a key.
 * This is a simplified example and should not be used in production
 * without proper error handling and security considerations.
 */
#include <openssl/evp.h>
#include <openssl/kdf.h>
#include <openssl/err.h>
#include <stdio.h>
#include <string.h>

int main(void) {
    EVP_PKEY_CTX *pctx, *kctx;
    unsigned char okm[32]; // Output keying material
    size_t okm_len = sizeof(okm);
    const char *ikm_str = "my-secret-input-key";
    const char *salt_str = "salt";
    const char *info_str = "my-application-context";
    int ret;

    // Setup for deriving a key
    pctx = EVP_PKEY_CTX_new(NULL, NULL);
    if (pctx == NULL) {
        fprintf(stderr, "EVP_PKEY_CTX_new failed\n");
        return 1;
    }

    if (EVP_PKEY_derive_init(pctx) <= 0) {
        fprintf(stderr, "EVP_PKEY_derive_init failed\n");
        goto err;
    }

    // Set HKDF parameters
    if (EVP_PKEY_CTX_set_hkdf_md(pctx, EVP_sha256()) <= 0) {
        fprintf(stderr, "EVP_PKEY_CTX_set_hkdf_md failed\n");
        goto err;
    }
    if (EVP_PKEY_CTX_set1_hkdf_salt(pctx, (unsigned char *)salt_str, strlen(salt_str)) <= 0) {
        fprintf(stderr, "EVP_PKEY_CTX_set1_hkdf_salt failed\n");
        goto err;
    }
    if (EVP_PKEY_CTX_set1_hkdf_key(pctx, (unsigned char *)ikm_str, strlen(ikm_str)) <= 0) {
        fprintf(stderr, "EVP_PKEY_CTX_set1_hkdf_key failed\n");
        goto err;
    }
    if (EVP_PKEY_CTX_add1_hkdf_info(pctx, (unsigned char *)info_str, strlen(info_str)) <= 0) {
        fprintf(stderr, "EVP_PKEY_CTX_add1_hkdf_info failed\n");
        goto err;
    }

    // Specify the desired output key length
    if (EVP_PKEY_derive_set_param(pctx, &okm_len) <= 0) {
        fprintf(stderr, "EVP_PKEY_derive_set_param failed\n");
        goto err;
    }

    // Derive the key
    ret = EVP_PKEY_derive(pctx, okm, &okm_len);
    if (ret <= 0) {
        fprintf(stderr, "EVP_PKEY_derive failed\n");
        goto err;
    }

    printf("Derived key (length %zu):\n", okm_len);
    for (size_t i = 0; i < okm_len; i++) {
        printf("%02x", okm[i]);
    }
    printf("\n");

    EVP_PKEY_CTX_free(pctx);
    return 0;

err:
    ERR_print_errors_fp(stderr);
    if (pctx) EVP_PKEY_CTX_free(pctx);
    return 1;
}
