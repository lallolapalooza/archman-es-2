libcaca-font - El formato de fuente de libcaca (versión 1)

Todos los tipos son big endian.

struct
{
magic:
   uint8_t caca_header[2];    // "\xCA\xCA"
   uint8_t caca_file_type[2]; // "FT"
font_header:
   uint32_t control_size;     // Tamaño del control (font_data - font_header)
   uint32_t data_size;        // Tamaño de los datos (EOF - font_data)
   uint16_t version;          // Versión del formato de fuente
                              //  bit 0: establecido a 1 si la fuente es compatible
                              //         con la versión 1 del formato
                              //  bits 1-15: no utilizados todavía, deben ser 0
   uint16_t blocks;           // Número de bloques en la fuente
   uint32_t glyphs;           // Número total de glifos en la fuente
   uint16_t bpp;              // Bits por píxel para los datos de glifo (válidos
                              // Valores son 1, 2, 4 y 8)
   uint16_t width;            // Ancho estándar del glifo
   uint16_t height;           // Altura estándar del glifo
   uint16_t maxwidth;         // Ancho máximo del glifo
   uint16_t maxheight;        // Altura máxima del glifo
   uint16_t flags;            // Indicadores de características
                              //  bit 0: establecido a 1 si la fuente es de ancho fijo
                              //  bits 1-15: no utilizados todavía, deben ser 0
block_info:
   struct
   {
      uint32_t start;         // Índice Unicode del primer glifo
      uint32_t stop;          // Índice Unicode del último glifo + 1
      uint32_t index;         // Índice de información de glifo del primer glifo
   }
   block_list[blocks];
glyph_info:
   struct
   {
      uint16_t width;         // Ancho del glifo en píxeles
      uint16_t height;        // Altura del glifo en píxeles
      uint32_t data_offset;   // Desplazamiento (comenzando desde los datos) a los datos
                              // para el primer carácter
   }
   glyph_list[glyphs];
control_extension_1:
control_extension_2:
   ...
control_extension_N:
   ...                        // reservado para uso futuro
font_data:
   uint8_t data[data_size];   // datos del glifo
data_extension_1:
data_extension_2:
   ...
data_extension_N:
   ...                        // reservado para uso futuro
};
