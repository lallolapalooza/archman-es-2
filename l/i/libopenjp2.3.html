NOMBRE

libopenjp2 - una biblioteca para leer y escribir archivos de imagen JPEG2000.

SINOPSIS

#include <openjpeg.h>

FORMATOS DE CONVERSIÓN

PGX: imagetopgx() / pgxtoimage()

PXM: imagetopnm() / pnmtoimage()

BMP: imagetobmp() / bmptoimage()

TIF: imagetotif() / tiftoimage()

RAW: imagetoraw() / rawtoimage()

TGA: imagetotga() / tgatoimage()

PNG: imagetopng() / pngtoimage()

YUV: imagetoyuv() / yuvtoimage() (MJ2)

PARÁMETROS DE DESCOMPRESIÓN

typedef struct opj_dparameters
{
/*
Establece el número de los niveles de resolución más altos a descartar.
La resolución de la imagen se divide efectivamente por 2 elevado a la potencia
del número de niveles descartados.
El factor de reducción está limitado por el número total más pequeño de
niveles de descomposición entre las teselas.
si != 0, entonces la dimensión original dividida por 2^(reduce);
si == 0 o no se usa, la imagen se decodifica a la resolución completa
*/
int cp_reduce;
/*
Establece el número máximo de capas de calidad a decodificar.
Si hay menos capas de calidad que el número especificado,
se decodifican todas las capas de calidad.
si != 0, entonces solo se decodifican las primeras "layer" capas;
si == 0 o no se usa, se decodifican todas las capas de calidad
*/
int cp_layer;


/* parámetros del codificador de línea de comandos (no utilizados dentro de la biblioteca) */
/* nombre del archivo de entrada */
char infile[OPJ_PATH_LEN];
/* nombre del archivo de salida */
char outfile[OPJ_PATH_LEN];
/* formato del archivo de entrada: ver OPJ_CODEC_FORMAT */
int decod_format;
/* formato del archivo de salida */
int cod_format;


/* Parámetros de decodificación JPWL */
/* activa las capacidades de corrección JPWL */
bool jpwl_correct;
/* número esperado de componentes */
int jpwl_exp_comps;
/* número máximo de teselas */
int jpwl_max_tiles;


/*
Especifica si la decodificación se realizará en el flujo de código completo,
o se limitará al encabezado principal.
Limitar la decodificación al encabezado principal permite
extraer las características del flujo de código.
si == NO_LIMITATION, se decodifica el flujo de código completo;
si == LIMIT_TO_MAIN_HEADER, solo se decodifica el encabezado principal;
*/
OPJ_LIMIT_DECODING cp_limit_decoding;
} opj_dparameters_t;
