NOMBRE

libopeniscsiusr.h - API de espacio de usuario iSCSI.

SINOPSIS

#include <libopeniscsiusr/libopeniscsiusr.h>

DESCRIPCIÓN

Todas las funciones públicas de libopeniscsiusr envían sus propias páginas de manual. Puede usar 'man -k iscsi' para averiguar y usar 'man 3 <nombre_función>' para verificar el uso detallado.

USO

Para usar libopeniscsiusr en su proyecto, sugerimos usar la forma 'pkg-config':


* Agregue esta línea a su configure.ac:


PKG_CHECK_MODULES([LIBISCSIUSR], [libopeniscsiusr])


* Agregue estas líneas a su Makefile.am:


foo_LDFLAGS += $(LIBISCSIUSR_LIBS)
foo_CFLAGS += $(LIBISCSIUSR_CFLAGS)

MANEJO DE REGISTROS

La función de manejo de registros se puede establecer a través de 'iscsi_context_log_func_set()'. La prioridad del registro se puede establecer a través de 'iscsi_context_log_priority_set()'.

Por defecto, las prioridades de registro son 'LIBISCSI_LOG_PRIORITY_WARNING'. Por defecto, el manejador de registros imprime el registro en STDERR, y su código se enumera a continuación en caso de que desee tomarlo como ejemplo para crear su propio manejador de registros.


#define _ISCSI_LOG_STRERR_ALIGN_WIDTH 80


void _iscsi_log_stderr(struct iscsi_context *ctx, int priority,
const char *file, int line,
const char *func_name,
const char *format, va_list args)
{
int printed_bytes = 0;


printed_bytes += fprintf(stderr, "iSCSI %s: ",
iscsi_log_priority_str(priority));
printed_bytes += vfprintf(stderr, format, args);


if (printed_bytes < _ISCSI_LOG_STRERR_ALIGN_WIDTH) {
fprintf(stderr, "%*s # %s:%s():%d0,
_ISCSI_LOG_STRERR_ALIGN_WIDTH - printed_bytes,
"", file, func_name, line);
} else {
fprintf(stderr, " # %s:%s():%d0, file, func_name,
line);
}
}

CÓDIGO DE EJEMPLO


struct iscsi_context *ctx = NULL;
struct iscsi_session **ses = NULL;
uint32_t se_count = 0;
uint32_t i = 0;
int rc = EXIT_SUCCESS;


ctx = iscsi_context_new();
iscsi_context_log_priority_set(ctx, LIBISCSI_LOG_PRIORITY_DEBUG);


if (iscsi_sessions_get(ctx, &ses, &se_count) != LIBISCSI_OK) {
printf("FALLÓ0);
rc = EXIT_FAILURE;
} else {
printf("Encontrados %" PRIu32 " sesiones iSCSI0, se_count);
for (i = 0; i < se_count; ++i)
printf("La SID es %" PRIu32 "0,
iscsi_session_sid_get(ses[i]));
iscsi_sessions_free(ses, se_count);
}
iscsi_context_free(ctx);
exit(rc);
