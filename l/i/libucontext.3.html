libucontext(3) Manual de Funciones de Biblioteca libucontext(3)

NOMBRE

libucontext - una biblioteca para intercambiar contextos en el espacio de usuario

SINOPSIS

#include <libucontext/libucontext.h>

typedef struct {
  /* depende de la arquitectura objetivo */
} libucontext_mcontext_t;
typedef struct {
  void *ss_sp;
  int ss_flags;
  size_t ss_size;
} libucontext_stack_t;
typedef struct libucontext_ucontext {
  unsigned int uc_flags;
  struct libucontext_ucontext *uc_link;
  libucontext_stack_t uc_stack;
  libucontext_mcontext_t uc_mcontext;
} libucontext_ucontext_t;

int libucontext_getcontext(libucontext_ucontext_t *ucp);

int libucontext_setcontext(const libucontext_ucontext_t *ucp);

void libucontext_makecontext(libucontext_ucontext_t *ucp, void (*func)(), int argc, ...);

int libucontext_swapcontext(libucontext_ucontext_t *oucp, const libucontext_ucontext_t *ucp);

DESCRIPCIÓN

La biblioteca libucontext proporciona una implementación de las funciones ucontext de SysV. Estas se utilizan tradicionalmente para implementar el intercambio de contextos en el espacio de usuario. Esto se logra utilizando las funciones libucontext_getcontext, libucontext_setcontext, libucontext_makecontext y libucontext_swapcontext según corresponda.

La función libucontext_getcontext inicializa una estructura apuntada por ucp con el contexto de usuario actual.

La función libucontext_setcontext establece el contexto de usuario actual a la estructura apuntada por ucp. Descarta el contexto de usuario actual.

La función libucontext_swapcontext guarda el contexto de usuario actual en una estructura apuntada por oucp y luego establece el contexto de usuario actual al nuevo contexto en una estructura apuntada por ucp.

La función libucontext_makecontext modifica un contexto de usuario en una estructura apuntada por ucp para ejecutar una función apuntada por func y configura una lista de argumentos de argc valores.

ADVERTENCIAS

En SysV, las funciones ucontext guardan y restauran las máscaras de señales. Sin embargo, la biblioteca libucontext no lo hace. En la práctica, esto no suele importar, ya que los usuarios de estas funciones rara vez cambian la máscara de señales entre contextos.

Otras implementaciones pueden o no guardar y restaurar registros de procesador adicionales que esta implementación no guarda y restaura. La biblioteca libucontext solo guarda y restaura los registros de propósito general. En la práctica, esto ha demostrado ser suficiente.

EJEMPLO

Un ejemplo práctico que muestra la multitarea cooperativa. Este programa está destinado solo a fines ilustrativos y se ha escrito de manera que favorece la simplicidad sobre el rendimiento y la robustez:

#include <assert.h>
#include <stddef.h>
#include <stdio.h>
#include <stdlib.h>
#include <string.h>
#include <time.h>
#include <libucontext/libucontext.h>
libucontext_ucontext_t mainctx = {};
libucontext_ucontext_t *curthr = &mainctx;
libucontext_ucontext_t *threads = NULL;
size_t thrcount = 0;
void
yieldto(libucontext_ucontext_t *target)
{
	libucontext_ucontext_t *oldthr = curthr;
	curthr = target;
	libucontext_swapcontext(oldthr, curthr);
}
void
yield(void)
{
	libucontext_ucontext_t *newthr;
	/* establecemos uc_flags a un valor distinto de cero para indicar la finalización del hilo. */
	do
		newthr = &threads[random() % thrcount];
	while (newthr == curthr || newthr->uc_flags);
	srandom(time(NULL));
	yieldto(newthr);
}
void
worker(size_t multiple)
{
	size_t accum = 1;
	for (size_t i = 0; i < 10; i++)
	{
		accum += (multiple * i);
		printf("[%p] acumulado %zun", curthr, accum);
		yield();
	}
	/* marcar el hilo como completado, para no volver aquí */
	curthr->uc_flags = 1;
}
void
create(size_t multiple)
{
	libucontext_ucontext_t *cursor;
	thrcount += 1;
	threads = realloc(threads, sizeof(*threads) * thrcount);
	cursor = &threads[thrcount - 1];
	memset(cursor, '0', sizeof *cursor);
	/* inicializar los valores del nuevo hilo a nuestro contexto actual */
	libucontext_getcontext(cursor);
	/* configurar uc_link */
	cursor->uc_link = thrcount > 1 ? &threads[thrcount - 2] : &mainctx;
	/* configurar una pila */
	cursor->uc_stack.ss_size = 8192;
	cursor->uc_stack.ss_sp = calloc(1, cursor->uc_stack.ss_size);
	/* configurar la llamada a la función */
	libucontext_makecontext(cursor, worker, 1, multiple);
}
int
main(int argc, const char *argv[])
{
	srandom(time(NULL));
	libucontext_getcontext(&mainctx);
	for (size_t i = 1; i < 4; i++)
		create(i);
	/* iniciar los hilos intercambiando al último */
	yieldto(&threads[thrcount - 1]);
	return EXIT_SUCCESS;
}
