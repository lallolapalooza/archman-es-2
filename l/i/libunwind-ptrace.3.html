NOMBRE

libunwind-ptrace -- Soporte de ptrace() en libunwind

SINOPSIS

#include <libunwind-ptrace.h>

unw_accessors_t _UPT_accessors;

void *_UPT_create(pid_t);
void _UPT_destroy(void *);

int _UPT_find_proc_info(unw_addr_space_t, unw_word_t, unw_proc_info_t *, int, void *);
void _UPT_put_unwind_info(unw_addr_space_t, unw_proc_info_t *, void *);
int _UPT_get_dyn_info_list_addr(unw_addr_space_t, unw_word_t *, void *);
int _UPT_access_mem(unw_addr_space_t, unw_word_t, unw_word_t *, int, void *);
int _UPT_access_reg(unw_addr_space_t, unw_regnum_t, unw_word_t *, int, void *);
int _UPT_access_fpreg(unw_addr_space_t, unw_regnum_t, unw_fpreg_t *, int, void *);
int _UPT_get_proc_name(unw_addr_space_t, unw_word_t, char *, size_t, unw_word_t *, void *);
int _UPT_resume(unw_addr_space_t, unw_cursor_t *, void *);

DESCRIPCIÓN

La llamada al sistema ptrace(2) permite que un proceso obtenga acceso al estado de la máquina y a la memoria virtual de *otro* proceso. Con el conjunto correcto de rutinas de devolución de llamada, es posible conectar libunwind a otro proceso a través de ptrace(2). Si bien no es muy difícil hacerlo directamente, libunwind facilita aún más esta tarea al proporcionar devoluciones de llamada listas para usar para este propósito. Las rutinas y variables que implementan esta funcionalidad utilizan un prefijo de nombre _UPT, que significa "unwind via ptrace".

Una aplicación que desee utilizar el acceso remoto ptrace de libunwind debe seguir los siguientes pasos.

1.
Cree un nuevo espacio de direcciones libunwind que represente el proceso objetivo. Esto se hace llamando a unw_create_addr_space(). En muchos casos, la aplicación simplemente querrá pasar la dirección de _UPT_accessors como primer argumento a esta rutina. Hacerlo asegurará que libunwind pueda desenrollar correctamente el proceso objetivo.
2.
Active el modo ptrace en el proceso objetivo, ya sea haciendo un fork de un nuevo proceso, invocando PTRACE_TRACEME y luego iniciando el programa objetivo (a través de execve(2)), o adjuntándose directamente a un proceso ya en ejecución (a través de PTRACE_ATTACH).
3.
Una vez que se conoce el ID del proceso (pid) del proceso objetivo, se puede crear una estructura de información UPT llamando a _UPT_create(), pasando el pid del proceso objetivo como único argumento.
4.
El puntero opaco devuelto luego debe pasarse como el puntero "argumento" (tercer argumento) a unw_init_remote().

En circunstancias especiales, una aplicación puede preferir usar solo partes del acceso remoto ptrace de libunwind. Por esta razón, las rutinas de devolución de llamada individuales (_UPT_find_proc_info(), _UPT_put_unwind_info(), etc.) también están disponibles para uso directo. Por supuesto, las direcciones de estas rutinas también podrían obtenerse de _UPT_accessors, pero hacerlo evitaría la inicialización estática. Además, al usar _UPT_accessors, *todas* las rutinas de devolución de llamada se enlazarán en la aplicación, incluso si nunca se llaman realmente.

La rutina _UPT_resume() se puede usar para reanudar la ejecución del proceso objetivo. Simplemente invoca ptrace(2) con un valor de comando de PTRACE_CONT.

Cuando la aplicación haya terminado de usar libunwind en el proceso objetivo, se debe llamar a _UPT_destroy(), pasándole el puntero opaco que se devolvió de la llamada a _UPT_create(). Esto asegura que se liberen todos los recursos, incluida la memoria.

DISPONIBILIDAD

Dado que ptrace(2) funciona solo dentro de una única máquina, el acceso remoto ptrace de libunwind, por definición, no está disponible en las versiones de libunwind configuradas para desenrollado cruzado.

SEGURIDAD DE HILOS

El acceso remoto ptrace de libunwind asume que una única estructura de información UPT nunca se comparte entre hilos. Debido a esto, no se utiliza ningún bloqueo explícito. Siempre que solo un hilo utilice una estructura de información UPT en un momento dado, esta funcionalidad es segura para hilos.

VALOR DE RETORNO

_UPT_create() puede devolver un puntero NULL si no puede crear la estructura de información UPT por alguna razón. Para la implementación actual, la única razón por la que esta llamada puede fallar es cuando el sistema se queda sin memoria.

ARCHIVOS

EJEMPLO

    #include <libunwind-ptrace.h>
    #include <stdio.h>
    #include <stdlib.h>
    int
    main (int argc, char **argv)
    {
      if (argc != 2) {
        fprintf (stderr, "uso: %s PID\n", argv[0]);
        exit (EXIT_FAILURE);
      }
      char *endptr;
      pid_t target_pid = strtoul (argv[1], &endptr, 10);
      if (target_pid == 0 && argv[1] == endptr) {
        fprintf (stderr, "uso: %s PID\n", argv[0]);
        exit (EXIT_FAILURE);
      }
      unw_addr_space_t as = unw_create_addr_space (&_UPT_accessors, 0);
      if (!as) {
        fprintf (stderr, "unw_create_addr_space() falló");
        exit (EXIT_FAILURE);
      }
      void *ui = _UPT_create (target_pid);
      if (!ui) {
        fprintf (stderr, "_UPT_create() falló");
        exit (EXIT_FAILURE);
      }
      unw_cursor_t cursor;
      int ret = unw_init_remote (&cursor, as, ui);
      if (ret < 0) {
        fprintf (stderr, "unw_init_remote() falló: ret=%d\n", ret);
        exit (EXIT_FAILURE);
      }
      do {
        unw_proc_info_t pi;
        ret = unw_get_proc_info (&cursor, &pi);
        if (ret == -UNW_ENOINFO) {
          fprintf (stdout, "sin información\n");
        } else if (ret >= 0) {
          printf ("\tproc=%#016lx-%#016lx\thandler=%#016lx lsda=%#016lx",
                  (long) pi.start_ip, (long) pi.end_ip,
                  (long) pi.handler, (long) pi.lsda);
        }
        ret = unw_step (&cursor);
      } while (ret > 0);
      if (ret < 0) {
        fprintf (stderr, "unwind falló con ret=%d\n", ret);
        exit (EXIT_FAILURE);
      }
      _UPT_destroy (ui);
      unw_destroy_addr_space (as);
      exit (EXIT_SUCCESS);
    }

NOMBRE

libunwind-ptrace - interfaz para obtener el registro de traza de pila para un hilo remoto

SINOPSIS

#include <libunwind.h>

unw_ptrace_get_reg(unw_cursor_t *cursor, int reg, void *value);

DESCRIPCIÓN

La función unw_ptrace_get_reg lee el valor de un registro específico de un hilo remoto en ejecución. Se utiliza principalmente cuando se depura un proceso remoto utilizando la interfaz de ptrace.

cursor: Un puntero a la estructura del cursor de libunwind. Este cursor debe haber sido inicializado previamente para apuntar al hilo remoto.

reg: El número del registro que se desea leer. Los números de registro estándar de libunwind se pueden encontrar en <libunwind.h> (por ejemplo, UNW_REG_IP, UNW_REG_SP, etc.).

value: Un puntero a un tipo adecuado (generalmente unw_word_t) donde se almacenará el valor del registro leído.

Esta función es una extensión específica de libunwind para interactuar con procesos remotos a través de ptrace. No forma parte del estándar POSIX para libunwind.

VALOR DE RETORNO

En caso de éxito, devuelve UNW_NORMAL.

En caso de error, devuelve un código de error negativo (por ejemplo, UNW_EBADREG si el número de registro no es válido).

EJEMPLO

```c

#include <libunwind.h>

#include <stdio.h>

#include <stdlib.h>

#include <sys/ptrace.h>

#include <sys/wait.h>

#include <unistd.h>

#include <errno.h>

int main() {

pid_t child_pid;

child_pid = fork();

if (child_pid == 0) {

// Código del proceso hijo

ptrace(PTRACE_TRACEME, 0, NULL, NULL);

raise(SIGSTOP); // Detener para que el padre pueda adjuntarse

printf("Hilo hijo ejecutándose...\n");

exit(0);

} else if (child_pid > 0) {

// Código del proceso padre

wait(NULL); // Esperar a que el hijo se detenga

ptrace(PTRACE_ATTACH, child_pid, NULL, NULL);

wait(NULL); // Esperar a que el hijo se detenga después de adjuntarse

unw_cursor_t cursor;

unw_addr_space_t addr_space = unw_create_addr_space(NULL, 0);

if (!addr_space) {

perror("unw_create_addr_space");

ptrace(PTRACE_DETACH, child_pid, NULL, NULL);

return 1;

}

// Inicializar el cursor para el proceso remoto

if (unw_pdcreate_init(&cursor, child_pid) < 0) {

fprintf(stderr, "Error al inicializar el cursor ptrace: %s\n", strerror(errno));

unw_destroy_addr_space(addr_space);

ptrace(PTRACE_DETACH, child_pid, NULL, NULL);

return 1;

}

unw_word_t rip_value;

// Leer el registro IP (Instruction Pointer)

if (unw_ptrace_get_reg(&cursor, UNW_REG_IP, &rip_value) < 0) {

fprintf(stderr, "Error al leer el registro IP: %s\n", strerror(errno));

} else {

printf("Valor del registro IP del proceso hijo: 0x%lx\n", (unsigned long)rip_value);

}

unw_destroy_addr_space(addr_space);

ptrace(PTRACE_DETACH, child_pid, NULL, NULL);

waitpid(child_pid, NULL, 0); // Asegurarse de que el hijo termine

} else {

perror("fork");

return 1;

}

return 0;

}

```
