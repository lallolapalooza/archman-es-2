NOMBRE

libdmmp.h - API de Device Mapper Multipath.

SINOPSIS

#include <libdmmp/libdmmp.h>

DESCRIPCIÓN

Todas las funciones públicas de libdmmp envían sus propias páginas de manual. Use 'man 3 <nombre_función>' para verificar el uso detallado.

USO

Para usar libdmmp en su proyecto, sugerimos usar el método 'pkg-config':


* Agregue esta línea a su configure.ac:


PKG_CHECK_MODULES([LIBDMMP], [libdmmp])


* Agregue estas líneas a su Makefile.am:


foo_LDFLAGS += $(LIBDMMP_LIBS)
foo_CFLAGS += $(LIBDMMP_CFLAGS)

MANEJO DE LOGS

La función manejadora de logs puede configurarse mediante 'dmmp_context_log_func_set()'. La prioridad del log puede configurarse mediante 'dmmp_context_log_priority_set()'.

Por defecto, las prioridades de log son 'DMMP_LOG_PRIORITY_WARNING'. Por defecto, el manejador de logs imprime el log en STDERR, y su código se lista a continuación en caso de que desee crear su propio manejador de logs.


static int _DMMP_LOG_STRERR_ALIGN_WIDTH = 80;


static void _log_stderr(struct dmmp_context *ctx,
enum dmmp_log_priority priority,
const char *file, int line,
const char *func_name,
const char *format, va_list args)
{
int printed_bytes = 0;


printed_bytes += fprintf(stderr, "libdmmp %s: ",
dmmp_log_priority_str(priority));
printed_bytes += vfprintf(stderr, format, args);
userdata = dmmp_context_userdata_get(ctx);
if (userdata != NULL)
fprintf(stderr, "(con datos de usuario en la dirección de memoria %p)",
userdata);


if (printed_bytes < _DMMP_LOG_STRERR_ALIGN_WIDTH) {
fprintf(stderr, " %*s # %s:%s():%d0,
_DMMP_LOG_STRERR_ALIGN_WIDTH - printed_bytes, "", file,
func_name, line);
} else {
fprintf(stderr, " # %s:%s():%d0, file, func_name, line);
}
}

CÓDIGO DE EJEMPLO


#include <libdmmp/libdmmp.h>


int main(int argc, char *argv[]) {
struct dmmp_context *ctx = NULL;
struct dmmp_mpath **dmmp_mps = NULL;
struct dmmp_path_group **dmmp_pgs = NULL;
struct dmmp_path **dmmp_ps = NULL;
uint32_t dmmp_mp_count = 0;
uint32_t dmmp_pg_count = 0;
uint32_t dmmp_p_count = 0;
const char *name = NULL;
const char *wwid = NULL;
uint32_t i = 0;
int rc = DMMP_OK;


ctx = dmmp_context_new();
dmmp_context_log_priority_set(ctx, DMMP_LOG_PRIORITY_DEBUG);
// Por defecto, el log se imprimirá en STDERR, puedes
// cambiarlo mediante dmmp_context_log_func_set()
rc = dmmp_mpath_array_get(ctx, &dmmp_mps, &dmmp_mp_count);
if (rc != DMMP_OK) {
printf("dmmp_mpath_array_get() falló con %d: %s", rc,
dmmp_strerror(rc));
goto out;
}
for (i = 0; i < dmmp_mp_count; ++i) {
name = dmmp_mpath_name_get(dmmp_mps[i]);
wwid = dmmp_mpath_wwid_get(dmmp_mps[i]);
printf("dmmp_mpath_array_get(): Obtenido mpath: %s %s0, name,
wwid);
// Puedes usar dmmp_path_group_array_get() para obtener
// información del grupo de rutas y luego invocar dmmp_path_array_get()
// para información de rutas.
}


out:
dmmp_context_free(ctx);
dmmp_mpath_array_free(dmmp_mps, dmmp_mp_count);
if (rc != DMMP_OK)
exit(1);
exit(0);
}
