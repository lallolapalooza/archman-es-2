DESCRIPCIÓN

ldns_rdf
Campo de datos del registro de recursos.

Los datos son un array de bytes en orden de red, cuyo tamaño se especifica por el campo (de 16 bits) size. Para analizarlo correctamente, use el tipo especificado en el campo (de 16 bits) type con un valor de \ref ldns_rdf_type.
struct ldns_struct_rdf
{
El tamaño de los datos (en octetos):
size_t _size;
El tipo de los datos:
ldns_rdf_type _type;
Puntero a los datos (octetos crudos):
void *_data;
};
typedef struct ldns_struct_rdf ldns_rdf;

Los diferentes tipos de campos RDATA.

LDNS_RDF_TYPE_ALG,

LDNS_RDF_TYPE_UNKNOWN,
tiempo (32 bits):
LDNS_RDF_TYPE_TIME,

LDNS_RDF_TYPE_PERIOD,
tiempo TSIG (48 bits):
LDNS_RDF_TYPE_TSIGTIME,
/** Representa el Algoritmo de Clave Pública, HIT y los campos de Clave Pública
para los tipos de registro HIP. Se utiliza un tipo de rdf específico de HIP debido

LDNS_RDF_TYPE_HIP,
/** rdf de longitud variable cualquiera donde la longitud

LDNS_RDF_TYPE_INT16_DATA,
mapas de bits de protocolo y puerto:
LDNS_RDF_TYPE_SERVICE,

LDNS_RDF_TYPE_LOC,

LDNS_RDF_TYPE_IPSECKEY,
sal de hash nsec3:
LDNS_RDF_TYPE_NSEC3_SALT,
cadena base32 nsec3 (con byte de longitud en la red:
LDNS_RDF_TYPE_NSEC3_NEXT_OWNER,

/** 4 enteros cortos representados como 4 * 16 números hexadecimales
* separados por dos puntos. Para NID y L64.
*/
LDNS_RDF_TYPE_ILNP64,

6 * números hexadecimales de 8 bits separados por guiones. Para EUI48.:
LDNS_RDF_TYPE_EUI48,
8 * números hexadecimales de 8 bits separados por guiones. Para EUI64.:
LDNS_RDF_TYPE_EUI64,

/** Una secuencia no nula de letras y números US-ASCII en minúsculas.
* Para CAA.
*/
LDNS_RDF_TYPE_TAG,

/** Una <cadena de caracteres> que codifica el campo de valor según
* [RFC1035], Sección 5.1., codificada como rdata restante.

*/
LDNS_RDF_TYPE_LONG_STR,

/** Desde RFC7218, los registros TLSA se pueden dar con mnemotecnia,
* de ahí estos tipos de campo rdata. Pero al igual que con DNSKEYs, la salida

LDNS_RDF_TYPE_CERTIFICATE_USAGE,
LDNS_RDF_TYPE_SELECTOR,
LDNS_RDF_TYPE_MATCHING_TYPE,

LDNS_RDF_TYPE_AMTRELAY,

borrador-ietf-dnsop-svcb-https *:
LDNS_RDF_TYPE_SVCPARAMS,

/* Alias */
LDNS_RDF_TYPE_BITMAP = LDNS_RDF_TYPE_NSEC
};
typedef enum ldns_enum_rdf_type ldns_rdf_type;
