ldns_rdf, ldns_rdf_type - tipo de campo rdata

SINOPSIS

#include <stdint.h>

DESCRIPCIÓN

ldns_rdf
Campo de datos del registro de recursos.

Los datos son un array de bytes en orden de red, cuyo tamaño se especifica por
el campo de tamaño (de 16 bits). Para analizarlo correctamente, use el tipo
especificado en el campo de tipo (de 16 bits) con un valor de \ref ldns_rdf_type.
struct ldns_struct_rdf
{
El tamaño de los datos (en octetos):
size_t _size;
El tipo de los datos:
ldns_rdf_type _type;
Puntero a los datos (octetos brutos):
void *_data;
};
typedef struct ldns_struct_rdf ldns_rdf;

ldns_rdf_type
Los diferentes tipos de campos RDATA.
enum ldns_enum_rdf_type
{
ninguno:
LDNS_RDF_TYPE_NONE,
nombre de dominio:
LDNS_RDF_TYPE_DNAME,
8 bits:
LDNS_RDF_TYPE_INT8,
16 bits:
LDNS_RDF_TYPE_INT16,
32 bits:
LDNS_RDF_TYPE_INT32,
Registro A:
LDNS_RDF_TYPE_A,
Registro AAAA:
LDNS_RDF_TYPE_AAAA,
cadena txt:
LDNS_RDF_TYPE_STR,
datos apl:
LDNS_RDF_TYPE_APL,
cadena b32:
LDNS_RDF_TYPE_B32_EXT,
cadena b64:
LDNS_RDF_TYPE_B64,
cadena hex:
LDNS_RDF_TYPE_HEX,
códigos de tipo nsec:
LDNS_RDF_TYPE_NSEC,
un tipo de RR:
LDNS_RDF_TYPE_TYPE,
una clase:
LDNS_RDF_TYPE_CLASS,
algoritmo de certificado:
LDNS_RDF_TYPE_CERT_ALG,
un algoritmo de clave:
LDNS_RDF_TYPE_ALG,
tipos desconocidos:
LDNS_RDF_TYPE_UNKNOWN,
tiempo (32 bits):
LDNS_RDF_TYPE_TIME,
período:
LDNS_RDF_TYPE_PERIOD,
tiempo tsig 48 bits:
LDNS_RDF_TYPE_TSIGTIME,
/** Representa el Algoritmo de Clave Pública, HIT y campos de Clave Pública
para los tipos RR de HIP. Se usa un tipo rdf específico de HIP debido a
su diseño inusual en formato de cable (ver RFC 5205 Sección 5) */
LDNS_RDF_TYPE_HIP,
/** rdata de tipo cualquiera de longitud variable donde la longitud se especifica por los 2 primeros bytes */
LDNS_RDF_TYPE_INT16_DATA,
mapas de bits de protocolo y puerto:
LDNS_RDF_TYPE_SERVICE,
datos de ubicación:
LDNS_RDF_TYPE_LOC,
servicios conocidos:
LDNS_RDF_TYPE_WKS,
NSAP:
LDNS_RDF_TYPE_NSAP,
ATMA:
LDNS_RDF_TYPE_ATMA,
IPSECKEY:
LDNS_RDF_TYPE_IPSECKEY,
sal de hash nsec3:
LDNS_RDF_TYPE_NSEC3_SALT,
cadena nsec3 base32 (con byte de longitud en el cable:
LDNS_RDF_TYPE_NSEC3_NEXT_OWNER,

/** 4 shorts representados como 4 números hexadecimales de 16 bits
* separados por dos puntos. Para NID y L64.
*/
LDNS_RDF_TYPE_ILNP64,

6 números hexadecimales de 8 bits separados por guiones. Para EUI48.:
LDNS_RDF_TYPE_EUI48,
8 números hexadecimales de 8 bits separados por guiones. Para EUI64.:
LDNS_RDF_TYPE_EUI64,

/** Una secuencia distinta de cero de letras y números US-ASCII en minúsculas.
* Para CAA.
*/
LDNS_RDF_TYPE_TAG,

/** Un <cadena de caracteres> que codifica el campo de valor como se especifica en
* [RFC1035], Sección 5.1., codificado como rdata restante.
* Para CAA.
*/
LDNS_RDF_TYPE_LONG_STR,

/** Desde RFC7218, los registros TLSA se pueden dar con mnemónicos,
* de ahí estos tipos de campo rdata. Pero al igual que con los DNSKEYs, la salida
* siempre es numérica.
*/
LDNS_RDF_TYPE_CERTIFICATE_USAGE,
LDNS_RDF_TYPE_SELECTOR,
LDNS_RDF_TYPE_MATCHING_TYPE,

descubrimiento de amt-driade borrador-ietf-mboned *:
LDNS_RDF_TYPE_AMTRELAY,

descubrimiento de https de borrador-ietf-dnsop-svcb *:
LDNS_RDF_TYPE_SVCPARAMS,

/* Alias */
LDNS_RDF_TYPE_BITMAP = LDNS_RDF_TYPE_NSEC
};
typedef enum ldns_enum_rdf_type ldns_rdf_type;
