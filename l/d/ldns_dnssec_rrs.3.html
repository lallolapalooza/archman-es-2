ldns_dnssec_name
Estructura que contiene todos los registros de recursos para un nombre de dominio
Incluyendo el NSEC3 derivado, si está presente
struct ldns_struct_dnssec_name
{
/**
* puntero a un dname que contiene el nombre.
* Normalmente apunta al nombre propietario del primer RR del primer RRset
*/
ldns_rdf *name;
/**
* Normalmente, el nombre es un puntero al nombre propietario del primer rr para
* este nombre, pero a veces no hay datos reales a los que apuntar,
* por ejemplo en
* nombres que representan terminales vacías. Si es así, establezca name_alloced a true para
* indicar que estos datos también deben liberarse cuando se libere el nombre.
*/
bool name_alloced;
/**
* Los rrsets para este nombre
*/
ldns_dnssec_rrsets *rrsets;
/**
* NSEC que apunta al siguiente nombre (o NSEC3 que apunta al siguiente NSEC3)
*/
ldns_rr *nsec;
/**
* firmas para el registro NSEC
*/
ldns_dnssec_rrs *nsec_signatures;
/**
* A diferencia de lo que sugiere el nombre is_glue, este campo se establece en true mediante
* ldns_dnssec_zone_mark_glue() o ldns_dnssec_zone_mark_and_get_glue()
* cuando el nombre, esta estructura dnssec_name representa, está ocluido.
* Los nombres que contienen otros rrsets ocluidos y registros con pegamento en
* el punto de delegación NO tendrán este booleano establecido en true.
* Este campo NO debe leerse directamente, sino solo a través de la función
* ldns_dnssec_name_is_glue()!
*/
bool is_glue;
/**
* puntero para almacenar el nombre hasheado (solo se usa cuando está en una zona NSEC3)
*/
ldns_rdf *hashed_name;
};

typedef struct ldns_struct_dnssec_name ldns_dnssec_name;

};

typedef struct ldns_struct_dnssec_rrs ldns_dnssec_rrs;

struct ldns_struct_dnssec_rrsets
{

};

typedef struct ldns_struct_dnssec_rrsets ldns_dnssec_rrsets;
