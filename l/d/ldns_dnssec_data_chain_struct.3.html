ldns_dnssec_data_chain
Estructura de cadena que contiene todos los datos DNSSEC necesarios para
verificar un rrset
struct ldns_dnssec_data_chain_struct
{
ldns_rr_list *rrset;
ldns_rr_list *signatures;
ldns_rr_type parent_type;
ldns_dnssec_data_chain *parent;
ldns_pkt_rcode packet_rcode;
ldns_rr_type packet_qtype;
bool packet_nodata;
};

typedef struct ldns_dnssec_data_chain_struct ldns_dnssec_data_chain;

ldns_dnssec_trust_tree
Estructura de árbol que contiene la relación de datos DNSSEC,
y su estado criptográfico.

Este árbol se deriva de una data_chain y puede utilizarse
para buscar si existe una conexión entre un RRSET
y una clave de confianza. El árbol solo contiene punteros a la
data_chain, y por lo tanto, uno *nunca* debe hacer free() de la
data_chain mientras todavía haya un trust tree derivado de
esa cadena.

Ejemplo de árbol:
key key key
\ | /
\ | /
\ | /
ds
|
key
|
key
|
rr

Para cada firma hay un padre; si el puntero al padre es nulo, no se pudo
encontrar y no hubo denegación; de lo contrario, es un árbol que contiene
una clave DNSKEY, una DS o un rr NSEC
struct ldns_dnssec_trust_tree_struct
{
ldns_rr *rr;
/* el rrset completo del que formaba parte este rr */
ldns_rr_list *rrset;
ldns_dnssec_trust_tree *parents[LDNS_DNSSEC_TRUST_TREE_MAX_PARENTS];
ldns_status parent_status[LDNS_DNSSEC_TRUST_TREE_MAX_PARENTS];
/** para depuración, añadir también firmas (podrías querer
aquellas si contienen errores) */
ldns_rr *parent_signature[LDNS_DNSSEC_TRUST_TREE_MAX_PARENTS];
size_t parent_count;
};

typedef struct ldns_dnssec_trust_tree_struct ldns_dnssec_trust_tree;
