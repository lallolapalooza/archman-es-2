#include <stdint.h>
#include <stdbool.h>

ldns_key* ldns_key_new(void);

ldns_key* ldns_key_new_frm_algorithm(ldns_signing_algorithm a, uint16_t size);

ldns_status ldns_key_new_frm_fp(ldns_key **k, FILE *fp);

DESCRIPCIÓN

ldns_key_new() Crea una nueva estructura de clave vacía
Devuelve una nueva estructura ldns_key *

ldns_key_new_frm_algorithm() Crea una nueva clave basada en el algoritmo

a: El algoritmo a usar
size: el número de bytes para el tamaño de la clave
Devuelve una nueva estructura ldns_key con la clave

ldns_key_new_frm_fp() Crea una nueva clave privada basada en el contenido del archivo apuntado por fp.

El archivo debe estar en formato de clave privada v1.x.

k: la nueva estructura ldns_key
fp: el puntero al archivo a usar
Devuelve un error o LDNS_STATUS_OK

ldns_key_new_frm_fp_l() Crea una nueva clave privada basada en el contenido del archivo apuntado por fp

El archivo debe estar en formato de clave privada v1.x.

k: la nueva estructura ldns_key
fp: el puntero al archivo a usar
line_nr: puntero a un entero que contiene el número de línea actual (con fines de depuración)
Devuelve un error o LDNS_STATUS_OK

ldns_key_new_frm_fp_rsa() Función auxiliar frm_fp. Esta función analiza el resto del archivo de clave privada (RSA) generado desde bind9
fp: el archivo a analizar
Devuelve NULL en caso de error, de lo contrario una estructura RSA

ldns_key_new_frm_fp_rsa_l() Función auxiliar frm_fp. Esta función analiza el resto del archivo de clave privada (RSA) generado desde bind9
fp: el archivo a analizar
line_nr: puntero a un entero que contiene el número de línea actual (con fines de depuración)
Devuelve NULL en caso de error, de lo contrario una estructura RSA

ldns_key_new_frm_fp_dsa() Función auxiliar frm_fp. Esta función analiza el resto del archivo de clave privada (DSA)
fp: el archivo a analizar
Devuelve NULL en caso de error, de lo contrario una estructura RSA

ldns_key_new_frm_fp_dsa_l() Función auxiliar frm_fp. Esta función analiza el resto del archivo de clave privada (DSA)
fp: el archivo a analizar
line_nr: puntero a un entero que contiene el número de línea actual (con fines de depuración)
Devuelve NULL en caso de error, de lo contrario una estructura RSA
