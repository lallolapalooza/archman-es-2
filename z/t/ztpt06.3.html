subroutine ztpt06 (double precision rcond, double precision rcondc, character uplo, character diag, integer n, complex*16, dimension( * ) ap, double precision, dimension( * ) rwork, double precision rat)

ZTPT06

Propósito:

!>
!> ZTPT06 calcula una relación de prueba que compara RCOND (el recíproco
!> del número de condición de la matriz triangular A) y RCONDC, la estimación
!> calculada por ZTPCON. Se utiliza información sobre la matriz triangular
!> si una estimación es cero y la otra no cero para decidir si la
!> subdesbordamiento en la estimación está justificado.
!>

Parámetros

RCOND 
!>          RCOND es DOUBLE PRECISION
!>          La estimación del recíproco del número de condición obtenida al
!>          formar la inversa explícita de la matriz A y calcular
!>          RCOND = 1/( norm(A) * norm(inv(A)) ).
!>

RCONDC

!>          RCONDC es DOUBLE PRECISION
!>          La estimación del recíproco del número de condición calculada por
!>          ZTPCON.
!>

UPLO

!>          UPLO es CHARACTER
!>          Especifica si la matriz A es triangular superior o inferior.
!>          = 'U':  Triangular superior
!>          = 'L':  Triangular inferior
!>

DIAG

!>          DIAG es CHARACTER
!>          Especifica si la matriz A es triangular unitaria o no.
!>          = 'N':  Triangular no unitaria
!>          = 'U':  Triangular unitaria
!>

N

!>          N es INTEGER
!>          El orden de la matriz A.  N >= 0.
!>

AP

!>          AP es un array COMPLEX*16, dimension (N*(N+1)/2)
!>          La matriz triangular superior o inferior A, empaquetada por columnas en
!>          un array lineal. La j-ésima columna de A se almacena en el array
!>          AP de la siguiente manera:
!>          si UPLO = 'U', AP((j-1)*j/2 + i) = A(i,j) para 1<=i<=j;
!>          si UPLO = 'L',
!>             AP((j-1)*(n-j) + j*(j+1)/2 + i-j) = A(i,j) para j<=i<=n.
!>

RWORK

!>          RWORK es un array DOUBLE PRECISION, dimension (N)
!>

RAT

!>          RAT es DOUBLE PRECISION
!>          La relación de prueba.  Si RCOND y RCONDC son ambos distintos de cero,
!>             RAT = MAX( RCOND, RCONDC )/MIN( RCOND, RCONDC ) - 1.
!>          Si RAT = 0, las dos estimaciones son exactamente las mismas.
!>
