subroutine ztbt03 (character uplo, character trans, character diag, integer n, integer kd, integer nrhs, complex*16, dimension( ldab, * ) ab, integer ldab, double precision scale, double precision, dimension( * ) cnorm, double precision tscal, complex*16, dimension( ldx, * ) x, integer ldx, complex*16, dimension( ldb, * ) b, integer ldb, complex*16, dimension( * ) work, double precision resid)

ZTBT03

Propósito:

!>
!> ZTBT03 calcula el residuo para la solución de un sistema triangular escalado
!> de ecuaciones  A*x = s*b,  A**T *x = s*b,  o  A**H *x = s*b
!> cuando A es una matriz de banda triangular. Aquí A**T denota la transpuesta
!> de A, A**H denota la transpuesta conjugada de A, s es un escalar, y
!> x y b son matrices N por NRHS. La relación de prueba es el máximo sobre
!> el número de lados derechos de
!>    norm(s*b - op(A)*x) / ( norm(op(A)) * norm(x) * EPS ),
!> donde op(A) denota A, A**T, o A**H, y EPS es la epsilon de la máquina.
!>

Parámetros

UPLO 
!>          UPLO es CHARACTER*1
!>          Especifica si la matriz A es triangular superior o inferior.
!>          = 'U':  Triangular superior
!>          = 'L':  Triangular inferior
!>

TRANS

!>          TRANS es CHARACTER*1
!>          Especifica la operación aplicada a A.
!>          = 'N':  A *x = s*b     (Sin transpuesta)
!>          = 'T':  A**T *x = s*b  (Transpuesta)
!>          = 'C':  A**H *x = s*b  (Transpuesta conjugada)
!>

DIAG

!>          DIAG es CHARACTER*1
!>          Especifica si la matriz A es unitaria triangular o no.
!>          = 'N':  Triangular no unitaria
!>          = 'U':  Triangular unitaria
!>

N

!>          N es INTEGER
!>          El orden de la matriz A.  N >= 0.
!>

KD

!>          KD es INTEGER
!>          El número de superdiagonales o subdiagonales de la
!>          matriz de banda triangular A.  KD >= 0.
!>

NRHS

!>          NRHS es INTEGER
!>          El número de lados derechos, es decir, el número de columnas
!>          de las matrices X y B.  NRHS >= 0.
!>

AB

!>          AB es COMPLEX*16 array, dimension (LDAB,N)
!>          La matriz de banda triangular superior o inferior A, almacenada en las
!>          primeras kd+1 filas del array. La j-ésima columna de A se almacena
!>          en la j-ésima columna del array AB como sigue:
!>          si UPLO = 'U', AB(kd+1+i-j,j) = A(i,j) para max(1,j-kd)<=i<=j;
!>          si UPLO = 'L', AB(1+i-j,j)    = A(i,j) para j<=i<=min(n,j+kd).
!>

LDAB

!>          LDAB es INTEGER
!>          La dimensión principal del array AB.  LDAB >= KD+1.
!>

SCALE

!>          SCALE es DOUBLE PRECISION
!>          El factor de escala s utilizado al resolver el sistema triangular.
!>

CNORM

!>          CNORM es DOUBLE PRECISION array, dimension (N)
!>          Las normas 1 de las columnas de A, sin contar la diagonal.
!>

TSCAL

!>          TSCAL es DOUBLE PRECISION
!>          El factor de escala utilizado al calcular las normas 1 en CNORM.
!>          CNORM en realidad contiene las normas de columna de TSCAL*A.
!>

X

!>          X es COMPLEX*16 array, dimension (LDX,NRHS)
!>          Los vectores de solución calculados para el sistema de ecuaciones
!>          lineales.
!>

LDX

!>          LDX es INTEGER
!>          La dimensión principal del array X.  LDX >= max(1,N).
!>

B

!>          B es COMPLEX*16 array, dimension (LDB,NRHS)
!>          Los vectores del lado derecho para el sistema de ecuaciones
!>          lineales.
!>

LDB

!>          LDB es INTEGER
!>          La dimensión principal del array B.  LDB >= max(1,N).
!>

WORK

!>          WORK es COMPLEX*16 array, dimension (N)
!>

RESID

!>          RESID es DOUBLE PRECISION
!>          El máximo sobre el número de lados derechos de
!>          norm(op(A)*x - s*b) / ( norm(op(A)) * norm(x) * EPS ).
!>
