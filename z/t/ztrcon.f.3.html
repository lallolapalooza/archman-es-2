subroutine ztrcon (character norm, character uplo, character diag, integer n, complex*16, dimension( lda, * ) a, integer lda, double precision rcond, complex*16, dimension( * ) work, double precision, dimension( * ) rwork, integer info)

ZTRCON

Propósito:

!>
!> ZTRCON estima el recíproco del número de condición de una
!> matriz triangular A, en norma 1 o norma infinito.
!>
!> Se calcula la norma de A y se obtiene una estimación de
!> norm(inv(A)), luego se calcula el recíproco del número de condición como
!>    RCOND = 1 / ( norm(A) * norm(inv(A)) ).
!>

Parámetros

NORM 
!>          NORM es CHARACTER*1
!>          Especifica si se requiere el número de condición en norma 1 o
!>          en norma infinito:
!>          = '1' o 'O':  norma 1;
!>          = 'I':         norma infinito.
!>

UPLO

!>          UPLO es CHARACTER*1
!>          = 'U':  A es triangular superior;
!>          = 'L':  A es triangular inferior.
!>

DIAG

!>          DIAG es CHARACTER*1
!>          = 'N':  A es triangular no unitaria;
!>          = 'U':  A es triangular unitaria.
!>

N

!>          N es INTEGER
!>          El orden de la matriz A.  N >= 0.
!>

A

!>          A es un array de COMPLEX*16, dimensión (LDA,N)
!>          La matriz triangular A.  Si UPLO = 'U', la parte superior
!>          triangular líder N-por-N del array A contiene la matriz
!>          triangular superior, y la parte estrictamente triangular inferior
!>          de A no se referencia.  Si UPLO = 'L', la parte inferior
!>          triangular líder N-por-N del array A contiene la matriz
!>          triangular inferior, y la parte estrictamente triangular superior
!>          de A no se referencia.  Si DIAG = 'U', los elementos diagonales
!>          de A tampoco se referencian y se asumen como 1.
!>

LDA

!>          LDA es INTEGER
!>          La dimensión líder del array A.  LDA >= max(1,N).
!>

RCOND

!>          RCOND es DOUBLE PRECISION
!>          El recíproco del número de condición de la matriz A,
!>          calculado como RCOND = 1/(norm(A) * norm(inv(A))).
!>

WORK

!>          WORK es un array de COMPLEX*16, dimensión (2*N)
!>

RWORK

!>          RWORK es un array de DOUBLE PRECISION, dimensión (N)
!>

INFO

!>          INFO es INTEGER
!>          = 0:  salida exitosa
!>          < 0:  si INFO = -i, el i-ésimo argumento tenía un valor ilegal
!>
