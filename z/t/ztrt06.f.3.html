subroutine ztrt06 (double precision rcond, double precision rcondc, character uplo, character diag, integer n, complex*16, dimension( lda, * ) a, integer lda, double precision, dimension( * ) rwork, double precision rat)

ZTRT06

Propósito:

!>
!> ZTRT06 calcula una relación de prueba que compara RCOND (el recíproco
!> del número de condición de una matriz triangular A) y RCONDC, la estimación
!> calculada por ZTRCON. Se utiliza información sobre la matriz triangular A
!> si una estimación es cero y la otra es distinta de cero para decidir si
!> el subdesbordamiento en la estimación está justificado.
!>

Parámetros

RCOND 
!>          RCOND es DOUBLE PRECISION
!>          La estimación del número de condición recíproco obtenida al
!>          formar la inversa explícita de la matriz A y calcular
!>          RCOND = 1/( norm(A) * norm(inv(A)) ).
!>

RCONDC

!>          RCONDC es DOUBLE PRECISION
!>          La estimación del número de condición recíproco calculada por
!>          ZTRCON.
!>

UPLO

!>          UPLO es CHARACTER
!>          Especifica si la matriz A es triangular superior o inferior.
!>          = 'U':  Triangular superior
!>          = 'L':  Triangular inferior
!>

DIAG

!>          DIAG es CHARACTER
!>          Especifica si la matriz A es unitaria triangular o no.
!>          = 'N':  Triangular no unitaria
!>          = 'U':  Triangular unitaria
!>

N

!>          N es INTEGER
!>          El orden de la matriz A.  N >= 0.
!>

A

!>          A es COMPLEX*16 array, dimension (LDA,N)
!>          La matriz triangular A.  Si UPLO = 'U', la parte triangular
!>          superior de n por n principal de la matriz A contiene la matriz
!>          triangular superior, y la parte estrictamente triangular inferior
!>          de A no se referencia.  Si UPLO = 'L', la parte triangular
!>          inferior de n por n principal de la matriz A contiene la matriz
!>          triangular inferior, y la parte estrictamente triangular superior
!>          de A no se referencia.  Si DIAG = 'U', los elementos diagonales de
!>          A tampoco se referencian y se asumen como 1.
!>

LDA

!>          LDA es INTEGER
!>          La dimensión principal de la matriz A.  LDA >= max(1,N).
!>

RWORK

!>          RWORK es DOUBLE PRECISION array, dimension (N)
!>

RAT

!>          RAT es DOUBLE PRECISION
!>          La relación de prueba.  Si tanto RCOND como RCONDC son distintos de
!>          cero,
!>             RAT = MAX( RCOND, RCONDC )/MIN( RCOND, RCONDC ) - 1.
!>          Si RAT = 0, las dos estimaciones son exactamente las mismas.
!>
