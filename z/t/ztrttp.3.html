subroutine ztrttp (character uplo, integer n, complex*16, dimension( lda, * ) a, integer lda, complex*16, dimension( * ) ap, integer info)

ZTRTTP copia una matriz triangular del formato completo estándar (TR) al formato empaquetado estándar (TP).

Propósito:

!>
!> ZTRTTP copia una matriz triangular A del formato completo (TR) al formato
!> estándar empaquetado (TP).
!>

Parámetros

UPLO 
!>          UPLO es CHARACTER*1
!>          = 'U':  A es triangular superior;
!>          = 'L':  A es triangular inferior.
!>

N

!>          N es INTEGER
!>          El orden de las matrices AP y A.  N >= 0.
!>

A

!>          A es un array COMPLEX*16, de dimensión (LDA,N)
!>          En la entrada, la matriz triangular A.  Si UPLO = 'U', la parte
!>          superior triangular de N por N de A contiene la parte triangular
!>          superior de la matriz A, y la parte estrictamente triangular inferior
!>          de A no se referencia.  Si UPLO = 'L', la parte inferior triangular
!>          de N por N de A contiene la parte triangular inferior de la matriz
!>          A, y la parte estrictamente triangular superior de A no se referencia.
!>

LDA

!>          LDA es INTEGER
!>          La dimensión principal del array A.  LDA >= max(1,N).
!>

AP

!>          AP es un array COMPLEX*16, de dimensión ( N*(N+1)/2 ),
!>          En la salida, la matriz triangular superior o inferior A, empaquetada
!>          por columnas en un array lineal. La j-ésima columna de A se almacena
!>          en el array AP de la siguiente manera:
!>          si UPLO = 'U', AP(i + (j-1)*j/2) = A(i,j) para 1<=i<=j;
!>          si UPLO = 'L', AP(i + (j-1)*(2n-j)/2) = A(i,j) para j<=i<=n.
!>

INFO

!>          INFO es INTEGER
!>          = 0:  salida exitosa
!>          < 0:  si INFO = -i, el i-ésimo argumento tenía un valor ilegal
!>
