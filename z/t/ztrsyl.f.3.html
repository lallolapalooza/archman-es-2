Documentación de la función/subrutina

subroutine ztrsyl (character trana, character tranb, integer isgn, integer m, integer n, complex*16, dimension( lda, * ) a, integer lda, complex*16, dimension( ldb, * ) b, integer ldb, complex*16, dimension( ldc, * ) c, integer ldc, double precision scale, integer info)

ZTRSYL

Propósito:

!>
!> ZTRSYL resuelve la ecuación matricial compleja de Sylvester:
!>
!>    op(A)*X + X*op(B) = scale*C o
!>    op(A)*X - X*op(B) = scale*C,
!>
!> donde op(A) = A o A**H, y A y B son ambas triangulares superiores. A
!> es M por M y B es N por N; el lado derecho C y la solución X son
!> M por N; y scale es un factor de escala de salida, establecido <= 1 para
!> evitar desbordamiento en X.
!>

Parámetros

TRANA 
!>          TRANA es CHARACTER*1
!>          Especifica la opción op(A):
!>          = 'N': op(A) = A    (Sin traspuesta)
!>          = 'C': op(A) = A**H (Traspuesta conjugada)
!>

TRANB

!>          TRANB es CHARACTER*1
!>          Especifica la opción op(B):
!>          = 'N': op(B) = B    (Sin traspuesta)
!>          = 'C': op(B) = B**H (Traspuesta conjugada)
!>

ISGN

!>          ISGN es INTEGER
!>          Especifica el signo en la ecuación:
!>          = +1: resuelve op(A)*X + X*op(B) = scale*C
!>          = -1: resuelve op(A)*X - X*op(B) = scale*C
!>

M

!>          M es INTEGER
!>          El orden de la matriz A, y el número de filas de las
!>          matrices X y C. M >= 0.
!>

N

!>          N es INTEGER
!>          El orden de la matriz B, y el número de columnas de las
!>          matrices X y C. N >= 0.
!>

A

!>          A es COMPLEX*16 array, dimension (LDA,M)
!>          La matriz triangular superior A.
!>

LDA

!>          LDA es INTEGER
!>          La dimensión principal del array A. LDA >= max(1,M).
!>

B

!>          B es COMPLEX*16 array, dimension (LDB,N)
!>          La matriz triangular superior B.
!>

LDB

!>          LDB es INTEGER
!>          La dimensión principal del array B. LDB >= max(1,N).
!>

C

!>          C es COMPLEX*16 array, dimension (LDC,N)
!>          En la entrada, la matriz C del lado derecho de M por N.
!>          En la salida, C se sobrescribe con la matriz solución X.
!>

LDC

!>          LDC es INTEGER
!>          La dimensión principal del array C. LDC >= max(1,M)
!>

SCALE

!>          SCALE es DOUBLE PRECISION
!>          El factor de escala, scale, establecido <= 1 para evitar
!>          desbordamiento en X.
!>

INFO

!>          INFO es INTEGER
!>          = 0: salida exitosa
!>          < 0: si INFO = -i, el i-ésimo argumento tuvo un valor ilegal
!>          = 1: A y B tienen valores propios comunes o muy cercanos; se
!>               utilizaron valores perturbados para resolver la ecuación
!>               (pero las matrices A y B no se modifican).
!>
