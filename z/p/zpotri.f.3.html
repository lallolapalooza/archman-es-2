subroutine zpotri (character uplo, integer n, complex*16, dimension( lda, * ) a, integer lda, integer info)

ZPOTRI

Propósito:

!>
!> ZPOTRI calcula la inversa de una matriz compleja Hermitiana definida positiva
!> A utilizando la factorización de Cholesky A = U**H*U o A = L*L**H
!> calculada por ZPOTRF.
!>

Parámetros

UPLO 
!>          UPLO es CHARACTER*1
!>          = 'U':  El triángulo superior de A está almacenado;
!>          = 'L':  El triángulo inferior de A está almacenado.
!>

N

!>          N es INTEGER
!>          El orden de la matriz A.  N >= 0.
!>

A

!>          A es un array COMPLEX*16, de dimensión (LDA,N)
!>          En la entrada, el factor triangular U o L de la factorización de Cholesky
!>          A = U**H*U o A = L*L**H, tal como se calculó por ZPOTRF.
!>          En la salida, el triángulo superior o inferior de la inversa
!>          (Hermitiana) de A, sobrescribiendo el factor de entrada U o L.
!>

LDA

!>          LDA es INTEGER
!>          La dimensión principal del array A.  LDA >= max(1,N).
!>

INFO

!>          INFO es INTEGER
!>          = 0:  salida exitosa
!>          < 0:  si INFO = -i, el i-ésimo argumento tuvo un valor ilegal
!>          > 0:  si INFO = i, el elemento (i,i) del factor U o L es
!>                cero, y la inversa no pudo ser calculada.
!>
