subroutine zpbt02 (character uplo, integer n, integer kd, integer nrhs, complex*16, dimension( lda, * ) a, integer lda, complex*16, dimension( ldx, * ) x, integer ldx, complex*16, dimension( ldb, * ) b, integer ldb, double precision, dimension( * ) rwork, double precision resid)

ZPBT02

Propósito:

!>
!> ZPBT02 calcula el residuo para una solución de un sistema hermitiano
!> de ecuaciones de banda  A*x = b:
!>    RESID = norm( B - A*X ) / ( norm(A) * norm(X) * EPS)
!> donde EPS es la precisión de la máquina.
!>

Parámetros

UPLO 
!>          UPLO es CHARACTER*1
!>          Especifica si la parte triangular superior o inferior de la
!>          matriz hermitiana A está almacenada:
!>          = 'U':  Triangular superior
!>          = 'L':  Triangular inferior
!>

N

!>          N es INTEGER
!>          El número de filas y columnas de la matriz A.  N >= 0.
!>

KD

!>          KD es INTEGER
!>          El número de superdiagonales de la matriz A si UPLO = 'U',
!>          o el número de subdiagonales si UPLO = 'L'.  KD >= 0.
!>

NRHS

!>          NRHS es INTEGER
!>          El número de lados derechos. NRHS >= 0.
!>

A

!>          A es COMPLEX*16 array, dimension (LDA,N)
!>          La matriz de banda hermitiana original A.  Si UPLO = 'U', la
!>          parte triangular superior de A se almacena como una matriz de
!>          banda; si UPLO = 'L', la parte triangular inferior de A se
!>          almacena.  Las columnas de la parte triangular apropiada se
!>          almacenan en las columnas de A y las diagonales de la
!>          triangular se almacenan en las filas de A.  Ver ZPBTRF para
!>          más detalles.
!>

LDA

!>          LDA es INTEGER.
!>          La dimensión principal del array A.  LDA >= max(1,KD+1).
!>

X

!>          X es COMPLEX*16 array, dimension (LDX,NRHS)
!>          Los vectores de solución calculados para el sistema de
!>          ecuaciones lineales.
!>

LDX

!>          LDX es INTEGER
!>          La dimensión principal del array X.   LDX >= max(1,N).
!>

B

!>          B es COMPLEX*16 array, dimension (LDB,NRHS)
!>          En la entrada, los vectores del lado derecho para el sistema de
!>          ecuaciones lineales.
!>          En la salida, B se sobrescribe con la diferencia B - A*X.
!>

LDB

!>          LDB es INTEGER
!>          La dimensión principal del array B.  LDB >= max(1,N).
!>

RWORK

!>          RWORK es DOUBLE PRECISION array, dimension (N)
!>

RESID

!>          RESID es DOUBLE PRECISION
!>          El máximo sobre el número de lados derechos de
!>          norm(B - A*X) / ( norm(A) * norm(X) * EPS ).
!>
