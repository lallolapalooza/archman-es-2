subroutine zpttrs (character uplo, integer n, integer nrhs, double precision, dimension( * ) d, complex*16, dimension( * ) e, complex*16, dimension( ldb, * ) b, integer ldb, integer info)

ZPTTRS

Propósito:

!>
!> ZPTTRS resuelve un sistema tridiagonal de la forma
!>    A * X = B
!> utilizando la factorización A = U**H *D* U o A = L*D*L**H calculada por ZPTTRF.
!> D es una matriz diagonal especificada en el vector D, U (o L) es una matriz unitaria
!> bidiagonal cuya superdiagonal (subdiagonal) está especificada en
!> el vector E, y X y B son matrices N por NRHS.
!>

Parámetros

UPLO 
!>          UPLO es CHARACTER*1
!>          Especifica la forma de la factorización y si el
!>          vector E es la superdiagonal del factor bidiagonal superior
!>          U o la subdiagonal del factor bidiagonal inferior L.
!>          = 'U':  A = U**H *D*U, E es la superdiagonal de U
!>          = 'L':  A = L*D*L**H, E es la subdiagonal de L
!>

N

!>          N es INTEGER
!>          El orden de la matriz tridiagonal A.  N >= 0.
!>

NRHS

!>          NRHS es INTEGER
!>          El número de lados derechos, es decir, el número de columnas
!>          de la matriz B.  NRHS >= 0.
!>

D

!>          D es un array de DOUBLE PRECISION, dimensión (N)
!>          Los n elementos diagonales de la matriz diagonal D de la
!>          factorización A = U**H *D*U o A = L*D*L**H.
!>

E

!>          E es un array de COMPLEX*16, dimensión (N-1)
!>          Si UPLO = 'U', los (n-1) elementos superdiagonales del factor unitario
!>          bidiagonal U de la factorización A = U**H*D*U.
!>          Si UPLO = 'L', los (n-1) elementos subdiagonales del factor unitario
!>          bidiagonal L de la factorización A = L*D*L**H.
!>

B

!>          B es un array de COMPLEX*16, dimensión (LDB,NRHS)
!>          En la entrada, los vectores del lado derecho B para el sistema de
!>          ecuaciones lineales.
!>          En la salida, los vectores solución, X.
!>

LDB

!>          LDB es INTEGER
!>          La dimensión principal del array B.  LDB >= max(1,N).
!>

INFO

!>          INFO es INTEGER
!>          = 0: salida exitosa
!>          < 0: si INFO = -k, el k-ésimo argumento tenía un valor ilegal
!>
