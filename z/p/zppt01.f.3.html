subroutine zppt01 (character uplo, integer n, complex*16, dimension( * ) a, complex*16, dimension( * ) afac, double precision, dimension( * ) rwork, double precision resid)

ZPPT01

Propósito:

!>
!> ZPPT01 reconstruye una matriz A Hermitiana definida positiva a partir de
!> su factorización L*L' o U'*U y calcula la norma residual
!>    norm( L*L' - A ) / ( N * norm(A) * EPS ) o
!>    norm( U'*U - A ) / ( N * norm(A) * EPS ),
!> donde EPS es la epsilon de la máquina, L' es la transpuesta conjugada de
!> L, y U' es la transpuesta conjugada de U.
!>

Parámetros

UPLO 
!>          UPLO es CHARACTER*1
!>          Especifica si la parte triangular superior o inferior de la
!>          matriz Hermitiana A está almacenada:
!>          = 'U':  Triangular superior
!>          = 'L':  Triangular inferior
!>

N

!>          N es INTEGER
!>          El número de filas y columnas de la matriz A.  N >= 0.
!>

A

!>          A es un array COMPLEX*16, dimension (N*(N+1)/2)
!>          La matriz Hermitiana original A, almacenada como una matriz
!>          triangular empaquetada.
!>

AFAC

!>          AFAC es un array COMPLEX*16, dimension (N*(N+1)/2)
!>          En la entrada, el factor L o U de la factorización L*L' o U'*U
!>          de A, almacenado como una matriz triangular empaquetada.
!>          Se sobrescribe con la matriz reconstruida, y luego con la
!>          diferencia L*L' - A (o U'*U - A).
!>

RWORK

!>          RWORK es un array DOUBLE PRECISION, dimension (N)
!>

RESID

!>          RESID es DOUBLE PRECISION
!>          Si UPLO = 'L', norm(L*L' - A) / ( N * norm(A) * EPS )
!>          Si UPLO = 'U', norm(U'*U - A) / ( N * norm(A) * EPS )
!>
