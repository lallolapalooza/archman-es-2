subroutine zpbequ (character uplo, integer n, integer kd, complex*16, dimension( ldab, * ) ab, integer ldab, double precision, dimension( * ) s, double precision scond, double precision amax, integer info)

ZPBEQU

Propósito

!>
!> ZPBEQU calcula escalados de fila y columna destinados a equilibrar una
!> matriz banda Hermítica definida positiva A y reducir su número de
!> condición (con respecto a la norma dos). S contiene los factores de
!> escala, S(i) = 1/sqrt(A(i,i)), elegidos de tal manera que la matriz
!> escalada B con elementos B(i,j) = S(i)*A(i,j)*S(j) tenga unos en la diagonal.
!> Esta elección de S pone el número de condición de B dentro de un factor N
!> del número de condición mínimo posible sobre todos los escalados diagonales
!> posibles.
!>

Parámetros

UPLO 
!>          UPLO es CHARACTER*1
!>          = 'U':  Triángulo superior de A está almacenado;
!>          = 'L':  Triángulo inferior de A está almacenado.
!>

N

!>          N es INTEGER
!>          El orden de la matriz A.  N >= 0.
!>

KD

!>          KD es INTEGER
!>          El número de superdiagonales de la matriz A si UPLO = 'U',
!>          o el número de subdiagonales si UPLO = 'L'.  KD >= 0.
!>

AB

!>          AB es COMPLEX*16 array, dimension (LDAB,N)
!>          El triángulo superior o inferior de la matriz banda Hermítica A,
!>          almacenado en las primeras KD+1 filas del array. La j-ésima columna
!>          de A está almacenada en la j-ésima columna del array AB de la siguiente
!>          manera:
!>          si UPLO = 'U', AB(kd+1+i-j,j) = A(i,j) para max(1,j-kd)<=i<=j;
!>          si UPLO = 'L', AB(1+i-j,j)    = A(i,j) para j<=i<=min(n,j+kd).
!>

LDAB

!>          LDAB es INTEGER
!>          La dimensión principal del array A.  LDAB >= KD+1.
!>

S

!>          S es DOUBLE PRECISION array, dimension (N)
!>          Si INFO = 0, S contiene los factores de escala para A.
!>

SCOND

!>          SCOND es DOUBLE PRECISION
!>          Si INFO = 0, S contiene la relación del S(i) más pequeño con
!>          el S(i) más grande. Si SCOND >= 0.1 y AMAX no es ni
!>          demasiado grande ni demasiado pequeño, no vale la pena escalar con S.
!>

AMAX

!>          AMAX es DOUBLE PRECISION
!>          Valor absoluto del elemento de matriz más grande. Si AMAX está muy
!>          cerca del desbordamiento o muy cerca de la subdesbordamiento,
!>          la matriz debe ser escalada.
!>

INFO

!>          INFO es INTEGER
!>          = 0:  salida exitosa
!>          < 0:  si INFO = -i, el i-ésimo argumento tuvo un valor ilegal.
!>          > 0:  si INFO = i, el i-ésimo elemento diagonal no es positivo.
!>
