subroutine zpst01 (character uplo, integer n, complex*16, dimension( lda, * ) a, integer lda, complex*16, dimension( ldafac, * ) afac, integer ldafac, complex*16, dimension( ldperm, * ) perm, integer ldperm, integer, dimension( * ) piv, double precision, dimension( * ) rwork, double precision resid, integer rank)

ZPST01

Propósito:

!>
!> ZPST01 reconstruye una matriz Hermitiana semidefinida positiva A
!> a partir de sus factores L o U y la matriz de permutación P y calcula
!> el residuo
!>    norm( P*L*L'*P' - A ) / ( N * norm(A) * EPS ) o
!>    norm( P*U'*U*P' - A ) / ( N * norm(A) * EPS ),
!> donde EPS es la epsilon de máquina, L' es la transpuesta conjugada de L,
!> y U' es la transpuesta conjugada de U.
!>

Parámetros

UPLO 
!>          UPLO es CHARACTER*1
!>          Especifica si la parte triangular superior o inferior de la
!>          matriz Hermitiana A está almacenada:
!>          = 'U':  Triangular superior
!>          = 'L':  Triangular inferior
!>

N

!>          N es INTEGER
!>          El número de filas y columnas de la matriz A.  N >= 0.
!>

A

!>          A es un array COMPLEX*16, de dimensión (LDA,N)
!>          La matriz Hermitiana original A.
!>

LDA

!>          LDA es INTEGER
!>          La dimensión principal del array A.  LDA >= max(1,N)
!>

AFAC

!>          AFAC es un array COMPLEX*16, de dimensión (LDAFAC,N)
!>          El factor L o U de la factorización L*L' o U'*U
!>          de A.
!>

LDAFAC

!>          LDAFAC es INTEGER
!>          La dimensión principal del array AFAC.  LDAFAC >= max(1,N).
!>

PERM

!>          PERM es un array COMPLEX*16, de dimensión (LDPERM,N)
!>          Sobrescrito con la matriz reconstruida, y luego con la
!>          diferencia P*L*L'*P' - A (o P*U'*U*P' - A)
!>

LDPERM

!>          LDPERM es INTEGER
!>          La dimensión principal del array PERM.
!>          LDAPERM >= max(1,N).
!>

PIV

!>          PIV es un array INTEGER, de dimensión (N)
!>          PIV es tal que las entradas distintas de cero son
!>          P( PIV( K ), K ) = 1.
!>

RWORK

!>          RWORK es un array DOUBLE PRECISION, de dimensión (N)
!>

RESID

!>          RESID es DOUBLE PRECISION
!>          Si UPLO = 'L', norm(L*L' - A) / ( N * norm(A) * EPS )
!>          Si UPLO = 'U', norm(U'*U - A) / ( N * norm(A) * EPS )
!>

RANK

!>          RANK es INTEGER
!>          número de valores singulares distintos de cero de A.
!>
