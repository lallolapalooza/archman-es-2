subroutine zpoequ (integer n, complex*16, dimension( lda, * ) a, integer lda, double precision, dimension( * ) s, double precision scond, double precision amax, integer info)

ZPOEQU

Propósito:

!>
!> ZPOEQU calcula escalados de fila y columna destinados a equilibrar una
!> matriz Hermitiana definida positiva A y reducir su número de condición
!> (con respecto a la norma dos). S contiene los factores de escala,
!> S(i) = 1/sqrt(A(i,i)), elegidos de modo que la matriz escalada B con
!> elementos B(i,j) = S(i)*A(i,j)*S(j) tenga unos en la diagonal. Esta
!> elección de S pone el número de condición de B dentro de un factor N del
!> menor número de condición posible sobre todos los escalados diagonales
!> posibles.
!>

Parámetros

N 
!>          N es INTEGER
!>          El orden de la matriz A. N >= 0.
!>

A

!>          A es COMPLEX*16 array, dimension (LDA,N)
!>          La matriz Hermitiana definida positiva N por N cuyos factores de
!>          escala se van a calcular. Solo se referencian los elementos
!>          diagonales de A.
!>

LDA

!>          LDA es INTEGER
!>          La dimensión principal del array A. LDA >= max(1,N).
!>

S

!>          S es DOUBLE PRECISION array, dimension (N)
!>          Si INFO = 0, S contiene los factores de escala para A.
!>

SCOND

!>          SCOND es DOUBLE PRECISION
!>          Si INFO = 0, S contiene la relación del S(i) más pequeño con
!>          el S(i) más grande. Si SCOND >= 0.1 y AMAX no es demasiado
!>          grande ni demasiado pequeño, no vale la pena escalar con S.
!>

AMAX

!>          AMAX es DOUBLE PRECISION
!>          Valor absoluto del elemento de matriz más grande. Si AMAX está muy
!>          cerca del desbordamiento o del subdesbordamiento, la matriz
!>          debe ser escalada.
!>

INFO

!>          INFO es INTEGER
!>          = 0:   Salida exitosa
!>          < 0:   si INFO = -i, el i-ésimo argumento tuvo un valor ilegal
!>          > 0:   si INFO = i, el i-ésimo elemento diagonal es no positivo.
!>
