subroutine zpocon (character uplo, integer n, complex*16, dimension( lda, * ) a, integer lda, double precision anorm, double precision rcond, complex*16, dimension( * ) work, double precision, dimension( * ) rwork, integer info)

ZPOCON

Propósito:

!>
!> ZPOCON estima el recíproco del número de condición (en la
!> norma 1) de una matriz Hermitiana positiva definida compleja utilizando
!> la factorización de Cholesky A = U**H*U o A = L*L**H calculada por ZPOTRF.
!>
!> Se obtiene una estimación de norm(inv(A)), y el recíproco del
!> número de condición se calcula como RCOND = 1 / (ANORM * norm(inv(A))).
!>

Parámetros

UPLO 
!>          UPLO es CHARACTER*1
!>          = 'U':  El triángulo superior de A está almacenado;
!>          = 'L':  El triángulo inferior de A está almacenado.
!>

N

!>          N es INTEGER
!>          El orden de la matriz A.  N >= 0.
!>

A

!>          A es un array COMPLEX*16, de dimensión (LDA,N)
!>          El factor triangular U o L de la factorización de Cholesky
!>          A = U**H*U o A = L*L**H, tal como se calcula por ZPOTRF.
!>

LDA

!>          LDA es INTEGER
!>          La dimensión principal del array A.  LDA >= max(1,N).
!>

ANORM

!>          ANORM es DOUBLE PRECISION
!>          La norma 1 (o norma infinito) de la matriz Hermitiana A.
!>

RCOND

!>          RCOND es DOUBLE PRECISION
!>          El recíproco del número de condición de la matriz A,
!>          calculado como RCOND = 1/(ANORM * AINVNM), donde AINVNM es una
!>          estimación de la norma 1 de inv(A) calculada en esta rutina.
!>

WORK

!>          WORK es un array COMPLEX*16, de dimensión (2*N)
!>

RWORK

!>          RWORK es un array DOUBLE PRECISION, de dimensión (N)
!>

INFO

!>          INFO es INTEGER
!>          = 0:  salida exitosa
!>          < 0:  si INFO = -i, el argumento i-ésimo tuvo un valor ilegal
!>
