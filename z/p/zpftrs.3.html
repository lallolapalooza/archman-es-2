subroutine zpftrs (character transr, character uplo, integer n, integer nrhs, complex*16, dimension( 0: * ) a, complex*16, dimension( ldb, * ) b, integer ldb, integer info)

ZPFTRS

Propósito:

!>
!> ZPFTRS resuelve un sistema de ecuaciones lineales A*X = B con una matriz
!> Hermitiana positiva definida A utilizando la factorización de Cholesky
!> A = U**H*U o A = L*L**H calculada por ZPFTRF.
!>

Parámetros

TRANSR 
!>          TRANSR es CHARACTER*1
!>          = 'N':  Se almacena la representación normal (TRANSR) de RFP A;
!>          = 'C':  Se almacena la representación transpuesta conjugada (TRANSR) de RFP A.
!>

UPLO

!>          UPLO es CHARACTER*1
!>          = 'U':  Se almacena el triángulo superior de RFP A;
!>          = 'L':  Se almacena el triángulo inferior de RFP A.
!>

N

!>          N es INTEGER
!>          El orden de la matriz A.  N >= 0.
!>

NRHS

!>          NRHS es INTEGER
!>          El número de lados derechos, es decir, el número de columnas
!>          de la matriz B.  NRHS >= 0.
!>

A

!>          A es un array COMPLEX*16, de dimensión ( N*(N+1)/2 );
!>          El factor triangular U o L de la factorización de Cholesky
!>          de RFP A = U**H*U o RFP A = L*L**H, tal como se calcula por ZPFTRF.
!>          Ver la nota a continuación para más detalles sobre RFP A.
!>

B

!>          B es un array COMPLEX*16, de dimensión (LDB,NRHS)
!>          En la entrada, la matriz del lado derecho B.
!>          En la salida, la matriz solución X.
!>

LDB

!>          LDB es INTEGER
!>          La dimensión principal del array B.  LDB >= max(1,N).
!>

INFO

!>          INFO es INTEGER
!>          = 0:  salida exitosa
!>          < 0:  si INFO = -i, el i-ésimo argumento tenía un valor ilegal
!>

Detalles adicionales:

!>
!>  Primero consideramos el Formato Empaquetado Estándar cuando N es par.
!>  Damos un ejemplo donde N = 6.
!>
!>      AP es Superior             AP es Inferior
!>
!>   00 01 02 03 04 05       00
!>      11 12 13 14 15       10 11
!>         22 23 24 25       20 21 22
!>            33 34 35       30 31 32 33
!>               44 45       40 41 42 43 44
!>                  55       50 51 52 53 54 55
!>
!>
!>  Sea TRANSR = 'N'. RFP almacena AP como sigue:
!>  Para UPLO = 'U', el trapecio superior A(0:5,0:2) consiste en las últimas
!>  tres columnas de la parte superior de AP. El triángulo inferior A(4:6,0:2) consiste en
!>  la transpuesta conjugada de las primeras tres columnas de la parte superior de AP.
!>  Para UPLO = 'L', el trapecio inferior A(1:6,0:2) consiste en las primeras
!>  tres columnas de la parte inferior de AP. El triángulo superior A(0:2,0:2) consiste en
!>  la transpuesta conjugada de las últimas tres columnas de la parte inferior de AP.
!>  Para denotar conjugado, colocamos -- encima del elemento. Esto cubre
!>  el caso N par y TRANSR = 'N'.
!>
!>         RFP A                   RFP A
!>
!>                                -- -- --
!>        03 04 05                33 43 53
!>                                   -- --
!>        13 14 15                00 44 54
!>                                      --
!>        23 24 25                10 11 55
!>
!>        33 34 35                20 21 22
!>        --
!>        00 44 45                30 31 32
!>        -- --
!>        01 11 55                40 41 42
!>        -- -- --
!>        02 12 22                50 51 52
!>
!>  Ahora, sea TRANSR = 'C'. RFP A en ambos casos de UPLO es solo la transpuesta
!>  conjugada de RFP A anterior. Por lo tanto, se obtiene:
!>
!>
!>           RFP A                   RFP A
!>
!>     -- -- -- --                -- -- -- -- -- --
!>     03 13 23 33 00 01 02    33 00 10 20 30 40 50
!>     -- -- -- -- --                -- -- -- -- --
!>     04 14 24 34 44 11 12    43 44 11 21 31 41 51
!>     -- -- -- -- -- --                -- -- -- --
!>     05 15 25 35 45 55 22    53 54 55 22 32 42 52
!>

!>  Damos un ejemplo donde N = 5.
!>
!>     AP es Superior              AP es Inferior
!>
!>   00 01 02 03 04              00
!>      11 12 13 14              10 11

!>            33 34              30 31 32 33
!>               44              40 41 42 43 44
!>
!>
!>  Sea TRANSR = 'N'. RFP almacena AP como sigue:

!>  tres columnas de la parte superior de AP. El triángulo inferior A(3:4,0:1) consiste en
!>  la transpuesta conjugada de las primeras dos columnas de la parte superior de AP.
!>  Para UPLO = 'L', el trapecio inferior A(0:4,0:2) consiste en las primeras
!>  tres columnas de la parte inferior de AP. El triángulo superior A(0:1,1:2) consiste en
!>  la transpuesta conjugada de las últimas dos columnas de la parte inferior de AP.
!>  Para denotar conjugado, colocamos -- encima del elemento. Esto cubre
!>  el caso N impar y TRANSR = 'N'.
