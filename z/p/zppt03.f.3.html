Documentación de función/subrutina

subroutine zppt03 (character uplo, integer n, complex*16, dimension( * ) a, complex*16, dimension( * ) ainv, complex*16, dimension( ldwork, * ) work, integer ldwork, double precision, dimension( * ) rwork, double precision rcond, double precision resid)

ZPPT03

Propósito:

!>
!> ZPPT03 calcula el residuo de una matriz hermítica empacada multiplicada por su
!> inversa:
!>    norm( I - A*AINV ) / ( N * norm(A) * norm(AINV) * EPS ),
!> donde EPS es la epsilon de la máquina.
!>

Parámetros

UPLO 
!>          UPLO es CHARACTER*1
!>          Especifica si la parte triangular superior o inferior de la
!>          matriz hermítica A está almacenada:
!>          = 'U':  Triangular superior
!>          = 'L':  Triangular inferior
!>

N

!>          N es INTEGER
!>          El número de filas y columnas de la matriz A.  N >= 0.
!>

A

!>          A es array COMPLEX*16, dimensión (N*(N+1)/2)
!>          La matriz hermítica original A, almacenada como una matriz
!>          triangular empacada.
!>

AINV

!>          AINV es array COMPLEX*16, dimensión (N*(N+1)/2)
!>          La inversa (hermítica) de la matriz A, almacenada como una matriz
!>          triangular empacada.
!>

WORK

!>          WORK es array COMPLEX*16, dimensión (LDWORK,N)
!>

LDWORK

!>          LDWORK es INTEGER
!>          La dimensión principal del array WORK.  LDWORK >= max(1,N).
!>

RWORK

!>          RWORK es array DOUBLE PRECISION, dimensión (N)
!>

RCOND

!>          RCOND es DOUBLE PRECISION
!>          El recíproco del número de condición de A, calculado como
!>          ( 1/norm(A) ) / norm(AINV).
!>

RESID

!>          RESID es DOUBLE PRECISION
!>          norm(I - A*AINV) / ( N * norm(A) * norm(AINV) * EPS )
!>
