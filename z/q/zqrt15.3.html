Documentación de la función/subrutina

subroutine zqrt15 (integer scale, integer rksel, integer m, integer n, integer nrhs, complex*16, dimension( lda, * ) a, integer lda, complex*16, dimension( ldb, * ) b, integer ldb, double precision, dimension( * ) s, integer rank, double precision norma, double precision normb, integer, dimension( 4 ) iseed, complex*16, dimension( lwork ) work, integer lwork)

ZQRT15

Propósito:

!>
!> ZQRT15 genera una matriz con rango completo o deficiente y de diversas
!> normas.
!>

Parámetros

SCALE 
!>          SCALE es INTEGER
!>          SCALE = 1: matriz con escala normal
!>          SCALE = 2: matriz escalada hacia arriba
!>          SCALE = 3: matriz escalada hacia abajo
!>

RKSEL

!>          RKSEL es INTEGER
!>          RKSEL = 1: matriz de rango completo
!>          RKSEL = 2: matriz de rango deficiente
!>

M

!>          M es INTEGER
!>          El número de filas de la matriz A.
!>

N

!>          N es INTEGER
!>          El número de columnas de A.
!>

NRHS

!>          NRHS es INTEGER
!>          El número de columnas de B.
!>

A

!>          A es una matriz COMPLEX*16, dimensión (LDA,N)
!>          La matriz A de M por N.
!>

LDA

!>          LDA es INTEGER
!>          La dimensión principal del array A.
!>

B

!>          B es una matriz COMPLEX*16, dimensión (LDB, NRHS)
!>          Una matriz que está en el espacio de rango de la matriz A.
!>

LDB

!>          LDB es INTEGER
!>          La dimensión principal del array B.
!>

S

!>          S es una matriz DOUBLE PRECISION, dimensión MIN(M,N)
!>          Valores singulares de A.
!>

RANK

!>          RANK es INTEGER
!>          número de valores singulares distintos de cero de A.
!>

NORMA

!>          NORMA es DOUBLE PRECISION
!>          norma uno de A.
!>

NORMB

!>          NORMB es DOUBLE PRECISION
!>          norma uno de B.
!>

ISEED

!>          ISEED es un array de enteros, dimensión (4)
!>          semilla para el generador de números aleatorios.
!>

WORK

!>          WORK es una matriz COMPLEX*16, dimensión (LWORK)
!>

LWORK

!>          LWORK es INTEGER
!>          longitud del espacio de trabajo requerido.
!>          LWORK >= MAX(M+MIN(M,N),NRHS*MIN(M,N),2*N+M)
!>
