subroutine zqrt02 (m, n, k, a, af, q, r, lda, tau, work, lwork, rwork, result)
ZQRT02

Documentación de función/subrutina

subrutina zqrt02 (integer m, integer n, integer k, complex*16, dimension( lda, * ) a, complex*16, dimension( lda, * ) af, complex*16, dimension( lda, * ) q, complex*16, dimension( lda, * ) r, integer lda, complex*16, dimension( * ) tau, complex*16, dimension( lwork ) work, integer lwork, double precision, dimension( * ) rwork, double precision, dimension( * ) result)

ZQRT02

Propósito:

!>
!> ZQRT02 prueba ZUNGQR, que genera una matriz m-por-n Q con
!> columnas ortonormales que se define como el producto de k
!> reflectores elementales.
!>
!> Dada la factorización QR de una matriz m-por-n A, ZQRT02 genera
!> la matriz ortogonal Q definida por la factorización de las primeras
!> k columnas de A; compara R(1:n,1:k) con Q(1:m,1:n)'*A(1:m,1:k),
!> y comprueba que las columnas de Q sean ortonormales.
!>

Parámetros

M 
!>          M es ENTERO
!>          El número de filas de la matriz Q a generar. M >= 0.
!>

N

!>          N es ENTERO
!>          El número de columnas de la matriz Q a generar.
!>          M >= N >= 0.
!>

K

!>          K es ENTERO
!>          El número de reflectores elementales cuyo producto define la
!>          matriz Q. N >= K >= 0.
!>

A

!>          A es un array COMPLEX*16, dimensión (LDA,N)
!>          La matriz m-por-n A que fue factorizada por ZQRT01.
!>

AF

!>          AF es un array COMPLEX*16, dimensión (LDA,N)
!>          Detalles de la factorización QR de A, como devuelve ZGEQRF.
!>          Ver ZGEQRF para más detalles.
!>

Q

!>          Q es un array COMPLEX*16, dimensión (LDA,N)
!>

R

!>          R es un array COMPLEX*16, dimensión (LDA,N)
!>

LDA

!>          LDA es ENTERO
!>          La dimensión principal de los arrays A, AF, Q y R. LDA >= M.
!>

TAU

!>          TAU es un array COMPLEX*16, dimensión (N)
!>          Los factores escalares de los reflectores elementales correspondientes
!>          a la factorización QR en AF.
!>

WORK

!>          WORK es un array COMPLEX*16, dimensión (LWORK)
!>

LWORK

!>          LWORK es ENTERO
!>          La dimensión del array WORK.
!>

RWORK

!>          RWORK es un array DOUBLE PRECISION, dimensión (M)
!>

RESULT

!>          RESULT es un array DOUBLE PRECISION, dimensión (2)
!>          Las razones de prueba:
!>          RESULT(1) = norm( R - Q'*A ) / ( M * norm(A) * EPS )
!>          RESULT(2) = norm( I - Q'*Q ) / ( M * EPS )
!>
