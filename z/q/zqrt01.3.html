Documentación de función/subrutina

subroutine zqrt01 (integer m, integer n, complex*16, dimension( lda, * ) a, complex*16, dimension( lda, * ) af, complex*16, dimension( lda, * ) q, complex*16, dimension( lda, * ) r, integer lda, complex*16, dimension( * ) tau, complex*16, dimension( lwork ) work, integer lwork, double precision, dimension( * ) rwork, double precision, dimension( * ) result)

ZQRT01

Propósito:

!>
!> ZQRT01 prueba ZGEQRF, que calcula la factorización QR de una matriz m por n
!> A, y prueba parcialmente ZUNGQR, que forma la matriz ortogonal m por m
!> Q.
!>
!> ZQRT01 compara R con Q'*A y comprueba que Q es ortogonal.
!>

Parámetros

M 
!>          M es INTEGER
!>          El número de filas de la matriz A.  M >= 0.
!>

N

!>          N es INTEGER
!>          El número de columnas de la matriz A.  N >= 0.
!>

A

!>          A es un array COMPLEX*16, dimensión (LDA,N)
!>          La matriz m por n A.
!>

AF

!>          AF es un array COMPLEX*16, dimensión (LDA,N)
!>          Detalles de la factorización QR de A, tal como devuelve ZGEQRF.
!>          Ver ZGEQRF para más detalles.
!>

Q

!>          Q es un array COMPLEX*16, dimensión (LDA,M)
!>          La matriz ortogonal m por m Q.
!>

R

!>          R es un array COMPLEX*16, dimensión (LDA,max(M,N))
!>

LDA

!>          LDA es INTEGER
!>          La dimensión principal de los arrays A, AF, Q y R.
!>          LDA >= max(M,N).
!>

TAU

!>          TAU es un array COMPLEX*16, dimensión (min(M,N))
!>          Los factores escalares de los reflectores elementales, tal como
!>          devuelve ZGEQRF.
!>

WORK

!>          WORK es un array COMPLEX*16, dimensión (LWORK)
!>

LWORK

!>          LWORK es INTEGER
!>          La dimensión del array WORK.
!>

RWORK

!>          RWORK es un array DOUBLE PRECISION, dimensión (M)
!>

RESULT

!>          RESULT es un array DOUBLE PRECISION, dimensión (2)
!>          Las razones de prueba:
!>          RESULT(1) = norm( R - Q'*A ) / ( M * norm(A) * EPS )
!>          RESULT(2) = norm( I - Q'*Q ) / ( M * EPS )
!>
