Documentación de función/subrutina

subroutine zqrt16 (character trans, integer m, integer n, integer nrhs, complex*16, dimension( lda, * ) a, integer lda, complex*16, dimension( ldx, * ) x, integer ldx, complex*16, dimension( ldb, * ) b, integer ldb, double precision, dimension( * ) rwork, double precision resid)

ZQRT16

Propósito:

!>
!> ZQRT16 calcula el residuo de una solución de un sistema de ecuaciones
!> lineales  A*x = b  o  A'*x = b:
!>    RESID = norm(B - A*X) / ( max(m,n) * norm(A) * norm(X) * EPS ),
!> donde EPS es la epsilon de la máquina.
!>

Parámetros

TRANS 
!>          TRANS es CHARACTER*1
!>          Especifica la forma del sistema de ecuaciones:
!>          = 'N':  A *x = b
!>          = 'T':  A^T*x = b, donde A^T es la transpuesta de A
!>          = 'C':  A^H*x = b, donde A^H es la traspuesta conjugada de A
!>

M

!>          M es INTEGER
!>          El número de filas de la matriz A.  M >= 0.
!>

N

!>          N es INTEGER
!>          El número de columnas de la matriz A.  N >= 0.
!>

NRHS

!>          NRHS es INTEGER
!>          El número de columnas de B, la matriz de lados derechos.
!>          NRHS >= 0.
!>

A

!>          A es COMPLEX*16 array, dimension (LDA,N)
!>          La matriz original A de M x N.
!>

LDA

!>          LDA es INTEGER
!>          La dimensión principal del array A.  LDA >= max(1,M).
!>

X

!>          X es COMPLEX*16 array, dimension (LDX,NRHS)
!>          La solución computada en vectores para el sistema de ecuaciones
!>          lineales.
!>

LDX

!>          LDX es INTEGER
!>          La dimensión principal del array X.  Si TRANS = 'N',
!>          LDX >= max(1,N); si TRANS = 'T' o 'C', LDX >= max(1,M).
!>

B

!>          B es COMPLEX*16 array, dimension (LDB,NRHS)
!>          En la entrada, los vectores del lado derecho para el sistema de
!>          ecuaciones lineales.
!>          En la salida, B se sobrescribe con la diferencia B - A*X.
!>

LDB

!>          LDB es INTEGER
!>          La dimensión principal del array B.  SI TRANS = 'N',
!>          LDB >= max(1,M); si TRANS = 'T' o 'C', LDB >= max(1,N).
!>

RWORK

!>          RWORK es DOUBLE PRECISION array, dimension (M)
!>

RESID

!>          RESID es DOUBLE PRECISION
!>          El máximo sobre el número de lados derechos de
!>          norm(B - A*X) / ( max(m,n) * norm(A) * norm(X) * EPS ).
!>
