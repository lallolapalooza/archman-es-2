subroutine zqrt13 (integer scale, integer m, integer n, complex*16, dimension( lda, * ) a, integer lda, double precision norma, integer, dimension( 4 ) iseed)

ZQRT13

Propósito:

!>
!> ZQRT13 genera una matriz de rango completo que puede escalarse para tener
!> una norma grande o pequeña.
!>

Parámetros

SCALE 
!>          SCALE es INTEGER
!>          SCALE = 1: matriz escalada normalmente
!>          SCALE = 2: matriz escalada hacia arriba
!>          SCALE = 3: matriz escalada hacia abajo
!>

M

!>          M es INTEGER
!>          El número de filas de la matriz A.
!>

N

!>          N es INTEGER
!>          El número de columnas de A.
!>

A

!>          A es un array COMPLEX*16, dimensión (LDA,N)
!>          La matriz M por N, A.
!>

LDA

!>          LDA es INTEGER
!>          La dimensión principal del array A.
!>

NORMA

!>          NORMA es DOUBLE PRECISION
!>          La norma uno de A.
!>

ISEED

!>          ISEED es un array de enteros, dimensión (4)
!>          Semilla para el generador de números aleatorios
!>
