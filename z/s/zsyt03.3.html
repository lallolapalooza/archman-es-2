subroutine zsyt03 (character uplo, integer n, complex*16, dimension( lda, * ) a, integer lda, complex*16, dimension( ldainv, * ) ainv, integer ldainv, complex*16, dimension( ldwork, * ) work, integer ldwork, double precision, dimension( * ) rwork, double precision rcond, double precision resid)

ZSYT03

Propósito:

!>
!> ZSYT03 calcula el residuo de una matriz compleja simétrica multiplicada
!> por su inversa:
!>    norm( I - A*AINV ) / ( N * norm(A) * norm(AINV) * EPS )
!> donde EPS es la epsilon de la máquina.
!>

Parámetros

UPLO 
!>          UPLO es CHARACTER*1
!>          Especifica si la parte triangular superior o inferior de la
!>          matriz simétrica compleja A está almacenada:
!>          = 'U':  Triangular superior
!>          = 'L':  Triangular inferior
!>

N

!>          N es INTEGER
!>          El número de filas y columnas de la matriz A.  N >= 0.
!>

A

!>          A es un array COMPLEX*16, dimension (LDA,N)
!>          La matriz simétrica compleja original A.
!>

LDA

!>          LDA es INTEGER
!>          La dimensión principal del array A.  LDA >= max(1,N)
!>

AINV

!>          AINV es un array COMPLEX*16, dimension (LDAINV,N)
!>          En la entrada, la inversa de la matriz A, almacenada como una
!>          matriz simétrica en el mismo formato que A.
!>          En esta versión, AINV se expande a una matriz completa y
!>          se multiplica por A, por lo que el triángulo opuesto de AINV
!>          se modificará; es decir, si la parte triangular superior de AINV
!>          está almacenada, la parte triangular inferior se utilizará como
!>          espacio de trabajo.
!>

LDAINV

!>          LDAINV es INTEGER
!>          La dimensión principal del array AINV.  LDAINV >= max(1,N).
!>

WORK

!>          WORK es un array COMPLEX*16, dimension (LDWORK,N)
!>

LDWORK

!>          LDWORK es INTEGER
!>          La dimensión principal del array WORK.  LDWORK >= max(1,N).
!>

RWORK

!>          RWORK es un array DOUBLE PRECISION, dimension (N)
!>

RCOND

!>          RCOND es DOUBLE PRECISION
!>          El recíproco del número de condición de A, calculado como
!>          RCOND = 1/ (norm(A) * norm(AINV)).
!>

RESID

!>          RESID es DOUBLE PRECISION
!>          norm(I - A*AINV) / ( N * norm(A) * norm(AINV) * EPS )
!>
