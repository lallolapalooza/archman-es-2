Documentación de la función/subrutina

subroutine zsytrs_aa (character uplo, integer n, integer nrhs, complex*16, dimension( lda, * ) a, integer lda, integer, dimension( * ) ipiv, complex*16, dimension( ldb, * ) b, integer ldb, complex*16, dimension( * ) work, integer lwork, integer info)

ZSYTRS_AA

Propósito:

!>
!> ZSYTRS_AA resuelve un sistema de ecuaciones lineales A*X = B con una matriz
!> simétrica compleja A utilizando la factorización A = U**T*T*U o
!> A = L*T*L**T calculada por ZSYTRF_AA.
!>

Parámetros

UPLO 
!>          UPLO es CHARACTER*1
!>          Especifica si los detalles de la factorización se almacenan
!>          como una matriz triangular superior o inferior.
!>          = 'U': Triangular superior, forma es A = U**T*T*U;
!>          = 'L': Triangular inferior, forma es A = L*T*L**T.
!>

N

!>          N es INTEGER
!>          El orden de la matriz A.  N >= 0.
!>

NRHS

!>          NRHS es INTEGER
!>          El número de lados derechos, es decir, el número de columnas
!>          de la matriz B.  NRHS >= 0.
!>

A

!>          A es un array COMPLEX*16, dimensión (LDA,N)
!>          Detalles de los factores calculados por ZSYTRF_AA.
!>

LDA

!>          LDA es INTEGER
!>          La dimensión principal del array A.  LDA >= max(1,N).
!>

IPIV

!>          IPIV es un array INTEGER, dimensión (N)
!>          Detalles de los intercambios calculados por ZSYTRF_AA.
!>

B

!>          B es un array COMPLEX*16, dimensión (LDB,NRHS)
!>          En la entrada, la matriz del lado derecho B.
!>          En la salida, la matriz solución X.
!>

LDB

!>          LDB es INTEGER
!>          La dimensión principal del array B.  LDB >= max(1,N).
!>

WORK

!>          WORK es un array COMPLEX*16, dimensión (MAX(1,LWORK))
!>

LWORK

!>          LWORK es INTEGER
!>          La dimensión del array WORK. LWORK >= max(1,3*N-2).
!>

INFO

!>          INFO es INTEGER
!>          = 0:  salida exitosa
!>          < 0:  si INFO = -i, el i-ésimo argumento tenía un valor ilegal
!>
