subroutine zsytrs_aa_2stage (character uplo, integer n, integer nrhs, complex*16, dimension( lda, * ) a, integer lda, complex*16, dimension( * ) tb, integer ltb, integer, dimension( * ) ipiv, integer, dimension( * ) ipiv2, complex*16, dimension( ldb, * ) b, integer ldb, integer info)

ZSYTRS_AA_2STAGE

Propósito:

!>
!> ZSYTRS_AA_2STAGE resuelve un sistema de ecuaciones lineales A*X = B con una matriz
!> simétrica compleja A usando la factorización A = U**T*T*U o
!> A = L*T*L**T calculada por ZSYTRF_AA_2STAGE.
!>

Parámetros

UPLO 
!>          UPLO es CHARACTER*1
!>          Especifica si los detalles de la factorización se almacenan
!>          como una matriz triangular superior o inferior.
!>          = 'U':  Triangular superior, la forma es A = U**T*T*U;
!>          = 'L':  Triangular inferior, la forma es A = L*T*L**T.
!>

N

!>          N es INTEGER
!>          El orden de la matriz A.  N >= 0.
!>

NRHS

!>          NRHS es INTEGER
!>          El número de lados derechos, es decir, el número de columnas
!>          de la matriz B.  NRHS >= 0.
!>

A

!>          A es un array COMPLEX*16, dimensión (LDA,N)
!>          Detalles de los factores calculados por ZSYTRF_AA_2STAGE.
!>

LDA

!>          LDA es INTEGER
!>          La dimensión principal del array A.  LDA >= max(1,N).
!>

TB

!>          TB es un array COMPLEX*16, dimensión (LTB)
!>          Detalles de los factores calculados por ZSYTRF_AA_2STAGE.
!>

LTB

!>          LTB es INTEGER
!>          El tamaño del array TB. LTB >= 4*N.
!>

IPIV

!>          IPIV es un array INTEGER, dimensión (N)
!>          Detalles de los intercambios calculados por
!>          ZSYTRF_AA_2STAGE.
!>

IPIV2

!>          IPIV2 es un array INTEGER, dimensión (N)
!>          Detalles de los intercambios calculados por
!>          ZSYTRF_AA_2STAGE.
!>

B

!>          B es un array COMPLEX*16, dimensión (LDB,NRHS)
!>          En la entrada, la matriz del lado derecho B.
!>          En la salida, la matriz solución X.
!>

LDB

!>          LDB es INTEGER
!>          La dimensión principal del array B.  LDB >= max(1,N).
!>

INFO

!>          INFO es INTEGER
!>          = 0:  salida exitosa
!>          < 0:  si INFO = -i, el i-ésimo argumento tenía un valor ilegal
!>
