subroutine zsymm (side, uplo, m, n, alpha, a, lda, b, ldb, beta, c, ldc)
ZSYMM

Documentación de la función/subrutina

subroutine zsymm (character side, character uplo, integer m, integer n, complex*16 alpha, complex*16, dimension(lda,*) a, integer lda, complex*16, dimension(ldb,*) b, integer ldb, complex*16 beta, complex*16, dimension(ldc,*) c, integer ldc)

ZSYMM

Propósito:

!>
!> ZSYMM  realiza una de las operaciones de matriz-matriz
!>
!>    C := alpha*A*B + beta*C,
!>
!> o
!>
!>    C := alpha*B*A + beta*C,
!>
!> donde  alpha y beta son escalares, A es una matriz simétrica y  B y
!> C son matrices m por n.
!>

Parámetros

SIDE 
!>          SIDE es CHARACTER*1
!>           En la entrada,  SIDE  especifica si la  matriz simétrica  A
!>           aparece a la  izquierda o a la derecha  en la  operación de la
!>           siguiente manera:
!>
!>              SIDE = 'L' o 'l'   C := alpha*A*B + beta*C,
!>
!>              SIDE = 'R' o 'r'   C := alpha*B*A + beta*C,
!>

UPLO

!>          UPLO es CHARACTER*1
!>           En la  entrada,   UPLO  especifica  si  la parte  superior  o
!>           inferior  triangular  de  la  matriz simétrica   A  será  a la
!>           que se hace referencia de la siguiente manera:
!>
!>              UPLO = 'U' o 'u'   Sólo se hará referencia a la parte triangular
!>                                  superior de la matriz simétrica.
!>
!>              UPLO = 'L' o 'l'   Sólo se hará referencia a la parte triangular
!>                                  inferior de la matriz simétrica.
!>

M

!>          M es INTEGER
!>           En la entrada,  M  especifica el número de filas de la matriz C.
!>           M  debe ser como mínimo cero.
!>

N

!>          N es INTEGER
!>           En la entrada, N especifica el número de columnas de la matriz C.
!>           N  debe ser como mínimo cero.
!>

ALPHA

!>          ALPHA es COMPLEX*16
!>           En la entrada, especifica el escalar alpha.
!>

A

!>          A es un array COMPLEX*16, de dimensión ( LDA, ka ), donde ka es
!>           m  cuando  SIDE = 'L' o 'l'  y es n  en caso contrario.
!>           Antes de la entrada con  SIDE = 'L' o 'l', la parte de m por m
!>           del array  A  debe contener la  matriz simétrica, de tal forma
!>           que cuando  UPLO = 'U' o 'u', la parte triangular superior de m por m
!>           que lidera el array  A  debe contener la parte triangular superior
!>           de la  matriz simétrica y la parte estrictamente triangular inferior
!>           de  A  no se referencia, y cuando  UPLO = 'L' o 'l',
!>           la parte triangular inferior de m por m que lidera el array  A
!>           debe  contener la parte triangular inferior  de la matriz simétrica
!>           y la parte estrictamente triangular superior de  A  no se referencia.
!>           Antes de la entrada con  SIDE = 'R' o 'r', la parte de n por n
!>           del array  A  debe contener la matriz simétrica, de tal forma
!>           que cuando  UPLO = 'U' o 'u', la parte triangular superior de n por n
!>           que lidera el array  A  debe contener la parte triangular superior
!>           de la matriz simétrica y la parte estrictamente triangular inferior
!>           de  A  no se referencia, y cuando  UPLO = 'L' o 'l',
!>           la parte triangular inferior de n por n que lidera el array  A
!>           debe  contener la parte triangular inferior  de la matriz simétrica
!>           y la parte estrictamente triangular superior de  A  no se referencia.
!>

LDA

!>          LDA es INTEGER
!>           En la entrada, LDA especifica la primera dimensión de A tal como
!>           se declaró en el programa (sub)llamador. Cuando  SIDE = 'L' o 'l'
!>           entonces  LDA  debe ser al menos  max( 1, m ), de lo contrario
!>           LDA debe ser al menos max( 1, n ).
!>

B

!>          B es un array COMPLEX*16, de dimensión ( LDB, N )
!>           Antes de la entrada, la parte de m por n que lidera el array  B  debe
!>           contener la matriz B.
!>

LDB

!>          LDB es INTEGER
!>           En la entrada, LDB especifica la primera dimensión de B tal como
!>           se declaró en el programa (sub)llamador.   LDB  debe ser al menos
!>           max( 1, m ).
!>

BETA

!>          BETA es COMPLEX*16
!>           En la entrada,  BETA  especifica el escalar  beta.  Cuando  BETA
!>           se suministra como cero, entonces C no necesita ser establecida en
!>           la entrada.
!>

C

!>          C es un array COMPLEX*16, de dimensión ( LDC, N )
!>           Antes de la entrada, la parte de m por n que lidera el array  C  debe
!>           contener la matriz  C,  excepto cuando beta es cero, en cuyo
!>           caso C no necesita ser establecida en la entrada.
!>           En la salida, el array  C  se sobrescribe con la  matriz actualizada
!>           de m por n.
!>

LDC

!>          LDC es INTEGER
!>           En la entrada, LDC especifica la primera dimensión de C tal como
!>           se declaró en el programa (sub)llamador.   LDC  debe ser al menos
!>           max( 1, m ).
!>
