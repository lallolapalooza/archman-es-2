subroutine zstt21 (integer n, integer kband, double precision, dimension( * ) ad, double precision, dimension( * ) ae, double precision, dimension( * ) sd, double precision, dimension( * ) se, complex*16, dimension( ldu, * ) u, integer ldu, complex*16, dimension( * ) work, double precision, dimension( * ) rwork, double precision, dimension( 2 ) result)

ZSTT21

Propósito:

!>
!> ZSTT21  comprueba una descomposición de la forma
!>
!>    A = U S U**H
!>
!> donde **H significa transpuesta conjugada, A es simétrica tridiagonal real,
!> U es unitaria y S es real y diagonal (si KBAND=0) o simétrica
!> tridiagonal (si KBAND=1).  Se realizan dos pruebas:
!>
!>    RESULT(1) = | A - U S U**H | / ( |A| n ulp )
!>
!>    RESULT(2) = | I - U U**H | / ( n ulp )
!>

Parámetros

N 
!>          N es INTEGER
!>          El tamaño de la matriz.  Si es cero, ZSTT21 no hace nada.
!>          Debe ser mayor o igual que cero.
!>

KBAND

!>          KBAND es INTEGER
!>          El ancho de banda de la matriz S.  Solo puede ser cero o uno.
!>          Si es cero, entonces S es diagonal y SE no se referencia.  Si
!>          es uno, entonces S es simétrica tridiagonal.
!>

AD

!>          AD es DOUBLE PRECISION array, dimension (N)
!>          La diagonal de la matriz A original (no factorizada). Se supone
!>          que A es simétrica tridiagonal real.
!>

AE

!>          AE es DOUBLE PRECISION array, dimension (N-1)
!>          La subdiagonal de la matriz A original (no factorizada). Se supone
!>          que A es simétrica tridiagonal.  AE(1) es el elemento (1,2)
!>          y (2,1), AE(2) es el elemento (2,3) y (3,2), etc.
!>

SD

!>          SD es DOUBLE PRECISION array, dimension (N)
!>          La diagonal de la matriz S real (diagonal simétrica tri-).
!>

SE

!>          SE es DOUBLE PRECISION array, dimension (N-1)
!>          La subdiagonal de la matriz S (diagonal simétrica tri-).
!>          No se referencia si KBSND=0.  Si KBAND=1, entonces AE(1) es el
!>          elemento (1,2) y (2,1), SE(2) es el elemento (2,3) y (3,2),
!>          etc.
!>

U

!>          U es COMPLEX*16 array, dimension (LDU, N)
!>          La matriz unitaria en la descomposición.
!>

LDU

!>          LDU es INTEGER
!>          La dimensión principal de U.  LDU debe ser al menos N.
!>

WORK

!>          WORK es COMPLEX*16 array, dimension (N**2)
!>

RWORK

!>          RWORK es DOUBLE PRECISION array, dimension (N)
!>

RESULT

!>          RESULT es DOUBLE PRECISION array, dimension (2)
!>          Los valores calculados por las dos pruebas descritas anteriormente.
!>          Los valores se limitan actualmente a 1/ulp, para evitar desbordamiento.
!>          RESULT(1) siempre se modifica.
!>
