subroutine zsptri (character uplo, integer n, complex*16, dimension( * ) ap, integer, dimension( * ) ipiv, complex*16, dimension( * ) work, integer info)

ZSPTRI

Propósito:

!>
!> ZSPTRI calcula la inversa de una matriz simétrica indefinida compleja
!> en almacenamiento empaquetado utilizando la factorización A = U*D*U**T o
!> A = L*D*L**T calculada por ZSPTRF.
!>

Parámetros

UPLO 
!>          UPLO es CHARACTER*1
!>          Especifica si los detalles de la factorización se almacenan
!>          como una matriz triangular superior o inferior.
!>          = 'U':  Triangular superior, la forma es A = U*D*U**T;
!>          = 'L':  Triangular inferior, la forma es A = L*D*L**T.
!>

N

!>          N es INTEGER
!>          El orden de la matriz A.  N >= 0.
!>

AP

!>          AP es un array COMPLEX*16, de dimensión (N*(N+1)/2)
!>          En la entrada, la matriz diagonal por bloques D y los multiplicadores
!>          utilizados para obtener el factor U o L calculados por ZSPTRF,
!>          almacenados como una matriz triangular empaquetada.
!>
!>          En la salida, si INFO = 0, la inversa (simétrica) de la matriz
!>          original, almacenada como una matriz triangular empaquetada. La j-ésima
!>          columna de inv(A) se almacena en el array AP de la siguiente manera:
!>          si UPLO = 'U', AP(i + (j-1)*j/2) = inv(A)(i,j) para 1<=i<=j;
!>          si UPLO = 'L',
!>             AP(i + (j-1)*(2n-j)/2) = inv(A)(i,j) para j<=i<=n.
!>

IPIV

!>          IPIV es un array INTEGER, de dimensión (N)
!>          Detalles de los intercambios y la estructura de bloques de D
!>          determinados por ZSPTRF.
!>

WORK

!>          WORK es un array COMPLEX*16, de dimensión (N)
!>

INFO

!>          INFO es INTEGER
!>          = 0: salida exitosa
!>          < 0: si INFO = -i, el i-ésimo argumento tenía un valor ilegal
!>          > 0: si INFO = i, D(i,i) = 0; la matriz es singular y su
!>               inversa no pudo ser calculada.
!>
