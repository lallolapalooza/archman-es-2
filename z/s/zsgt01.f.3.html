subroutine zsgt01 (integer itype, character uplo, integer n, integer m, complex*16, dimension( lda, * ) a, integer lda, complex*16, dimension( ldb, * ) b, integer ldb, complex*16, dimension( ldz, * ) z, integer ldz, double precision, dimension( * ) d, complex*16, dimension( * ) work, double precision, dimension( * ) rwork, double precision, dimension( * ) result)

ZSGT01

Propósito:

!>
!> CDGT01 comprueba una descomposición de la forma
!>
!>    A Z   =  B Z D o
!>    A B Z =  Z D o
!>    B A Z =  Z D
!>
!> donde A es una matriz Hermítica, B es Hermítica positiva definida,
!> Z es unitaria y D es diagonal.
!>
!> Se calcula uno de los siguientes ratios de prueba:
!>
!> ITYPE = 1:  RESULT(1) = | A Z - B Z D | / ( |A| |Z| n ulp )
!>
!> ITYPE = 2:  RESULT(1) = | A B Z - Z D | / ( |A| |Z| n ulp )
!>
!> ITYPE = 3:  RESULT(1) = | B A Z - Z D | / ( |A| |Z| n ulp )
!>

Parámetros

ITYPE 
!>          ITYPE es INTEGER
!>          La forma del problema generalizado Hermítico de valores propios.
!>          = 1:  A*z = (lambda)*B*z
!>          = 2:  A*B*z = (lambda)*z
!>          = 3:  B*A*z = (lambda)*z
!>

UPLO

!>          UPLO es CHARACTER*1
!>          Especifica si la parte triangular superior o inferior de las
!>          matrices Hermíticas A y B está almacenada.
!>          = 'U':  Triangular superior
!>          = 'L':  Triangular inferior
!>

N

!>          N es INTEGER
!>          El orden de la matriz A.  N >= 0.
!>

M

!>          M es INTEGER
!>          El número de valores propios encontrados.  M >= 0.
!>

A

!>          A es COMPLEX*16 array, dimension (LDA, N)
!>          La matriz Hermítica original A.
!>

LDA

!>          LDA es INTEGER
!>          La dimensión principal del array A.  LDA >= max(1,N).
!>

B

!>          B es COMPLEX*16 array, dimension (LDB, N)
!>          La matriz Hermítica positiva definida original B.
!>

LDB

!>          LDB es INTEGER
!>          La dimensión principal del array B.  LDB >= max(1,N).
!>

Z

!>          Z es COMPLEX*16 array, dimension (LDZ, M)
!>          Los vectores propios calculados del problema generalizado de
!>          valores propios.
!>

LDZ

!>          LDZ es INTEGER
!>          La dimensión principal del array Z.  LDZ >= max(1,N).
!>

D

!>          D es DOUBLE PRECISION array, dimension (M)
!>          Los valores propios calculados del problema generalizado de
!>          valores propios.
!>

WORK

!>          WORK es COMPLEX*16 array, dimension (N*N)
!>

RWORK

!>          RWORK es DOUBLE PRECISION array, dimension (N)
!>

RESULT

!>          RESULT es DOUBLE PRECISION array, dimension (1)
!>          El ratio de prueba como se describe arriba.
!>
