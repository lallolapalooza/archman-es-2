subroutine zsyswapr (character uplo, integer n, complex*16, dimension( lda, * ) a, integer lda, integer i1, integer i2)

ZSYSWAPR

Propósito:

!>
!> ZSYSWAPR aplica una permutación elemental en las filas y columnas de
!> una matriz simétrica.
!>

Parámetros

UPLO 
!>          UPLO es CHARACTER*1
!>          Especifica si los detalles de la factorización se almacenan
!>          como una matriz triangular superior o inferior.
!>          = 'U':  Triangular superior, la forma es A = U*D*U**T;
!>          = 'L':  Triangular inferior, la forma es A = L*D*L**T.
!>

N

!>          N es INTEGER
!>          El orden de la matriz A.  N >= 0.
!>

A

!>          A es un array COMPLEX*16, dimensión (LDA,*)
!>          En la entrada, la matriz A de N por N. En la salida, la matriz
!>          permutada donde las filas I1 e I2 y las columnas I1 e I2 son
!>          intercambiadas.
!>          Si UPLO = 'U', los intercambios se aplican a la parte triangular
!>          superior y la parte triangular estrictamente inferior de A no
!>          se referencia; si UPLO = 'L', los intercambios se aplican a la
!>          parte triangular inferior y la parte de A por encima de la
!>          diagonal no se referencia.
!>

LDA

!>          LDA es INTEGER
!>          La dimensión principal del array A.  LDA >= max(1,N).
!>

I1

!>          I1 es INTEGER
!>          Índice de la primera fila a intercambiar
!>

I2

!>          I2 es INTEGER
!>          Índice de la segunda fila a intercambiar
!>
