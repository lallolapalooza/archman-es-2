subroutine zsytrs2 (character uplo, integer n, integer nrhs, complex*16, dimension( lda, * ) a, integer lda, integer, dimension( * ) ipiv, complex*16, dimension( ldb, * ) b, integer ldb, complex*16, dimension( * ) work, integer info)

ZSYTRS2

Propósito:

!>
!> ZSYTRS2 resuelve un sistema de ecuaciones lineales A*X = B con una matriz simétrica compleja A
!> usando la factorización A = U*D*U**T o A = L*D*L**T calculada por ZSYTRF y convertida por ZSYCONV.
!>

Parámetros

UPLO 
!>          UPLO es CHARACTER*1
!>          Especifica si los detalles de la factorización se almacenan
!>          como una matriz triangular superior o inferior.
!>          = 'U':  Triangular superior, forma es A = U*D*U**T;
!>          = 'L':  Triangular inferior, forma es A = L*D*L**T.
!>

N

!>          N es INTEGER
!>          El orden de la matriz A.  N >= 0.
!>

NRHS

!>          NRHS es INTEGER
!>          El número de lados derechos, es decir, el número de columnas
!>          de la matriz B.  NRHS >= 0.
!>

A

!>          A es un array COMPLEX*16, dimensión (LDA,N)
!>          La matriz diagonal por bloques D y los multiplicadores utilizados para
!>          obtener el factor U o L calculado por ZSYTRF.
!>          Tenga en cuenta que A es de entrada/salida. Esto puede ser contraintuitivo,
!>          y uno puede pensar que A es solo de entrada. A es de entrada/salida.
!>          Esto se debe a que, al inicio de la subrutina, permutamos A en una
!>          forma y luego la permutamos de vuelta a su forma original al
!>          final.
!>

LDA

!>          LDA es INTEGER
!>          La dimensión principal del array A.  LDA >= max(1,N).
!>

IPIV

!>          IPIV es un array INTEGER, dimensión (N)
!>          Detalles de los intercambios y la estructura de bloques de D
!>          según lo determinado por ZSYTRF.
!>

B

!>          B es un array COMPLEX*16, dimensión (LDB,NRHS)
!>          En la entrada, la matriz del lado derecho B.
!>          En la salida, la matriz solución X.
!>

LDB

!>          LDB es INTEGER
!>          La dimensión principal del array B.  LDB >= max(1,N).
!>

WORK

!>          WORK es un array COMPLEX*16, dimensión (N)
!>

INFO

!>          INFO es INTEGER
!>          = 0:  salida exitosa
!>          < 0:  si INFO = -i, el i-ésimo argumento tenía un valor ilegal
!>
