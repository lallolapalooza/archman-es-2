subroutine zunhr_col02 (integer m, integer n, integer mb1, integer nb1, integer nb2, double precision, dimension(6) result)

ZUNHR_COL02

Propósito:

!>
!> ZUNHR_COL02 prueba ZUNGTSQR_ROW y ZUNHR_COL dentro de ZGETSQRHRT
!> (que llama a ZLATSQR, ZUNGTSQR_ROW y ZUNHR_COL) usando ZGEMQRT.
!> Por lo tanto, ZLATSQR (parte de ZGEQR), ZGEMQRT (parte de ZGEMQR)
!> deben ser probados antes de esta prueba.
!>
!>

Parámetros

M 
!>          M es ENTERO
!>          Número de filas en la matriz de prueba.
!>

N

!>          N es ENTERO
!>          Número de columnas en la matriz de prueba.
!>

MB1

!>          MB1 es ENTERO
!>          Número de filas en el bloque de filas en una matriz de entrada de prueba.
!>

NB1

!>          NB1 es ENTERO
!>          Número de columnas en el bloque de columnas de una matriz de entrada de prueba.
!>

NB2

!>          NB2 es ENTERO
!>          Número de columnas en el bloque de columnas en una matriz de salida.
!>

RESULT

!>          RESULT es un array de DOBLE PRECISIÓN, de dimensión (6)
!>          Resultados de cada una de las seis pruebas siguientes.
!>
!>            A es una matriz de prueba de entrada m-por-n a ser factorizada.
!>            de tal manera que A = Q_gr * ( R )
!>                                     ( 0 ),
!>
!>            Q_qr es una matriz unitaria implícita m-por-m, el resultado
!>            de la factorización en representación WY bloqueada,
!>            almacenada en el formato de salida de ZGEQRT.
!>
!>            R es una matriz triangular superior n-por-n,
!>
!>            0 es una matriz cero (m-n)-por-n,
!>
!>            Q es una matriz unitaria explícita m-por-m Q = Q_gr * I
!>
!>            C es una matriz aleatoria m-por-n,
!>
!>            D es una matriz aleatoria n-por-m.
!>
!>          Las seis pruebas son:
!>
!>          RESULT(1) = |R - (Q**H) * A| / ( eps * m * |A| )
!>            es equivalente a la prueba de | A - Q * R | / (eps * m * |A|),
!>
!>          RESULT(2) = |I - (Q**H) * Q| / ( eps * m ),
!>
!>          RESULT(3) = | Q_qr * C - Q * C | / (eps * m * |C|),
!>
!>          RESULT(4) = | (Q_gr**H) * C - (Q**H) * C | / (eps * m * |C|)
!>
!>          RESULT(5) = | D * Q_qr - D * Q | / (eps * m * |D|)
!>
!>          RESULT(6) = | D * (Q_qr**H) - D * (Q**H) | / (eps * m * |D|),
!>
!>          donde:
!>            Q_qr * C, (Q_gr**H) * C, D * Q_qr, D * (Q_qr**H) se
!>            computan usando ZGEMQRT,
!>
!>            Q * C, (Q**H) * C, D * Q, D * (Q**H) se
!>            computan usando ZGEMM.
!>
