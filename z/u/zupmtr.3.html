subroutine zupmtr (character side, character uplo, character trans, integer m, integer n, complex*16, dimension( * ) ap, complex*16, dimension( * ) tau, complex*16, dimension( ldc, * ) c, integer ldc, complex*16, dimension( * ) work, integer info)

ZUPMTR

Propósito:

!>
!> ZUPMTR sobrescribe la matriz general compleja M por N, C, con
!>
!>                 SIDE = 'L'     SIDE = 'R'
!> TRANS = 'N':      Q * C          C * Q
!> TRANS = 'C':      Q**H * C       C * Q**H
!>
!> donde Q es una matriz unitaria compleja de orden nq, con nq = m si
!> SIDE = 'L' y nq = n si SIDE = 'R'. Q se define como el producto de
!> nq-1  reflectores elementales, como se devuelve por ZHPTRD usando almacenamiento empaquetado:
!>
!> si UPLO = 'U', Q = H(nq-1) . . . H(2) H(1);
!>
!> si UPLO = 'L', Q = H(1) H(2) . . . H(nq-1).
!>

Parámetros

SIDE 
!>          SIDE es CHARACTER*1
!>          = 'L': aplicar Q o Q**H desde la Izquierda;
!>          = 'R': aplicar Q o Q**H desde la Derecha.
!>

UPLO

!>          UPLO es CHARACTER*1
!>          = 'U': almacenamiento empaquetado triangular superior usado en la
!>                 llamada previa a ZHPTRD;
!>          = 'L': almacenamiento empaquetado triangular inferior usado en la
!>                 llamada previa a ZHPTRD.
!>

TRANS

!>          TRANS es CHARACTER*1
!>          = 'N': Sin transponer, aplicar Q;
!>          = 'C': Transpuesta conjugada, aplicar Q**H.
!>

M

!>          M es INTEGER
!>          El número de filas de la matriz C. M >= 0.
!>

N

!>          N es INTEGER
!>          El número de columnas de la matriz C. N >= 0.
!>

AP

!>          AP es un array COMPLEX*16, de dimensión
!>                               (M*(M+1)/2) si SIDE = 'L'
!>                               (N*(N+1)/2) si SIDE = 'R'
!>          Los vectores que definen los reflectores elementales, como
!>          se devuelve por ZHPTRD. AP es modificado por la rutina pero
!>          restaurado a la salida.
!>

TAU

!>          TAU es un array COMPLEX*16, de dimensión (M-1) si SIDE = 'L'
!>                                     o (N-1) si SIDE = 'R'
!>          TAU(i) debe contener el factor escalar del reflector
!>          elemental H(i), como se devuelve por ZHPTRD.
!>

C

!>          C es un array COMPLEX*16, de dimensión (LDC,N)
!>          A la entrada, la matriz C de M por N.
!>          A la salida, C es sobrescrito por Q*C o Q**H*C o C*Q**H o C*Q.
!>

LDC

!>          LDC es INTEGER
!>          La dimensión principal del array C. LDC >= max(1,M).
!>

WORK

!>          WORK es un array COMPLEX*16, de dimensión
!>                                   (N) si SIDE = 'L'
!>                                   (M) si SIDE = 'R'
!>

INFO

!>          INFO es INTEGER
!>          = 0:  salida exitosa
!>          < 0:  si INFO = -i, el i-ésimo argumento tenía un valor ilegal
!>
