subroutine zupgtr (character uplo, integer n, complex*16, dimension( * ) ap, complex*16, dimension( * ) tau, complex*16, dimension( ldq, * ) q, integer ldq, complex*16, dimension( * ) work, integer info)

ZUPGTR

Propósito:

!>
!> ZUPGTR genera una matriz unitaria compleja Q que se define como el
!> producto de n-1   reflectores elementales H(i) de orden n, como se devuelve
!> por ZHPTRD usando almacenamiento empaquetado:
!>
!> si UPLO = 'U', Q = H(n-1) . . . H(2) H(1),
!>
!> si UPLO = 'L', Q = H(1) H(2) . . . H(n-1).
!>

Parámetros

UPLO 
!>          UPLO es CHARACTER*1
!>          = 'U': Almacenamiento empaquetado triangular superior utilizado en
!>                 la llamada anterior a ZHPTRD;
!>          = 'L': Almacenamiento empaquetado triangular inferior utilizado en
!>                 la llamada anterior a ZHPTRD.
!>

N

!>          N es INTEGER
!>          El orden de la matriz Q. N >= 0.
!>

AP

!>          AP es un array COMPLEX*16, dimension (N*(N+1)/2)
!>          Los vectores que definen los reflectores elementales, como
!>          se devuelve por ZHPTRD.
!>

TAU

!>          TAU es un array COMPLEX*16, dimension (N-1)
!>          TAU(i) debe contener el factor escalar del reflector
!>          elemental H(i), como se devuelve por ZHPTRD.
!>

Q

!>          Q es un array COMPLEX*16, dimension (LDQ,N)
!>          La matriz unitaria Q de N por N.
!>

LDQ

!>          LDQ es INTEGER
!>          La dimensión principal del array Q. LDQ >= max(1,N).
!>

WORK

!>          WORK es un array COMPLEX*16, dimension (N-1)
!>

INFO

!>          INFO es INTEGER
!>          = 0:  salida exitosa
!>          < 0:  si INFO = -i, el argumento i tenía un valor ilegal
!>
