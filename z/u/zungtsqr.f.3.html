DOCUMENTACIÓN DE LA FUNCIÓN/SUBRUTINA

subroutine zungtsqr (integer m, integer n, integer mb, integer nb, complex*16, dimension( lda, * ) a, integer lda, complex*16, dimension( ldt, * ) t, integer ldt, complex*16, dimension( * ) work, integer lwork, integer info)

ZUNGTSQR

Propósito:

!>
!> ZUNGTSQR genera una matriz compleja M por N, Q_out, con columnas
!> ortonormales, que son las primeras N columnas de un producto de matrices
!> unitarias complejas de orden M que son devueltas por ZLATSQR.
!>
!>      Q_out = primeras_N_columnas( Q(1)_in * Q(2)_in * ... * Q(k)_in ).
!>
!> Ver la documentación de ZLATSQR.
!>

Parámetros

M 
!>          M es INTEGER
!>          El número de filas de la matriz A.  M >= 0.
!>

N

!>          N es INTEGER
!>          El número de columnas de la matriz A. M >= N >= 0.
!>

MB

!>          MB es INTEGER
!>          El tamaño de bloque de filas utilizado por ZLATSQR para devolver
!>          los arrays A y T. MB > N.
!>          (Nota: si MB > M, entonces se utiliza M en lugar de MB
!>          como tamaño de bloque de filas).
!>

NB

!>          NB es INTEGER
!>          El tamaño de bloque de columnas utilizado por ZLATSQR para devolver
!>          los arrays A y T. NB >= 1.
!>          (Nota: si NB > N, entonces se utiliza N en lugar de NB
!>          como tamaño de bloque de columnas).
!>

A

!>          A es array COMPLEX*16, dimension (LDA,N)
!>
!>          En la entrada:
!>
!>             Los elementos en y por encima de la diagonal no se acceden.
!>             Los elementos por debajo de la diagonal representan la matriz
!>             bloque triangular inferior unitaria V calculada por ZLATSQR
!>             que define las matrices de entrada Q_in(k) (las unos en la
!>             diagonal no se almacenan) (mismo formato que la salida A
!>             por debajo de la diagonal en ZLATSQR).
!>
!>          En la salida:
!>
!>             El array A contiene una matriz ortonormal M por N, Q_out,
!>             es decir, las columnas de A son vectores unitarios ortogonales.
!>

LDA

!>          LDA es INTEGER
!>          La dimensión principal del array A.  LDA >= max(1,M).
!>

T

!>          T es array COMPLEX*16,
!>          dimension (LDT, N * NIRB)
!>          donde NIRB = Número_de_bloques_de_filas_de_entrada
!>                     = MAX( 1, CEIL((M-N)/(MB-N)) )
!>          Sea NICB = Número_de_bloques_de_columnas_de_entrada
!>                   = CEIL(N/NB)
!>
!>          Los reflectores de bloque triangulares superiores que definen
!>          las matrices de entrada Q_in(k), k=(1:NIRB*NICB). Los reflectores
!>          de bloque se almacenan en forma compacta en N bloques de columnas
!>          de tamaño NB y consisten en NICB bloques de columnas más pequeños
!>          de tamaño NBxNB triangulares superiores.
!>          (mismo formato que la salida T en ZLATSQR).
!>

LDT

!>          LDT es INTEGER
!>          La dimensión principal del array T.
!>          LDT >= max(1,min(NB1,N)).
!>

WORK

!>          (espacio de trabajo) array COMPLEX*16, dimension (MAX(2,LWORK))
!>          En la salida, si INFO = 0, WORK(1) devuelve el LWORK óptimo.
!>

LWORK

!>          LWORK es INTEGER
!>          La dimensión del array WORK.  LWORK >= (M+NB)*N.
!>          Si LWORK = -1, se asume una consulta de espacio de trabajo.
!>          La rutina solo calcula el tamaño óptimo del array WORK,
!>          devuelve este valor como la primera entrada del array WORK,
!>          y XERBLA no emite ningún mensaje de error relacionado con LWORK.
!>

INFO

!>          INFO es INTEGER
!>          = 0:  salida exitosa
!>          < 0:  si INFO = -i, el i-ésimo argumento tenía un valor ilegal
!>
