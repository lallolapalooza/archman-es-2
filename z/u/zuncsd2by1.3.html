subroutine zuncsd2by1 (character jobu1, character jobu2, character jobv1t, integer m, integer p, integer q, complex*16, dimension(ldx11,*) x11, integer ldx11, complex*16, dimension(ldx21,*) x21, integer ldx21, double precision, dimension(*) theta, complex*16, dimension(ldu1,*) u1, integer ldu1, complex*16, dimension(ldu2,*) u2, integer ldu2, complex*16, dimension(ldv1t,*) v1t, integer ldv1t, complex*16, dimension(*) work, integer lwork, double precision, dimension(*) rwork, integer lrwork, integer, dimension(*) iwork, integer info)

ZUNCSD2BY1

Propósito:

!>
!> ZUNCSD2BY1 calcula la descomposición CS de una matriz M x Q, X, con
!> columnas ortonormales que ha sido particionada en una estructura de bloque
!> 2x1:
!>
!>                                [  I1 0  0 ]
!>                                [  0  C  0 ]
!>          [ X11 ]   [ U1 |    ] [  0  0  0 ]
!>      X = [-----] = [---------] [----------] V1**T .
!>          [ X21 ]   [    | U2 ] [  0  0  0 ]
!>                                [  0  S  0 ]
!>                                [  0  0  I2]
!>
!> X11 es PxQ. Las matrices unitarias U1, U2 y V1 son PxP,
!> (M-P)x(M-P) y QxQ, respectivamente. C y S son matrices diagonales
!> no negativas RxR que satisfacen C^2 + S^2 = I, donde
!> R = MIN(P,M-P,Q,M-Q). I1 es una matriz identidad K1xK1 e I2 es una
!> matriz identidad K2xK2, donde K1 = MAX(Q+P-M,0) y K2 = MAX(Q-P,0).
!>

Parámetros

JOBU1 
!>          JOBU1 es CHARACTER
!>          = 'Y':      Se calcula U1;
!>          en caso contrario:  No se calcula U1.
!>

JOBU2

!>          JOBU2 es CHARACTER
!>          = 'Y':      Se calcula U2;
!>          en caso contrario:  No se calcula U2.
!>

JOBV1T

!>          JOBV1T es CHARACTER
!>          = 'Y':      Se calcula V1T;
!>          en caso contrario:  No se calcula V1T.
!>

M

!>          M es INTEGER
!>          El número de filas de X.
!>

P

!>          P es INTEGER
!>          El número de filas de X11. 0 <= P <= M.
!>

Q

!>          Q es INTEGER
!>          El número de columnas de X11 y X21. 0 <= Q <= M.
!>

X11

!>          X11 es array COMPLEX*16, dimensión (LDX11,Q)
!>          En la entrada, parte de la matriz unitaria cuya CSD se desea.
!>

LDX11

!>          LDX11 es INTEGER
!>          La dimensión principal de X11. LDX11 >= MAX(1,P).
!>

X21

!>          X21 es array COMPLEX*16, dimensión (LDX21,Q)
!>          En la entrada, parte de la matriz unitaria cuya CSD se desea.
!>

LDX21

!>          LDX21 es INTEGER
!>          La dimensión principal de X21. LDX21 >= MAX(1,M-P).
!>

THETA

!>          THETA es array DOUBLE PRECISION, dimensión (R), donde R =
!>          MIN(P,M-P,Q,M-Q).
!>          C = DIAG( COS(THETA(1)), ... , COS(THETA(R)) ) y
!>          S = DIAG( SIN(THETA(1)), ... , SIN(THETA(R)) ).
!>

U1

!>          U1 es array COMPLEX*16, dimensión (P)
!>          Si JOBU1 = 'Y', U1 contiene la matriz unitaria PxP, U1.
!>

LDU1

!>          LDU1 es INTEGER
!>          La dimensión principal de U1. Si JOBU1 = 'Y', LDU1 >=
!>          MAX(1,P).
!>

U2

!>          U2 es array COMPLEX*16, dimensión (M-P)
!>          Si JOBU2 = 'Y', U2 contiene la matriz unitaria (M-P)x(M-P),
!>          U2.
!>

LDU2

!>          LDU2 es INTEGER
!>          La dimensión principal de U2. Si JOBU2 = 'Y', LDU2 >=
!>          MAX(1,M-P).
!>

V1T

!>          V1T es array COMPLEX*16, dimensión (Q)
!>          Si JOBV1T = 'Y', V1T contiene la matriz unitaria QxQ,
!>          V1**T.
!>

LDV1T

!>          LDV1T es INTEGER
!>          La dimensión principal de V1T. Si JOBV1T = 'Y', LDV1T >=
!>          MAX(1,Q).
!>

WORK

!>          WORK es array COMPLEX*16, dimensión (MAX(1,LWORK))
!>          En la salida, si INFO = 0, WORK(1) devuelve el LWORK óptimo.
!>

LWORK

!>          LWORK es INTEGER
!>          La dimensión del array WORK.
!>
!>          Si LWORK = -1, entonces se asume una consulta de espacio de
!>          trabajo; la rutina solo calcula el tamaño óptimo de los arrays
!>          WORK y RWORK, devuelve este valor como la primera entrada de los
!>          arrays WORK y RWORK respectivamente, y XERBLA no emite ningún
!>          mensaje de error relacionado con LWORK o LRWORK.
!>

RWORK

!>          RWORK es array DOUBLE PRECISION, dimensión (MAX(1,LRWORK))
!>          En la salida, si INFO = 0, RWORK(1) devuelve el LRWORK óptimo.
!>          Si INFO > 0 en la salida, RWORK(2:R) contiene los valores PHI(1),
!>          ..., PHI(R-1) que, junto con THETA(1), ..., THETA(R),
!>          definen la matriz en forma bidiagonal-bloque intermedia
!>          que queda tras la no convergencia. INFO especifica el número
!>          de PHI's distintos de cero.
!>

LRWORK

!>          LRWORK es INTEGER
!>          La dimensión del array RWORK.
!>
!>          Si LRWORK=-1, entonces se asume una consulta de espacio de
!>          trabajo; la rutina solo calcula el tamaño óptimo de los arrays
!>          WORK y RWORK, devuelve este valor como la primera entrada de los
!>          arrays WORK y RWORK respectivamente, y XERBLA no emite ningún
!>          mensaje de error relacionado con LWORK o LRWORK.
!>

IWORK

!>          IWORK es array INTEGER, dimensión (M-MIN(P,M-P,Q,M-Q))
!>

INFO

!>          INFO es INTEGER
!>          = 0:  salida exitosa.
!>          < 0:  si INFO = -i, el i-ésimo argumento tenía un valor ilegal.
!>          > 0:  ZBBCSD no convergió. Consulte la descripción de WORK
!>                anterior para obtener detalles.
!>

Referencias:

[1] Brian D. Sutton. Computing the complete CS decomposition. Numer. Algorithms, 50(1):33-65, 2009.
