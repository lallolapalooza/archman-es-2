Documentación de función/subrutina

subroutine zbdt01 (integer m, integer n, integer kd, complex*16, dimension( lda, * ) a, integer lda, complex*16, dimension( ldq, * ) q, integer ldq, double precision, dimension( * ) d, double precision, dimension( * ) e, complex*16, dimension( ldpt, * ) pt, integer ldpt, complex*16, dimension( * ) work, double precision, dimension( * ) rwork, double precision resid)

ZBDT01

Propósito:

!>
!> ZBDT01 reconstruye una matriz general A a partir de su forma bidiagonal
!>    A = Q * B * P**H
!> donde Q (m por min(m,n)) y P**H (min(m,n) por n) son matrices unitarias
!> y B es bidiagonal.
!>
!> La relación de prueba para verificar la reducción es
!>    RESID = norm(A - Q * B * P**H) / ( n * norm(A) * EPS )
!> donde EPS es la precisión de la máquina.
!>

Parámetros

M 
!>          M es INTEGER
!>          El número de filas de las matrices A y Q.
!>

N

!>          N es INTEGER
!>          El número de columnas de las matrices A y P**H.
!>

KD

!>          KD es INTEGER
!>          Si KD = 0, B es diagonal y el array E no se referencia.
!>          Si KD = 1, la reducción fue realizada por xGEBRD; B es
!>          bidiagonal superior si M >= N, y bidiagonal inferior si M < N.
!>          Si KD = -1, la reducción fue realizada por xGBBRD; B es
!>          siempre bidiagonal superior.
!>

A

!>          A es array COMPLEX*16, dimension (LDA,N)
!>          La matriz m por n A.
!>

LDA

!>          LDA es INTEGER
!>          La dimensión principal del array A.  LDA >= max(1,M).
!>

Q

!>          Q es array COMPLEX*16, dimension (LDQ,N)
!>          La matriz unitaria m por min(m,n) Q en la reducción
!>          A = Q * B * P**H.
!>

LDQ

!>          LDQ es INTEGER
!>          La dimensión principal del array Q.  LDQ >= max(1,M).
!>

D

!>          D es array DOUBLE PRECISION, dimension (min(M,N))
!>          Los elementos diagonales de la matriz bidiagonal B.
!>

E

!>          E es array DOUBLE PRECISION, dimension (min(M,N)-1)
!>          Los elementos superdiagonales de la matriz bidiagonal B si
!>          m >= n, o los elementos subdiagonales de B si m < n.
!>

PT

!>          PT es array COMPLEX*16, dimension (LDPT,N)
!>          La matriz unitaria min(m,n) por n P**H en la reducción
!>          A = Q * B * P**H.
!>

LDPT

!>          LDPT es INTEGER
!>          La dimensión principal del array PT.
!>          LDPT >= max(1,min(M,N)).
!>

WORK

!>          WORK es array COMPLEX*16, dimension (M+N)
!>

RWORK

!>          RWORK es array DOUBLE PRECISION, dimension (M)
!>

RESID

!>          RESID es DOUBLE PRECISION
!>          La relación de prueba:
!>          norm(A - Q * B * P**H) / ( n * norm(A) * EPS )
!>
