subroutine zget02 (character trans, integer m, integer n, integer nrhs, complex*16, dimension( lda, * ) a, integer lda, complex*16, dimension( ldx, * ) x, integer ldx, complex*16, dimension( ldb, * ) b, integer ldb, double precision, dimension( * ) rwork, double precision resid)

ZGET02

Propósito:

!>
!> ZGET02 calcula el residuo para una solución de un sistema de ecuaciones
!> lineales op(A)*X = B:
!>    RESID = norm(B - op(A)*X) / ( norm(op(A)) * norm(X) * EPS ),
!> donde op(A) = A, A**T, o A**H, dependiendo de TRANS, y EPS es la
!> epsilon de la máquina.
!>

Parámetros

TRANS 
!>          TRANS es CHARACTER*1
!>          Especifica la forma del sistema de ecuaciones:
!>          = 'N':  A    * X = B  (Sin transposición)
!>          = 'T':  A**T * X = B  (Transpuesta)
!>          = 'C':  A**H * X = B  (Transpuesta conjugada)
!>

M

!>          M es INTEGER
!>          El número de filas de la matriz A.  M >= 0.
!>

N

!>          N es INTEGER
!>          El número de columnas de la matriz A.  N >= 0.
!>

NRHS

!>          NRHS es INTEGER
!>          El número de columnas de B, la matriz de lados derechos.
!>          NRHS >= 0.
!>

A

!>          A es un array COMPLEX*16, dimensión (LDA,N)
!>          La matriz original A de M x N.
!>

LDA

!>          LDA es INTEGER
!>          La dimensión principal del array A.  LDA >= max(1,M).
!>

X

!>          X es un array COMPLEX*16, dimensión (LDX,NRHS)
!>          Las soluciones computadas de los vectores para el sistema de
!>          ecuaciones lineales.
!>

LDX

!>          LDX es INTEGER
!>          La dimensión principal del array X.  Si TRANS = 'N',
!>          LDX >= max(1,N); si TRANS = 'T' o 'C', LDX >= max(1,M).
!>

B

!>          B es un array COMPLEX*16, dimensión (LDB,NRHS)
!>          En la entrada, los vectores del lado derecho para el sistema de
!>          ecuaciones lineales.
!>          En la salida, B se sobrescribe con la diferencia B - op(A)*X.
!>

LDB

!>          LDB es INTEGER
!>          La dimensión principal del array B.  SI TRANS = 'N',
!>          LDB >= max(1,M); si TRANS = 'T' o 'C', LDB >= max(1,N).
!>

RWORK

!>          RWORK es DOUBLE PRECISION, dimensión (M)
!>

RESID

!>          RESID es DOUBLE PRECISION
!>          El máximo sobre el número de lados derechos de
!>          norm(B - op(A)*X) / ( norm(op(A)) * norm(X) * EPS ).
!>
