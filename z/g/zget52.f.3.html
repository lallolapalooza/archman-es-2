subroutine zget52 (logical left, integer n, complex*16, dimension( lda, * ) a, integer lda, complex*16, dimension( ldb, * ) b, integer ldb, complex*16, dimension( lde, * ) e, integer lde, complex*16, dimension( * ) alpha, complex*16, dimension( * ) beta, complex*16, dimension( * ) work, double precision, dimension( * ) rwork, double precision, dimension( 2 ) result)

ZGET52

Propósito:

!>
!> ZGET52  realiza una comprobación de autovectores para el problema de autovalores
!> generalizado.
!>
!> La prueba básica para autovectores derechos es:
!>
!>                           | b(i) A E(i) -  a(i) B E(i) |
!>         RESULT(1) = max   -------------------------------
!>                      i    n ulp max( |b(i) A|, |a(i) B| )
!>
!> usando la norma 1. Aquí, a(i)/b(i) = w es el i-ésimo autovalor generalizado
!> de A - w B, o, equivalentemente, b(i)/a(i) = m es el i-ésimo
!> autovalor generalizado de m A - B.
!>
!>                         H   H  _      _
!> Para autovectores izquierdos, se utilizan A , B , a, y b .
!>
!> ZGET52 también prueba la normalización de E. Se supone que cada autovector está
!> normalizado de tal manera que el máximo de sus elementos sea 1, donde en este caso,
!> la norma de un valor complejo x es  |Re(x)| + |Im(x)| ; llamemos a esta
!> norma máxima de un vector v  M(v).
!> Si a(i)=b(i)=0, entonces el autovector se establece como el vector de coordenadas j.
!> La prueba de normalización es:
!>
!>         RESULT(2) =      max       | M(v(i)) - 1 | / ( n ulp )
!>                    eigenvectors v(i)
!>
!>

Parámetros

LEFT 
!>          LEFT es LOGICAL
!>          =.TRUE.:  Se supone que los autovectores en las columnas de E son
!>                    autovectores *izquierdos*.
!>          =.FALSE.: Se supone que los autovectores en las columnas de E son
!>                    autovectores *derechos*.
!>

N

!>          N es INTEGER
!>          El tamaño de las matrices. Si es cero, ZGET52 no hace nada.
!>          Debe ser al menos cero.
!>

A

!>          A es un array COMPLEX*16, dimension (LDA, N)
!>          La matriz A.
!>

LDA

!>          LDA es INTEGER
!>          La dimensión principal de A. Debe ser al menos 1
!>          y al menos N.
!>

B

!>          B es un array COMPLEX*16, dimension (LDB, N)
!>          La matriz B.
!>

LDB

!>          LDB es INTEGER
!>          La dimensión principal de B. Debe ser al menos 1
!>          y al menos N.
!>

E

!>          E es un array COMPLEX*16, dimension (LDE, N)
!>          La matriz de autovectores. Debe ser O( 1 ).
!>

LDE

!>          LDE es INTEGER
!>          La dimensión principal de E. Debe ser al menos 1 y
!>          al menos N.
!>

ALPHA

!>          ALPHA es un array COMPLEX*16, dimension (N)
!>          Los valores a(i) como se describe arriba, que, junto con b(i),
!>          definen los autovalores generalizados.
!>

BETA

!>          BETA es un array COMPLEX*16, dimension (N)
!>          Los valores b(i) como se describe arriba, que, junto con a(i),
!>          definen los autovalores generalizados.
!>

WORK

!>          WORK es un array COMPLEX*16, dimension (N**2)
!>

RWORK

!>          RWORK es un array DOUBLE PRECISION, dimension (N)
!>

RESULT

!>          RESULT es un array DOUBLE PRECISION, dimension (2)
!>          Los valores calculados por la prueba descrita anteriormente. Si A E o
!>          B E es probable que desborde, entonces RESULT(1:2) se establece en
!>          10 / ulp.
!>
