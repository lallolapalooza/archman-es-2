subroutine zgeev (character jobvl, character jobvr, integer n, complex*16, dimension( lda, * ) a, integer lda, complex*16, dimension( * ) w, complex*16, dimension( ldvl, * ) vl, integer ldvl, complex*16, dimension( ldvr, * ) vr, integer ldvr, complex*16, dimension( * ) work, integer lwork, double precision, dimension( * ) rwork, integer info)

ZGEEV calcula los valores propios y, opcionalmente, los vectores propios izquierdos y/o derechos para matrices GE

Propósito:

!>
!> ZGEEV calcula para una matriz compleja no simétrica N-por-N A, los
!> valores propios y, opcionalmente, los vectores propios izquierdos y/o derechos.
!>
!> El vector propio derecho v(j) de A satisface
!>                  A * v(j) = lambda(j) * v(j)
!> donde lambda(j) es su valor propio.
!> El vector propio izquierdo u(j) de A satisface
!>               u(j)**H * A = lambda(j) * u(j)**H
!> donde u(j)**H denota la transpuesta conjugada de u(j).
!>
!> Los vectores propios calculados se normalizan para tener una norma euclidiana
!> igual a 1 y el componente más grande real.
!>

Parámetros

JOBVL 
!>          JOBVL es CHARACTER*1
!>          = 'N': los vectores propios izquierdos de A no se calculan;
!>          = 'V': los vectores propios izquierdos de se calculan.
!>

JOBVR

!>          JOBVR es CHARACTER*1
!>          = 'N': los vectores propios derechos de A no se calculan;
!>          = 'V': los vectores propios derechos de A se calculan.
!>

N

!>          N es INTEGER
!>          El orden de la matriz A. N >= 0.
!>

A

!>          A es una matriz COMPLEX*16, dimension (LDA,N)
!>          En la entrada, la matriz N-por-N A.
!>          En la salida, A ha sido sobrescrita.
!>

LDA

!>          LDA es INTEGER
!>          La dimensión principal de la matriz A. LDA >= max(1,N).
!>

W

!>          W es una matriz COMPLEX*16, dimension (N)
!>          W contiene los valores propios calculados.
!>

VL

!>          VL es una matriz COMPLEX*16, dimension (LDVL,N)
!>          Si JOBVL = 'V', los vectores propios izquierdos u(j) se almacenan uno
!>          después de otro en las columnas de VL, en el mismo orden
!>          que sus valores propios.
!>          Si JOBVL = 'N', VL no se referencia.
!>          u(j) = VL(:,j), la columna j de VL.
!>

LDVL

!>          LDVL es INTEGER
!>          La dimensión principal de la matriz VL. LDVL >= 1; si
!>          JOBVL = 'V', LDVL >= N.
!>

VR

!>          VR es una matriz COMPLEX*16, dimension (LDVR,N)
!>          Si JOBVR = 'V', los vectores propios derechos v(j) se almacenan uno
!>          después de otro en las columnas de VR, en el mismo orden
!>          que sus valores propios.
!>          Si JOBVR = 'N', VR no se referencia.
!>          v(j) = VR(:,j), la columna j de VR.
!>

LDVR

!>          LDVR es INTEGER
!>          La dimensión principal de la matriz VR. LDVR >= 1; si
!>          JOBVR = 'V', LDVR >= N.
!>

WORK

!>          WORK es una matriz COMPLEX*16, dimension (MAX(1,LWORK))
!>          En la salida, si INFO = 0, WORK(1) devuelve el LWORK óptimo.
!>

LWORK

!>          LWORK es INTEGER
!>          La dimensión de la matriz WORK. LWORK >= max(1,2*N).
!>          Para un buen rendimiento, LWORK generalmente debe ser mayor.
!>
!>          Si LWORK = -1, entonces se asume una consulta de espacio de trabajo; la rutina
!>          solo calcula el tamaño óptimo de la matriz WORK, devuelve
!>          este valor como la primera entrada de la matriz WORK, y XERBLA no emite
!>          ningún mensaje de error relacionado con LWORK.
!>

RWORK

!>          RWORK es una matriz DOUBLE PRECISION, dimension (2*N)
!>

INFO

!>          INFO es INTEGER
!>          = 0:  salida exitosa
!>          < 0:  si INFO = -i, el i-ésimo argumento tenía un valor ilegal.
!>          > 0:  si INFO = i, el algoritmo QR no pudo calcular todos los
!>                valores propios, y no se han calculado vectores propios;
!>                los elementos i+1:N de W contienen valores propios que han
!>                convergido.
!>
