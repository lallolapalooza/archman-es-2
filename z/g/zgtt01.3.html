Documentación de Función/Subrutina

subroutine zgtt01 (integer n, complex*16, dimension( * ) dl, complex*16, dimension( * ) d, complex*16, dimension( * ) du, complex*16, dimension( * ) dlf, complex*16, dimension( * ) df, complex*16, dimension( * ) duf, complex*16, dimension( * ) du2, integer, dimension( * ) ipiv, complex*16, dimension( ldwork, * ) work, integer ldwork, double precision, dimension( * ) rwork, double precision resid)

ZGTT01

Propósito:

!>
!> ZGTT01 reconstruye una matriz tridiagonal A a partir de su factorización LU
!> y calcula el residuo
!>    norm(L*U - A) / ( norm(A) * EPS ),
!> donde EPS es la epsilon de la máquina.
!>

Parámetros

N 
!>          N es INTEGER
!>          El orden de la matriz A.  N >= 0.
!>

DL

!>          DL es COMPLEX*16 array, dimension (N-1)
!>          Los (n-1) elementos subdiagonales de A.
!>

D

!>          D es COMPLEX*16 array, dimension (N)
!>          Los elementos diagonales de A.
!>

DU

!>          DU es COMPLEX*16 array, dimension (N-1)
!>          Los (n-1) elementos superdiagonales de A.
!>

DLF

!>          DLF es COMPLEX*16 array, dimension (N-1)
!>          Los (n-1) multiplicadores que definen la matriz L de la
!>          factorización LU de A.
!>

DF

!>          DF es COMPLEX*16 array, dimension (N)
!>          Los n elementos diagonales de la matriz triangular superior U de
!>          la factorización LU de A.
!>

DUF

!>          DUF es COMPLEX*16 array, dimension (N-1)
!>          Los (n-1) elementos de la primera superdiagonal de U.
!>

DU2

!>          DU2 es COMPLEX*16 array, dimension (N-2)
!>          Los (n-2) elementos de la segunda superdiagonal de U.
!>

IPIV

!>          IPIV es INTEGER array, dimension (N)
!>          Los índices de pivote; para 1 <= i <= n, la fila i de la matriz fue
!>          intercambiada con la fila IPIV(i).  IPIV(i) siempre será i o
!>          i+1; IPIV(i) = i indica que no se requirió un intercambio de fila.
!>

WORK

!>          WORK es COMPLEX*16 array, dimension (LDWORK,N)
!>

LDWORK

!>          LDWORK es INTEGER
!>          La dimensión principal del array WORK.  LDWORK >= max(1,N).
!>

RWORK

!>          RWORK es DOUBLE PRECISION array, dimension (N)
!>

RESID

!>          RESID es DOUBLE PRECISION
!>          El residuo escalado:  norm(L*U - A) / (norm(A) * EPS)
!>
