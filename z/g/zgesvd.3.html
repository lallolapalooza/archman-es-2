subroutine zgesvd (character jobu, character jobvt, integer m, integer n, complex*16, dimension( lda, * ) a, integer lda, double precision, dimension( * ) s, complex*16, dimension( ldu, * ) u, integer ldu, complex*16, dimension( ldvt, * ) vt, integer ldvt, complex*16, dimension( * ) work, integer lwork, double precision, dimension( * ) rwork, integer info)

ZGESVD calcula la descomposición en valores singulares (SVD) para matrices GE

Propósito:

!>
!> ZGESVD calcula la descomposición en valores singulares (SVD) de una matriz compleja
!> M por N, A, calculando opcionalmente los vectores singulares izquierdos y/o derechos.
!> La SVD se escribe
!>
!>      A = U * SIGMA * transpuesta conjugada(V)
!>
!> donde SIGMA es una matriz M por N que es cero excepto por sus
!> elementos diagonales min(m,n), U es una matriz unitaria M por M, y
!> V es una matriz unitaria N por N. Los elementos diagonales de SIGMA
!> son los valores singulares de A; son reales y no negativos, y
!> se devuelven en orden descendente. Las primeras min(m,n) columnas de
!> U y V son los vectores singulares izquierdos y derechos de A.
!>
!> ¡Tenga en cuenta que la rutina devuelve V**H, no V.
!>

Parámetros

JOBU 
!>          JOBU es CHARACTER*1
!>          Especifica opciones para calcular todo o parte de la matriz U:
!>          = 'A':  se devuelven las M columnas de U en el array U:
!>          = 'S':  se devuelven las primeras min(m,n) columnas de U (los vectores singulares
!>                  izquierdos) en el array U;
!>          = 'O':  se sobrescriben las primeras min(m,n) columnas de U (los vectores singulares
!>                  izquierdos) en el array A;
!>          = 'N':  no se calculan columnas de U (no hay vectores singulares izquierdos).
!>

JOBVT

!>          JOBVT es CHARACTER*1
!>          Especifica opciones para calcular todo o parte de la matriz
!>          V**H:
!>          = 'A':  se devuelven las N filas de V**H en el array VT;
!>          = 'S':  se devuelven las primeras min(m,n) filas de V**H (los vectores singulares
!>                  derechos) en el array VT;
!>          = 'O':  se sobrescriben las primeras min(m,n) filas de V**H (los vectores singulares
!>                  derechos) en el array A;
!>          = 'N':  no se calculan filas de V**H (no hay vectores singulares derechos).
!>
!>          JOBVT y JOBU no pueden ser ambos 'O'.
!>

M

!>          M es INTEGER
!>          El número de filas de la matriz de entrada A.  M >= 0.
!>

N

!>          N es INTEGER
!>          El número de columnas de la matriz de entrada A.  N >= 0.
!>

A

!>          A es un array COMPLEX*16, dimension (LDA,N)
!>          En la entrada, la matriz A de M por N.
!>          En la salida,
!>          si JOBU = 'O',  A se sobrescribe con las primeras min(m,n)
!>                          columnas de U (los vectores singulares izquierdos,
!>                          almacenados por columnas);
!>          si JOBVT = 'O', A se sobrescribe con las primeras min(m,n)
!>                          filas de V**H (los vectores singulares derechos,
!>                          almacenados por filas);
!>          si JOBU .ne. 'O' y JOBVT .ne. 'O', el contenido de A
!>                          se destruye.
!>

LDA

!>          LDA es INTEGER
!>          La dimensión principal del array A.  LDA >= max(1,M).
!>

S

!>          S es un array DOUBLE PRECISION, dimension (min(M,N))
!>          Los valores singulares de A, ordenados de tal manera que S(i) >= S(i+1).
!>

U

!>          U es un array COMPLEX*16, dimension (LDU,UCOL)
!>          (LDU,M) si JOBU = 'A' o (LDU,min(M,N)) si JOBU = 'S'.
!>          Si JOBU = 'A', U contiene la matriz unitaria M por M, U;
!>          Si JOBU = 'S', U contiene las primeras min(m,n) columnas de
!>          U (los vectores singulares izquierdos, almacenados por columnas);
!>          Si JOBU = 'N' o 'O', U no se referencia.
!>

LDU

!>          LDU es INTEGER
!>          La dimensión principal del array U.  LDU >= 1; si
!>          JOBU = 'S' o 'A', LDU >= M.
!>

VT

!>          VT es un array COMPLEX*16, dimension (LDVT,N)
!>          Si JOBVT = 'A', VT contiene la matriz unitaria N por N
!>          V**H;
!>          Si JOBVT = 'S', VT contiene las primeras min(m,n) filas de
!>          V**H (los vectores singulares derechos, almacenados por filas);
!>          Si JOBVT = 'N' o 'O', VT no se referencia.
!>

LDVT

!>          LDVT es INTEGER
!>          La dimensión principal del array VT.  LDVT >= 1; si
!>          JOBVT = 'A', LDVT >= N; si JOBVT = 'S', LDVT >= min(M,N).
!>

WORK

!>          WORK es un array COMPLEX*16, dimension (MAX(1,LWORK))
!>          En la salida, si INFO = 0, WORK(1) devuelve el LWORK óptimo.
!>

LWORK

!>          LWORK es INTEGER
!>          La dimensión del array WORK.
!>          LWORK >=  MAX(1,2*MIN(M,N)+MAX(M,N)).
!>          Para un buen rendimiento, LWORK debería ser generalmente mayor.
!>
!>          Si LWORK = -1, se asume una consulta de espacio de trabajo; la rutina
!>          sólo calcula el tamaño óptimo del array WORK, devuelve
!>          este valor como la primera entrada del array WORK, y XERBLA no emite ningún mensaje de error
!>          relacionado con LWORK.
!>

RWORK

!>          RWORK es un array DOUBLE PRECISION, dimension (5*min(M,N))
!>          En la salida, si INFO > 0, RWORK(1:MIN(M,N)-1) contiene los
!>          elementos superdiagonales no convergentes de una matriz bidiagonal intermedia B
!>          cuya diagonal está en S (no necesariamente ordenada).
!>          B satisface A = U * B * VT, por lo que tiene los mismos valores singulares
!>          que A, y vectores singulares relacionados por U y VT.
!>

INFO

!>          INFO es INTEGER
!>          = 0:  salida exitosa.
!>          < 0:  si INFO = -i, el i-ésimo argumento tuvo un valor ilegal.
!>          > 0:  si ZBDSQR no convergió, INFO especifica cuántos
!>                superdiagonales de una forma bidiagonal intermedia B
!>                no convergieron a cero. Consulte la descripción de RWORK
!>                arriba para obtener detalles.
!>
