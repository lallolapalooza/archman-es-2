Documentación de Función/Subrutina

subroutine zgtt05 (character trans, integer n, integer nrhs, complex*16, dimension( * ) dl, complex*16, dimension( * ) d, complex*16, dimension( * ) du, complex*16, dimension( ldb, * ) b, integer ldb, complex*16, dimension( ldx, * ) x, integer ldx, complex*16, dimension( ldxact, * ) xact, integer ldxact, double precision, dimension( * ) ferr, double precision, dimension( * ) berr, double precision, dimension( * ) reslts)

ZGTT05

Propósito:

!>
!> ZGTT05 prueba los límites de error del refinamiento iterativo para la
!> solución calculada de un sistema de ecuaciones A*X = B, donde A es una
!> matriz tridiagonal general de orden n y op(A) = A o A**T,
!> dependiendo de TRANS.
!>
!> RESLTS(1) = prueba del límite de error
!>           = norm(X - XACT) / ( norm(X) * FERR )
!>
!> Se devuelve un valor grande si esta relación no es menor que uno.
!>
!> RESLTS(2) = residuo del subprograma de refinamiento iterativo
!>           = el máximo de BERR / ( NZ*EPS + (*) ), donde
!>             (*) = NZ*UNFL / (min_i (abs(op(A))*abs(X) +abs(b))_i )
!>             y NZ = número máximo de no ceros en cualquier fila de A, más 1
!>

Parámetros

TRANS 
!>          TRANS es CHARACTER*1
!>          Especifica la forma del sistema de ecuaciones.
!>          = 'N':  A * X = B     (Sin transponer)
!>          = 'T':  A**T * X = B  (Transpuesta)
!>          = 'C':  A**H * X = B  (Transpuesta conjugada = Transpuesta)
!>

N

!>          N es INTEGER
!>          El número de filas de las matrices X y XACT.  N >= 0.
!>

NRHS

!>          NRHS es INTEGER
!>          El número de columnas de las matrices X y XACT.  NRHS >= 0.
!>

DL

!>          DL es una matriz COMPLEX*16, dimensión (N-1)
!>          Los elementos subdiagonales (n-1) de A.
!>

D

!>          D es una matriz COMPLEX*16, dimensión (N)
!>          Los elementos diagonales de A.
!>

DU

!>          DU es una matriz COMPLEX*16, dimensión (N-1)
!>          Los elementos superdiagonales (n-1) de A.
!>

B

!>          B es una matriz COMPLEX*16, dimensión (LDB,NRHS)
!>          Los vectores del lado derecho para el sistema de ecuaciones
!>          lineales.
!>

LDB

!>          LDB es INTEGER
!>          La dimensión principal del array B.  LDB >= max(1,N).
!>

X

!>          X es una matriz COMPLEX*16, dimensión (LDX,NRHS)
!>          Los vectores de solución calculada. Cada vector se almacena como
!>          una columna de la matriz X.
!>

LDX

!>          LDX es INTEGER
!>          La dimensión principal del array X.  LDX >= max(1,N).
!>

XACT

!>          XACT es una matriz COMPLEX*16, dimensión (LDX,NRHS)
!>          Los vectores de solución exactos. Cada vector se almacena como
!>          una columna de la matriz XACT.
!>

LDXACT

!>          LDXACT es INTEGER
!>          La dimensión principal del array XACT.  LDXACT >= max(1,N).
!>

FERR

!>          FERR es una matriz DOUBLE PRECISION, dimensión (NRHS)
!>          Los límites de error hacia adelante estimados para cada vector
!>          de solución X. Si XTRUE es la solución verdadera, FERR limita la
!>          magnitud de la entrada más grande en (X - XTRUE) dividida por
!>          la magnitud de la entrada más grande en X.
!>

BERR

!>          BERR es una matriz DOUBLE PRECISION, dimensión (NRHS)
!>          El error residual relativo componente a componente de cada vector
!>          de solución (es decir, el cambio relativo más pequeño en cualquier
!>          entrada de A o B que hace que X sea una solución exacta).
!>

RESLTS

!>          RESLTS es una matriz DOUBLE PRECISION, dimensión (2)
!>          El máximo sobre los vectores de solución NRHS de las relaciones:
!>          RESLTS(1) = norm(X - XACT) / ( norm(X) * FERR )
!>          RESLTS(2) = BERR / ( NZ*EPS + (*) )
!>
