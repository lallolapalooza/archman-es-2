subroutine zgeev (character jobvl, character jobvr, integer n, complex*16, dimension( lda, * ) a, integer lda, complex*16, dimension( * ) w, complex*16, dimension( ldvl, * ) vl, integer ldvl, complex*16, dimension( ldvr, * ) vr, integer ldvr, complex*16, dimension( * ) work, integer lwork, double precision, dimension( * ) rwork, integer info)

ZGEEV calcula los valores propios y, opcionalmente, los vectores propios izquierdos y/o derechos para matrices GE

Propósito:

!>
!> ZGEEV calcula para una matriz compleja no simétrica A de N por N, los
!> valores propios y, opcionalmente, los vectores propios izquierdos y/o derechos.
!>
!> El vector propio derecho v(j) de A satisface
!>                  A * v(j) = lambda(j) * v(j)
!> donde lambda(j) es su valor propio.
!> El vector propio izquierdo u(j) de A satisface
!>               u(j)**H * A = lambda(j) * u(j)**H
!> donde u(j)**H denota la transpuesta conjugada de u(j).
!>
!> Los vectores propios calculados se normalizan para tener norma Euclidiana
!> igual a 1 y el componente real más grande.
!>

Parámetros

JOBVL 
!>          JOBVL es CHARACTER*1
!>          = 'N': no se calculan los vectores propios izquierdos de A;
!>          = 'V': se calculan los vectores propios izquierdos de A.
!>

JOBVR

!>          JOBVR es CHARACTER*1
!>          = 'N': no se calculan los vectores propios derechos de A;
!>          = 'V': se calculan los vectores propios derechos de A.
!>

N

!>          N es INTEGER
!>          El orden de la matriz A. N >= 0.
!>

A

!>          A es un array COMPLEX*16, dimensión (LDA,N)
!>          En la entrada, la matriz A de N por N.
!>          En la salida, A ha sido sobrescrita.
!>

LDA

!>          LDA es INTEGER
!>          La dimensión principal del array A. LDA >= max(1,N).
!>

W

!>          W es un array COMPLEX*16, dimensión (N)
!>          W contiene los valores propios calculados.
!>

VL

!>          VL es un array COMPLEX*16, dimensión (LDVL,N)
!>          Si JOBVL = 'V', los vectores propios izquierdos u(j) se almacenan uno
!>          tras otro en las columnas de VL, en el mismo orden
!>          que sus valores propios.
!>          Si JOBVL = 'N', VL no se referencia.
!>          u(j) = VL(:,j), la j-ésima columna de VL.
!>

LDVL

!>          LDVL es INTEGER
!>          La dimensión principal del array VL. LDVL >= 1; si
!>          JOBVL = 'V', LDVL >= N.
!>

VR

!>          VR es un array COMPLEX*16, dimensión (LDVR,N)
!>          Si JOBVR = 'V', los vectores propios derechos v(j) se almacenan uno
!>          tras otro en las columnas de VR, en el mismo orden
!>          que sus valores propios.
!>          Si JOBVR = 'N', VR no se referencia.
!>          v(j) = VR(:,j), la j-ésima columna de VR.
!>

LDVR

!>          LDVR es INTEGER
!>          La dimensión principal del array VR. LDVR >= 1; si
!>          JOBVR = 'V', LDVR >= N.
!>

WORK

!>          WORK es un array COMPLEX*16, dimensión (MAX(1,LWORK))
!>          En la salida, si INFO = 0, WORK(1) devuelve el LWORK óptimo.
!>

LWORK

!>          LWORK es INTEGER
!>          La dimensión del array WORK. LWORK >= max(1,2*N).
!>          Para un buen rendimiento, LWORK debe ser generalmente mayor.
!>
!>          Si LWORK = -1, entonces se asume una consulta de espacio de trabajo; la rutina
!>          solo calcula el tamaño óptimo del array WORK, devuelve
!>          este valor como la primera entrada del array WORK, y no se emite ningún mensaje
!>          de error relacionado con LWORK por XERBLA.
!>

RWORK

!>          RWORK es un array DOUBLE PRECISION, dimensión (2*N)
!>

INFO

!>          INFO es INTEGER
!>          = 0:  salida exitosa
!>          < 0:  si INFO = -i, el i-ésimo argumento tuvo un valor ilegal.
!>          > 0:  si INFO = i, el algoritmo QR no logró calcular todos los
!>                valores propios, y no se han calculado vectores propios;
!>                los elementos i+1:N de W contienen valores propios que han
!>                convergido.
!>
