DOCUMENTACIÓN DE FUNCIÓN/SUBRUTINA

subroutine zget04 (integer n, integer nrhs, complex*16, dimension( ldx, * ) x, integer ldx, complex*16, dimension( ldxact, * ) xact, integer ldxact, double precision rcond, double precision resid)

ZGET04

Propósito

!>
!> ZGET04 calcula la diferencia entre una solución calculada y la
!> solución verdadera de un sistema de ecuaciones lineales.
!>
!> RESID =  ( norm(X-XACT) * RCOND ) / ( norm(XACT) * EPS ),
!> donde RCOND es la inversa del número de condición y EPS es la
!> epsilon de la máquina.
!>

Parámetros

N 
!>          N es ENTERO
!>          El número de filas de las matrices X y XACT.  N >= 0.
!>

NRHS

!>          NRHS es ENTERO
!>          El número de columnas de las matrices X y XACT.  NRHS >= 0.
!>

X

!>          X es una matriz de tipo COMPLEX*16, dimensión (LDX,NRHS)
!>          Los vectores solución calculados. Cada vector se almacena como una
!>          columna de la matriz X.
!>

LDX

!>          LDX es ENTERO
!>          La dimensión principal del array X.  LDX >= max(1,N).
!>

XACT

!>          XACT es una matriz de tipo COMPLEX*16, dimensión (LDX,NRHS)
!>          Los vectores solución exactos. Cada vector se almacena como una
!>          columna de la matriz XACT.
!>

LDXACT

!>          LDXACT es ENTERO
!>          La dimensión principal del array XACT.  LDXACT >= max(1,N).
!>

RCOND

!>          RCOND es DOUBLE PRECISION
!>          La inversa del número de condición de la matriz de coeficientes
!>          en el sistema de ecuaciones.
!>

RESID

!>          RESID es DOUBLE PRECISION
!>          El máximo sobre los NRHS vectores solución de
!>          ( norm(X-XACT) * RCOND ) / ( norm(XACT) * EPS )
!>
