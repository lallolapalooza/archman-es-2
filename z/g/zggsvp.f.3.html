subroutine zggsvp (character jobu, character jobv, character jobq, integer m, integer p, integer n, complex*16, dimension( lda, * ) a, integer lda, complex*16, dimension( ldb, * ) b, integer ldb, double precision tola, double precision tolb, integer k, integer l, complex*16, dimension( ldu, * ) u, integer ldu, complex*16, dimension( ldv, * ) v, integer ldv, complex*16, dimension( ldq, * ) q, integer ldq, integer, dimension( * ) iwork, double precision, dimension( * ) rwork, complex*16, dimension( * ) tau, complex*16, dimension( * ) work, integer info)

ZGGSVP

Propósito:

!>
!> Esta subrutina está obsoleta y ha sido reemplazada por la subrutina ZGGSVP3.
!>
!> ZGGSVP calcula matrices unitarias U, V y Q tales que
!>
!>                    N-K-L  K    L
!>  U**H*A*Q =     K ( 0    A12  A13 )  si M-K-L >= 0;
!>                 L ( 0     0   A23 )
!>             M-K-L ( 0     0    0  )
!>
!>                  N-K-L  K    L
!>         =     K ( 0    A12  A13 )  si M-K-L < 0;
!>             M-K ( 0     0   A23 )
!>
!>                  N-K-L  K    L
!>  V**H*B*Q =   L ( 0     0   B13 )
!>             P-L ( 0     0    0  )
!>
!> donde la matriz A12 de KxK y la matriz B13 de LxL no son singulares
!> triangulares superiores; A23 es triangular superior de LxL si M-K-L >= 0,
!> de lo contrario A23 es trapezoidal superior de (M-K)xL. K+L = el rango
!> numérico efectivo de la matriz (A**H,B**H)**H de (M+P)xN.
!>
!> Esta descomposición es el paso de preprocesamiento para calcular la
!> Descomposición en Valores Singulares Generalizada (GSVD), véase la subrutina
!> ZGGSVD.
!>

Parámetros

JOBU 
!>          JOBU es CHARACTER*1
!>          = 'U':  se calcula la matriz unitaria U;
!>          = 'N':  U no se calcula.
!>

JOBV

!>          JOBV es CHARACTER*1
!>          = 'V':  se calcula la matriz unitaria V;
!>          = 'N':  V no se calcula.
!>

JOBQ

!>          JOBQ es CHARACTER*1
!>          = 'Q':  se calcula la matriz unitaria Q;
!>          = 'N':  Q no se calcula.
!>

M

!>          M es INTEGER
!>          El número de filas de la matriz A. M >= 0.
!>

P

!>          P es INTEGER
!>          El número de filas de la matriz B. P >= 0.
!>

N

!>          N es INTEGER
!>          El número de columnas de las matrices A y B. N >= 0.
!>

A

!>          A es un array de COMPLEX*16, dimensión (LDA,N)
!>          En la entrada, A contiene la matriz A de M por N.
!>          En la salida, A contiene la matriz triangular (o trapezoidal)
!>          descrita en la sección Propósito.
!>

LDA

!>          LDA es INTEGER
!>          La dimensión principal del array A. LDA >= max(1,M).
!>

B

!>          B es un array de COMPLEX*16, dimensión (LDB,N)
!>          En la entrada, B contiene la matriz P por N.
!>          En la salida, B contiene la matriz triangular descrita en
!>          la sección Propósito.
!>

LDB

!>          LDB es INTEGER
!>          La dimensión principal del array B. LDB >= max(1,P).
!>

TOLA

!>          TOLA es DOUBLE PRECISION
!>

TOLB

!>          TOLB es DOUBLE PRECISION
!>
!>          TOLA y TOLB son los umbrales para determinar el rango
!>          numérico efectivo de la matriz B y un subbloque de A. Generalmente,
!>          se establecen como
!>             TOLA = MAX(M,N)*norm(A)*MAZHEPS,
!>             TOLB = MAX(P,N)*norm(B)*MAZHEPS.
!>          El tamaño de TOLA y TOLB puede afectar los errores hacia atrás
!>          de la descomposición.
!>

K

!>          K es INTEGER
!>

L

!>          L es INTEGER
!>
!>          En la salida, K y L especifican la dimensión de los subbloques
!>          descritos en la sección Propósito.
!>          K + L = rango numérico efectivo de (A**H,B**H)**H.
!>

U

!>          U es un array de COMPLEX*16, dimensión (LDU,M)
!>          Si JOBU = 'U', U contiene la matriz unitaria U.
!>          Si JOBU = 'N', U no se referencia.
!>

LDU

!>          LDU es INTEGER
!>          La dimensión principal del array U. LDU >= max(1,M) si
!>          JOBU = 'U'; LDU >= 1 en caso contrario.
!>

V

!>          V es un array de COMPLEX*16, dimensión (LDV,P)
!>          Si JOBV = 'V', V contiene la matriz unitaria V.
!>          Si JOBV = 'N', V no se referencia.
!>

LDV

!>          LDV es INTEGER
!>          La dimensión principal del array V. LDV >= max(1,P) si
!>          JOBV = 'V'; LDV >= 1 en caso contrario.
!>

Q

!>          Q es un array de COMPLEX*16, dimensión (LDQ,N)
!>          Si JOBQ = 'Q', Q contiene la matriz unitaria Q.
!>          Si JOBQ = 'N', Q no se referencia.
!>

LDQ

!>          LDQ es INTEGER
!>          La dimensión principal del array Q. LDQ >= max(1,N) si
!>          JOBQ = 'Q'; LDQ >= 1 en caso contrario.
!>

IWORK

!>          IWORK es un array de INTEGER, dimensión (N)
!>

RWORK

!>          RWORK es un array de DOUBLE PRECISION, dimensión (2*N)
!>

TAU

!>          TAU es un array de COMPLEX*16, dimensión (N)
!>

WORK

!>          WORK es un array de COMPLEX*16, dimensión (max(3*N,M,P))
!>

INFO

!>          INFO es INTEGER
!>          = 0:  salida exitosa
!>          < 0:  si INFO = -i, el i-ésimo argumento tenía un valor ilegal.
!>
