DOCUMENTACIÓN DE LA FUNCIÓN/SUBRUTINA

subroutine zgelqs (integer m, integer n, integer nrhs, complex*16, dimension( lda, * ) a, integer lda, complex*16, dimension( * ) tau, complex*16, dimension( ldb, * ) b, integer ldb, complex*16, dimension( lwork ) work, integer lwork, integer info)

ZGELQS

Propósito:

!>
!> Calcula una solución de norma mínima
!>     min || A*X - B ||
!> utilizando la factorización LQ
!>     A = L*Q
!> calculada por ZGELQF.
!>

Parámetros

M 
!>          M es INTEGER
!>          El número de filas de la matriz A.  M >= 0.
!>

N

!>          N es INTEGER
!>          El número de columnas de la matriz A.  N >= M >= 0.
!>

NRHS

!>          NRHS es INTEGER
!>          El número de columnas de B.  NRHS >= 0.
!>

A

!>          A es una matriz COMPLEX*16, dimensión (LDA,N)
!>          Detalles de la factorización LQ de la matriz original A devueltos
!>          por ZGELQF.
!>

LDA

!>          LDA es INTEGER
!>          La dimensión principal de la matriz A.  LDA >= M.
!>

TAU

!>          TAU es una matriz COMPLEX*16, dimensión (M)
!>          Detalles de la matriz ortogonal Q.
!>

B

!>          B es una matriz COMPLEX*16, dimensión (LDB,NRHS)
!>          En la entrada, la matriz B de mano derecha de m por nrhs.
!>          En la salida, la matriz solución X de n por nrhs.
!>

LDB

!>          LDB es INTEGER
!>          La dimensión principal de la matriz B. LDB >= N.
!>

WORK

!>          WORK es una matriz COMPLEX*16, dimensión (LWORK)
!>

LWORK

!>          LWORK es INTEGER
!>          La longitud de la matriz WORK.  LWORK debe ser al menos NRHS,
!>          y debería ser al menos NRHS*NB, donde NB es el tamaño del bloque
!>          para este entorno.
!>

INFO

!>          INFO es INTEGER
!>          = 0: salida exitosa
!>          < 0: si INFO = -i, el i-ésimo argumento tuvo un valor ilegal
!>
