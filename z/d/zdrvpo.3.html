subrutina zdrvpo (logical, dimension( * ) dotype, integer nn, integer, dimension( * ) nval, integer nrhs, double precision thresh, logical tsterr, integer nmax, complex*16, dimension( * ) a, complex*16, dimension( * ) afac, complex*16, dimension( * ) asav, complex*16, dimension( * ) b, complex*16, dimension( * ) bsav, complex*16, dimension( * ) x, complex*16, dimension( * ) xact, double precision, dimension( * ) s, complex*16, dimension( * ) work, double precision, dimension( * ) rwork, integer nout)

ZDRVPO

Propósito

!>
!> ZDRVPO prueba las rutinas de controlador ZPOSV y -SVX.
!>

Parámetros

DOTYPE 
!>          DOTYPE es un array LOGICAL, dimension (NTYPES)
!>          Los tipos de matriz a usar para las pruebas. Las matrices de tipo j
!>          (para 1 <= j <= NTYPES) se usan para las pruebas si DOTYPE(j) =
!>          .TRUE.; si DOTYPE(j) = .FALSE., entonces el tipo j no se usa.
!>

NN

!>          NN es INTEGER
!>          El número de valores de N contenidos en el vector NVAL.
!>

NVAL

!>          NVAL es un array INTEGER, dimension (NN)
!>          Los valores de la dimensión de la matriz N.
!>

NRHS

!>          NRHS es INTEGER
!>          El número de vectores del lado derecho a generar para
!>          cada sistema lineal.
!>

THRESH

!>          THRESH es DOUBLE PRECISION
!>          El valor umbral para las relaciones de prueba. Un resultado se
!>          incluye en el archivo de salida si RESULT >= THRESH. Para tener
!>          cada relación de prueba impresa, use THRESH = 0.
!>

TSTERR

!>          TSTERR es LOGICAL
!>          Bandera que indica si las salidas de error deben ser probadas.
!>

NMAX

!>          NMAX es INTEGER
!>          El valor máximo permitido para N, usado en la dimensión de los
!>          arrays de trabajo.
!>

A

!>          A es un array COMPLEX*16, dimension (NMAX*NMAX)
!>

AFAC

!>          AFAC es un array COMPLEX*16, dimension (NMAX*NMAX)
!>

ASAV

!>          ASAV es un array COMPLEX*16, dimension (NMAX*NMAX)
!>

B

!>          B es un array COMPLEX*16, dimension (NMAX*NRHS)
!>

BSAV

!>          BSAV es un array COMPLEX*16, dimension (NMAX*NRHS)
!>

X

!>          X es un array COMPLEX*16, dimension (NMAX*NRHS)
!>

XACT

!>          XACT es un array COMPLEX*16, dimension (NMAX*NRHS)
!>

S

!>          S es un array DOUBLE PRECISION, dimension (NMAX)
!>

WORK

!>          WORK es un array COMPLEX*16, dimension
!>                      (NMAX*max(3,NRHS))
!>

RWORK

!>          RWORK es un array DOUBLE PRECISION, dimension (NMAX+2*NRHS)
!>

NOUT

!>          NOUT es INTEGER
!>          El número de unidad para la salida.
!>
