Documentación de Función/Subrutina

subroutine zdrvac (logical, dimension( * ) dotype, integer nm, integer, dimension( * ) mval, integer nns, integer, dimension( * ) nsval, double precision thresh, integer nmax, complex*16, dimension( * ) a, complex*16, dimension( * ) afac, complex*16, dimension( * ) b, complex*16, dimension( * ) x, complex*16, dimension( * ) work, double precision, dimension( * ) rwork, complex, dimension(*) swork, integer nout)

ZDRVAC

Propósito:

!>
!> ZDRVAC prueba ZCPOSV.
!>

Parámetros

DOTYPE 
!>          DOTYPE es un array LOGICAL, de dimensión (NTYPES)
!>          Los tipos de matrices a utilizar para las pruebas. Las matrices de tipo j
!>          (para 1 <= j <= NTYPES) se utilizan para las pruebas si DOTYPE(j) =
!>          .TRUE.; si DOTYPE(j) = .FALSE., entonces el tipo j no se utiliza.
!>

NM

!>          NM es INTEGER
!>          El número de valores de N contenidos en el vector MVAL.
!>

MVAL

!>          MVAL es un array INTEGER, de dimensión (NM)
!>          Los valores de la dimensión de la matriz N.
!>

NNS

!>          NNS es INTEGER
!>          El número de valores de NRHS contenidos en el vector NSVAL.
!>

NSVAL

!>          NSVAL es un array INTEGER, de dimensión (NNS)
!>          Los valores del número de lados derechos NRHS.
!>

THRESH

!>          THRESH es DOUBLE PRECISION
!>          El valor umbral para las ratios de prueba. Un resultado se
!>          incluye en el archivo de salida si RESULT >= THRESH. Para que
!>          se imprima cada ratio de prueba, use THRESH = 0.
!>

NMAX

!>          NMAX es INTEGER
!>          El valor máximo permitido para N, utilizado en la dimensionación de los
!>          arrays de trabajo.
!>

A

!>          A es un array COMPLEX*16, de dimensión (NMAX*NMAX)
!>

AFAC

!>          AFAC es un array COMPLEX*16, de dimensión (NMAX*NMAX)
!>

B

!>          B es un array COMPLEX*16, de dimensión (NMAX*NSMAX)
!>

X

!>          X es un array COMPLEX*16, de dimensión (NMAX*NSMAX)
!>

WORK

!>          WORK es un array COMPLEX*16, de dimensión
!>                      (NMAX*max(3,NSMAX))
!>

RWORK

!>          RWORK es un array DOUBLE PRECISION, de dimensión
!>                      (max(2*NMAX,2*NSMAX+NWORK))
!>

SWORK

!>          SWORK es un array COMPLEX, de dimensión
!>                      (NMAX*(NSMAX+NMAX))
!>

NOUT

!>          NOUT es INTEGER
!>          El número de unidad para la salida.
!>
