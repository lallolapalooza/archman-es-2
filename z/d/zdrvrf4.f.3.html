subroutine zdrvrf4 (integer nout, integer nn, integer, dimension( nn ) nval, double precision thresh, complex*16, dimension( ldc, * ) c1, complex*16, dimension( ldc, *) c2, integer ldc, complex*16, dimension( * ) crf, complex*16, dimension( lda, * ) a, integer lda, double precision, dimension( * ) d_work_zlange)

ZDRVRF4

Propósito

!>
!> ZDRVRF4 prueba las rutinas RFP de LAPACK:
!>     ZHFRK
!>

Parámetros

NOUT 
!>          NOUT es INTEGER
!>                El número de unidad para la salida.
!>

NN

!>          NN es INTEGER
!>                El número de valores de N contenidos en el vector NVAL.
!>

NVAL

!>          NVAL es un array de INTEGER, dimensión (NN)
!>                Los valores de la dimensión de la matriz N.
!>

THRESH

!>          THRESH es DOUBLE PRECISION
!>                El valor umbral para las ratios de prueba. Un resultado se
!>                incluye en el archivo de salida si RESULT >= THRESH. Para
!>                imprimir cada ratio de prueba, use THRESH = 0.
!>

C1

!>          C1 es un array de COMPLEX*16, dimensión (LDC,NMAX)
!>

C2

!>          C2 es un array de COMPLEX*16, dimensión (LDC,NMAX)
!>

LDC

!>          LDC es INTEGER
!>                La dimensión principal del array A. LDA >= max(1,NMAX).
!>

CRF

!>          CRF es un array de COMPLEX*16, dimensión ((NMAX*(NMAX+1))/2).
!>

A

!>          A es un array de COMPLEX*16, dimensión (LDA,NMAX)
!>

LDA

!>          LDA es INTEGER
!>                La dimensión principal del array A. LDA >= max(1,NMAX).
!>

D_WORK_ZLANGE

!>          D_WORK_ZLANGE es un array de DOUBLE PRECISION, dimensión (NMAX)
!>
