subrutina zdrvls (logical, dimension( * ) dotype, integer nm, integer, dimension( * ) mval, integer nn, integer, dimension( * ) nval, integer nns, integer, dimension( * ) nsval, integer nnb, integer, dimension( * ) nbval, integer, dimension( * ) nxval, double precision thresh, logical tsterr, complex*16, dimension( * ) a, complex*16, dimension( * ) copya, complex*16, dimension( * ) b, complex*16, dimension( * ) copyb, complex*16, dimension( * ) c, double precision, dimension( * ) s, double precision, dimension( * ) copys, integer nout)

ZDRVLS

Propósito:

!>
!> ZDRVLS prueba las subrutinas de controladores de mínimos cuadrados ZGELS, ZGELST,
!> ZGETSLS, ZGELSS, ZGELSY y ZGELSD.
!>

Parámetros

DOTYPE 
!>          DOTYPE es un array LOGICAL, dimension (NTYPES)
!>          Los tipos de matriz a utilizar para las pruebas. Las matrices de tipo j
!>          (para 1 <= j <= NTYPES) se utilizan para las pruebas si DOTYPE(j) =
!>          .TRUE.; si DOTYPE(j) = .FALSE., entonces el tipo j no se utiliza.
!>          La matriz de tipo j se genera de la siguiente manera:
!>          j=1: A = U*D*V donde U y V son matrices unitarias aleatorias
!>               y D tiene entradas aleatorias (> 0.1) tomadas de una distribución
!>               uniforme (0,1). A tiene rango completo.
!>          j=2: Lo mismo que 1, pero A se escala hacia arriba.
!>          j=3: Lo mismo que 1, pero A se escala hacia abajo.
!>          j=4: A = U*D*V donde U y V son matrices unitarias aleatorias
!>               y D tiene 3*min(M,N)/4 entradas aleatorias (> 0.1) tomadas
!>               de una distribución uniforme (0,1) y las entradas restantes
!>               se establecen en 0. A tiene rango deficiente.
!>          j=5: Lo mismo que 4, pero A se escala hacia arriba.
!>          j=6: Lo mismo que 5, pero A se escala hacia abajo.
!>

NM

!>          NM es INTEGER
!>          El número de valores de M contenidos en el vector MVAL.
!>

MVAL

!>          MVAL es un array INTEGER, dimension (NM)
!>          Los valores de la dimensión de fila de la matriz M.
!>

NN

!>          NN es INTEGER
!>          El número de valores de N contenidos en el vector NVAL.
!>

NVAL

!>          NVAL es un array INTEGER, dimension (NN)
!>          Los valores de la dimensión de columna de la matriz N.
!>

NNB

!>          NNB es INTEGER
!>          El número de valores de NB y NX contenidos en los
!>          vectores NBVAL y NXVAL. Los parámetros de bloqueo se utilizan
!>          en pares (NB,NX).
!>

NBVAL

!>          NBVAL es un array INTEGER, dimension (NNB)
!>          Los valores del tamaño de bloque NB.
!>

NXVAL

!>          NXVAL es un array INTEGER, dimension (NNB)
!>          Los valores del punto de cruce NX.
!>

NNS

!>          NNS es INTEGER
!>          El número de valores de NRHS contenidos en el vector NSVAL.
!>

NSVAL

!>          NSVAL es un array INTEGER, dimension (NNS)
!>          Los valores del número de lados derechos NRHS.
!>

THRESH

!>          THRESH es DOUBLE PRECISION
!>          El valor umbral para las relaciones de prueba. Un resultado se
!>          incluye en el archivo de salida si RESULT >= THRESH. Para que
!>          se imprima cada relación de prueba, use THRESH = 0.
!>

TSTERR

!>          TSTERR es LOGICAL
!>          Indicador que señala si se deben probar las salidas de error.
!>

A

!>          A es un array COMPLEX*16, dimension (MMAX*NMAX)
!>          donde MMAX es el valor máximo de M en MVAL y NMAX es el
!>          valor máximo de N en NVAL.
!>

COPYA

!>          COPYA es un array COMPLEX*16, dimension (MMAX*NMAX)
!>

B

!>          B es un array COMPLEX*16, dimension (MMAX*NSMAX)
!>          donde MMAX es el valor máximo de M en MVAL y NSMAX es el
!>          valor máximo de NRHS en NSVAL.
!>

COPYB

!>          COPYB es un array COMPLEX*16, dimension (MMAX*NSMAX)
!>

C

!>          C es un array COMPLEX*16, dimension (MMAX*NSMAX)
!>

S

!>          S es un array DOUBLE PRECISION, dimension
!>                      (min(MMAX,NMAX))
!>

COPYS

!>          COPYS es un array DOUBLE PRECISION, dimension
!>                      (min(MMAX,NMAX))
!>

NOUT

!>          NOUT es INTEGER
!>          El número de unidad para la salida.
!>
