DOCUMENTACIÓN DE FUNCIÓN/SUBRUTINA

subrutina zdrvpp (logical, dimension( * ) dotype, integer nn, integer, dimension( * ) nval, integer nrhs, double precision thresh, logical tsterr, integer nmax, complex*16, dimension( * ) a, complex*16, dimension( * ) afac, complex*16, dimension( * ) asav, complex*16, dimension( * ) b, complex*16, dimension( * ) bsav, complex*16, dimension( * ) x, complex*16, dimension( * ) xact, double precision, dimension( * ) s, complex*16, dimension( * ) work, double precision, dimension( * ) rwork, integer nout)

ZDRVPP

Propósito

!>
!> ZDRVPP prueba las rutinas de controlador ZPPSV y -SVX.
!>

Parámetros

DOTYPE 
!>          DOTYPE es un array LOGICAL, de dimensión (NTYPES)
!>          Los tipos de matriz a utilizar para las pruebas. Las matrices de tipo j
!>          (para 1 <= j <= NTYPES) se utilizan para las pruebas si DOTYPE(j) =
!>          .TRUE.; si DOTYPE(j) = .FALSE., entonces el tipo j no se utiliza.
!>

NN

!>          NN es INTEGER
!>          El número de valores de N contenidos en el vector NVAL.
!>

NVAL

!>          NVAL es un array INTEGER, de dimensión (NN)
!>          Los valores de la dimensión de la matriz N.
!>

NRHS

!>          NRHS es INTEGER
!>          El número de vectores del lado derecho que se generarán para
!>          cada sistema lineal.
!>

THRESH

!>          THRESH es DOUBLE PRECISION
!>          El valor umbral para las relaciones de prueba. Un resultado se
!>          incluye en el archivo de salida si RESULT >= THRESH. Para que
!>          cada relación de prueba se imprima, use THRESH = 0.
!>

TSTERR

!>          TSTERR es LOGICAL
!>          Bandera que indica si se deben probar las salidas de error.
!>

NMAX

!>          NMAX es INTEGER
!>          El valor máximo permitido para N, utilizado en la dimensionamiento de los
!>          arrays de trabajo.
!>

A

!>          A es un array COMPLEX*16, de dimensión (NMAX*(NMAX+1)/2)
!>

AFAC

!>          AFAC es un array COMPLEX*16, de dimensión (NMAX*(NMAX+1)/2)
!>

ASAV

!>          ASAV es un array COMPLEX*16, de dimensión (NMAX*(NMAX+1)/2)
!>

B

!>          B es un array COMPLEX*16, de dimensión (NMAX*NRHS)
!>

BSAV

!>          BSAV es un array COMPLEX*16, de dimensión (NMAX*NRHS)
!>

X

!>          X es un array COMPLEX*16, de dimensión (NMAX*NRHS)
!>

XACT

!>          XACT es un array COMPLEX*16, de dimensión (NMAX*NRHS)
!>

S

!>          S es un array DOUBLE PRECISION, de dimensión (NMAX)
!>

WORK

!>          WORK es un array COMPLEX*16, de dimensión
!>                      (NMAX*max(3,NRHS))
!>

RWORK

!>          RWORK es un array DOUBLE PRECISION, de dimensión (NMAX+2*NRHS)
!>

NOUT

!>          NOUT es INTEGER
!>          El número de unidad para la salida.
!>
