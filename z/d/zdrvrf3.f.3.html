Documentación de Función/Subrutina

subroutine zdrvrf3 (integer nout, integer nn, integer, dimension( nn ) nval, double precision thresh, complex*16, dimension( lda, * ) a, integer lda, complex*16, dimension( * ) arf, complex*16, dimension( lda, * ) b1, complex*16, dimension( lda, * ) b2, double precision, dimension( * ) d_work_zlange, complex*16, dimension( * ) z_work_zgeqrf, complex*16, dimension( * ) tau)

ZDRVRF3

Propósito:

!>
!> ZDRVRF3 prueba las rutinas RFP de LAPACK:
!>     ZTFSM
!>

Parámetros

NOUT 
!>          NOUT es INTEGER
!>                El número de unidad para la salida.
!>

NN

!>          NN es INTEGER
!>                El número de valores de N contenidos en el vector NVAL.
!>

NVAL

!>          NVAL es un array INTEGER, dimension (NN)
!>                Los valores de la dimensión de la matriz N.
!>

THRESH

!>          THRESH es DOUBLE PRECISION
!>                El valor umbral para las ratios de prueba. Un resultado se
!>                incluye en el archivo de salida si RESULT >= THRESH. Para que
!>                cada ratio de prueba se imprima, use THRESH = 0.
!>

A

!>          A es un array COMPLEX*16, dimension (LDA,NMAX)
!>

LDA

!>          LDA es INTEGER
!>                La dimensión principal del array A. LDA >= max(1,NMAX).
!>

ARF

!>          ARF es un array COMPLEX*16, dimension ((NMAX*(NMAX+1))/2).
!>

B1

!>          B1 es un array COMPLEX*16, dimension (LDA,NMAX)
!>

B2

!>          B2 es un array COMPLEX*16, dimension (LDA,NMAX)
!>

D_WORK_ZLANGE

!>          D_WORK_ZLANGE es un array DOUBLE PRECISION, dimension (NMAX)
!>

Z_WORK_ZGEQRF

!>          Z_WORK_ZGEQRF es un array COMPLEX*16, dimension (NMAX)
!>

TAU

!>          TAU es un array COMPLEX*16, dimension (NMAX)
!>
