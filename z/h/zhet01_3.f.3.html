subrutina zhet01_3 (carácter uplo, entero n, complejo*16, dimensión( lda, * ) a, entero lda, complejo*16, dimensión( ldafac, * ) afac, entero ldafac, complejo*16, dimensión( * ) e, entero, dimensión( * ) ipiv, complejo*16, dimensión( ldc, * ) c, entero ldc, doble_precisión, dimensión( * ) rwork, doble_precisión resid)

ZHET01_3

Propósito:

!>
!> ZHET01_3 reconstruye una matriz Hermítica indefinida A a partir de su
!> factorización por bloques L*D*L' o U*D*U' calculada por ZHETRF_RK
!> (o ZHETRF_BK) y calcula el residuo
!>    norm( C - A ) / ( N * norm(A) * EPS ),
!> donde C es la matriz reconstruida y EPS es la epsilon de la máquina.
!>

Parámetros

UPLO 
!>          UPLO es CARACTER*1
!>          Especifica si la parte triangular superior o inferior de la
!>          matriz Hermítica A está almacenada:
!>          = 'U':  Triangular superior
!>          = 'L':  Triangular inferior
!>

N

!>          N es ENTERO
!>          El número de filas y columnas de la matriz A. N >= 0.
!>

A

!>          A es un array COMPLEX*16, dimensión (LDA,N)
!>          La matriz Hermítica original A.
!>

LDA

!>          LDA es ENTERO
!>          La dimensión principal del array A. LDA >= max(1,N)
!>

AFAC

!>          AFAC es un array COMPLEX*16, dimensión (LDAFAC,N)
!>          Diagonal de la matriz diagonal por bloques D y factores U o L
!>          calculados por ZHETRF_RK y ZHETRF_BK:
!>            a) SÓLO los elementos diagonales de la matriz Hermítica
!>               diagonal por bloques D en la diagonal de A, es decir,
!>               D(k,k) = A(k,k);
!>               (los elementos de la supradiagonal (o subdiagonal) de D
!>                deben proporcionarse en la entrada en el array E), y
!>            b) Si UPLO = 'U': factor U en la parte supradiagonal de A.
!>               Si UPLO = 'L': factor L en la parte subdiagonal de A.
!>

LDAFAC

!>          LDAFAC es ENTERO
!>          La dimensión principal del array AFAC.
!>          LDAFAC >= max(1,N).
!>

E

!>          E es un array COMPLEX*16, dimensión (N)
!>          En la entrada, contiene los elementos de la supradiagonal (o
!>          subdiagonal) de la matriz Hermítica diagonal por bloques D
!>          con bloques diagonales de 1x1 o 2x2, donde
!>          Si UPLO = 'U': E(i) = D(i-1,i), i=2:N, E(1) no referenciado;
!>          Si UPLO = 'L': E(i) = D(i+1,i), i=1:N-1, E(N) no referenciado.
!>

IPIV

!>          IPIV es un array ENTERO, dimensión (N)
!>          Los índices de pivote de ZHETRF_RK (o ZHETRF_BK).
!>

C

!>          C es un array COMPLEX*16, dimensión (LDC,N)
!>

LDC

!>          LDC es ENTERO
!>          La dimensión principal del array C. LDC >= max(1,N).
!>

RWORK

!>          RWORK es un array DOUBLE PRECISION, dimensión (N)
!>

RESID

!>          RESID es DOUBLE PRECISION
!>          Si UPLO = 'L', norm(L*D*L' - A) / ( N * norm(A) * EPS )
!>          Si UPLO = 'U', norm(U*D*U' - A) / ( N * norm(A) * EPS )
!>
