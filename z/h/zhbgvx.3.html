subroutine zhbgvx (character jobz, character range, character uplo, integer n, integer ka, integer kb, complex*16, dimension( ldab, * ) ab, integer ldab, complex*16, dimension( ldbb, * ) bb, integer ldbb, complex*16, dimension( ldq, * ) q, integer ldq, double precision vl, double precision vu, integer il, integer iu, double precision abstol, integer m, double precision, dimension( * ) w, complex*16, dimension( ldz, * ) z, integer ldz, complex*16, dimension( * ) work, double precision, dimension( * ) rwork, integer, dimension( * ) iwork, integer, dimension( * ) ifail, integer info)

ZHBGVX

Propósito:

!>
!> ZHBGVX calcula todos los valores propios y, opcionalmente, los vectores propios
!> de un problema de valores propios bandeados hermitianos-definidos generalizados complejos, de
!> la forma A*x=(lambda)*B*x. Aquí se asume que A y B son hermitianas
!> y bandeadas, y B también es definida positiva. Los valores propios y
!> vectores propios pueden seleccionarse especificando ya sea todos los valores propios,
!> un rango de valores o un rango de índices para los valores propios deseados.
!>

Parámetros

JOBZ 
!>          JOBZ es CHARACTER*1
!>          = 'N':  Calcular solo valores propios;
!>          = 'V':  Calcular valores propios y vectores propios.
!>

RANGE

!>          RANGE es CHARACTER*1
!>          = 'A': se encontrarán todos los valores propios;
!>          = 'V': se encontrarán todos los valores propios en el intervalo semiabierto (VL,VU]
!>                 ;
!>          = 'I': se encontrarán los valores propios desde el IL-ésimo hasta el IU-ésimo.
!>

UPLO

!>          UPLO es CHARACTER*1
!>          = 'U':  Se almacenan los triángulos superiores de A y B;
!>          = 'L':  Se almacenan los triángulos inferiores de A y B.
!>

N

!>          N es INTEGER
!>          El orden de las matrices A y B.  N >= 0.
!>

KA

!>          KA es INTEGER
!>          El número de superdiagonales de la matriz A si UPLO = 'U',
!>          o el número de subdiagonales si UPLO = 'L'. KA >= 0.
!>

KB

!>          KB es INTEGER
!>          El número de superdiagonales de la matriz B si UPLO = 'U',
!>          o el número de subdiagonales si UPLO = 'L'. KB >= 0.
!>

AB

!>          AB es un arreglo COMPLEX*16, dimensión (LDAB, N)
!>          Al entrar, el triángulo superior o inferior de la matriz bandeada
!>          hermitiana A, almacenada en las primeras ka+1 filas del arreglo. La
!>          j-ésima columna de A se almacena en la j-ésima columna del arreglo AB
!>          de la siguiente manera:
!>          si UPLO = 'U', AB(ka+1+i-j,j) = A(i,j) para max(1,j-ka)<=i<=j;
!>          si UPLO = 'L', AB(1+i-j,j)    = A(i,j) para j<=i<=min(n,j+ka).
!>
!>          Al salir, el contenido de AB se destruye.
!>

LDAB

!>          LDAB es INTEGER
!>          La dimensión principal del arreglo AB.  LDAB >= KA+1.
!>

BB

!>          BB es un arreglo COMPLEX*16, dimensión (LDBB, N)
!>          Al entrar, el triángulo superior o inferior de la matriz bandeada
!>          hermitiana B, almacenada en las primeras kb+1 filas del arreglo. La
!>          j-ésima columna de B se almacena en la j-ésima columna del arreglo BB
!>          de la siguiente manera:
!>          si UPLO = 'U', BB(kb+1+i-j,j) = B(i,j) para max(1,j-kb)<=i<=j;
!>          si UPLO = 'L', BB(1+i-j,j)    = B(i,j) para j<=i<=min(n,j+kb).
!>
!>          Al salir, el factor S de la factorización de Cholesky dividida
!>          B = S**H*S, tal como lo devuelve ZPBSTF.
!>

LDBB

!>          LDBB es INTEGER
!>          La dimensión principal del arreglo BB.  LDBB >= KB+1.
!>

Q

!>          Q es un arreglo COMPLEX*16, dimensión (LDQ, N)
!>          Si JOBZ = 'V', la matriz n-por-n usada en la reducción de
!>          A*x = (lambda)*B*x a forma estándar, es decir, C*x = (lambda)*x,
!>          y consecuentemente C a forma tridiagonal.
!>          Si JOBZ = 'N', el arreglo Q no se referencia.
!>

LDQ

!>          LDQ es INTEGER
!>          La dimensión principal del arreglo Q.  Si JOBZ = 'N',
!>          LDQ >= 1. Si JOBZ = 'V', LDQ >= max(1,N).
!>

VL

!>          VL es DOUBLE PRECISION
!>
!>          Si RANGE='V', el límite inferior del intervalo a
!>          buscar para valores propios. VL < VU.
!>          No se referencia si RANGE = 'A' o 'I'.
!>

VU

!>          VU es DOUBLE PRECISION
!>
!>          Si RANGE='V', el límite superior del intervalo a
!>          buscar para valores propios. VL < VU.
!>          No se referencia si RANGE = 'A' o 'I'.
!>

IL

!>          IL es INTEGER
!>
!>          Si RANGE='I', el índice del
!>          valor propio más pequeño a devolver.
!>          1 <= IL <= IU <= N, si N > 0; IL = 1 e IU = 0 si N = 0.
!>          No se referencia si RANGE = 'A' o 'V'.
!>

IU

!>          IU es INTEGER
!>
!>          Si RANGE='I', el índice del
!>          valor propio más grande a devolver.
!>          1 <= IL <= IU <= N, si N > 0; IL = 1 e IU = 0 si N = 0.
!>          No se referencia si RANGE = 'A' o 'V'.
!>

ABSTOL

!>          ABSTOL es DOUBLE PRECISION
!>          La tolerancia de error absoluto para los valores propios.
!>          Un valor propio aproximado se acepta como convergido
!>          cuando se determina que se encuentra en un intervalo [a,b]
!>          de ancho menor o igual a
!>
!>                  ABSTOL + EPS *   max( |a|,|b| ) ,
!>
!>          donde EPS es la precisión de la máquina.  Si ABSTOL es menor o
!>          igual a cero, entonces se usará EPS*|T| en su lugar,
!>          donde |T| es la 1-norma de la matriz tridiagonal obtenida
!>          al reducir AP a forma tridiagonal.
!>
!>          Los valores propios se calcularán con mayor precisión cuando ABSTOL se
!>          establezca en el doble del umbral de subdesbordamiento 2*DLAMCH('S'), no cero.
!>          Si esta rutina regresa con INFO>0, indicando que algunos
!>          vectores propios no convergieron, intente establecer ABSTOL en
!>          2*DLAMCH('S').
!>

M

!>          M es INTEGER
!>          El número total de valores propios encontrados.  0 <= M <= N.
!>          Si RANGE = 'A', M = N, y si RANGE = 'I', M = IU-IL+1.
!>

W

!>          W es un arreglo DOUBLE PRECISION, dimensión (N)
!>          Si INFO = 0, los valores propios en orden ascendente.
!>

Z

!>          Z es un arreglo COMPLEX*16, dimensión (LDZ, N)
!>          Si JOBZ = 'V', entonces si INFO = 0, Z contiene la matriz Z de
!>          vectores propios, con la i-ésima columna de Z conteniendo el
!>          vector propio asociado con W(i). Los vectores propios están
!>          normalizados de modo que Z**H*B*Z = I.
!>          Si JOBZ = 'N', entonces Z no se referencia.
!>

LDZ

!>          LDZ es INTEGER
!>          La dimensión principal del arreglo Z.  LDZ >= 1, y si
!>          JOBZ = 'V', LDZ >= N.
!>

WORK

!>          WORK es un arreglo COMPLEX*16, dimensión (N)
!>

RWORK

!>          RWORK es un arreglo DOUBLE PRECISION, dimensión (7*N)
!>

IWORK

!>          IWORK es un arreglo INTEGER, dimensión (5*N)
!>

IFAIL

!>          IFAIL es un arreglo INTEGER, dimensión (N)
!>          Si JOBZ = 'V', entonces si INFO = 0, los primeros M elementos de
!>          IFAIL son cero.  Si INFO > 0, entonces IFAIL contiene los
!>          índices de los vectores propios que fallaron en converger.
!>          Si JOBZ = 'N', entonces IFAIL no se referencia.
!>

INFO

!>          INFO es INTEGER
!>          = 0:  salida exitosa
!>          < 0:  si INFO = -i, el i-ésimo argumento tenía un valor ilegal
!>          > 0:  si INFO = i, e i es:
!>             <= N:  entonces i vectores propios fallaron en converger. Sus
!>                    índices se almacenan en el arreglo IFAIL.
!>             > N:   si INFO = N + i, para 1 <= i <= N, entonces ZPBSTF
!>                    devolvió INFO = i: B no es definida positiva.
!>                    La factorización de B no pudo completarse y
!>                    no se calcularon valores propios ni vectores propios.
!>
