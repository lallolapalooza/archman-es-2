subroutine zhseqr (character job, character compz, integer n, integer ilo, integer ihi, complex*16, dimension( ldh, * ) h, integer ldh, complex*16, dimension( * ) w, complex*16, dimension( ldz, * ) z, integer ldz, complex*16, dimension( * ) work, integer lwork, integer info)

ZHSEQR

Propósito:

!>
!>    ZHSEQR calcula los valores propios de una matriz de Hessenberg H
!>    y, opcionalmente, las matrices T y Z de la descomposición de Schur
!>    H = Z T Z**H, donde T es una matriz triangular superior (la
!>    forma de Schur), y Z es la matriz unitaria de los vectores de Schur.
!>
!>    Opcionalmente, Z puede ser postmultiplicada en una matriz unitaria
!>    de entrada Q para que esta subrutina pueda dar la factorización de Schur
!>    de una matriz A que ha sido reducida a la forma de Hessenberg
!>    H por la matriz unitaria Q:  A = Q*H*Q**H = (QZ)*T*(QZ)**H.
!>

Parámetros

JOB 
!>          JOB es CHARACTER*1
!>           = 'E':  calcular solo los valores propios;
!>           = 'S':  calcular los valores propios y la forma de Schur T.
!>

COMPZ

!>          COMPZ es CHARACTER*1
!>           = 'N':  no se calculan vectores de Schur;
!>           = 'I':  Z se inicializa a la matriz identidad y se devuelve la matriz Z
!>                   de vectores de Schur de H;
!>           = 'V':  Z debe contener una matriz unitaria Q en la entrada, y
!>                   el producto Q*Z se devuelve.
!>

N

!>          N es INTEGER
!>           El orden de la matriz H.  N >= 0.
!>

ILO

!>          ILO es INTEGER
!>

IHI

!>          IHI es INTEGER
!>
!>           Se asume que H ya es triangular superior en las filas
!>           y columnas 1:ILO-1 y IHI+1:N. ILO e IHI se establecen normalmente
!>           por una llamada previa a ZGEBAL, y luego se pasan a ZGEHRD
!>           cuando la matriz de salida de ZGEBAL se reduce a la forma de Hessenberg.
!>           De lo contrario, ILO e IHI deben establecerse a 1 y N
!>           respectivamente.  Si N > 0, entonces 1 <= ILO <= IHI <= N.
!>           Si N = 0, entonces ILO = 1 e IHI = 0.
!>

H

!>          H es una matriz COMPLEX*16, dimension (LDH,N)
!>           En la entrada, la matriz de Hessenberg superior H.
!>           En la salida, si INFO = 0 y JOB = 'S', H contiene la matriz
!>           triangular superior T de la descomposición de Schur (la
!>           forma de Schur). Si INFO = 0 y JOB = 'E', el contenido de
!>           H no está especificado en la salida.  (El valor de salida de H cuando
!>           INFO > 0 se describe en la descripción de INFO a continuación.)
!>
!>           A diferencia de versiones anteriores de ZHSEQR, esta subrutina puede
!>           explicitamente establecer H(i,j) = 0 para i > j y j = 1, 2, ... ILO-1
!>           o j = IHI+1, IHI+2, ... N.
!>

LDH

!>          LDH es INTEGER
!>           La dimensión principal de la matriz H. LDH >= max(1,N).
!>

W

!>          W es una matriz COMPLEX*16, dimension (N)
!>           Los valores propios calculados. Si JOB = 'S', los valores propios están
!>           almacenados en el mismo orden que en la diagonal de la forma
!>           de Schur devuelta en H, con W(i) = H(i,i).
!>

Z

!>          Z es una matriz COMPLEX*16, dimension (LDZ,N)
!>           Si COMPZ = 'N', Z no se referencia.
!>           Si COMPZ = 'I', en la entrada Z no necesita estar definida y en la salida,
!>           si INFO = 0, Z contiene la matriz unitaria Z de los vectores de Schur de H.
!>           Si COMPZ = 'V', en la entrada Z debe contener una matriz N-por-N Q,
!>           que se asume igual a la matriz identidad excepto por la submatriz Z(ILO:IHI,ILO:IHI).
!>           En la salida, si INFO = 0, Z contiene Q*Z.
!>           Normalmente Q es la matriz unitaria generada por ZUNGHR
!>           después de la llamada a ZGEHRD que formó la matriz de Hessenberg H.
!>           (El valor de salida de Z cuando INFO > 0 se describe en la descripción de INFO a continuación.)
!>

LDZ

!>          LDZ es INTEGER
!>           La dimensión principal de la matriz Z.  Si COMPZ = 'I' o
!>           COMPZ = 'V', entonces LDZ >= MAX(1,N).  De lo contrario, LDZ >= 1.
!>

WORK

!>          WORK es una matriz COMPLEX*16, dimension (LWORK)
!>           En la salida, si INFO = 0, WORK(1) devuelve una estimación de
!>           el valor óptimo para LWORK.
!>

LWORK

!>          LWORK es INTEGER
!>           La dimensión de la matriz WORK.  LWORK >= max(1,N)
!>           es suficiente y proporciona un rendimiento muy bueno y a veces
!>           óptimo.  Sin embargo, LWORK de hasta 11*N
!>           puede ser necesario para un rendimiento óptimo.
!>           Se recomienda una consulta de espacio de trabajo para determinar el tamaño óptimo del espacio de trabajo.
!>
!>           Si LWORK = -1, entonces ZHSEQR realiza una consulta de espacio de trabajo.
!>           En este caso, ZHSEQR comprueba los parámetros de entrada y
!>           estima el tamaño óptimo del espacio de trabajo para los
!>           valores dados de N, ILO y IHI.  La estimación se devuelve
!>           en WORK(1).  XERBLA no emite ningún mensaje de error relacionado con LWORK.
!>           Ni H ni Z se acceden.
!>

INFO

!>          INFO es INTEGER
!>             = 0:  salida exitosa
!>             < 0:  si INFO = -i, el i-ésimo argumento tuvo un valor ilegal
!>                    valor
!>             > 0:  si INFO = i, ZHSEQR no pudo calcular todos los
!>                valores propios.  Los elementos 1:ilo-1 y i+1:n de W
!>                contienen aquellos valores propios que se han
!>                calculado con éxito.  (Los fallos son raros.)
!>
!>                Si INFO > 0 y JOB = 'E', entonces en la salida, los
!>                valores propios restantes no convergidos son los valores
!>                propios de la matriz de Hessenberg superior de las filas y
!>                columnas ILO hasta INFO del valor final de H en la salida.
!>
!>                Si INFO > 0 y JOB   = 'S', entonces en la salida
!>
!>           (*)  (valor inicial de H)*U  = U*(valor final de H)
!>
!>                donde U es una matriz unitaria.  El valor final
!>                de H es de Hessenberg superior y triangular en
!>                las filas y columnas INFO+1 hasta IHI.
!>
!>                Si INFO > 0 y COMPZ = 'V', entonces en la salida
!>
!>                  (valor final de Z)  =  (valor inicial de Z)*U
!>
!>                donde U es la matriz unitaria en (*) (independientemente
!>                del valor de JOB.)
!>
!>                Si INFO > 0 y COMPZ = 'I', entonces en la salida
!>                      (valor final de Z)  = U
!>                donde U es la matriz unitaria en (*) (independientemente
!>                del valor de JOB.)
!>
!>                Si INFO > 0 y COMPZ = 'N', entonces Z no se
!>                accede.
!>

SYNOPSIS

 CALL zhseqr( JOB, COMPZ, N, ILO, IHI, H, LDH, WR, WI, Z, LDZ, WORK, LWORK, INFO )

Function/Subroutine Documentation

zhseqr is a LAPACK routine that implements the QR algorithm for reducing a complex upper Hessenberg matrix to upper triangular form. It can also compute the eigenvectors of the matrix.

Parameters
JOB (CHARACTER*1)
Specifies whether to compute the eigenvectors.
JOB = 'N': Eigenvectors are not computed.
JOB = 'V': Eigenvectors are computed.
COMPZ (CHARACTER*1)
Specifies the form of the eigenvectors to be computed.
COMPZ = 'N': Eigenvectors are not computed.
COMPZ = 'I': Initialize Z to the identity matrix and compute
eigenvectors Z*H*Z**H = D. 
COMPZ = 'V': Initialize Z to the identity matrix and compute
eigenvectors V such that Z*H*Z**H = D. 
COMPZ = 'B': Z is supplied on entry with the unitary matrix Q
such that H = Q*Z, and Z is replaced by the
product Q*Z.
Note: JOB must be 'V' if COMPZ is 'I', 'V', or 'B'.
N (INTEGER)
The order of the matrix H. N >= 0.
ILO (INTEGER)
ILO and IHI are integers such that a submatrix
H(ILO:IHI, ILO:IHI) contains all the eigenvalues. Typically
ILO = 1, IHI = N. Contained on output are the diagonal
entries of the triangular matrix.
IHI (INTEGER)
See ILO.
H (COMPLEX*16 array, dimension (LDH,n))
On entry, the N-by-N upper Hessenberg matrix H.
On exit, H has been hessenberg-reduced to upper triangular
form. The elements below the first subdiagonal and
above the second subdiagonal with modifiers shift
parameters and subdiagonal entries are not necessarily
preserved.
LDH (INTEGER)
The leading dimension of the array H. LDH >= max(1, N).
WR (DOUBLE PRECISION array, dimension (n))
WR and WI contain the real and imaginary parts,
respectively, of the eigenvalues of H. The eigenvalues
are returned in the order they occur in the Harwell
matrix.
WI (DOUBLE PRECISION array, dimension (n))
See WR.
Z (COMPLEX*16 array, dimension (LDZ,n))
If JOB = 'V', Z is a square unitary matrix of order N.
On exit, if COMPZ = 'I', Z = eye(N). If COMPZ = 'V',
Z contains the normalized eigenvectors H*v = lambda*v,
Z<0> and Z<1> contain the corresponding
eigenvectors.
LDZ (INTEGER)
The leading dimension of the array Z. LDZ >= max(1, N).
WORK (COMPLEX*16 array, dimension (LWORK))
Workspace array. See LWORK.
LWORK (INTEGER)
The dimension of the array WORK. LWORK >= N. If
LWORK = -1, then the optimal dimension is calculated and
returned in WORK(1).
INFO (INTEGER)
= 0: Successful exit
< 0: If INFO = -i, the i-th argument had an illegal value.
> 0: If INFO = i > 0, the QR algorithm failed to compute all
eigenvalues, and i off-diagonal elements of H remain
non-zero.

Additional Parameters

The following parameters are used to control the implementation of the QR algorithm. These are primarily intended for use by LAPACK developers.

            ISPEC=1:  Flag that K. Braman, R. Byers and R. Mathias have
                                    developed a more advanced implementation of the QR
                                    algorithm that exploits structure in the matrix.
                                    The default for ISPEC=1 is 0. ISPEC=1 is only
                                    meaningful if JOB=V.
            ISPEC=2:  Maximum number of iterations allowed in the main
                      loop of the QR algorithm. The default is 30*N.
            ISPEC=3:  Minimum number of iterations per submatrix of
size 30, 40, 50, 60, 70, 80, 90, 100, 110, 120, 130, 140, 150, 160, 170, 180, 190, 200. 
            ISPEC=4:  Selects the method of deflation. Default is 3.
                      (See IPARMQ for details.)
            ISPEC=5:  Selects the deflation window size.
                      (See IPARMQ for details.) Default is 50.
            ISPEC=6:  Selects the shifting strategy. Default is 1.
                      (See IPARMQ for details.)
            ISPEC=7:  Selects the size of the deflation window.
                      Default: 25.
            ISPEC=8:  Selects the size of the deflation window.
                      Default: 50.
            ISPEC=9:  Selects the size of the deflation window.
                      Default: 100.
            ISPEC=10: Selects the size of the deflation window.
                      Default: 200.
            ISPEC=11: Selects the size of the deflation window.
                      Default: 400.
            ISPEC=12: Selects the size of the deflation window.
                      Default: 600.
            ISPEC=13: Number of simultaneous shifts to be returned
by ILAENV(ISPEC=15). (See ISPEC=15 below.) The default for (IHI-ILO+1) <= 500 is NS. The default for (IHI-ILO+1) > 500 is 3*NS/2. 
            ISPEC=14: Nibble crossover point. (See IPARMQ for
details.) Default: 14% of deflation window size. 
            ISPEC=15: Number of simultaneous shifts in a multishift
QR iteration. 
                      If IHI-ILO+1 is ...
                      greater than ...but less ... the
                      or equal to ...      than        default is
                           1               30          NS =   2(+)
                          30               60          NS =   4(+)
                          60              150          NS =  10(+)
                         150              590          NS =  **
                         590             3000          NS =  64
                        3000             6000          NS = 128
                        6000             infinity      NS = 256
                  (+)  By default some or all matrices of this order
are passed to the implicit double shift routine ZLAHQR and this parameter is ignored. See ISPEC=12 above and comments in IPARMQ for details.
                 (**)  The asterisks (**) indicate an ad-hoc
function of N increasing from 10 to 64.
            ISPEC=16: Select structured matrix multiply.
If the number of simultaneous shifts (specified by ISPEC=15) is less than 14, then the default for ISPEC=16 is 0. Otherwise the default for ISPEC=16 is 2.
