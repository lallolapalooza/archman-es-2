subroutine zhet01_rook (character uplo, integer n, complex*16, dimension( lda, * ) a, integer lda, complex*16, dimension( ldafac, * ) afac, integer ldafac, integer, dimension( * ) ipiv, complex*16, dimension( ldc, * ) c, integer ldc, double precision, dimension( * ) rwork, double precision resid)

ZHET01_ROOK

Propósito:

!>
!> ZHET01_ROOK reconstruye una matriz hermítica indefinida compleja A a partir de su
!> factorización de bloque L*D*L' o U*D*U' y calcula el residuo
!>    norm( C - A ) / ( N * norm(A) * EPS ),
!> donde C es la matriz reconstruida, EPS es la epsilon de la máquina,
!> L' es la transpuesta de L, y U' es la transpuesta de U.
!>

Parámetros

UPLO 
!>          UPLO es CHARACTER*1
!>          Especifica si la parte triangular superior o inferior de la
!>          matriz hermítica compleja A está almacenada:
!>          = 'U':  Triangular superior
!>          = 'L':  Triangular inferior
!>

N

!>          N es INTEGER
!>          El número de filas y columnas de la matriz A.  N >= 0.
!>

A

!>          A es un array COMPLEX*16, dimensión (LDA,N)
!>          La matriz hermítica compleja original A.
!>

LDA

!>          LDA es INTEGER
!>          La dimensión principal del array A.  LDA >= max(1,N)
!>

AFAC

!>          AFAC es un array COMPLEX*16, dimensión (LDAFAC,N)
!>          La forma factorizada de la matriz A. AFAC contiene la matriz
!>          diagonal de bloques D y los multiplicadores utilizados para
!>          obtener L o U de la factorización de bloque L*D*L' o U*D*U'
!>          computada por CSYTRF_ROOK.
!>

LDAFAC

!>          LDAFAC es INTEGER
!>          La dimensión principal del array AFAC.  LDAFAC >= max(1,N).
!>

IPIV

!>          IPIV es un array INTEGER, dimensión (N)
!>          Los índices de pivote de CSYTRF_ROOK.
!>

C

!>          C es un array COMPLEX*16, dimensión (LDC,N)
!>

LDC

!>          LDC es INTEGER
!>          La dimensión principal del array C.  LDC >= max(1,N).
!>

RWORK

!>          RWORK es un array DOUBLE PRECISION, dimensión (N)
!>

RESID

!>          RESID es DOUBLE PRECISION
!>          Si UPLO = 'L', norm(L*D*L' - A) / ( N * norm(A) * EPS )
!>          Si UPLO = 'U', norm(U*D*U' - A) / ( N * norm(A) * EPS )
!>
