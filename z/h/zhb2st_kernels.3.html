subroutine zhb2st_kernels (character uplo, logical wantz, integer ttype, integer st, integer ed, integer sweep, integer n, integer nb, integer ib, complex*16, dimension( lda, * ) a, integer lda, complex*16, dimension( * ) v, complex*16, dimension( * ) tau, integer ldvt, complex*16, dimension( * ) work)

ZHB2ST_KERNELS

Propósito:

!>
!> ZHB2ST_KERNELS es una rutina interna utilizada por la subrutina ZHETRD_HB2ST.
!>

Parámetros

UPLO 
!>          UPLO es CHARACTER*1
!>

WANTZ

!>          WANTZ es LOGICAL, indica si se solicitan valores propios o ambos
!>          valores/vectores propios.
!>

TTYPE

!>          TTYPE es INTEGER
!>

ST

!>          ST es INTEGER
!>          parámetro interno para índices.
!>

ED

!>          ED es INTEGER
!>          parámetro interno para índices.
!>

SWEEP

!>          SWEEP es INTEGER
!>          parámetro interno para índices.
!>

N

!>          N es INTEGER. El orden de la matriz A.
!>

NB

!>          NB es INTEGER. El tamaño de la banda.
!>

IB

!>          IB es INTEGER.
!>

A

!>          A es un array COMPLEX*16. Un puntero a la matriz A.
!>

LDA

!>          LDA es INTEGER. La dimensión principal de la matriz A.
!>

V

!>          V es un array COMPLEX*16, de dimensión 2*n si solo se solicitan
!>          valores propios o si se consultan vectores.
!>

TAU

!>          TAU es un array COMPLEX*16, de dimensión (2*n).
!>          Los factores escalares de los reflectores de Householder se
!>          almacenan en este array.
!>

LDVT

!>          LDVT es INTEGER.
!>

WORK

!>          WORK es un array COMPLEX*16. Espacio de trabajo de tamaño nb.
!>
