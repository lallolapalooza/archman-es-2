Documentación de la función/subrutina

subroutine zhbtrd (character vect, character uplo, integer n, integer kd, complex*16, dimension( ldab, * ) ab, integer ldab, double precision, dimension( * ) d, double precision, dimension( * ) e, complex*16, dimension( ldq, * ) q, integer ldq, complex*16, dimension( * ) work, integer info)

ZHBTRD

Propósito:

!>
!> ZHBTRD reduce una matriz de banda Hermitiana compleja A a la forma
!> tridiagonal real simétrica T mediante una transformación unitaria de
!> similitud:
!> Q**H * A * Q = T.
!>

Parámetros

VECT 
!>          VECT es CHARACTER*1
!>          = 'N':  no formar Q;
!>          = 'V':  formar Q;
!>          = 'U':  actualizar una matriz X, formando X*Q.
!>

UPLO

!>          UPLO es CHARACTER*1
!>          = 'U':  se almacena el triángulo superior de A;
!>          = 'L':  se almacena el triángulo inferior de A.
!>

N

!>          N es INTEGER
!>          El orden de la matriz A.  N >= 0.
!>

KD

!>          KD es INTEGER
!>          El número de superdiagonales de la matriz A si UPLO = 'U',
!>          o el número de subdiagonales si UPLO = 'L'.  KD >= 0.
!>

AB

!>          AB es un array COMPLEX*16, de dimensión (LDAB,N)
!>          En la entrada, el triángulo superior o inferior de la matriz de
!>          banda Hermitiana A, almacenado en las primeras KD+1 filas del
!>          array. La columna j de A se almacena en la columna j del array AB
!>          como sigue:
!>          si UPLO = 'U', AB(kd+1+i-j,j) = A(i,j) para max(1,j-kd)<=i<=j;
!>          si UPLO = 'L', AB(1+i-j,j)    = A(i,j) para j<=i<=min(n,j+kd).
!>          En la salida, los elementos diagonales de AB se sobrescriben con
!>          los elementos diagonales de la matriz tridiagonal T; si KD > 0,
!>          los elementos de la primera superdiagonal (si UPLO = 'U') o la
!>          primera subdiagonal (si UPLO = 'L') se sobrescriben con los
!>          elementos fuera de la diagonal de T; el resto de AB se sobrescribe
!>          con valores generados durante la reducción.
!>

LDAB

!>          LDAB es INTEGER
!>          La dimensión principal del array AB.  LDAB >= KD+1.
!>

D

!>          D es un array DOUBLE PRECISION, de dimensión (N)
!>          Los elementos diagonales de la matriz tridiagonal T.
!>

E

!>          E es un array DOUBLE PRECISION, de dimensión (N-1)
!>          Los elementos fuera de la diagonal de la matriz tridiagonal T:
!>          E(i) = T(i,i+1) si UPLO = 'U'; E(i) = T(i+1,i) si UPLO = 'L'.
!>

Q

!>          Q es un array COMPLEX*16, de dimensión (LDQ,N)
!>          En la entrada, si VECT = 'U', entonces Q debe contener una matriz
!>          X de tamaño N por N; si VECT = 'N' o 'V', entonces no es necesario
!>          configurar Q.
!>
!>          En la salida:
!>          si VECT = 'V', Q contiene la matriz unitaria Q de tamaño N por N;
!>          si VECT = 'U', Q contiene el producto X*Q;
!>          si VECT = 'N', el array Q no se referencia.
!>

LDQ

!>          LDQ es INTEGER
!>          La dimensión principal del array Q.
!>          LDQ >= 1, y LDQ >= N si VECT = 'V' o 'U'.
!>

WORK

!>          WORK es un array COMPLEX*16, de dimensión (N)
!>

INFO

!>          INFO es INTEGER
!>          = 0:  salida exitosa
!>          < 0:  si INFO = -i, el i-ésimo argumento tuvo un valor ilegal
!>
