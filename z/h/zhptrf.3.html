subroutine zhptrf (character uplo, integer n, complex*16, dimension( * ) ap, integer, dimension( * ) ipiv, integer info)

ZHPTRF

Propósito:

!>
!> ZHPTRF calcula la factorización de una matriz hermítica empaquetada compleja A
!> utilizando el método de pivoteo diagonal de Bunch-Kaufman:
!>
!>    A = U*D*U**H  o  A = L*D*L**H
!>
!> donde U (o L) es un producto de matrices de permutación y triangulares unitarias
!> superiores (inferiores), y D es una matriz hermítica y diagonal por bloques
!> con bloques diagonales de 1x1 y 2x2.
!>

Parámetros

UPLO 
!>          UPLO es CHARACTER*1
!>          = 'U':  El triángulo superior de A está almacenado;
!>          = 'L':  El triángulo inferior de A está almacenado.
!>

N

!>          N es INTEGER
!>          El orden de la matriz A.  N >= 0.
!>

AP

!>          AP es una matriz COMPLEX*16, dimensión (N*(N+1)/2)
!>          En la entrada, el triángulo superior o inferior de la matriz hermítica
!>          A, empaquetado por columnas en una matriz lineal. La j-ésima columna de A
!>          se almacena en la matriz AP de la siguiente manera:
!>          si UPLO = 'U', AP(i + (j-1)*j/2) = A(i,j) para 1<=i<=j;
!>          si UPLO = 'L', AP(i + (j-1)*(2n-j)/2) = A(i,j) para j<=i<=n.
!>
!>          En la salida, la matriz diagonal por bloques D y los multiplicadores utilizados
!>          para obtener el factor U o L, almacenados como una matriz triangular empaquetada
!>          que sobrescribe A (ver más abajo para más detalles).
!>

IPIV

!>          IPIV es una matriz INTEGER, dimensión (N)
!>          Detalles de los intercambios y la estructura de bloques de D.
!>          Si IPIV(k) > 0, entonces las filas y columnas k e IPIV(k) fueron
!>          intercambiadas y D(k,k) es un bloque diagonal de 1x1.
!>          Si UPLO = 'U' e IPIV(k) = IPIV(k-1) < 0, entonces las filas y
!>          columnas k-1 e -IPIV(k) fueron intercambiadas y D(k-1:k,k-1:k)
!>          es un bloque diagonal de 2x2. Si UPLO = 'L' e IPIV(k) =
!>          IPIV(k+1) < 0, entonces las filas y columnas k+1 e -IPIV(k) fueron
!>          intercambiadas y D(k:k+1,k:k+1) es un bloque diagonal de 2x2.
!>

INFO

!>          INFO es INTEGER
!>          = 0: salida exitosa
!>          < 0: si INFO = -i, el i-ésimo argumento tenía un valor ilegal
!>          > 0: si INFO = i, D(i,i) es exactamente cero. La factorización
!>               se ha completado, pero la matriz diagonal por bloques D es
!>               exactamente singular, y ocurrirá una división por cero si se
!>               utiliza para resolver un sistema de ecuaciones.
!>

Detalles Adicionales:

!>
!>  Si UPLO = 'U', entonces A = U*D*U**H, donde
!>     U = P(n)*U(n)* ... *P(k)U(k)* ...,
!>  es decir, U es un producto de términos P(k)*U(k), donde k disminuye de n a
!>  1 en pasos de 1 o 2, y D es una matriz diagonal por bloques con bloques
!>  diagonales de 1x1 y 2x2, D(k). P(k) es una matriz de permutación como
!>  se define por IPIV(k), y U(k) es una matriz triangular superior unitaria, tal
!>  que si el bloque diagonal D(k) es de orden s (s = 1 o 2), entonces
!>
!>             (   I    v    0   )   k-s
!>     U(k) =  (   0    I    0   )   s
!>             (   0    0    I   )   n-k
!>                k-s   s   n-k
!>
!>  Si s = 1, D(k) sobrescribe A(k,k), y v sobrescribe A(1:k-1,k).
!>  Si s = 2, el triángulo superior de D(k) sobrescribe A(k-1,k-1), A(k-1,k),
!>  y A(k,k), y v sobrescribe A(1:k-2,k-1:k).
!>
!>  Si UPLO = 'L', entonces A = L*D*L**H, donde
!>     L = P(1)*L(1)* ... *P(k)*L(k)* ...,
!>  es decir, L es un producto de términos P(k)*L(k), donde k aumenta de 1 a
!>  n en pasos de 1 o 2, y D es una matriz diagonal por bloques con bloques
!>  diagonales de 1x1 y 2x2, D(k). P(k) es una matriz de permutación como
!>  se define por IPIV(k), y L(k) es una matriz triangular inferior unitaria, tal
!>  que si el bloque diagonal D(k) es de orden s (s = 1 o 2), entonces
!>
!>             (   I    0     0   )  k-1
!>     L(k) =  (   0    I     0   )  s
!>             (   0    v     I   )  n-k-s+1
!>                k-1   s  n-k-s+1
!>
!>  Si s = 1, D(k) sobrescribe A(k,k), y v sobrescribe A(k+1:n,k).
!>  Si s = 2, el triángulo inferior de D(k) sobrescribe A(k,k), A(k+1,k),
!>  y A(k+1,k+1), y v sobrescribe A(k+2:n,k:k+1).
!>
