subroutine zhsein (carácter side, carácter eigsrc, carácter initv, lógico, dimensión( * ) select, entero n, complejo*16, dimensión( ldh, * ) h, entero ldh, complejo*16, dimensión( * ) w, complejo*16, dimensión( ldvl, * ) vl, entero ldvl, complejo*16, dimensión( ldvr, * ) vr, entero ldvr, entero mm, entero m, complejo*16, dimensión( * ) work, doble precisión, dimensión( * ) rwork, entero, dimensión( * ) ifaill, entero, dimensión( * ) ifailr, entero info)

ZHSEIN

Propósito:

!>
!> ZHSEIN utiliza la iteración inversa para encontrar los
!> (auto)vectores derechos y/o izquierdos especificados de una matriz
!> compleja de Hessenberg superior H.
!>
!> El (auto)vector derecho x y el (auto)vector izquierdo y de la matriz H
!> correspondientes a un (auto)valor propio w se definen por:
!>
!>              H * x = w * x,     y**h * H = w * y**h
!>
!> donde y**h denota la transpuesta conjugada del vector y.
!>

Parámetros

SIDE 
!>          SIDE es CHARACTER*1
!>          = 'R': calcular solo (auto)vectores derechos;
!>          = 'L': calcular solo (auto)vectores izquierdos;
!>          = 'B': calcular ambos (auto)vectores derechos e izquierdos.
!>

EIGSRC

!>          EIGSRC es CHARACTER*1
!>          Especifica la fuente de (auto)valores propios proporcionados en W:
!>          = 'Q': los (auto)valores propios se encontraron usando ZHSEQR; por lo tanto, si
!>                 H tiene elementos subdiagonales cero, y por lo tanto es
!>                 triangular por bloques, entonces el j-ésimo (auto)valor propio puede
!>                 asumirse como un (auto)valor propio del bloque que
!>                 contiene la j-ésima fila/columna.  Esta propiedad permite a ZHSEIN
!>                 realizar iteración inversa solo en un bloque diagonal.
!>          = 'N': no se hacen suposiciones sobre la correspondencia
!>                 entre (auto)valores propios y bloques diagonales.  En este
!>                 caso, ZHSEIN siempre debe realizar iteración inversa
!>                 utilizando toda la matriz H.
!>

INITV

!>          INITV es CHARACTER*1
!>          = 'N': no se proporcionan (auto)vectores iniciales;
!>          = 'U': se proporcionan (auto)vectores iniciales definidos por el usuario en los arreglos
!>                 VL y/o VR.
!>

SELECT

!>          SELECT es un arreglo LÓGICO, dimensión (N)
!>          Especifica los (auto)vectores a calcular. Para seleccionar el
!>          (auto)vector correspondiente al (auto)valor propio W(j),
!>          SELECT(j) debe ser establecido a .TRUE..
!>

N

!>          N es ENTERO
!>          El orden de la matriz H.  N >= 0.
!>

H

!>          H es un arreglo COMPLEJO*16, dimensión (LDH,N)
!>          La matriz de Hessenberg superior H.
!>          Si se detecta un NaN en H, la rutina retornará con INFO=-6.
!>

LDH

!>          LDH es ENTERO
!>          La dimensión principal del arreglo H.  LDH >= max(1,N).
!>

W

!>          W es un arreglo COMPLEJO*16, dimensión (N)
!>          En la entrada, los (auto)valores propios de H.
!>          En la salida, las partes reales de W pueden haber sido alteradas ya que
!>          (auto)valores propios cercanos se perturban ligeramente al buscar
!>          (auto)vectores independientes.
!>

VL

!>          VL es un arreglo COMPLEJO*16, dimensión (LDVL,MM)
!>          En la entrada, si INITV = 'U' y SIDE = 'L' o 'B', VL debe
!>          contener (auto)vectores de inicio para la iteración inversa para
!>          los (auto)vectores izquierdos; el (auto)vector de inicio para cada (auto)vector
!>          debe estar en la misma columna en la que se almacenará el (auto)vector.
!>          En la salida, si SIDE = 'L' o 'B', los (auto)vectores izquierdos
!>          especificados por SELECT se almacenarán consecutivamente en
!>          las columnas de VL, en el mismo orden que sus (auto)valores propios.
!>          Si SIDE = 'R', VL no se referencia.
!>

LDVL

!>          LDVL es ENTERO
!>          La dimensión principal del arreglo VL.
!>          LDVL >= max(1,N) si SIDE = 'L' o 'B'; LDVL >= 1 en caso contrario.
!>

VR

!>          VR es un arreglo COMPLEJO*16, dimensión (LDVR,MM)
!>          En la entrada, si INITV = 'U' y SIDE = 'R' o 'B', VR debe
!>          contener (auto)vectores de inicio para la iteración inversa para
!>          los (auto)vectores derechos; el (auto)vector de inicio para cada (auto)vector
!>          debe estar en la misma columna en la que se almacenará el (auto)vector.
!>          En la salida, si SIDE = 'R' o 'B', los (auto)vectores derechos
!>          especificados por SELECT se almacenarán consecutivamente en
!>          las columnas de VR, en el mismo orden que sus (auto)valores propios.
!>          Si SIDE = 'L', VR no se referencia.
!>

LDVR

!>          LDVR es ENTERO
!>          La dimensión principal del arreglo VR.
!>          LDVR >= max(1,N) si SIDE = 'R' o 'B'; LDVR >= 1 en caso contrario.
!>

MM

!>          MM es ENTERO
!>          El número de columnas en los arreglos VL y/o VR. MM >= M.
!>

M

!>          M es ENTERO
!>          El número de columnas en los arreglos VL y/o VR requerido para
!>          almacenar los (auto)vectores (= el número de elementos .TRUE. en
!>          SELECT).
!>

WORK

!>          WORK es un arreglo COMPLEJO*16, dimensión (N*N)
!>

RWORK

!>          RWORK es un arreglo DOBLE PRECISIÓN, dimensión (N)
!>

IFAILL

!>          IFAILL es un arreglo ENTERO, dimensión (MM)
!>          Si SIDE = 'L' o 'B', IFAILL(i) = j > 0 si el (auto)vector izquierdo en la i-ésima
!>          columna de VL (correspondiente al (auto)valor propio w(j)) no convergió;
!>          IFAILL(i) = 0 si el (auto)vector convergió satisfactoriamente.
!>          Si SIDE = 'R', IFAILL no se referencia.
!>

IFAILR

!>          IFAILR es un arreglo ENTERO, dimensión (MM)
!>          Si SIDE = 'R' o 'B', IFAILR(i) = j > 0 si el (auto)vector derecho en la i-ésima
!>          columna de VR (correspondiente al (auto)valor propio w(j)) no convergió;
!>          IFAILR(i) = 0 si el (auto)vector convergió satisfactoriamente.
!>          Si SIDE = 'L', IFAILR no se referencia.
!>

INFO

!>          INFO es ENTERO
!>          = 0:  salida exitosa
!>          < 0:  si INFO = -i, el i-ésimo argumento tuvo un valor ilegal
!>          > 0:  si INFO = i, i es el número de (auto)vectores que no convergieron; ver IFAILL
!>                e IFAILR para más detalles.
!>
