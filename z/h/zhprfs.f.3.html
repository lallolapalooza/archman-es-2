subroutine zhprfs (character uplo, integer n, integer nrhs, complex*16, dimension( * ) ap, complex*16, dimension( * ) afp, integer, dimension( * ) ipiv, complex*16, dimension( ldb, * ) b, integer ldb, complex*16, dimension( ldx, * ) x, integer ldx, double precision, dimension( * ) ferr, double precision, dimension( * ) berr, complex*16, dimension( * ) work, double precision, dimension( * ) rwork, integer info)

ZHPRFS

Propósito:

!>
!> ZHPRFS mejora la solución calculada para un sistema de ecuaciones
!> lineales cuando la matriz de coeficientes es hermitiana indefinida
!> y empaquetada, y proporciona límites de error y estimaciones de error
!> hacia atrás para la solución.
!>

Parámetros

UPLO 
!>          UPLO es CHARACTER*1
!>          = 'U':  El triángulo superior de A está almacenado;
!>          = 'L':  El triángulo inferior de A está almacenado.
!>

N

!>          N es INTEGER
!>          El orden de la matriz A.  N >= 0.
!>

NRHS

!>          NRHS es INTEGER
!>          El número de lados derechos, es decir, el número de columnas
!>          de las matrices B y X.  NRHS >= 0.
!>

AP

!>          AP es una matriz COMPLEX*16, dimensionada (N*(N+1)/2)
!>          El triángulo superior o inferior de la matriz hermitiana A,
!>          empaquetado columna a columna en una matriz lineal. La j-ésima
!>          columna de A se almacena en la matriz AP de la siguiente manera:
!>          si UPLO = 'U', AP(i + (j-1)*j/2) = A(i,j) para 1<=i<=j;
!>          si UPLO = 'L', AP(i + (j-1)*(2*n-j)/2) = A(i,j) para j<=i<=n.
!>

AFP

!>          AFP es una matriz COMPLEX*16, dimensionada (N*(N+1)/2)
!>          La forma factorizada de la matriz A.  AFP contiene la matriz
!>          diagonal de bloques D y los multiplicadores utilizados para
!>          obtener U o L de la factorización A = U*D*U**H o
!>          A = L*D*L**H, calculada por ZHPTRF, almacenada como una matriz
!>          triangular empaquetada.
!>

IPIV

!>          IPIV es una matriz INTEGER, dimensionada (N)
!>          Detalles de los intercambios y la estructura de bloques de D
!>          determinada por ZHPTRF.
!>

B

!>          B es una matriz COMPLEX*16, dimensionada (LDB,NRHS)
!>          La matriz del lado derecho B.
!>

LDB

!>          LDB es INTEGER
!>          La dimensión principal de la matriz B.  LDB >= max(1,N).
!>

X

!>          X es una matriz COMPLEX*16, dimensionada (LDX,NRHS)
!>          En la entrada, la matriz solución X, tal como se calculó por ZHPTRS.
!>          En la salida, la matriz solución mejorada X.
!>

LDX

!>          LDX es INTEGER
!>          La dimensión principal de la matriz X.  LDX >= max(1,N).
!>

FERR

!>          FERR es una matriz DOUBLE PRECISION, dimensionada (NRHS)
!>          El límite de error hacia adelante estimado para cada vector solución
!>          X(j) (la j-ésima columna de la matriz solución X).
!>          Si XTRUE es la solución verdadera correspondiente a X(j), FERR(j)
!>          es un límite superior estimado para la magnitud del elemento más
!>          grande en (X(j) - XTRUE) dividido por la magnitud del elemento más
!>          grande en X(j). La estimación es tan confiable como
!>          la estimación de RCOND, y es casi siempre una ligera
!>          sobreestimación del error verdadero.
!>

BERR

!>          BERR es una matriz DOUBLE PRECISION, dimensionada (NRHS)
!>          El error hacia atrás relativo componente a componente de cada vector
!>          solución X(j) (es decir, el menor cambio relativo en
!>          cualquier elemento de A o B que haga que X(j) sea una solución exacta).
!>

WORK

!>          WORK es una matriz COMPLEX*16, dimensionada (2*N)
!>

RWORK

!>          RWORK es una matriz DOUBLE PRECISION, dimensionada (N)
!>

INFO

!>          INFO es INTEGER
!>          = 0:  salida exitosa
!>          < 0:  si INFO = -i, el i-ésimo argumento tenía un valor ilegal
!>

Parámetros internos:

!>  ITMAX es el número máximo de pasos de refinamiento iterativo.
!>
