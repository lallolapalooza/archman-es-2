subroutine zhpgvx (integer itype, character jobz, character range, character uplo, integer n, complex*16, dimension( * ) ap, complex*16, dimension( * ) bp, double precision vl, double precision vu, integer il, integer iu, double precision abstol, integer m, double precision, dimension( * ) w, complex*16, dimension( ldz, * ) z, integer ldz, complex*16, dimension( * ) work, double precision, dimension( * ) rwork, integer, dimension( * ) iwork, integer, dimension( * ) ifail, integer info)

ZHPGVX

Propósito:

!>
!> ZHPGVX calcula valores propios seleccionados y, opcionalmente,
!> vectores propios de un problema de valores propios Hermítico-definido
!> generalizado complejo, de la forma A*x=(lambda)*B*x, A*Bx=(lambda)*x,
!> o B*A*x=(lambda)*x. Aquí, se asume que A y B son Hermíticos, almacenados
!> en formato empaquetado, y B también es positivo definido. Los valores
!> propios y vectores propios pueden ser seleccionados especificando un
!> rango de valores o un rango de índices para los valores propios deseados.
!>

Parámetros

ITYPE 
!>          ITYPE es INTEGER
!>          Especifica el tipo de problema a resolver:
!>          = 1:  A*x = (lambda)*B*x
!>          = 2:  A*B*x = (lambda)*x
!>          = 3:  B*A*x = (lambda)*x
!>

JOBZ

!>          JOBZ es CHARACTER*1
!>          = 'N':  Calcular solo valores propios;
!>          = 'V':  Calcular valores propios y vectores propios.
!>

RANGE

!>          RANGE es CHARACTER*1
!>          = 'A': se encontrarán todos los valores propios;
!>          = 'V': se encontrarán todos los valores propios en el intervalo semiabierto (VL,VU]
!>                 ;
!>          = 'I': se encontrarán el IL-ésimo hasta el IU-ésimo valor propio.
!>

UPLO

!>          UPLO es CHARACTER*1
!>          = 'U':  Se almacenan las triángulos superiores de A y B;
!>          = 'L':  Se almacenan las triángulos inferiores de A y B.
!>

N

!>          N es INTEGER
!>          El orden de las matrices A y B.  N >= 0.
!>

AP

!>          AP es un array COMPLEX*16, de dimensión (N*(N+1)/2)
!>          En la entrada, la triángulo superior o inferior de la matriz Hermítica
!>          A, empaquetada columna por columna en un array lineal. La j-ésima columna de A
!>          se almacena en el array AP como sigue:
!>          si UPLO = 'U', AP(i + (j-1)*j/2) = A(i,j) para 1<=i<=j;
!>          si UPLO = 'L', AP(i + (j-1)*(2*n-j)/2) = A(i,j) para j<=i<=n.
!>
!>          En la salida, el contenido de AP se destruye.
!>

BP

!>          BP es un array COMPLEX*16, de dimensión (N*(N+1)/2)
!>          En la entrada, la triángulo superior o inferior de la matriz Hermítica
!>          B, empaquetada columna por columna en un array lineal. La j-ésima columna de B
!>          se almacena en el array BP como sigue:
!>          si UPLO = 'U', BP(i + (j-1)*j/2) = B(i,j) para 1<=i<=j;
!>          si UPLO = 'L', BP(i + (j-1)*(2*n-j)/2) = B(i,j) para j<=i<=n.
!>
!>          En la salida, el factor triangular U o L de la factorización de Cholesky
!>          B = U**H*U o B = L*L**H, en el mismo formato de almacenamiento
!>          que B.
!>

VL

!>          VL es DOUBLE PRECISION
!>
!>          Si RANGE='V', el límite inferior del intervalo a
!>          buscar para los valores propios. VL < VU.
!>          No se referencia si RANGE = 'A' o 'I'.
!>

VU

!>          VU es DOUBLE PRECISION
!>
!>          Si RANGE='V', el límite superior del intervalo a
!>          buscar para los valores propios. VL < VU.
!>          No se referencia si RANGE = 'A' o 'I'.
!>

IL

!>          IL es INTEGER
!>
!>          Si RANGE='I', el índice del
!>          valor propio más pequeño a devolver.
!>          1 <= IL <= IU <= N, si N > 0; IL = 1 y IU = 0 si N = 0.
!>          No se referencia si RANGE = 'A' o 'V'.
!>

IU

!>          IU es INTEGER
!>
!>          Si RANGE='I', el índice del
!>          valor propio más grande a devolver.
!>          1 <= IL <= IU <= N, si N > 0; IL = 1 y IU = 0 si N = 0.
!>          No se referencia si RANGE = 'A' o 'V'.
!>

ABSTOL

!>          ABSTOL es DOUBLE PRECISION
!>          La tolerancia de error absoluto para los valores propios.
!>          Un valor propio aproximado se acepta como convergente
!>          cuando se determina que yace en un intervalo [a,b]
!>          de anchura menor o igual a
!>
!>                  ABSTOL + EPS *   max( |a|,|b| ) ,
!>
!>          donde EPS es la precisión de la máquina. Si ABSTOL es menor o
!>          igual a cero, entonces se usará EPS*|T| en su lugar,
!>          donde |T| es la norma 1 de la matriz tridiagonal obtenida
!>          al reducir AP a forma tridiagonal.
!>
!>          Los valores propios se calcularán con mayor precisión cuando ABSTOL se
!>          establezca en el doble del umbral de subdesbordamiento 2*DLAMCH('S'), no cero.
!>          Si esta rutina retorna con INFO>0, indicando que algunos
!>          vectores propios no convergieron, intente establecer ABSTOL a
!>          2*DLAMCH('S').
!>

M

!>          M es INTEGER
!>          El número total de valores propios encontrados.  0 <= M <= N.
!>          Si RANGE = 'A', M = N, y si RANGE = 'I', M = IU-IL+1.
!>

W

!>          W es un array DOUBLE PRECISION, de dimensión (N)
!>          En salida normal, los primeros M elementos contienen los valores propios
!>          seleccionados en orden ascendente.
!>

Z

!>          Z es un array COMPLEX*16, de dimensión (LDZ, N)
!>          Si JOBZ = 'N', entonces Z no se referencia.
!>          Si JOBZ = 'V', entonces si INFO = 0, las primeras M columnas de Z
!>          contienen los vectores propios ortonormales de la matriz A
!>          correspondientes a los valores propios seleccionados, con la i-ésima
!>          columna de Z conteniendo el vector propio asociado a W(i).
!>          Los vectores propios se normalizan como sigue:
!>          si ITYPE = 1 o 2, Z**H*B*Z = I;
!>          si ITYPE = 3, Z**H*inv(B)*Z = I.
!>
!>          Si un vector propio falla en converger, entonces esa columna de Z
!>          contiene la última aproximación del vector propio, y el
!>          índice del vector propio se devuelve en IFAIL.
!>          Nota: el usuario debe asegurarse de que se suministren al menos
!>          max(1,M) columnas en el array Z; si RANGE = 'V', el valor exacto de M
!>          no se conoce de antemano y se debe usar un límite superior.
!>

LDZ

!>          LDZ es INTEGER
!>          La dimensión principal del array Z.  LDZ >= 1, y si
!>          JOBZ = 'V', LDZ >= max(1,N).
!>

WORK

!>          WORK es un array COMPLEX*16, de dimensión (2*N)
!>

RWORK

!>          RWORK es un array DOUBLE PRECISION, de dimensión (7*N)
!>

IWORK

!>          IWORK es un array INTEGER, de dimensión (5*N)
!>

IFAIL

!>          IFAIL es un array INTEGER, de dimensión (N)
!>          Si JOBZ = 'V', entonces si INFO = 0, los primeros M elementos de
!>          IFAIL son cero.  Si INFO > 0, entonces IFAIL contiene los
!>          índices de los vectores propios que fallaron en converger.
!>          Si JOBZ = 'N', entonces IFAIL no se referencia.
!>

INFO

!>

 CALL ZHPGVX( ITYPE, JOBZ, RANGE, UPLO, N, AP, VB, VI, AB, B, WORK, RWORK,

Function/Subroutine Documentation

 SUBROUTINE ZHPGVX( ITYPE, JOBZ, RANGE, UPLO, N, AP, VB, VI, AB, B, WORK,
              RWORK, LDAB, LDAB, IA, JA, SLAVE, CLIQUE, FACT,
              INFO )
  
  
  >     CHARACTER*1 ITYPE, JOBZ, RANGE, UPLO
  >     INTEGER N, LDAB, LDAB, IA, JA, SLAVE, CLIQUE, FACT, INFO
  >     COMPLEX*16 AP( * ), VB( * ), VI( * ), AB( * ), B( * ), WORK( * )
  >     DOUBLE PRECISION RWORK( * )
  
  >     .. Scalar Arguments ..
  
  >     INTEGER FACT, IA, INFO, ITYPE, JA, JOBZ, LDAB, N, SLAVE, CLIQUE
  >     CHARACTER*1 RANGE, UPLO
  >     ..
  >     .. Array Arguments ..
  
  >     COMPLEX*16 AB( LDAB, * ), AP( * ), B( * ), VI( * ), VB( * ),
  >                WORK( * )
  >     DOUBLE PRECISION RWORK( * )
  >     ..
  
  >
  >  Purpose
  >  =======
  >
  >  ZHPGVX computes the eigenvalues and, optionally, eigenvectors of a
  >  complex general matrix A in the case where B is a real symmetric matrix
  >  of the same block size as A, using the "cholesky" or "llt" decomposition
  >  of B.  A is a Hermitian matrix if ITYPE = 1 or 2, or a general matrix
  >  if ITYPE = 3.
  >
  >  Arguments
  >  =========
  >
  >  ITYPE   (input) CHARACTER*1
  >          Specifies the problem type to be solved:
  >          = '1':  A is Hermitian and B is Hermitian positive definite.
  >          = '2':  A is Hermitian and B is real symmetric positive definite.
  >          = '3':  A is general and B is real symmetric positive definite.
  >
  >  JOBZ    (input) CHARACTER*1
  >          = 'N':  Compute only the eigenvalues of the
  >                  multiple-valued matrix A.
  >          = 'V':  Compute the eigenvalues and eigenvectors, or the
  >                  eigenvectors of the multiple-valued matrix A.
  >
  >  RANGE   (input) CHARACTER*1
  >          = 'A': all eigenvalues will be found.
  >          = 'V': all eigenvalues between VL and VU will be found.
  >          = 'I': if the record is supplied between IL and IU.
  >
  >  UPLO    (input) CHARACTER*1
  >          Specifies whether the matrix B is upper-triangular or
  >          lower-triangular:
  >          = 'U': Upper triangle of B is stored and the diagonal
  >                 is stored in the upper triangle of B.
  >          = 'L': Lower triangle of B is stored and the diagonal
  >                 is stored in the lower triangle of B.
  >
  >  N       (input) INTEGER
  >          The order of the matrices A and B. N >= 0.
  >
  >  AP      (input) COMPLEX*16 array, dimension (N*(N+1)/2)
  >          On entry with ITYPE = 1 or 3, the upper or lower
  >          triangular part of the Hermitian matrix A, the band matrix
  >          storage format is described below.
  >          On entry with ITYPE = 2, the upper or lower triangular part
  >          of the Hermitian matrix A, the band matrix storage format
  >          is described below.
  >
  >  VB      (input) COMPLEX*16 array, dimension (N)
  >          On entry with ITYPE = 1, the diagonal elements of the
  >          Hermitian matrix B.
  >          On entry with ITYPE = 2 or 3, the diagonal elements of the
  >          real symmetric matrix B.
  >
  >  VI      (input) COMPLEX*16 array, dimension (N)
  >          On entry with ITYPE = 1, the imaginary parts of the
  >          diagonal elements of the Hermitian matrix B.
  >          On entry with ITYPE = 2 or 3, the imaginary parts of the
  >          diagonal elements of the real symmetric matrix B.
  >
  >  AB      (input) COMPLEX*16 array, dimension (LDAB, N)
  >          On entry with ITYPE = 1, the upper or lower triangular part
  >          of the Hermitian matrix A, the band matrix storage format
  >          is described below.
  >          On entry with ITYPE = 2 or 3, the upper or lower triangular
  >          part of the general matrix A, the band matrix storage format
  >          is described below.
  >
  >  B       (input) COMPLEX*16 array, dimension (N)
  >          On entry with ITYPE = 1, the diagonal elements of the
  >          Hermitian matrix B.
  >          On entry with ITYPE = 2 or 3, the diagonal elements of the
  >          real symmetric matrix B.
  >
  >  WORK    (workspace) COMPLEX*16 array, dimension (LWORK)
  >
  >  RWORK   (workspace) DOUBLE PRECISION array, dimension (LRWORK)
  >
  >  LDAB    (input) INTEGER
  >          The leading dimension of the array AB.  LDAB >= max(1, N).
  >
  >  LDAB    (input) INTEGER
  >          The leading dimension of the array B. LDAB >= max(1, N).
  >
  >  IA      (input) INTEGER
  >          The row index of the first sub-diagonal of the band matrix A
  >          if ITYPE = 1 or 3.
  >
  >  JA      (input) INTEGER
  >          The column index of the first sub-diagonal of the band matrix A
  >          if ITYPE = 1 or 3.
  >
  >  SLAVE   (input) INTEGER
  >          The number of slave processes available.
  >
  >  CLIQUE  (input) INTEGER
  >          The number of cliques available.
  >
  >  FACT    (input) INTEGER
  >          = 0:  On entry, B has not been factorized. On exit, B has
  >                been factorized.
  >          = 1:  On entry, B has been factorized by the current call.
  >
  >  INFO    (output) INTEGER
  >          = 0:  successful exit
  >          < 0:  if INFO = -i, the i-th argument had an illegal value
  >          > 0:  ZPPTRF or ZHPEVX returned an error code:
  >             <= N:  if INFO = i, ZHPEVX failed to converge;
  >                    i eigenvectors failed to converge.  Their indices
  >                    are stored in array IFAIL.
  >             > N:   if INFO = N + i, for 1 <= i <= n, then the leading
  >                    principal minor of order i of B is not positive.
  >                    The factorization of B could not be completed and
  >                    no eigenvalues or eigenvectors were computed.
  >
