subroutine zhgeqz (character job, character compq, character compz, integer n, integer ilo, integer ihi, complex*16, dimension( ldh, * ) h, integer ldh, complex*16, dimension( ldt, * ) t, integer ldt, complex*16, dimension( * ) alpha, complex*16, dimension( * ) beta, complex*16, dimension( ldq, * ) q, integer ldq, complex*16, dimension( ldz, * ) z, integer ldz, complex*16, dimension( * ) work, integer lwork, double precision, dimension( * ) rwork, integer info)

ZHGEQZ

Propósito:

!>
!> ZHGEQZ calcula los valores propios de un par de matrices complejas (H,T),
!> donde H es una matriz de Hessenberg superior y T es triangular superior,
!> utilizando el método QZ de desplazamiento único.
!> Los pares de matrices de este tipo se producen a partir de la reducción a
!> forma de Hessenberg generalizada de un par de matrices complejas (A,B):
!>
!>    A = Q1*H*Z1**H,  B = Q1*T*Z1**H,
!>
!> como se calcula por ZGGHRD.
!>
!> Si JOB='S', entonces el par de Hessenberg-triangular (H,T) se
!> reduce también a forma de Schur generalizada,
!>
!>    H = Q*S*Z**H,  T = Q*P*Z**H,
!>
!> donde Q y Z son matrices unitarias y S y P son triangulares superiores.
!>
!> Opcionalmente, la matriz unitaria Q de la factorización de Schur
!> generalizada puede ser postmultiplicada en una matriz de entrada Q1, y la
!> matriz unitaria Z puede ser postmultiplicada en una matriz de entrada Z1.
!> Si Q1 y Z1 son las matrices unitarias de ZGGHRD que redujeron
!> el par de matrices (A,B) a forma de Hessenberg generalizada, entonces las
!> matrices de salida Q1*Q y Z1*Z son los factores unitarios de la
!> factorización de Schur generalizada de (A,B):
!>
!>    A = (Q1*Q)*S*(Z1*Z)**H,  B = (Q1*Q)*P*(Z1*Z)**H.
!>
!> Para evitar el desbordamiento, los valores propios del par de matrices (H,T)
!> (equivalente, de (A,B)) se calculan como un par de valores complejos
!> (alpha,beta). Si beta es distinto de cero, lambda = alpha / beta es un
!> valor propio del problema de valores propios generalizado no simétrico (GNEP)
!>    A*x = lambda*B*x
!> y si alpha es distinto de cero, mu = beta / alpha es un valor propio de la
!> forma alternativa del GNEP
!>    mu*A*y = B*y.
!> Los valores de alpha y beta para el i-ésimo valor propio pueden leerse
!> directamente de la forma de Schur generalizada: alpha = S(i,i),
!> beta = P(i,i).
!>
!> Ref: C.B. Moler & G.W. Stewart, , SIAM J. Numer. Anal., 10(1973),
!>      pp. 241--256.
!>

Parámetros

JOB 
!>          JOB es CHARACTER*1
!>          = 'E': Calcula solo los valores propios;
!>          = 'S': Calcula los valores propios y la forma de Schur.
!>

COMPQ

!>          COMPQ es CHARACTER*1
!>          = 'N': Los vectores de Schur izquierdos (Q) no se calculan;
!>          = 'I': Q se inicializa a la matriz unidad y se devuelve la matriz Q
!>                 de vectores de Schur izquierdos de (H,T);
!>          = 'V': Q debe contener una matriz unitaria Q1 en la entrada y
!>                 se devuelve el producto Q1*Q.
!>

COMPZ

!>          COMPZ es CHARACTER*1
!>          = 'N': Los vectores de Schur derechos (Z) no se calculan;
!>          = 'I': Q se inicializa a la matriz unidad y se devuelve la matriz Z
!>                 de vectores de Schur derechos de (H,T);
!>          = 'V': Z debe contener una matriz unitaria Z1 en la entrada y
!>                 se devuelve el producto Z1*Z.
!>

N

!>          N es INTEGER
!>          El orden de las matrices H, T, Q y Z. N >= 0.
!>

ILO

!>          ILO es INTEGER
!>

IHI

!>          IHI es INTEGER
!>          ILO e IHI marcan las filas y columnas de H que están en
!>          forma de Hessenberg. Se asume que A ya es triangular superior
!>          en las filas y columnas 1:ILO-1 y IHI+1:N.
!>          Si N > 0, 1 <= ILO <= IHI <= N; si N = 0, ILO=1 y IHI=0.
!>

H

!>          H es una matriz COMPLEX*16, dimensión (LDH, N)
!>          En la entrada, la matriz de Hessenberg superior N por N, H.
!>          En la salida, si JOB = 'S', H contiene la matriz triangular superior
!>          S de la factorización de Schur generalizada.
!>          Si JOB = 'E', la diagonal de H coincide con la de S, pero
!>          el resto de H no está especificado.
!>

LDH

!>          LDH es INTEGER
!>          La dimensión principal de la matriz H. LDH >= max( 1, N ).
!>

T

!>          T es una matriz COMPLEX*16, dimensión (LDT, N)
!>          En la entrada, la matriz triangular superior N por N, T.
!>          En la salida, si JOB = 'S', T contiene la matriz triangular superior
!>          P de la factorización de Schur generalizada.
!>          Si JOB = 'E', la diagonal de T coincide con la de P, pero
!>          el resto de T no está especificado.
!>

LDT

!>          LDT es INTEGER
!>          La dimensión principal de la matriz T. LDT >= max( 1, N ).
!>

ALPHA

!>          ALPHA es una matriz COMPLEX*16, dimensión (N)
!>          Los escalares complejos alpha que definen los valores propios de
!>          GNEP. ALPHA(i) = S(i,i) en la factorización de Schur
!>          generalizada.
!>

BETA

!>          BETA es una matriz COMPLEX*16, dimensión (N)
!>          Los escalares reales no negativos beta que definen los
!>          valores propios de GNEP. BETA(i) = P(i,i) en la factorización
!>          de Schur generalizada.
!>
!>          Juntos, las cantidades alpha = ALPHA(j) y beta = BETA(j)
!>          representan el j-ésimo valor propio del par de matrices (A,B), en
!>          una de las formas lambda = alpha/beta o mu = beta/alpha.
!>          Dado que lambda o mu pueden desbordarse, en general no deben
!>          calcularse.
!>

Q

!>          Q es una matriz COMPLEX*16, dimensión (LDQ, N)
!>          En la entrada, si COMPQ = 'V', la matriz unitaria Q1 utilizada en la
!>          reducción de (A,B) a forma de Hessenberg generalizada.
!>          En la salida, si COMPQ = 'I', la matriz unitaria de vectores de Schur
!>          izquierdos de (H,T), y si COMPQ = 'V', la matriz unitaria de
!>          vectores de Schur izquierdos de (A,B).
!>          No referenciada si COMPQ = 'N'.
!>

LDQ

!>          LDQ es INTEGER
!>          La dimensión principal de la matriz Q. LDQ >= 1.
!>          Si COMPQ='V' o 'I', entonces LDQ >= N.
!>

Z

!>          Z es una matriz COMPLEX*16, dimensión (LDZ, N)
!>          En la entrada, si COMPZ = 'V', la matriz unitaria Z1 utilizada en la
!>          reducción de (A,B) a forma de Hessenberg generalizada.
!>          En la salida, si COMPZ = 'I', la matriz unitaria de vectores de Schur
!>          derechos de (H,T), y si COMPZ = 'V', la matriz unitaria de
!>          vectores de Schur derechos de (A,B).
!>          No referenciada si COMPZ = 'N'.
!>

LDZ

!>          LDZ es INTEGER
!>          La dimensión principal de la matriz Z. LDZ >= 1.
!>          Si COMPZ='V' o 'I', entonces LDZ >= N.
!>

WORK

!>          WORK es una matriz COMPLEX*16, dimensión (MAX(1,LWORK))
!>          En la salida, si INFO >= 0, WORK(1) devuelve el LWORK óptimo.
!>

LWORK

!>          LWORK es INTEGER
!>          La dimensión de la matriz WORK. LWORK >= max(1,N).
!>
!>          Si LWORK = -1, entonces se asume una consulta de espacio de trabajo; la rutina
!>          solo calcula el tamaño óptimo de la matriz WORK, devuelve
!>          este valor como la primera entrada de la matriz WORK y no se produce ningún error

NAME

zhgeqz - The complex version of the QR algorithm for computing the generalized Schur factorization of a complex pair (A, B)

SYNOPSIS

 subroutine zhgeqz(JOBQZ, COMPQ, COMPZ, N, ILOQ, ILOQ, A, LDA, B, LDB,
  ALPHA, BETA, Q, LDQ, Z, LDZ, WORK, LWORK, INFO)
                        character JOBQZ, COMPQ, COMPZ
                        integer N, ILOQ, LDA, LDB, LDQ, LDZ, LWORK, INFO
                        complex*16 A(LDA,*), B(LDB,*), Q(LDQ,*), Z(LDZ,*)
                        complex*16 ALPHA(*), BETA(*)
                        complex*16 WORK(*)

Function/Subroutine Documentation

zhgeqz computes the generalized Schur factorization of a complex pair of general matrices (A,B).

A and B are first reduced to upper triangular form (or quasi-upper triangular form if the matrices are not of full rank) by means of unitary transformations.

A = Q * S * ZH

B = Q * T * ZH

where S and T are upper triangular matrices, and Q and Z are unitary matrices.

Then, the routine computes the generalized eigenvalues and eigenvectors of the upper triangular matrices S and T.

Parameters:

JOBQZ
CHARACTER*1.
= 'E': compute the generalized eigenvalues only.
= 'S': compute the generalized Schur factorization; the matrices A and B will be overwritten by the upper triangular matrices S and T.
= 'X': compute the generalized Schur factorization and the matrices Q and Z, so that A = Q*S*Z**H and B = Q*T*Z**H.
COMPQ
CHARACTER*1.
= 'N': do not compute Q.
= 'I': compute Q.
= 'V': compute Q and Z.
COMPZ
CHARACTER*1.
= 'N': do not compute Z.
= 'I': compute Z.
= 'V': compute Q and Z.
N
INTEGER.
The order of the matrices A and B. N >= 0.
ILOQ
INTEGER.
ILOQ and ILQ specify the rows and columns of the submatrix of A and B for which the generalized Schur factorization is required. The Schur factorization is computed for the submatrix A(ILOQ:N, ILOQ:N) and B(ILOQ:N, ILOQ:N). ILOQ and ILQ are such that A(1:ILOQ-1, ILOQ:N) and B(1:ILOQ-1, ILOQ:N) are zero.
ILOQ
INTEGER.
ILOQ and ILQ specify the rows and columns of the submatrix of A and B for which the generalized Schur factorization is required. The Schur factorization is computed for the submatrix A(ILOQ:N, ILOQ:N) and B(ILOQ:N, ILOQ:N). ILOQ and ILQ are such that A(ILOQ:N, 1:ILOQ-1) and B(ILOQ:N, 1:ILOQ-1) are zero.
A
COMPLEX*16 array, dimension (LDA, *)
On entry, the first dimension of A is LDA and the second dimension is at least max(1, N). On exit, A is overwritten by its upper triangular matrix S, or by the quasi-upper triangular matrix if the matrix is not of full rank.
LDA
INTEGER.
The first dimension of A. LDA >= max(1, N).
B
COMPLEX*16 array, dimension (LDB, *)
On entry, the first dimension of B is LDB and the second dimension is at least max(1, N). On exit, B is overwritten by its upper triangular matrix T, or by the quasi-upper triangular matrix if the matrix is not of full rank.
LDB
INTEGER.
The first dimension of B. LDB >= max(1, N).
ALPHA
COMPLEX*16 array, dimension (N).
On exit, ALPHA(I) and BETA(I) contain the generalized eigenvalues of the pair (A,B) for I=1, ..., N.
BETA
COMPLEX*16 array, dimension (N).
On exit, ALPHA(I) and BETA(I) contain the generalized eigenvalues of the pair (A,B) for I=1, ..., N.
Q
COMPLEX*16 array, dimension (LDQ, *)
On exit, if COMPQ = 'I', Q is the unitary matrix Q.
If COMPQ = 'V', Q is the unitary matrix Q.
If COMPQ = 'N', Q is not referenced.
LDQ
INTEGER.
The first dimension of Q. LDQ >= 1.
If COMPQ = 'I' or 'V', LDQ >= N.
Z
COMPLEX*16 array, dimension (LDZ, *)
On exit, if COMPZ = 'I', Z is the unitary matrix Z.
If COMPZ = 'V', Z is the unitary matrix Z.
If COMPZ = 'N', Z is not referenced.
LDZ
INTEGER.
The first dimension of Z. LDZ >= 1.
If COMPZ = 'I' or 'V', LDZ >= N.
WORK
COMPLEX*16 array, dimension (LWORK).
Workspace array.
LWORK
INTEGER.
The dimension of the array WORK. LWORK >= 1.
For optimal performance, LWORK is N*NB, where NB is the block size for the routines called by zhgeqz.
If LWORK = -1, then only a workspace query is performed. In this case, WORK(1) returns the optimal value of LWORK.
INFO
INTEGER.
= 0: successful exit
< 0: if INFO = -i, the i-th argument had an illegal value
= 1,...,N: the QZ iteration did not converge. (H,T) is not
in Schur form, but ALPHA(i) and BETA(i),
i=INFO+1,...,N should be correct.
= N+1,...,2*N: the shift calculation failed. (H,T) is not
in Schur form, but ALPHA(i) and BETA(i),
i=INFO-N+1,...,N should be correct.

RWORK

 RWORK is DOUBLE PRECISION array, dimension (N)

INFO

 INFO is INTEGER
 = 0: successful exit
 < 0: if INFO = -i, the i-th argument had an illegal value
 = 1,...,N: the QZ iteration did not converge.  (H,T) is not
                    in Schur form, but ALPHA(i) and BETA(i),
                    i=INFO+1,...,N should be correct.
 = N+1,...,2*N: the shift calculation failed.  (H,T) is not
                    in Schur form, but ALPHA(i) and BETA(i),
                    i=INFO-N+1,...,N should be correct.
