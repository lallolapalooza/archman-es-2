DOCUMENTACIÓN DE FUNCIÓN/SUBRUTINA

subroutine zhemm (character side, character uplo, integer m, integer n, complex*16 alpha, complex*16, dimension(lda,*) a, integer lda, complex*16, dimension(ldb,*) b, integer ldb, complex*16 beta, complex*16, dimension(ldc,*) c, integer ldc)

ZHEMM

Propósito:

!>
!> ZHEMM  realiza una de las operaciones de matriz-matriz
!>
!>    C := alpha*A*B + beta*C,
!>
!> o
!>
!>    C := alpha*B*A + beta*C,
!>
!> donde alpha y beta son escalares, A es una matriz hermítica y B y
!> C son matrices de m por n.
!>

Parámetros

SIDE 
!>          SIDE es CHARACTER*1
!>           A la entrada,  SIDE  especifica si la matriz hermítica A
!>           aparece a la izquierda o derecha en la operación, de la siguiente manera:
!>
!>              SIDE = 'L' o 'l'   C := alpha*A*B + beta*C,
!>
!>              SIDE = 'R' o 'r'   C := alpha*B*A + beta*C,
!>

UPLO

!>          UPLO es CHARACTER*1
!>           A la entrada,  UPLO  especifica si se debe hacer referencia
!>           a la parte triangular superior o inferior de la matriz hermítica A, de la siguiente manera:
!>
!>              UPLO = 'U' o 'u'   Sólo se hará referencia a la parte triangular superior de la matriz hermítica.
!>
!>              UPLO = 'L' o 'l'   Sólo se hará referencia a la parte triangular inferior de la matriz hermítica.
!>

M

!>          M es INTEGER
!>           A la entrada,  M  especifica el número de filas de la matriz C.
!>           M  debe ser al menos cero.
!>

N

!>          N es INTEGER
!>           A la entrada, N especifica el número de columnas de la matriz C.
!>           N  debe ser al menos cero.
!>

ALPHA

!>          ALPHA es COMPLEX*16
!>           A la entrada,  ALPHA  especifica el escalar alpha.
!>

A

!>          A es un array COMPLEX*16, dimensión ( LDA, ka ), donde ka es
!>           m  cuando  SIDE = 'L' o 'l'  y es n en caso contrario.
!>           Antes de la entrada con  SIDE = 'L' o 'l',  la parte de m por m
!>           del array  A  debe contener la matriz hermítica, de modo que
!>           cuando  UPLO = 'U' o 'u', la parte triangular superior principal de m por m
!>           del array  A  debe contener la parte triangular superior
!>           de la matriz hermítica y la parte triangular estrictamente inferior
!>           de  A  no se hace referencia, y cuando  UPLO = 'L' o 'l',
!>           la parte triangular inferior principal de n por n del array  A
!>           debe contener la parte triangular inferior de la matriz hermítica
!>           y la parte triangular estrictamente superior de  A  no se hace referencia.
!>           Antes de la entrada con  SIDE = 'R' o 'r',  la parte de n por n
!>           del array  A  debe contener la matriz hermítica, de modo que
!>           cuando  UPLO = 'U' o 'u', la parte triangular superior principal de n por n
!>           del array  A  debe contener la parte triangular superior
!>           de la matriz hermítica y la parte triangular estrictamente inferior
!>           de  A  no se hace referencia, y cuando  UPLO = 'L' o 'l',
!>           la parte triangular inferior principal de n por n del array  A
!>           debe contener la parte triangular inferior de la matriz hermítica
!>           y la parte triangular estrictamente superior de  A  no se hace referencia.
!>           Tenga en cuenta que las partes imaginarias de los elementos diagonales
!>           no necesitan ser establecidas, se asumen que son cero.
!>

LDA

!>          LDA es INTEGER
!>           A la entrada, LDA especifica la primera dimensión de A tal como se declara
!>           en el programa (sub)llamador. Cuando  SIDE = 'L' o 'l'  entonces
!>           LDA debe ser al menos  max( 1, m ), de lo contrario  LDA debe ser
!>           al menos max( 1, n ).
!>

B

!>          B es un array COMPLEX*16, dimensión ( LDB, N )
!>           Antes de la entrada, la parte principal de m por n del array  B  debe
!>           contener la matriz B.
!>

LDB

!>          LDB es INTEGER
!>           A la entrada, LDB especifica la primera dimensión de B tal como se declara
!>           en el programa (sub)llamador.  LDB  debe ser al menos
!>           max( 1, m ).
!>

BETA

!>          BETA es COMPLEX*16
!>           A la entrada,  BETA  especifica el escalar  beta.  Cuando  BETA  se
!>           suministra como cero, entonces C no necesita ser establecida a la entrada.
!>

C

!>          C es un array COMPLEX*16, dimensión ( LDC, N )
!>           Antes de la entrada, la parte principal de m por n del array  C debe
!>           contener la matriz C,  excepto cuando beta es cero, en cuyo caso C
!>           no necesita ser establecida a la entrada.
!>           A la salida, el array  C  se sobrescribe con la matriz actualizada de m por n.
!>

LDC

!>          LDC es INTEGER
!>           A la entrada, LDC especifica la primera dimensión de C tal como se declara
!>           en el programa (sub)llamador.  LDC  debe ser al menos
!>           max( 1, m ).
!>
