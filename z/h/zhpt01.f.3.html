subroutine zhpt01 (character uplo, integer n, complex*16, dimension( * ) a, complex*16, dimension( * ) afac, integer, dimension( * ) ipiv, complex*16, dimension( ldc, * ) c, integer ldc, double precision, dimension( * ) rwork, double precision resid)

ZHPT01

Propósito:

!>
!> ZHPT01 reconstruye una matriz empaquetada indefinida Hermitiana A a partir de su
!> factorización en bloques L*D*L' o U*D*U' y calcula el residuo
!>    norm( C - A ) / ( N * norm(A) * EPS ),
!> donde C es la matriz reconstruida, EPS es la epsilon de la máquina,
!> L' es la transpuesta conjugada de L, y U' es la transpuesta conjugada
!> de U.
!>

Parámetros

UPLO 
!>          UPLO es CHARACTER*1
!>          Especifica si la parte triangular superior o inferior de la
!>          matriz Hermitiana A está almacenada:
!>          = 'U':  Triangular superior
!>          = 'L':  Triangular inferior
!>

N

!>          N es INTEGER
!>          El número de filas y columnas de la matriz A.  N >= 0.
!>

A

!>          A es un array COMPLEX*16, dimensión (N*(N+1)/2)
!>          La matriz Hermitiana original A, almacenada como una matriz
!>          triangular empaquetada.
!>

AFAC

!>          AFAC es un array COMPLEX*16, dimensión (N*(N+1)/2)
!>          La forma factorizada de la matriz A, almacenada como una matriz
!>          triangular empaquetada. AFAC contiene la matriz diagonal de bloques D
!>          y los multiplicadores utilizados para obtener el factor L o U de la
!>          factorización en bloques L*D*L' o U*D*U' tal como la calcula ZHPTRF.
!>

IPIV

!>          IPIV es un array INTEGER, dimensión (N)
!>          Los índices de pivote de ZHPTRF.
!>

C

!>          C es un array COMPLEX*16, dimensión (LDC,N)
!>

LDC

!>          LDC es INTEGER
!>          La dimensión principal del array C.  LDC >= max(1,N).
!>

RWORK

!>          RWORK es un array DOUBLE PRECISION, dimensión (N)
!>

RESID

!>          RESID es DOUBLE PRECISION
!>          Si UPLO = 'L', norm(L*D*L' - A) / ( N * norm(A) * EPS )
!>          Si UPLO = 'U', norm(U*D*U' - A) / ( N * norm(A) * EPS )
!>
