subroutine zhbgst (character vect, character uplo, integer n, integer ka, integer kb, complex*16, dimension( ldab, * ) ab, integer ldab, complex*16, dimension( ldbb, * ) bb, integer ldbb, complex*16, dimension( ldx, * ) x, integer ldx, complex*16, dimension( * ) work, double precision, dimension( * ) rwork, integer info)

ZHBGST

Propósito:

!>
!> ZHBGST reduce un problema generalizado de autovalores hermítico-definido complejo
!> A*x = lambda*B*x  a la forma estándar  C*y = lambda*y,
!> tal que C tiene la misma banda que A.
!>
!> B debe haber sido previamente factorizado como S**H*S por ZPBSTF, usando una
!> factorización de Cholesky dividida. A se sobrescribe con C = X**H*A*X, donde
!> X = S**(-1)*Q y Q es una matriz unitaria elegida para preservar la
!> banda de A.
!>

Parámetros

VECT 
!>          VECT es CHARACTER*1
!>          = 'N':  no formar la matriz de transformación X;
!>          = 'V':  formar X.
!>

UPLO

!>          UPLO es CHARACTER*1
!>          = 'U':  el triángulo superior de A está almacenado;
!>          = 'L':  el triángulo inferior de A está almacenado.
!>

N

!>          N es INTEGER
!>          El orden de las matrices A y B.  N >= 0.
!>

KA

!>          KA es INTEGER
!>          El número de superdiagonales de la matriz A si UPLO = 'U',
!>          o el número de subdiagonales si UPLO = 'L'.  KA >= 0.
!>

KB

!>          KB es INTEGER
!>          El número de superdiagonales de la matriz B si UPLO = 'U',
!>          o el número de subdiagonales si UPLO = 'L'.  KA >= KB >= 0.
!>

AB

!>          AB es un array COMPLEX*16, dimension (LDAB,N)
!>          A la entrada, el triángulo superior o inferior de la matriz
!>          banda Hermitiana A, almacenado en las primeras ka+1 filas del array.
!>          La j-ésima columna de A se almacena en la j-ésima columna del array AB
!>          como sigue:
!>          si UPLO = 'U', AB(ka+1+i-j,j) = A(i,j) para max(1,j-ka)<=i<=j;
!>          si UPLO = 'L', AB(1+i-j,j)    = A(i,j) para j<=i<=min(n,j+ka).
!>
!>          A la salida, la matriz transformada X**H*A*X, almacenada en el mismo
!>          formato que A.
!>

LDAB

!>          LDAB es INTEGER
!>          La dimensión principal del array AB.  LDAB >= KA+1.
!>

BB

!>          BB es un array COMPLEX*16, dimension (LDBB,N)
!>          El factor de banda S de la factorización de Cholesky dividida
!>          de B, tal como lo devuelve ZPBSTF, almacenado en las primeras kb+1
!>          filas del array.
!>

LDBB

!>          LDBB es INTEGER
!>          La dimensión principal del array BB.  LDBB >= KB+1.
!>

X

!>          X es un array COMPLEX*16, dimension (LDX,N)
!>          Si VECT = 'V', la matriz X de n por n.
!>          Si VECT = 'N', el array X no se referencia.
!>

LDX

!>          LDX es INTEGER
!>          La dimensión principal del array X.
!>          LDX >= max(1,N) si VECT = 'V'; LDX >= 1 en caso contrario.
!>

WORK

!>          WORK es un array COMPLEX*16, dimension (N)
!>

RWORK

!>          RWORK es un array DOUBLE PRECISION, dimension (N)
!>

INFO

!>          INFO es INTEGER
!>          = 0:  salida exitosa
!>          < 0:  si INFO = -i, el i-ésimo argumento tenía un valor ilegal.
!>
