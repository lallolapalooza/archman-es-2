subroutine zchkpt (logical, dimension( * ) dotype, integer nn, integer, dimension( * ) nval, integer nns, integer, dimension( * ) nsval, double precision thresh, logical tsterr, complex*16, dimension( * ) a, double precision, dimension( * ) d, complex*16, dimension( * ) e, complex*16, dimension( * ) b, complex*16, dimension( * ) x, complex*16, dimension( * ) xact, complex*16, dimension( * ) work, double precision, dimension( * ) rwork, integer nout)

ZCHKPT

Propósito:

!>
!> ZCHKPT prueba ZPTTRF, -TRS, -RFS y -CON
!>

Parámetros

DOTYPE 
!>          DOTYPE es un array LOGICAL, de dimensión (NTYPES)
!>          Los tipos de matriz a usar para las pruebas. Las matrices de tipo j
!>          (para 1 <= j <= NTYPES) se usan para las pruebas si DOTYPE(j) =
!>          .TRUE.; si DOTYPE(j) = .FALSE., entonces el tipo j no se usa.
!>

NN

!>          NN es INTEGER
!>          El número de valores de N contenidos en el vector NVAL.
!>

NVAL

!>          NVAL es un array INTEGER, de dimensión (NN)
!>          Los valores de la dimensión de la matriz N.
!>

NNS

!>          NNS es INTEGER
!>          El número de valores de NRHS contenidos en el vector NSVAL.
!>

NSVAL

!>          NSVAL es un array INTEGER, de dimensión (NNS)
!>          Los valores del número de lados derechos NRHS.
!>

THRESH

!>          THRESH es DOUBLE PRECISION
!>          El valor umbral para las relaciones de prueba. Un resultado se
!>          incluye en el archivo de salida si RESULT >= THRESH. Para que se
!>          imprima cada relación de prueba, use THRESH = 0.
!>

TSTERR

!>          TSTERR es LOGICAL
!>          Indicador que señala si se deben probar las salidas de error.
!>

A

!>          A es un array COMPLEX*16, de dimensión (NMAX*2)
!>

D

!>          D es un array DOUBLE PRECISION, de dimensión (NMAX*2)
!>

E

!>          E es un array COMPLEX*16, de dimensión (NMAX*2)
!>

B

!>          B es un array COMPLEX*16, de dimensión (NMAX*NSMAX)
!>          donde NSMAX es la entrada más grande en NSVAL.
!>

X

!>          X es un array COMPLEX*16, de dimensión (NMAX*NSMAX)
!>

XACT

!>          XACT es un array COMPLEX*16, de dimensión (NMAX*NSMAX)
!>

WORK

!>          WORK es un array COMPLEX*16, de dimensión
!>                      (NMAX*max(3,NSMAX))
!>

RWORK

!>          RWORK es un array DOUBLE PRECISION, de dimensión
!>                      (max(NMAX,2*NSMAX))
!>

NOUT

!>          NOUT es INTEGER
!>          El número de unidad para la salida.
!>
