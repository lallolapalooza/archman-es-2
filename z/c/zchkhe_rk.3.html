subrutina zchkhe_rk (logical, dimension( * ) dotype, integer nn, integer, dimension( * ) nval, integer nnb, integer, dimension( * ) nbval, integer nns, integer, dimension( * ) nsval, double precision thresh, logical tsterr, integer nmax, complex*16, dimension( * ) a, complex*16, dimension( * ) afac, complex*16, dimension( * ) e, complex*16, dimension( * ) ainv, complex*16, dimension( * ) b, complex*16, dimension( * ) x, complex*16, dimension( * ) xact, complex*16, dimension( * ) work, double precision, dimension( * ) rwork, integer, dimension( * ) iwork, integer nout)

ZCHKHE_RK

Propósito:

!>
!> ZCHKHE_RK prueba ZHETRF_RK, -TRI_3, -TRS_3,
!> y -CON_3.
!>

Parámetros

DOTYPE 
!>          DOTYPE es una matriz LOGICAL, dimension (NTYPES)
!>          Los tipos de matriz a utilizar para las pruebas. Las matrices
!>          de tipo j (para 1 <= j <= NTYPES) se utilizan para las pruebas
!>          si DOTYPE(j) = .TRUE.; si DOTYPE(j) = .FALSE., entonces el tipo j
!>          no se utiliza.
!>

NN

!>          NN es INTEGER
!>          El número de valores de N contenidos en el vector NVAL.
!>

NVAL

!>          NVAL es una matriz INTEGER, dimension (NN)
!>          Los valores de la dimensión de la matriz N.
!>

NNB

!>          NNB es INTEGER
!>          El número de valores de NB contenidos en el vector NBVAL.
!>

NBVAL

!>          NBVAL es una matriz INTEGER, dimension (NNB)
!>          Los valores del tamaño del bloque NB.
!>

NNS

!>          NNS es INTEGER
!>          El número de valores de NRHS contenidos en el vector NSVAL.
!>

NSVAL

!>          NSVAL es una matriz INTEGER, dimension (NNS)
!>          Los valores del número de segundos miembros NRHS.
!>

THRESH

!>          THRESH es DOUBLE PRECISION
!>          El valor umbral para las relaciones de prueba. Un resultado se
!>          incluye en el archivo de salida si RESULT >= THRESH. Para que
!>          cada relación de prueba se imprima, use THRESH = 0.
!>

TSTERR

!>          TSTERR es LOGICAL
!>          Indicador que señala si se van a probar las salidas de error.
!>

NMAX

!>          NMAX es INTEGER
!>          El valor máximo permitido para N, utilizado en la dimensionamiento
!>          de los arreglos de trabajo.
!>

A

!>          A es una matriz CCOMPLEX*16, dimension (NMAX*NMAX)
!>

AFAC

!>          AFAC es una matriz COMPLEX*16, dimension (NMAX*NMAX)
!>

E

!>          E es una matriz COMPLEX*16, dimension (NMAX)
!>

AINV

!>          AINV es una matriz COMPLEX*16, dimension (NMAX*NMAX)
!>

B

!>          B es una matriz CCOMPLEX*16, dimension (NMAX*NSMAX)
!>          donde NSMAX es la entrada más grande en NSVAL.
!>

X

!>          X es una matriz COMPLEX*16, dimension (NMAX*NSMAX)
!>

XACT

!>          XACT es una matriz COMPLEX*16, dimension (NMAX*NSMAX)
!>

WORK

!>          WORK es una matriz COMPLEX*16, dimension (NMAX*max(3,NSMAX))
!>

RWORK

!>          RWORK es una matriz DOUBLE PRECISION, dimension (max(NMAX,2*NSMAX)
!>

IWORK

!>          IWORK es una matriz INTEGER, dimension (2*NMAX)
!>

NOUT

!>          NOUT es INTEGER
!>          El número de unidad para la salida.
!>
