subroutine zchkbb (integer nsizes, integer, dimension( * ) mval, integer, dimension( * ) nval, integer nwdths, integer, dimension( * ) kk, integer ntypes, logical, dimension( * ) dotype, integer nrhs, integer, dimension( 4 ) iseed, double precision thresh, integer nounit, complex*16, dimension( lda, * ) a, integer lda, complex*16, dimension( ldab, * ) ab, integer ldab, double precision, dimension( * ) bd, double precision, dimension( * ) be, complex*16, dimension( ldq, * ) q, integer ldq, complex*16, dimension( ldp, * ) p, integer ldp, complex*16, dimension( ldc, * ) c, integer ldc, complex*16, dimension( ldc, * ) cc, complex*16, dimension( * ) work, integer lwork, double precision, dimension( * ) rwork, double precision, dimension( * ) result, integer info)

ZCHKBB

Propósito:

!>
!> ZCHKBB prueba la reducción de una matriz de banda rectangular compleja general
!> a forma bidiagonal real.
!>
!> ZGBBRD factoriza una matriz de banda general A como Q B P* , donde * significa
!> transpuesta conjugada, B es bidiagonal superior, y Q y P son unitarias;
!> ZGBBRD también puede sobrescribir una matriz C dada con Q* C .
!>
!> Para cada par de dimensiones de matriz (M,N) y cada tipo de matriz seleccionado,
!> se generan una matriz A de M por N y una matriz C de M por NRHS.
!> Las dimensiones del problema son las siguientes
!>    A:          M x N
!>    Q:          M x M
!>    P:          N x N
!>    B:          min(M,N) x min(M,N)
!>    C:          M x NRHS
!>
!> Para cada matriz generada, se realizan 4 pruebas:
!>
!> (1)   | A - Q B PT | / ( |A| max(M,N) ulp ), PT = P'
!>
!> (2)   | I - Q' Q | / ( M ulp )
!>
!> (3)   | I - PT PT' | / ( N ulp )
!>
!> (4)   | Y - Q' C | / ( |Y| max(M,NRHS) ulp ), donde Y = Q' C.
!>
!> Los se especifican mediante un array lógico DOTYPE( 1:NTYPES );
!> si DOTYPE(j) es .TRUE., entonces se generará la matriz de tipo j.
!> Actualmente, la lista de tipos posibles es:
!>
!> Los tipos de matriz posibles son
!>
!> (1)  La matriz cero.
!> (2)  La matriz identidad.
!>
!> (3)  Una matriz diagonal con entradas espaciadas uniformemente
!>      1, ..., ULP  y signos aleatorios.
!>      (ULP = (primer número mayor que 1) - 1 )
!> (4)  Una matriz diagonal con entradas espaciadas geométricamente
!>      1, ..., ULP  y signos aleatorios.
!> (5)  Una matriz diagonal con entradas 1, ULP, ..., ULP
!>      y signos aleatorios.
!>
!> (6)  Igual que (3), pero multiplicado por SQRT( umbral de desbordamiento )
!> (7)  Igual que (3), pero multiplicado por SQRT( umbral de subdesbordamiento )
!>
!> (8)  Una matriz de la forma U D V, donde U y V son ortogonales y
!>      D tiene entradas espaciadas uniformemente 1, ..., ULP con signos aleatorios
!>      en la diagonal.
!>
!> (9)  Una matriz de la forma U D V, donde U y V son ortogonales y
!>      D tiene entradas espaciadas geométricamente 1, ..., ULP con signos aleatorios
!>      en la diagonal.
!>
!> (10) Una matriz de la forma U D V, donde U y V son ortogonales y
!>      D tiene entradas 1, ULP,..., ULP con signos aleatorios
!>      en la diagonal.
!>
!> (11) Igual que (8), pero multiplicado por SQRT( umbral de desbordamiento )
!> (12) Igual que (8), pero multiplicado por SQRT( umbral de subdesbordamiento )
!>
!> (13) Matriz rectangular con entradas aleatorias elegidas de (-1,1).
!> (14) Igual que (13), pero multiplicado por SQRT( umbral de desbordamiento )
!> (15) Igual que (13), pero multiplicado por SQRT( umbral de subdesbordamiento )
!>

Parámetros

NSIZES 
!>          NSIZES es INTEGER
!>          El número de valores de M y N contenidos en los vectores
!>          MVAL y NVAL. Los tamaños de matriz se usan en pares (M,N).
!>          Si NSIZES es cero, ZCHKBB no hace nada. NSIZES debe ser
!>          al menos cero.
!>

MVAL

!>          MVAL es un array INTEGER, dimension (NSIZES)
!>          Los valores de la dimensión de fila de matriz M.
!>

NVAL

!>          NVAL es un array INTEGER, dimension (NSIZES)
!>          Los valores de la dimensión de columna de matriz N.
!>

NWDTHS

!>          NWDTHS es INTEGER
!>          El número de anchos de banda a utilizar. Si es cero,
!>          ZCHKBB no hace nada. Debe ser al menos cero.
!>

KK

!>          KK es un array INTEGER, dimension (NWDTHS)
!>          Un array que contiene los anchos de banda a utilizar para las matrices
!>          de banda. Los valores deben ser al menos cero.
!>

NTYPES

!>          NTYPES es INTEGER
!>          El número de elementos en DOTYPE. Si es cero, ZCHKBB
!>          no hace nada. Debe ser al menos cero. Si es MAXTYP+1
!>          y NSIZES es 1, entonces se define un tipo adicional, MAXTYP+1,
!>          que es usar la matriz que sea que esté en A. Esto
!>          sólo es útil si DOTYPE(1:MAXTYP) es .FALSE. y
!>          DOTYPE(MAXTYP+1) es .TRUE. .
!>

DOTYPE

!>          DOTYPE es un array LOGICAL, dimension (NTYPES)
!>          Si DOTYPE(j) es .TRUE., entonces para cada tamaño en NN se generará
!>          una matriz de ese tamaño y de tipo j.
!>          Si NTYPES es menor que el número máximo de tipos
!>          definidos (PARAMETER MAXTYP), entonces los tipos NTYPES+1 hasta
!>          MAXTYP no se generarán. Si NTYPES es mayor
!>          que MAXTYP, DOTYPE(MAXTYP+1) hasta DOTYPE(NTYPES)
!>          serán ignorados.
!>

NRHS

!>          NRHS es INTEGER
!>          El número de columnas en la matriz C.
!>          Si NRHS = 0, entonces las operaciones en el lado derecho
!>          no se probarán. NRHS debe ser al menos 0.
!>

ISEED

!>          ISEED es un array INTEGER, dimension (4)
!>          En la entrada, ISEED especifica la semilla del generador de números
!>          aleatorios. Los elementos del array deben estar entre 0 y 4095;
!>          si no lo están, se reducirán módulo 4096. Además, ISEED(4) debe
!>          ser impar. El generador de números aleatorios utiliza una secuencia
!>          congruencial lineal limitada a enteros pequeños, y por lo tanto
!>          debe producir números aleatorios independientes de la máquina. Los
!>          valores de ISEED se modifican a la salida y se pueden usar en la
!>          siguiente llamada a ZCHKBB para continuar la misma secuencia de números
!>          aleatorios.
!>

THRESH

!>          THRESH es DOUBLE PRECISION
!>          Una prueba se contará como si el error calculado,
!>          como se describe anteriormente, excede THRESH. Tenga en cuenta que el
!>          error se escala para ser O(1), por lo que THRESH debe ser un múltiplo
!>          razonablemente pequeño de 1, por ejemplo, 10 o 100. En particular,
!>          no debe depender de la precisión (simple o doble)
!>          o del tamaño de la matriz. Debe ser al menos cero.
!>

NOUNIT

!>          NOUNIT es INTEGER
!>          El número de unidad FORTRAN para imprimir mensajes de error
!>          (por ejemplo, si una rutina devuelve IINFO distinto de 0).
!>
