subroutine zchkqrt (double precision thresh, logical tsterr, integer nm, integer, dimension( * ) mval, integer nn, integer, dimension( * ) nval, integer nnb, integer, dimension( * ) nbval, integer nout)

ZCHKQRT

Propósito

!>
!> ZCHKQRT prueba ZGEQRT y ZGEMQRT.
!>

Parámetros

THRESH 
!>          THRESH es DOUBLE PRECISION
!>          El valor umbral para las relaciones de prueba. Un resultado se
!>          incluye en el archivo de salida si RESULT >= THRESH. Para que
!>          cada relación de prueba se imprima, use THRESH = 0.
!>

TSTERR

!>          TSTERR es LOGICAL
!>          Indicador que señala si se van a probar las salidas de error.
!>

NM

!>          NM es INTEGER
!>          El número de valores de M contenidos en el vector MVAL.
!>

MVAL

!>          MVAL es un array INTEGER, dimensión (NM)
!>          Los valores de la dimensión de fila de la matriz M.
!>

NN

!>          NN es INTEGER
!>          El número de valores de N contenidos en el vector NVAL.
!>

NVAL

!>          NVAL es un array INTEGER, dimensión (NN)
!>          Los valores de la dimensión de columna de la matriz N.
!>

NNB

!>          NNB es INTEGER
!>          El número de valores de NB contenidos en el vector NBVAL.
!>

NBVAL

!>          NBVAL es un array INTEGER, dimensión (NNB)
!>          Los valores del tamaño del bloque NB.
!>

NOUT

!>          NOUT es INTEGER
!>          El número de unidad para la salida.
!>
