Documentación de la función/subrutina

subroutine zchktp (logical, dimension( * ) dotype, integer nn, integer, dimension( * ) nval, integer nns, integer, dimension( * ) nsval, double precision thresh, logical tsterr, integer nmax, complex*16, dimension( * ) ap, complex*16, dimension( * ) ainvp, complex*16, dimension( * ) b, complex*16, dimension( * ) x, complex*16, dimension( * ) xact, complex*16, dimension( * ) work, double precision, dimension( * ) rwork, integer nout)

ZCHKTP

Propósito:

!>
!> ZCHKTP prueba ZTPTRI, -TRS, -RFS, y -CON, y ZLATPS
!>

Parámetros

DOTYPE 
!>          DOTYPE es un array LOGICAL, dimension (NTYPES)
!>          Los tipos de matriz a utilizar para las pruebas. Matrices del tipo j
!>          (para 1 <= j <= NTYPES) se utilizan para las pruebas si DOTYPE(j) =
!>          .TRUE.; si DOTYPE(j) = .FALSE., entonces el tipo j no se utiliza.
!>

NN

!>          NN es INTEGER
!>          El número de valores de N contenidos en el vector NVAL.
!>

NVAL

!>          NVAL es un array INTEGER, dimension (NN)
!>          Los valores de la dimensión de columna de la matriz N.
!>

NNS

!>          NNS es INTEGER
!>          El número de valores de NRHS contenidos en el vector NSVAL.
!>

NSVAL

!>          NSVAL es un array INTEGER, dimension (NNS)
!>          Los valores del número de lados derechos NRHS.
!>

THRESH

!>          THRESH es DOUBLE PRECISION
!>          El valor umbral para las relaciones de prueba. Un resultado se
!>          incluye en el archivo de salida si RESULT >= THRESH. Para que
!>          cada relación de prueba se imprima, use THRESH = 0.
!>

TSTERR

!>          TSTERR es LOGICAL
!>          Indicador que señala si las salidas de error deben ser probadas.
!>

NMAX

!>          NMAX es INTEGER
!>          La dimensión principal de los arrays de trabajo. NMAX >= el
!>          valor máximo de N en NVAL.
!>

AP

!>          AP es un array COMPLEX*16, dimension (NMAX*(NMAX+1)/2)
!>

AINVP

!>          AINVP es un array COMPLEX*16, dimension (NMAX*(NMAX+1)/2)
!>

B

!>          B es un array COMPLEX*16, dimension (NMAX*NSMAX)
!>          donde NSMAX es la entrada más grande en NSVAL.
!>

X

!>          X es un array COMPLEX*16, dimension (NMAX*NSMAX)
!>

XACT

!>          XACT es un array COMPLEX*16, dimension (NMAX*NSMAX)
!>

WORK

!>          WORK es un array COMPLEX*16, dimension
!>                      (NMAX*max(3,NSMAX))
!>

RWORK

!>          RWORK es un array DOUBLE PRECISION, dimension
!>                      (max(NMAX,2*NSMAX))
!>

NOUT

!>          NOUT es INTEGER
!>          El número de unidad para la salida.
!>
