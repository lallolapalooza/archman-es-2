subrutina zchkpb (lógico, dimensión( * ) dotype, entero nn, entero, dimensión( * ) nval, entero nnb, entero, dimensión( * ) nbval, entero nns, entero, dimensión( * ) nsval, doble precisión thresh, lógico tsterr, entero nmax, complejo*16, dimensión( * ) a, complejo*16, dimensión( * ) afac, complejo*16, dimensión( * ) ainv, complejo*16, dimensión( * ) b, complejo*16, dimensión( * ) x, complejo*16, dimensión( * ) xact, complejo*16, dimensión( * ) work, doble precisión, dimensión( * ) rwork, entero nout)

ZCHKPB

Propósito

!>
!> ZCHKPB prueba ZPBTRF, -TRS, -RFS y -CON.
!>

Parámetros

DOTYPE 
!>          DOTYPE es un array lógico, de dimensión (NTYPES)
!>          Los tipos de matrices a utilizar para las pruebas. Las matrices de tipo j
!>          (para 1 <= j <= NTYPES) se utilizan para las pruebas si DOTYPE(j) =
!>          .TRUE.; si DOTYPE(j) = .FALSE., entonces el tipo j no se utiliza.
!>

NN

!>          NN es un ENTERO
!>          El número de valores de N contenidos en el vector NVAL.
!>

NVAL

!>          NVAL es un array de ENTEROS, de dimensión (NN)
!>          Los valores de la dimensión de la matriz N.
!>

NNB

!>          NNB es un ENTERO
!>          El número de valores de NB contenidos en el vector NBVAL.
!>

NBVAL

!>          NBVAL es un array de ENTEROS, de dimensión (NNB)
!>          Los valores del tamaño de bloque NB.
!>

NNS

!>          NNS es un ENTERO
!>          El número de valores de NRHS contenidos en el vector NSVAL.
!>

NSVAL

!>          NSVAL es un array de ENTEROS, de dimensión (NNS)
!>          Los valores del número de segundos miembros NRHS.
!>

THRESH

!>          THRESH es DOBLE PRECISIÓN
!>          El valor umbral para las relaciones de prueba. Un resultado se
!>          incluye en el archivo de salida si RESULT >= THRESH. Para que
!>          cada relación de prueba se imprima, use THRESH = 0.
!>

TSTERR

!>          TSTERR es LÓGICO
!>          Indicador que señala si se deben probar las salidas de error.
!>

NMAX

!>          NMAX es un ENTERO
!>          El valor máximo permitido para N, utilizado en la dimensionamiento de los
!>          arrays de trabajo.
!>

A

!>          A es un array de DOBLE PRECISIÓN, de dimensión (NMAX*NMAX)
!>

AFAC

!>          AFAC es un array de DOBLE PRECISIÓN, de dimensión (NMAX*NMAX)
!>

AINV

!>          AINV es un array de DOBLE PRECISIÓN, de dimensión (NMAX*NMAX)
!>

B

!>          B es un array de DOBLE PRECISIÓN, de dimensión (NMAX*NSMAX)
!>          donde NSMAX es la entrada más grande en NSVAL.
!>

X

!>          X es un array de DOBLE PRECISIÓN, de dimensión (NMAX*NSMAX)
!>

XACT

!>          XACT es un array de DOBLE PRECISIÓN, de dimensión (NMAX*NSMAX)
!>

WORK

!>          WORK es un array de DOBLE PRECISIÓN, de dimensión
!>                      (NMAX*max(3,NSMAX))
!>

RWORK

!>          RWORK es un array de DOBLE PRECISIÓN, de dimensión
!>                      (max(NMAX,2*NSMAX))
!>

NOUT

!>          NOUT es un ENTERO
!>          El número de unidad para la salida.
!>
