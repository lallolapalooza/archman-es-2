subrutina zchkpt (lógico, dimensión( * ) dotype, entero nn, entero, dimensión( * ) nval, entero nns, entero, dimensión( * ) nsval, doble precisión thresh, lógico tsterr, complejo*16, dimensión( * ) a, doble precisión, dimensión( * ) d, complejo*16, dimensión( * ) e, complejo*16, dimensión( * ) b, complejo*16, dimensión( * ) x, complejo*16, dimensión( * ) xact, complejo*16, dimensión( * ) work, doble precisión, dimensión( * ) rwork, entero nout)

ZCHKPT

Propósito

!>
!> ZCHKPT prueba ZPTTRF, -TRS, -RFS y -CON
!>

Parámetros

DOTYPE 
!>          DOTYPE es un array LOGICO, dimension (NTYPES)
!>          Los tipos de matrices a utilizar para las pruebas. Las matrices de tipo j
!>          (para 1 <= j <= NTYPES) se utilizan para las pruebas si DOTYPE(j) =
!>          .TRUE.; si DOTYPE(j) = .FALSE., entonces el tipo j no se utiliza.
!>

NN

!>          NN es un ENTERO
!>          El número de valores de N contenidos en el vector NVAL.
!>

NVAL

!>          NVAL es un array ENTERO, dimension (NN)
!>          Los valores de la dimensión de la matriz N.
!>

NNS

!>          NNS es un ENTERO
!>          El número de valores de NRHS contenidos en el vector NSVAL.
!>

NSVAL

!>          NSVAL es un array ENTERO, dimension (NNS)
!>          Los valores del número de segundos miembros NRHS.
!>

THRESH

!>          THRESH es DOBLE PRECISION
!>          El valor umbral para las relaciones de prueba. Un resultado se
!>          incluye en el archivo de salida si RESULT >= THRESH. Para que
!>          se imprima cada relación de prueba, utilice THRESH = 0.
!>

TSTERR

!>          TSTERR es LOGICO
!>          Indicador que señala si se van a probar las salidas de error.
!>

A

!>          A es un array COMPLEJO*16, dimension (NMAX*2)
!>

D

!>          D es un array DOBLE PRECISION, dimension (NMAX*2)
!>

E

!>          E es un array COMPLEJO*16, dimension (NMAX*2)
!>

B

!>          B es un array COMPLEJO*16, dimension (NMAX*NSMAX)
!>          donde NSMAX es la entrada más grande en NSVAL.
!>

X

!>          X es un array COMPLEJO*16, dimension (NMAX*NSMAX)
!>

XACT

!>          XACT es un array COMPLEJO*16, dimension (NMAX*NSMAX)
!>

WORK

!>          WORK es un array COMPLEJO*16, dimension
!>                      (NMAX*max(3,NSMAX))
!>

RWORK

!>          RWORK es un array DOBLE PRECISION, dimension
!>                      (max(NMAX,2*NSMAX))
!>

NOUT

!>          NOUT es un ENTERO
!>          El número de unidad para la salida.
!>
