subroutine zlaqz1 (logical, intent(in) ilq, logical, intent(in) ilz, integer, intent(in) k, integer, intent(in) istartm, integer, intent(in) istopm, integer, intent(in) ihi, complex*16, dimension( lda, * ) a, integer, intent(in) lda, complex*16, dimension( ldb, * ) b, integer, intent(in) ldb, integer, intent(in) nq, integer, intent(in) qstart, complex*16, dimension( ldq, * ) q, integer, intent(in) ldq, integer, intent(in) nz, integer, intent(in) zstart, complex*16, dimension( ldz, * ) z, integer, intent(in) ldz)

ZLAQZ1

Propósito:

!>
!>      ZLAQZ1 persigue una protuberancia de desplazamiento 1x1 en un lápiz de matriz hacia abajo una sola posición
!>

Parámetros

ILQ 
!>          ILQ es LOGICAL
!>              Determina si se actualiza o no la matriz Q
!>

ILZ

!>          ILZ es LOGICAL
!>              Determina si se actualiza o no la matriz Z
!>

K

!>          K es INTEGER
!>              Índice que indica la posición de la protuberancia.
!>              En la entrada, la protuberancia se encuentra en
!>              (A(k+1,k),B(k+1,k)).
!>              En la salida, la protuberancia se encuentra en
!>              (A(k+2,k+1),B(k+2,k+1)).
!>

ISTARTM

!>          ISTARTM es INTEGER
!>

ISTOPM

!>          ISTOPM es INTEGER
!>              Las actualizaciones a (A,B) se restringen a
!>              (istartm:k+2,k:istopm). Se asume
!>              sin verificación que istartm <= k+1 y
!>              k+2 <= istopm
!>

IHI

!>          IHI es INTEGER
!>

A

!>          A es un array COMPLEX*16, dimensión (LDA,N)
!>

LDA

!>          LDA es INTEGER
!>              La dimensión principal de A tal como se declara en
!>              el procedimiento de llamada.
!>

B

!>          B es un array COMPLEX*16, dimensión (LDB,N)
!>

LDB

!>          LDB es INTEGER
!>              La dimensión principal de B tal como se declara en
!>              el procedimiento de llamada.
!>

NQ

!>          NQ es INTEGER
!>              El orden de la matriz Q
!>

QSTART

!>          QSTART es INTEGER
!>              Índice de inicio de la matriz Q. Las rotaciones se aplican
!>              a las columnas k+2-qStart:k+3-qStart de Q.
!>

Q

!>          Q es un array COMPLEX*16, dimensión (LDQ,NQ)
!>

LDQ

!>          LDQ es INTEGER
!>              La dimensión principal de Q tal como se declara en
!>              el procedimiento de llamada.
!>

NZ

!>          NZ es INTEGER
!>              El orden de la matriz Z
!>

ZSTART

!>          ZSTART es INTEGER
!>              Índice de inicio de la matriz Z. Las rotaciones se aplican
!>              a las columnas k+1-qStart:k+2-qStart de Z.
!>

Z

!>          Z es un array COMPLEX*16, dimensión (LDZ,NZ)
!>

LDZ

!>          LDZ es INTEGER
!>              La dimensión principal de Q tal como se declara en
!>              el procedimiento de llamada.
!>
