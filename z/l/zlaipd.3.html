subroutine zlaipd (integer n, complex*16, dimension( * ) a, integer inda, integer vinda)

ZLAIPD

Propósito:

!>
!> ZLAIPD establece la parte imaginaria de los elementos diagonales de una
!> matriz compleja A a un valor grande. Esto se utiliza para probar rutinas
!> LAPACK para matrices Hermitianas complejas, que no deben acceder ni usar
!> las partes imaginarias de las diagonales.
!>

Parámetros

N 
!>          N es ENTERO
!>         El número de elementos diagonales de A.
!>

A

!>          A es un array COMPLEX*16, dimension
!>                        (1+(N-1)*INDA+(N-2)*VINDA)
!>         A la entrada, la matriz compleja (Hermitiana) A.
!>         A la salida, las partes imaginarias de los elementos diagonales se
!>         establecen a BIGNUM = EPS / SAFMIN, donde EPS es el epsilon de la
!>         máquina y SAFMIN es el mínimo seguro.
!>

INDA

!>          INDA es ENTERO
!>         El incremento entre A(1) y el siguiente elemento diagonal de A.
!>         Los valores típicos son
!>         = LDA+1: matrices cuadradas con dimensión principal LDA
!>         = 2: matriz triangular superior empaquetada, comenzando en A(1,1)
!>         = N: matriz triangular inferior empaquetada, comenzando en A(1,1)
!>

VINDA

!>          VINDA es ENTERO
!>         El cambio en el incremento diagonal entre columnas de A.
!>         Los valores típicos son
!>         = 0: sin cambios, los incrementos de fila y columna en A son fijos
!>         = 1: matriz triangular superior empaquetada
!>         = -1: matriz triangular inferior empaquetada
!>
