subroutine zlaqz3 (logical, intent(in) ilschur, logical, intent(in) ilq, logical, intent(in) ilz, integer, intent(in) n, integer, intent(in) ilo, integer, intent(in) ihi, integer, intent(in) nshifts, integer, intent(in) nblock_desired, complex*16, dimension( * ), intent(inout) alpha, complex*16, dimension( * ), intent(inout) beta, complex*16, dimension( lda, * ), intent(inout) a, integer, intent(in) lda, complex*16, dimension( ldb, * ), intent(inout) b, integer, intent(in) ldb, complex*16, dimension( ldq, * ), intent(inout) q, integer, intent(in) ldq, complex*16, dimension( ldz, * ), intent(inout) z, integer, intent(in) ldz, complex*16, dimension( ldqc, * ), intent(inout) qc, integer, intent(in) ldqc, complex*16, dimension( ldzc, * ), intent(inout) zc, integer, intent(in) ldzc, complex*16, dimension( * ), intent(inout) work, integer, intent(in) lwork, integer, intent(out) info)

ZLAQZ3

Propósito:

!>
!> ZLAQZ3 Ejecuta un barrido QZ multieje.
!>

Parámetros

ILSCHUR 
!>          ILSCHUR es LOGICAL
!>              Determina si se debe actualizar la forma de Schur completa o no.
!>

ILQ

!>          ILQ es LOGICAL
!>              Determina si se debe actualizar la matriz Q o no.
!>

ILZ

!>          ILZ es LOGICAL
!>              Determina si se debe actualizar la matriz Z o no.
!>

N

!>          N es INTEGER
!>          El orden de las matrices A, B, Q y Z. N >= 0.
!>

ILO

!>          ILO es INTEGER
!>

IHI

!>          IHI es INTEGER
!>

NSHIFTS

!>          NSHIFTS es INTEGER
!>          El número deseado de desplazamientos a utilizar.
!>

NBLOCK_DESIRED

!>          NBLOCK_DESIRED es INTEGER
!>          El tamaño deseado de las ventanas computacionales.
!>

ALPHA

!>          ALPHA es un array COMPLEX*16. SR contiene
!>          las partes alfa de los desplazamientos a utilizar.
!>

BETA

!>          BETA es un array COMPLEX*16. SS contiene
!>          la escala de los desplazamientos a utilizar.
!>

A

!>          A es un array COMPLEX*16, dimensión (LDA, N)
!>

LDA

!>          LDA es INTEGER
!>          La dimensión principal del array A. LDA >= max( 1, N ).
!>

B

!>          B es un array COMPLEX*16, dimensión (LDB, N)
!>

LDB

!>          LDB es INTEGER
!>          La dimensión principal del array B. LDB >= max( 1, N ).
!>

Q

!>          Q es un array COMPLEX*16, dimensión (LDQ, N)
!>

LDQ

!>          LDQ es INTEGER
!>

Z

!>          Z es un array COMPLEX*16, dimensión (LDZ, N)
!>

LDZ

!>          LDZ es INTEGER
!>

QC

!>          QC es un array COMPLEX*16, dimensión (LDQC, NBLOCK_DESIRED)
!>

LDQC

!>          LDQC es INTEGER
!>

ZC

!>          ZC es un array COMPLEX*16, dimensión (LDZC, NBLOCK_DESIRED)
!>

LDZC

!>          LDZ es INTEGER
!>

WORK

!>          WORK es un array COMPLEX*16, dimensión (MAX(1,LWORK))
!>          A la salida, si INFO >= 0, WORK(1) devuelve el LWORK óptimo.
!>

LWORK

!>          LWORK es INTEGER
!>          La dimensión del array WORK. LWORK >= max(1,N).
!>
!>          Si LWORK = -1, entonces se asume una consulta de espacio de trabajo; la rutina
!>          solo calcula el tamaño óptimo del array WORK, devuelve
!>          este valor como la primera entrada del array WORK, y XERBLA no emite
!>          ningún mensaje de error relacionado con LWORK.
!>

INFO

!>          INFO es INTEGER
!>          = 0: salida exitosa
!>          < 0: si INFO = -i, el i-ésimo argumento tenía un valor ilegal.
!>
