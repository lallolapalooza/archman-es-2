subroutine zlatme (integer n, character dist, integer, dimension( 4 ) iseed, complex*16, dimension( * ) d, integer mode, double precision cond, complex*16 dmax, character rsign, character upper, character sim, double precision, dimension( * ) ds, integer modes, double precision conds, integer kl, integer ku, double precision anorm, complex*16, dimension( lda, * ) a, integer lda, complex*16, dimension( * ) work, integer info)

ZLATME

Propósito:

!>
!>    ZLATME genera matrices cuadradas no simétricas aleatorias con
!>    valores propios especificados para probar programas LAPACK.
!>
!>    ZLATME opera aplicando la siguiente secuencia de
!>    operaciones:
!>
!>    1. Establece la diagonal en D, donde D puede ser de entrada o
!>         calculada según MODE, COND, DMAX y RSIGN
!>         como se describe a continuación.
!>
!>    2. Si UPPER='T', el triángulo superior de A se establece con valores
!>         aleatorios de la distribución DIST.
!>
!>    3. Si SIM='T', A se multiplica por la izquierda por una matriz
!>         aleatoria X, cuyos valores singulares están especificados por DS, MODES y
!>         CONDS, y por la derecha por X inversa.
!>
!>    4. Si KL < N-1, la banda inferior se reduce a KL usando
!>         transformaciones de Householder. Si KU < N-1, la banda superior
!>         se reduce a KU.
!>
!>    5. Si ANORM no es negativo, la matriz se escala para tener
!>         norma de elemento máximo ANORM.
!>
!>    (Nota: dado que la matriz no puede reducirse más allá de la forma de Hessenberg,
!>     no hay opciones de empaquetado disponibles.)
!>

Parámetros

N 
!>          N es INTEGER
!>           El número de columnas (o filas) de A. No modificado.
!>

DIST

!>          DIST es CHARACTER*1
!>           En la entrada, DIST especifica el tipo de distribución que se usará
!>           para generar los valores propios/singulares aleatorios y en el
!>           triángulo superior (ver UPPER).
!>           'U' => UNIFORME( 0, 1 )  ( 'U' por uniforme )
!>           'S' => UNIFORME( -1, 1 ) ( 'S' por simétrico )
!>           'N' => NORMAL( 0, 1 )   ( 'N' por normal )
!>           'D' => uniforme en el disco complejo |z| < 1.
!>           No modificado.
!>

ISEED

!>          ISEED es un array INTEGER, dimensión ( 4 )
!>           En la entrada, ISEED especifica la semilla del generador de números
!>           aleatorios. Deben estar entre 0 y 4095 inclusive,
!>           y ISEED(4) debe ser impar. El generador de números
!>           aleatorios utiliza una secuencia congruencial lineal limitada a
!>           enteros pequeños, y por lo tanto debe producir números
!>           aleatorios independientes de la máquina. Los valores de ISEED se
!>           cambian a la salida y pueden usarse en la siguiente llamada a ZLATME
!>           para continuar la misma secuencia de números aleatorios.
!>           Modificado a la salida.
!>

D

!>          D es un array COMPLEX*16, dimensión ( N )
!>           Este array se utiliza para especificar los valores propios de A. Si
!>           MODE=0, entonces se asume que D contiene los valores propios
!>           en caso contrario, se calcularán según MODE, COND,
!>           DMAX y RSIGN y se colocarán en D.
!>           Modificado si MODE es distinto de cero.
!>

MODE

!>          MODE es INTEGER
!>           En la entrada, esto describe cómo se especificarán los valores propios:
!>           MODE = 0 significa usar D como entrada
!>           MODE = 1 establece D(1)=1 y D(2:N)=1.0/COND
!>           MODE = 2 establece D(1:N-1)=1 y D(N)=1.0/COND
!>           MODE = 3 establece D(I)=COND**(-(I-1)/(N-1))
!>           MODE = 4 establece D(i)=1 - (i-1)/(N-1)*(1 - 1/COND)
!>           MODE = 5 establece D con números aleatorios en el rango
!>                    ( 1/COND , 1 ) tales que sus logaritmos
!>                    están distribuidos uniformemente.
!>           MODE = 6 establece D con números aleatorios de la misma distribución
!>                    que el resto de la matriz.
!>           MODE < 0 tiene el mismo significado que ABS(MODE), excepto que
!>              el orden de los elementos de D se invierte.
!>           Por lo tanto, si MODE está entre 1 y 4, D tiene entradas que van
!>              de 1 a 1/COND, si está entre -1 y -4, D tiene entradas
!>              que van de 1/COND a 1,
!>           No modificado.
!>

COND

!>          COND es DOUBLE PRECISION
!>           En la entrada, se utiliza como se describe en MODE arriba.
!>           Si se utiliza, debe ser >= 1. No modificado.
!>

DMAX

!>          DMAX es COMPLEX*16
!>           Si MODE no es -6, 0 ni 6, el contenido de D, tal como se
!>           calcula según MODE y COND, se escalará por
!>           DMAX / max(abs(D(i))). Tenga en cuenta que DMAX no necesita
!>           ser positivo o real: si DMAX es negativo o complejo (o cero),
!>           D se escalará por un número negativo o complejo (o cero).
!>           Si RSIGN='F', entonces el valor propio más grande (en valor absoluto) será
!>           igual a DMAX.
!>           No modificado.
!>

RSIGN

!>          RSIGN es CHARACTER*1
!>           Si MODE no es 0, 6 ni -6, y RSIGN='T', entonces los
!>           elementos de D, tal como se calculan según MODE y COND, se
!>           multiplicarán por un número complejo aleatorio del círculo unitario
!>           |z| = 1. Si RSIGN='F', no se multiplicarán. RSIGN solo puede
!>           tener los valores 'T' o 'F'.
!>           No modificado.
!>

UPPER

!>          UPPER es CHARACTER*1
!>           Si UPPER='T', entonces los elementos de A por encima de la diagonal
!>           se establecerán con números aleatorios de DIST. Si UPPER='F',
!>           no lo harán. UPPER solo puede tener los valores 'T' o 'F'.
!>           No modificado.
!>

SIM

!>          SIM es CHARACTER*1
!>           Si SIM='T', entonces A será operado por , es decir, multiplicado por la izquierda por una matriz X y
!>           por la derecha por X inversa. X = U S V, donde U y V son
!>           matrices unitarias aleatorias y S es una matriz (diagonal) de
!>           valores singulares especificados por DS, MODES y CONDS. Si
!>           SIM='F', entonces A no será transformado.
!>           No modificado.
!>

DS

!>          DS es un array DOUBLE PRECISION, dimensión ( N )
!>           Este array se utiliza para especificar los valores singulares de X,
!>           de la misma manera que D especifica los valores propios de A.
!>           Si MODE=0, DS contiene los valores singulares, que
!>           no pueden ser cero.
!>           Modificado si MODE es distinto de cero.
!>

MODES

!>          MODES es INTEGER
!>

CONDS

!>          CONDS es DOUBLE PRECISION
!>           Similar a MODE y COND, pero para especificar la diagonal
!>           de S. MODES=-6 y +6 no están permitidos (ya que resultarían
!>           en valores propios aleatoriamente mal condicionados).
!>

KL

!>          KL es INTEGER
!>           Esto especifica el ancho de banda inferior de la matriz. KL=1
!>           especifica la forma superior de Hessenberg. Si KL es al menos N-1,
!>           entonces A tendrá un ancho de banda inferior completo.
!>           No modificado.
!>

KU

!>          KU es INTEGER
!>           Esto especifica el ancho de banda superior de la matriz. KU=1
!>           especifica la forma inferior de Hessenberg. Si KU es al menos N-1,
!>           entonces A tendrá un ancho de banda superior completo; si KU y KL

zlatme.f(3) — LAPACK

NAME

zlatme — La subrutina ZLATME construye una matriz de prueba compleja para la reducción a la forma de Hessenberg.

SYNOPSIS

 SUBROUTINE ZLATME (
                M, N, DIST, DIAG, MODE, UPLO, NSEQ, SEQ,
                KL, KU, DL, DR, CL, CR, ANORM, A, LDA, WORK, INFO )
  CHARACTER*1   DIST, DIAG, MODE, UPLO, NSEQ, SEQ
  INTEGER       M, N, KL, KU, LDA, INFO
  COMPLEX*16    DL, DR, CL, CR
  DOUBLE PRECISION ANORM
  COMPLEX*16    A( LDA, N ), WORK( * )

Function/Subroutine Documentation

Esta subrutina es utilizada por el conjunto de pruebas de LAPACK para generar matrices de prueba complejas A de tamaño M por N.

Se pueden especificar las siguientes características de la matriz:

  • La distribución de los valores singulares o los valores propios (si es simétrica).
  • Si la matriz es diagonal, triangular superior, triangular inferior o general.
  • Si la matriz es simétrica o no.
  • Si la matriz es de banda o general.
  • El número de condicionalidad o el número de condicionalidad del subespacio de los valores propios o de los valores singulares.
  • La norma del elemento máximo.

No todas las combinaciones de parámetros de entrada son posibles. Por ejemplo, si la matriz es triangular, no se especifica su contraparte (superior o inferior). Si la matriz es de banda, entonces KL y KU especifican el número de sub y superdiagonales.

Cuando MODE=0, la matriz A se genera para tener un valor propio dado igual a DL. Si MODE=1, A se genera para tener un valor propio dado igual a DL y otro igual a DR. Si MODE=2, A se genera para tener un valor propio dado igual a DL y un subespacio propio con condicionalidad CL. Si MODE=3, A se genera para tener un subespacio propio con condicionalidad CL y un subespacio propio con condicionalidad CR. Si MODE=4, A se genera para tener un subespacio propio con condicionalidad CL y un subespacio propio con condicionalidad CR y un valor propio dado igual a DL.

Cuando MODE=5, A se genera para tener un valor singular dado igual a DL. Si MODE=6, A se genera para tener un valor singular dado igual a DL y otro igual a DR. Si MODE=7, A se genera para tener un valor singular dado igual a DL y un subespacio de valores singulares con condicionalidad CL.

Parámetros

M

!>          M is INTEGER
!>           The number of rows of A. M >= 0.
!>           Not modified.

N

!>          N is INTEGER
!>           The number of columns of A. N >= 0.
!>           Not modified.

DIST

!>          DIST is CHARACTER*1
!>           On entry, DIST specifies the type of distribution of the
!>           singular values or eigenvalues.
!>           DIST = 'U' =>uniformly distributed
!>           DIST = 'S' =>scaled diagonally dominant
!>           Not modified.

DIAG

!>          DIAG is CHARACTER*1
!>           On entry, DIAG specifies the type of diagonal degree.
!>           DIAG = 'N' => a normal degree
!>           DIAG = 'S' => a small degree
!>           DIAG = 'L' => a large degree
!>           DIAG = 'S' => a scaled diagonally dominant degree
!>           Not modified.

MODE

!>          MODE is INTEGER
!>           On entry, MODE specifies the type of matrix to be generated.
!>           Not modified.

UPLO

!>          UPLO is CHARACTER*1
!>           On entry, UPLO specifies whether A is upper or lower triangular.
!>           UPLO = 'U' => upper triangular
!>           UPLO = 'L' => lower triangular
!>           Not modified.

NSEQ

!>          NSEQ is INTEGER
!>           On entry, NSEQ specifies the number of sequences of conditions.
!>           Not modified.

SEQ

!>          SEQ is INTEGER array, dimension (NSEQ)
!>           On entry, SEQ specifies the type of sequence of conditions.
!>           Not modified.

KL

!>          KL is INTEGER
!>           On entry, KL is the number of required subnodiagonals in
!>           the matrix.
!>           Not modified.

KU

!>          KU is INTEGER
!>           On entry, KU is the number of required supernodiagonals in
!>           the matrix.
!>           Not modified.

DL

!>          DL is COMPLEX*16
!>           On entry, DL is the value of the first diagonal element or
!>           eigenvalue or singular value.
!>           Not modified.

DR

!>          DR is COMPLEX*16
!>           On entry, DR is the value of the second diagonal element or
!>           eigenvalue or singular value.
!>           Not modified.

CL

!>          CL is COMPLEX*16
!>           On entry, CL is the condition number of the first subpace.
!>           Not modified.

CR

!>          CR is COMPLEX*16
!>           On entry, CR is the condition number of the second subpace.
!>           Not modified.

ANORM

!>          ANORM is DOUBLE PRECISION
!>           If ANORM is not negative, then A will be scaled by a non-
!>           negative real number to make the maximum-element-norm of A
!>           to be ANORM.
!>           Not modified.

A

!>          A is COMPLEX*16 array, dimension ( LDA, N )
!>           On exit A is the desired test matrix.
!>           Modified.

LDA

!>          LDA is INTEGER
!>           LDA specifies the first dimension of A as declared in the
!>           calling program.  LDA must be at least M.
!>           Not modified.

WORK

!>          WORK is COMPLEX*16 array, dimension ( 3*N )
!>           Workspace.
!>           Modified.

INFO

!>          INFO is INTEGER
!>           Error code.  On exit, INFO will be set to one of the
!>           following values:
!>             0 => normal return
!>            -1 => N negative
!>            -2 => DIST illegal string
!>            -5 => MODE not in range -6 to 6
!>            -6 => COND less than 1.0, and MODE neither -6, 0 nor 6
!>            -9 => RSIGN is not 'T' or 'F'
!>           -10 => UPPER is not 'T' or 'F'
!>           -11 => SIM   is not 'T' or 'F'
!>           -12 => MODES=0 and DS has a zero singular value.
!>           -13 => MODES is not in the range -5 to 5.
!>           -14 => MODES is nonzero and CONDS is less than 1.
!>           -15 => KL is less than 1.
!>           -16 => KU is less than 1, or KL and KU are both less than
!>                  N-1.
!>           -19 => LDA is less than M.
!>            1  => Error return from ZLATM1 (computing D)
!>            2  => Cannot scale to DMAX (max. eigenvalue is 0)
!>            3  => Error return from DLATM1 (computing DS)
!>            4  => Error return from ZLARGE
!>            5  => Zero singular value from DLATM1.
