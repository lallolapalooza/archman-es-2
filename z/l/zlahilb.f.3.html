subroutine zlahilb (integer n, integer nrhs, complex*16, dimension(lda,n) a, integer lda, complex*16, dimension(ldx, nrhs) x, integer ldx, complex*16, dimension(ldb, nrhs) b, integer ldb, double precision, dimension(n) work, integer info, character*3 path)

ZLAHILB

Propósito:

!>
!> ZLAHILB genera una matriz de Hilbert escalada N por N en A junto con
!> NRHS lados derechos en B y soluciones en X tales que A*X=B.
!>
!> La matriz de Hilbert se escala por M = MCM(1, 2, ..., 2*N-1) para que todas
!> las entradas sean enteros. Los lados derechos son las primeras NRHS
!> columnas de la matriz identidad M, y las soluciones son las
!> primeras NRHS columnas de la inversa de la matriz de Hilbert.
!>
!> El número de condición de la matriz de Hilbert crece exponencialmente con
!> su tamaño, aproximadamente como O(e ** (3.5*N)). Adicionalmente, las inversas
!> de matrices de Hilbert más allá de una dimensión relativamente pequeña no pueden
!> generarse exactamente sin precisión adicional. La precisión se agota
!> cuando la entrada más grande en la matriz inversa de Hilbert es mayor que
!> 2 elevado a la potencia del número de bits en la fracción del tipo de dato
!> utilizado más uno, que es 24 para precisión simple.
!>
!> En precisión simple, la solución generada es exacta para N <= 6 y tiene
!> un error componente a componente pequeño para 7 <= N <= 11.
!>

Parámetros

N 
!>          N es ENTERO
!>          La dimensión de la matriz A.
!>

NRHS

!>          NRHS es ENTERO
!>          El número solicitado de lados derechos.
!>

A

!>          A es un arreglo COMPLEX, dimensión (LDA, N)
!>          La matriz de Hilbert escalada generada.
!>

LDA

!>          LDA es ENTERO
!>          La dimensión principal del arreglo A.  LDA >= N.
!>

X

!>          X es un arreglo COMPLEX, dimensión (LDX, NRHS)
!>          Las soluciones exactas generadas.  Actualmente, las primeras NRHS
!>          columnas de la inversa de la matriz de Hilbert.
!>

LDX

!>          LDX es ENTERO
!>          La dimensión principal del arreglo X.  LDX >= N.
!>

B

!>          B es un arreglo REAL, dimensión (LDB, NRHS)
!>          Los lados derechos generados.  Actualmente, las primeras NRHS
!>          columnas de MCM(1, 2, ..., 2*N-1) * la matriz identidad.
!>

LDB

!>          LDB es ENTERO
!>          La dimensión principal del arreglo B.  LDB >= N.
!>

WORK

!>          WORK es un arreglo REAL, dimensión (N)
!>

INFO

!>          INFO es ENTERO
!>          = 0: salida exitosa
!>          = 1: N es demasiado grande; los datos todavía se generan pero puede que no
!>               sean exactos.
!>          < 0: si INFO = -i, el i-ésimo argumento tenía un valor ilegal
!>

PATH

!>          PATH es CHARACTER*3
!>          El nombre de la ruta LAPACK.
!>
