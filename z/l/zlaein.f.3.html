subroutine zlaein (logical rightv, logical noinit, integer n, complex*16, dimension( ldh, * ) h, integer ldh, complex*16 w, complex*16, dimension( * ) v, complex*16, dimension( ldb, * ) b, integer ldb, double precision, dimension( * ) rwork, double precision eps3, double precision smlnum, integer info)

ZLAEIN calcula un vector propio derecho o izquierdo especificado de una matriz de Hessenberg superior mediante iteración inversa.

Propósito

!>
!> ZLAEIN utiliza la iteración inversa para encontrar un vector propio
!> derecho o izquierdo correspondiente al valor propio W de una matriz
!> de Hessenberg superior compleja H.
!>

Parámetros

RIGHTV 
!>          RIGHTV es LOGICAL
!>          = .TRUE. : calcular vector propio derecho;
!>          = .FALSE.: calcular vector propio izquierdo.
!>

NOINIT

!>          NOINIT es LOGICAL
!>          = .TRUE. : no se suministró vector inicial en V
!>          = .FALSE.: se suministró vector inicial en V.
!>

N

!>          N es INTEGER
!>          El orden de la matriz H.  N >= 0.
!>

H

!>          H es array COMPLEX*16, dimension (LDH,N)
!>          La matriz de Hessenberg superior H.
!>

LDH

!>          LDH es INTEGER
!>          La dimensión principal del array H.  LDH >= max(1,N).
!>

W

!>          W es COMPLEX*16
!>          El valor propio de H cuyo vector propio derecho o izquierdo
!>          correspondiente se va a calcular.
!>

V

!>          V es array COMPLEX*16, dimension (N)
!>          A la entrada, si NOINIT = .FALSE., V debe contener un vector
!>          de inicio para la iteración inversa; de lo contrario, V no
!>          necesita ser establecido.
!>          A la salida, V contiene el vector propio calculado, normalizado
!>          de modo que el componente de mayor magnitud tenga magnitud 1;
!>          aquí, la magnitud de un número complejo (x,y) se toma como
!>          |x| + |y|.
!>

B

!>          B es array COMPLEX*16, dimension (LDB,N)
!>

LDB

!>          LDB es INTEGER
!>          La dimensión principal del array B.  LDB >= max(1,N).
!>

RWORK

!>          RWORK es array DOUBLE PRECISION, dimension (N)
!>

EPS3

!>          EPS3 es DOUBLE PRECISION
!>          Un pequeño valor dependiente de la máquina que se utiliza para
!>          perturbar valores propios cercanos y para reemplazar pivotes
!>          cero.
!>

SMLNUM

!>          SMLNUM es DOUBLE PRECISION
!>          Un valor dependiente de la máquina cercano al umbral de
!>          subdesbordamiento.
!>

INFO

!>          INFO es INTEGER
!>          = 0:  salida exitosa
!>          = 1:  la iteración inversa no convergió; V se establece en el
!>                último iterado.
!>
