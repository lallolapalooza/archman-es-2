subroutine zlatsy (character uplo, integer n, complex*16, dimension( ldx, * ) x, integer ldx, integer, dimension( * ) iseed)

ZLATSY

Propósito:

!>
!> ZLATSY genera una matriz de prueba especial para la factorización compleja simétrica
!> (indefinida). Los bloques pivote de la matriz generada estarán en el siguiente orden:
!>    Bloque pivote de 2x2, no diagonalizable
!>    Bloque pivote de 1x1
!>    Bloque pivote de 2x2, diagonalizable
!>    (el ciclo se repite)
!> Se requiere un intercambio de filas para cada bloque de 2x2 no diagonalizable.
!>

Parámetros

UPLO 
!>          UPLO es CHARACTER
!>          Especifica si la matriz generada será triangular superior o
!>          inferior.
!>          = 'U':  Triangular superior
!>          = 'L':  Triangular inferior
!>

N

!>          N es INTEGER
!>          La dimensión de la matriz a generar.
!>

X

!>          X es una matriz COMPLEX*16, dimensión (LDX,N)
!>          La matriz generada, que consta de bloques diagonales de 3x3 y 2x2
!>          que dan como resultado la secuencia de pivotes mencionada anteriormente.
!>          La matriz fuera de estos bloques diagonales es cero.
!>

LDX

!>          LDX es INTEGER
!>          La dimensión principal del arreglo X.
!>

ISEED

!>          ISEED es un arreglo INTEGER, dimensión (4)
!>          En la entrada, la semilla para el generador de números aleatorios. El último
!>          de los cuatro enteros debe ser impar. (modificado en la salida)
!>
