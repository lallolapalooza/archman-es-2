subroutine zlamswlq (character side, character trans, integer m, integer n, integer k, integer mb, integer nb, complex*16, dimension( lda, * ) a, integer lda, complex*16, dimension( ldt, * ) t, integer ldt, complex*16, dimension( ldc, * ) c, integer ldc, complex*16, dimension( * ) work, integer lwork, integer info)

ZLAMSWLQ

Propósito:

!>
!>    ZLAMSWLQ sobrescribe la matriz compleja general M por N, C, con
!>
!>
!>                    SIDE = 'L'     SIDE = 'R'
!>    TRANS = 'N':      Q * C          C * Q
!>    TRANS = 'C':      Q**H * C       C * Q**H
!>    donde Q es una matriz unitaria compleja definida como el producto de
!>    reflectores elementales bloqueados calculados por la factorización LQ
!>    de ancho corto (ZLASWLQ)
!>

Parámetros

SIDE 
!>          SIDE es CHARACTER*1
!>          = 'L': aplicar Q o Q**H desde la Izquierda;
!>          = 'R': aplicar Q o Q**H desde la Derecha.
!>

TRANS

!>          TRANS es CHARACTER*1
!>          = 'N': Sin transponer, aplicar Q;
!>          = 'C': Transpuesta conjugada, aplicar Q**H.
!>

M

!>          M es INTEGER
!>          El número de filas de la matriz C.  M >=0.
!>

N

!>          N es INTEGER
!>          El número de columnas de la matriz C. N >= 0.
!>

K

!>          K es INTEGER
!>          El número de reflectores elementales cuyo producto define
!>          la matriz Q.
!>          M >= K >= 0;
!>
!>

MB

!>          MB es INTEGER
!>          El tamaño del bloque de filas que se utilizará en la LQ bloqueada.
!>          M >= MB >= 1
!>

NB

!>          NB es INTEGER
!>          El tamaño del bloque de columnas que se utilizará en la LQ bloqueada.
!>          NB > M.
!>

A

!>          A es COMPLEX*16 array, dimension
!>                               (LDA,M) si SIDE = 'L',
!>                               (LDA,N) si SIDE = 'R'
!>          La i-ésima fila debe contener el vector que define el reflector
!>          bloqueado H(i), para i = 1,2,...,k, tal como se devuelve por
!>          ZLASWLQ en las primeras k filas de su argumento de array A.
!>

LDA

!>          LDA es INTEGER
!>          La dimensión principal del array A. LDA >= MAX(1,K).
!>

T

!>          T es COMPLEX*16 array, dimension
!>          ( M * Number of blocks(CEIL(N-K/NB-K)),
!>          Los reflectores de bloque triangulares superiores bloqueados
!>          almacenados en forma compacta como una secuencia de bloques
!>          triangulares superiores. Consulte a continuación para más
!>          detalles.
!>

LDT

!>          LDT es INTEGER
!>          La dimensión principal del array T. LDT >= MB.
!>

C

!>          C es COMPLEX*16 array, dimension (LDC,N)
!>          En la entrada, la matriz C de M por N.
!>          En la salida, C se sobrescribe con Q*C o Q**H*C o C*Q**H o C*Q.
!>

LDC

!>          LDC es INTEGER
!>          La dimensión principal del array C. LDC >= max(1,M).
!>

WORK

!>          (espacio de trabajo) COMPLEX*16 array, dimension (MAX(1,LWORK))
!>          En la salida, si INFO = 0, WORK(1) devuelve el LWORK mínimo.
!>

LWORK

!>          LWORK es INTEGER
!>          La dimensión del array WORK.
!>          Si MIN(M,N,K) = 0, LWORK >= 1.
!>          Si SIDE = 'L', LWORK >= max(1,NB*MB).
!>          Si SIDE = 'R', LWORK >= max(1,M*MB).
!>
!>          Si LWORK = -1, se asume una consulta de espacio de trabajo; la rutina
!>          solo calcula el tamaño mínimo del array WORK, devuelve
!>          este valor como la primera entrada del array WORK, y XERBLA no
!>          emite ningún mensaje de error relacionado con LWORK.
!>

INFO

!>          INFO es INTEGER
!>          = 0:  salida exitosa
!>          < 0:  si INFO = -i, el i-ésimo argumento tenía un valor ilegal
!>

Detalles adicionales:

!> LQ de Ancho Corto (SWLQ) realiza LQ mediante una secuencia de transformaciones
!> unitarias, representando Q como un producto de otras matrices unitarias
!>   Q = Q(1) * Q(2) * . . . * Q(k)
!> donde cada Q(i) pone a cero las entradas de la diagonal superior de un bloque
!> de NB filas de A:
!>   Q(1) pone a cero las entradas de la diagonal superior de las filas 1:NB de A
!>   Q(2) pone a cero las filas inferiores MB-N de las filas [1:M,NB+1:2*NB-M] de A
!>   Q(3) pone a cero las filas inferiores MB-N de las filas [1:M,2*NB-M+1:3*NB-2*M] de A
!>   . . .
!>
!> Q(1) se calcula con GELQT, que representa Q(1) mediante vectores de Householder
!> almacenados debajo de la diagonal de las filas 1:MB de A, y mediante
!> reflectores de bloque triangulares superiores, almacenados en el array
!> T(1:LDT,1:N).
!> Para más información, consulte Detalles adicionales en GELQT.
!>
!> Q(i) para i>1 se calcula con TPLQT, que representa Q(i) mediante vectores de
!> Householder almacenados en las columnas [(i-1)*(NB-M)+M+1:i*(NB-M)+M] de A,
!> y mediante reflectores de bloque triangulares superiores, almacenados en el
!> array T(1:LDT,(i-1)*M+1:i*M).
!> La última Q(k) puede usar menos filas.
!> Para más información, consulte Detalles adicionales en TPLQT.
!>
!> Para más detalles sobre el algoritmo general, consulte la descripción de
!> TSQR Secuencial en la Sección 2.2 de [1].
!>
!> [1] “Communication-Optimal Parallel and Sequential QR and LU Factorizations,”
!>     J. Demmel, L. Grigori, M. Hoemmen, J. Langou,
!>     SIAM J. Sci. Comput, vol. 34, no. 1, 2012
!>
