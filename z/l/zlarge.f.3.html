subroutine zlarge (integer n, complex*16, dimension( lda, * ) a, integer lda, integer, dimension( 4 ) iseed, complex*16, dimension( * ) work, integer info)

ZLARGE

Propósito:

!>
!> ZLARGE pre- y post-multiplica una matriz compleja general n por n A
!> con una matriz unitaria aleatoria: A = U*D*U'.
!>

Parámetros

N 
!>          N es INTEGER
!>          El orden de la matriz A.  N >= 0.
!>

A

!>          A es array COMPLEX*16, dimension (LDA,N)
!>          En la entrada, la matriz original A de n por n.
!>          En la salida, A se sobrescribe con U*A*U' para alguna
!>          matriz unitaria aleatoria U.
!>

LDA

!>          LDA es INTEGER
!>          La dimensión principal del array A.  LDA >= N.
!>

ISEED

!>          ISEED es array INTEGER, dimension (4)
!>          En la entrada, la semilla del generador de números aleatorios; los elementos
!>          del array deben estar entre 0 y 4095, y ISEED(4) debe ser
!>          impar.
!>          En la salida, la semilla se actualiza.
!>

WORK

!>          WORK es array COMPLEX*16, dimension (2*N)
!>

INFO

!>          INFO es INTEGER
!>          = 0: salida exitosa
!>          < 0: si INFO = -i, el i-ésimo argumento tenía un valor ilegal
!>
