subroutine zla_gerfsx_extended (integer prec_type, integer trans_type, integer n, integer nrhs, complex*16, dimension( lda, * ) a, integer lda, complex*16, dimension( ldaf, * ) af, integer ldaf, integer, dimension( * ) ipiv, logical colequ, double precision, dimension( * ) c, complex*16, dimension( ldb, * ) b, integer ldb, complex*16, dimension( ldy, * ) y, integer ldy, double precision, dimension( * ) berr_out, integer n_norms, double precision, dimension( nrhs, * ) errs_n, double precision, dimension( nrhs, * ) errs_c, complex*16, dimension( * ) res, double precision, dimension( * ) ayb, complex*16, dimension( * ) dy, complex*16, dimension( * ) y_tail, double precision rcond, integer ithresh, double precision rthresh, double precision dz_ub, logical ignore_cwise, integer info)

ZLA_GERFSX_EXTENDED

Propósito:

!>
!> ZLA_GERFSX_EXTENDED mejora la solución calculada a un sistema de
!> ecuaciones lineales realizando un refinamiento iterativo de precisión extra
!> y proporciona cotas de error y estimaciones de error hacia atrás para la solución.
!> Esta subrutina es llamada por ZGERFSX para realizar el refinamiento iterativo.
!> Además de la cota de error "normwise", el código proporciona una cota de error
!> máxima "componentwise" si es posible. Ver los comentarios para ERRS_N
!> y ERRS_C para detalles de las cotas de error. Tenga en cuenta que esta
!> subrutina solo es responsable de establecer los segundos campos de
!> ERRS_N y ERRS_C.
!>

Parámetros

PREC_TYPE 
!>          PREC_TYPE es INTEGER
!>     Especifica la precisión intermedia a utilizar en el refinamiento.
!>     El valor se define por ILAPREC(P) donde P es un CARACTER y P
!>          = 'S':  Simple (Single)
!>          = 'D':  Doble (Double)
!>          = 'I':  Indígena (Indigenous)
!>          = 'X' o 'E':  Extra (Extended)
!>

TRANS_TYPE

!>          TRANS_TYPE es INTEGER
!>     Especifica la operación de transposición sobre A.
!>     El valor se define por ILATRANS(T) donde T es un CARACTER y T
!>          = 'N':  Sin transposición
!>          = 'T':  Transpuesta
!>          = 'C':  Transpuesta conjugada
!>

N

!>          N es INTEGER
!>     El número de ecuaciones lineales, es decir, el orden de la
!>     matriz A.  N >= 0.
!>

NRHS

!>          NRHS es INTEGER
!>     El número de lados derechos, es decir, el número de columnas de la
!>     matriz B.
!>

A

!>          A es array COMPLEX*16, dimensión (LDA,N)
!>     A la entrada, la matriz N-por-N A.
!>

LDA

!>          LDA es INTEGER
!>     La dimensión principal del array A.  LDA >= max(1,N).
!>

AF

!>          AF es array COMPLEX*16, dimensión (LDAF,N)
!>     Los factores L y U de la factorización
!>     A = P*L*U calculada por ZGETRF.
!>

LDAF

!>          LDAF es INTEGER
!>     La dimensión principal del array AF.  LDAF >= max(1,N).
!>

IPIV

!>          IPIV es array INTEGER, dimensión (N)
!>     Los índices de pivote de la factorización A = P*L*U
!>     calculada por ZGETRF; la fila i de la matriz se intercambió
!>     con la fila IPIV(i).
!>

COLEQU

!>          COLEQU es LOGICAL
!>     Si es .TRUE., se realizó la   equilibración de columnas a A antes de llamar
!>     a esta rutina. Esto es necesario para calcular la solución y las cotas de error
!>     correctamente.
!>

C

!>          C es array DOUBLE PRECISION, dimensión (N)
!>     Los factores de escala de columna para A. Si COLEQU = .FALSE., C
!>     no se accede. Si C es de entrada, cada elemento de C debe ser una potencia
!>     de la base para asegurar una solución y estimaciones de error fiables.
!>     La escala por potencias de la base no causa errores de redondeo a menos
!>     que el resultado cause desbordamiento inferior (underflow) o superior (overflow).
!>     Los errores de redondeo durante la escala
!>     conducen a refinar con una matriz que no es equivalente a la
!>     matriz de entrada, produciendo estimaciones de error que pueden no ser
!>     fiables.
!>

B

!>          B es array COMPLEX*16, dimensión (LDB,NRHS)
!>     La matriz del lado derecho B.
!>

LDB

!>          LDB es INTEGER
!>     La dimensión principal del array B.  LDB >= max(1,N).
!>

Y

!>          Y es array COMPLEX*16, dimensión (LDY,NRHS)
!>     A la entrada, la matriz solución X, calculada por ZGETRS.
!>     A la salida, la matriz solución mejorada Y.
!>

LDY

!>          LDY es INTEGER
!>     La dimensión principal del array Y.  LDY >= max(1,N).
!>

BERR_OUT

!>          BERR_OUT es array DOUBLE PRECISION, dimensión (NRHS)
!>     A la salida, BERR_OUT(j) contiene el error relativo hacia atrás "componentwise"
!>     para el lado derecho j de la fórmula
!>         max(i) ( abs(RES(i)) / ( abs(op(A_s))*abs(Y) + abs(B_s) )(i) )
!>     donde abs(Z) es el valor absoluto "componentwise" de la matriz
!>     o vector Z. Esto se calcula mediante ZLA_LIN_BERR.
!>

N_NORMS

!>          N_NORMS es INTEGER
!>     Determina qué cotas de error se devuelven (ver ERRS_N
!>     y ERRS_C).
!>     Si N_NORMS >= 1, se devuelven cotas de error "normwise".
!>     Si N_NORMS >= 2, se devuelven cotas de error "componentwise".
!>

ERRS_N

!>          ERRS_N es array DOUBLE PRECISION, dimensión (NRHS, N_ERR_BNDS)
!>     Para cada lado derecho, esta matriz contiene información sobre
!>     varias cotas de error y números de condición correspondientes al
!>     error relativo "normwise", que se define de la siguiente manera:
!>
!>     Error relativo "normwise" en el i-ésimo vector solución:
!>             max_j (abs(XTRUE(j,i) - X(j,i)))
!>            ------------------------------
!>                  max_j abs(X(j,i))
!>
!>     La matriz se indexa por el tipo de información de error según se describe
!>     a continuación. Actualmente hay hasta tres piezas de información
!>     devueltas.
!>
!>     El primer índice en ERRS_N(i,:) corresponde al i-ésimo
!>     lado derecho.
!>
!>     El segundo índice en ERRS_N(:,err) contiene los siguientes
!>     tres campos:
!>     err = 1  booleano. Confiar en la respuesta si el
!>              número de condición recíproco es menor que el umbral
!>              sqrt(n) * slamch('Epsilon').
!>
!>     err = 2  cota de error: El error de avance estimado,
!>              casi con seguridad dentro de un factor de 10 del error verdadero
!>              siempre que la siguiente entrada sea mayor que el umbral
!>              sqrt(n) * slamch('Epsilon'). Esta cota de error solo debe
!>              confiarse si el booleano anterior es verdadero.
!>
!>
err = 3 Número de condición recíproco: Estimado número de condición !> recíproco norma a norma. Comparado con el umbral !> sqrt(n) * slamch('Epsilon') para determinar si la estimación !> del error es . Estos números de condición recíprocos !> son 1 / (norm(Z^{-1},inf) * norm(Z,inf)) para alguna !> matriz Z apropiadamente escalada. !> Sea Z = S*A, donde S escala cada fila por una potencia de la !> base para que todas las sumas de filas absolutas de Z sean !> aproximadamente 1. !> !> Esta subrutina solo es responsable de establecer el segundo campo !> anterior. !> Consulte la Nota de Trabajo 165 de Lapack para más detalles y !> precauciones adicionales. !>

ERRS_C

!>          ERRS_C es un array de DOBLE PRECISIÓN, de dimensión (NRHS, N_ERR_BNDS)
!>     Para cada lado derecho, este array contiene información sobre
!>     varios límites de error y números de condición correspondientes al
!>     error relativo componente a componente, que se define como sigue:
!>
!>     Error relativo componente a componente en el i-ésimo vector solución:
!>                    abs(XTRUE(j,i) - X(j,i))
!>             max_j ----------------------
!>                         abs(X(j,i))
!>
!>     El array se indexa por el lado derecho i (del que depende el
!>     error relativo componente a componente), y por el tipo de información
!>     de error, como se describe a continuación. Actualmente hay hasta tres
!>     piezas de información devueltas para cada lado derecho. Si
!>     no se solicita precisión componente a componente (PARAMS(3) = 0.0),
!>     entonces ERRS_C no se accede. Si N_ERR_BNDS < 3, entonces se devuelven
!>     como máximo las primeras entradas (:,N_ERR_BNDS).
!>
!>     El primer índice en ERRS_C(i,:) corresponde al i-ésimo
!>     lado derecho.
!>
!>     El segundo índice en ERRS_C(:,err) contiene los siguientes
!>     tres campos:
!>     err = 1  booleano. Confía en la respuesta si el
!>              número de condición recíproco es menor que el umbral
!>              sqrt(n) * slamch('Epsilon').
!>
!>     err = 2  límite de error: El error directo estimado,
!>              casi con toda seguridad dentro de un factor de 10 del error
!>              verdadero siempre que la siguiente entrada sea mayor que el umbral
!>              sqrt(n) * slamch('Epsilon'). Este límite de error solo debe
!>              confiarse si el booleano anterior es verdadero.
!>
!>     err = 3  Número de condición recíproco: Número de condición
!>              recíproco componente a componente estimado. Comparado con el umbral
!>              sqrt(n) * slamch('Epsilon') para determinar si la estimación
!>              del error es . Estos números de condición
!>              recíprocos son 1 / (norm(Z^{-1},inf) * norm(Z,inf)) para alguna
!>              matriz Z apropiadamente escalada.
!>              Sea Z = S*(A*diag(x)), donde x es la solución para el
!>              lado derecho actual y S escala cada fila de
!>              A*diag(x) por una potencia de la base para que todas las sumas
!>              de filas absolutas de Z sean aproximadamente 1.
!>
!>     Esta subrutina solo es responsable de establecer el segundo campo
!>     anterior.
!>     Consulte la Nota de Trabajo 165 de Lapack para más detalles y
!>     precauciones adicionales.
!>

RES

!>          RES es un array de COMPLEX*16, de dimensión (N)
!>     Espacio de trabajo para contener el residuo intermedio.
!>

AYB

!>          AYB es un array de DOBLE PRECISIÓN, de dimensión (N)
!>     Espacio de trabajo.
!>

DY

!>          DY es un array de COMPLEX*16, de dimensión (N)
!>     Espacio de trabajo para contener la solución intermedia.
!>

Y_TAIL

!>          Y_TAIL es un array de COMPLEX*16, de dimensión (N)
!>     Espacio de trabajo para contener los bits finales de la solución intermedia.
!>

RCOND

!>          RCOND es DOBLE PRECISIÓN
!>     Número de condición escalado recíproco. Esta es una estimación del
!>     número de condición de Skeel recíproco de la matriz A después de
!>     la  equilibración (si se realizó). Si es menor que la precisión de la
!>     máquina (en particular, si es cero), la matriz es singular
!>     a la precisión de trabajo. Tenga en cuenta que el error aún puede ser pequeño incluso
!>     si este número es muy pequeño y la matriz parece mal
!>     condicionada.
!>

ITHRESH

!>          ITHRESH es INTEGER
!>     El número máximo de cómputos de residuo permitidos para el
!>     refinamiento. El valor predeterminado es 10. Para 'agresivo',
!>     establecerlo en 100 para permitir la convergencia utilizando
!>     factorizaciones aproximadas o factorizaciones distintas de LU. Si la factorización
!>     utiliza una técnica distinta de la eliminación gaussiana, las garantías en
!>     ERRS_N y ERRS_C pueden dejar de ser confiables.
!>

RTHRESH

!>          RTHRESH es DOBLE PRECISIÓN
!>     Determina cuándo detener el refinamiento si la estimación del error deja de
!>     disminuir. El refinamiento se detendrá cuando la siguiente solución ya no
!>     satisfaga norm(dx_{i+1}) < RTHRESH * norm(dx_i), donde norm(Z) es
!>     la norma del infinito de Z. RTHRESH satisface 0 < RTHRESH <= 1. El
!>     valor predeterminado es 0.5. Para 'agresivo', establecerlo en 0.9 para permitir
!>     la convergencia en matrices extremadamente mal condicionadas. Consulte LAWN 165
!>     para más detalles.
!>

DZ_UB

!>          DZ_UB es DOBLE PRECISIÓN
!>     Determina cuándo se debe comenzar a considerar la convergencia componente a
!>     componente. La convergencia componente a componente solo se considera después de que cada
!>     componente de la solución Y sea estable, lo que definimos como que el cambio relativo
!>     en cada componente sea menor que DZ_UB. El valor predeterminado
!>     es 0.25, requiriendo que el primer bit sea estable. Consulte LAWN 165 para
!>     más detalles.
!>

IGNORE_CWISE

!>          IGNORE_CWISE es LOGICAL
!>     Si .TRUE. entonces ignora la convergencia componente a componente. El valor
!>     predeterminado es .FALSE..
!>

INFO

!>          INFO es INTEGER
!>       = 0: Salida exitosa.
!>       < 0: Si INFO = -i, el i-ésimo argumento de ZGETRS tuvo un valor
!>             ilegal.
!>
