subroutine zlaqhb (character uplo, integer n, integer kd, complex*16, dimension( ldab, * ) ab, integer ldab, double precision, dimension( * ) s, double precision scond, double precision amax, character equed)

ZLAQHB escala una matriz de banda Hermitiana, utilizando factores de escala calculados por cpbequ.

Propósito:

!>
!> ZLAQHB equilibra una matriz de banda Hermitiana A
!> utilizando los factores de escala en el vector S.
!>

Parámetros

UPLO 
!>          UPLO es CHARACTER*1
!>          Especifica si la parte triangular superior o inferior de la
!>          matriz simétrica A está almacenada.
!>          = 'U':  Triangular superior
!>          = 'L':  Triangular inferior
!>

N

!>          N es INTEGER
!>          El orden de la matriz A.  N >= 0.
!>

KD

!>          KD es INTEGER
!>          El número de super-diagonales de la matriz A si UPLO = 'U',
!>          o el número de sub-diagonales si UPLO = 'L'.  KD >= 0.
!>

AB

!>          AB es un array COMPLEX*16, dimensionado (LDAB,N)
!>          En la entrada, la parte triangular superior o inferior de la
!>          matriz de banda simétrica A, almacenada en las primeras KD+1
!>          filas del array. La columna j-ésima de A se almacena en la
!>          columna j-ésima del array AB como sigue:
!>          si UPLO = 'U', AB(kd+1+i-j,j) = A(i,j) para max(1,j-kd)<=i<=j;
!>          si UPLO = 'L', AB(1+i-j,j)    = A(i,j) para j<=i<=min(n,j+kd).
!>
!>          En la salida, si INFO = 0, el factor triangular U o L de la
!>          factorización de Cholesky A = U**H *U o A = L*L**H de la matriz
!>          de banda A, en el mismo formato de almacenamiento que A.
!>

LDAB

!>          LDAB es INTEGER
!>          La dimensión principal del array AB.  LDAB >= KD+1.
!>

S

!>          S es un array DOUBLE PRECISION, dimensionado (N)
!>          Los factores de escala para A.
!>

SCOND

!>          SCOND es DOUBLE PRECISION
!>          Relación del S(i) más pequeño con el S(i) más grande.
!>

AMAX

!>          AMAX es DOUBLE PRECISION
!>          Valor absoluto del elemento de matriz más grande.
!>

EQUED

!>          EQUED es CHARACTER*1
!>          Especifica si se realizó o no la equilibración.
!>          = 'N':  No se realizó la equilibración.
!>          = 'Y':  Se realizó la equilibración, es decir, A ha sido
!>                  reemplazado por diag(S) * A * diag(S).
!>

Parámetros internos:

!>  THRESH es un valor umbral utilizado para decidir si se debe escalar
!>  basado en la relación de los factores de escala. Si SCOND < THRESH,
!>  se realiza el escalado.
!>
!>  LARGE y SMALL son valores umbral utilizados para decidir si se debe
!>  escalar basado en el tamaño absoluto del elemento de matriz más
!>  grande. Si AMAX > LARGE o AMAX < SMALL, se realiza el escalado.
!>
