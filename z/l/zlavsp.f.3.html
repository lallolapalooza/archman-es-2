subroutine zlavsp (character uplo, character trans, character diag, integer n, integer nrhs, complex*16, dimension( * ) a, integer, dimension( * ) ipiv, complex*16, dimension( ldb, * ) b, integer ldb, integer info)

ZLAVSP

Propósito:

!>
!>    ZLAVSP  realiza una de las operaciones de matriz-vector
!>       x := A*x  o  x := A^T*x,
!>    donde x es un vector de N elementos y  A es uno de los factores
!>    de la factorización simétrica calculada por ZSPTRF.
!>    ZSPTRF produce una factorización de la forma
!>         U * D * U^T     o     L * D * L^T,
!>    donde U (o L) es un producto de permutación y matrices triangulares
!>    unitarias superiores (inferiores), U^T (o L^T) es la transpuesta de
!>    U (o L), y D es simétrica y diagonal por bloques con bloques
!>    diagonales de 1x1 y 2x2. Los multiplicadores para las transformaciones
!>    y las partes triangulares superiores o inferiores de los bloques
!>    diagonales se almacenan por columnas en formato empaquetado en el
!>    array lineal A.
!>
!>    Si TRANS = 'N' o 'n', ZLAVSP multiplica por U o U * D
!>    (o L o L * D).
!>    Si TRANS = 'C' o 'c', ZLAVSP multiplica por U^T o D * U^T
!>    (o L^T o D * L^T ).
!> 
!>  UPLO   - CHARACTER*1
!>           En la entrada, UPLO especifica si la matriz triangular
!>           almacenada en A es superior o inferior triangular.
!>              UPLO = 'U' o 'u'   La matriz es triangular superior.
!>              UPLO = 'L' o 'l'   La matriz es triangular inferior.
!>           Sin cambios a la salida.
!>
!>  TRANS  - CHARACTER*1
!>           En la entrada, TRANS especifica la operación a realizar como
!>           sigue:
!>              TRANS = 'N' o 'n'   x := A*x.
!>              TRANS = 'T' o 't'   x := A^T*x.
!>           Sin cambios a la salida.
!>
!>  DIAG   - CHARACTER*1
!>           En la entrada, DIAG especifica si se asume que los bloques
!>           diagonales son matrices unitarias, como sigue:
!>              DIAG = 'U' o 'u'   Los bloques diagonales son matrices unitarias.
!>              DIAG = 'N' o 'n'   Los bloques diagonales no son unitarios.
!>           Sin cambios a la salida.
!>
!>  N      - INTEGER
!>           En la entrada, N especifica el orden de la matriz A.
!>           N debe ser mayor o igual que cero.
!>           Sin cambios a la salida.
!>
!>  NRHS   - INTEGER
!>           En la entrada, NRHS especifica el número de lados derechos,
!>           es decir, el número de vectores x a multiplicar por A.
!>           NRHS debe ser mayor o igual que cero.
!>           Sin cambios a la salida.
!>
!>  A      - Arreglo COMPLEX*16, dimension( N*(N+1)/2 )
!>           En la entrada, A contiene una matriz diagonal por bloques y los
!>           multiplicadores de las transformaciones usadas para obtenerla,
!>           almacenados como una matriz triangular empaquetada.
!>           Sin cambios a la salida.
!>
!>  IPIV   - Arreglo INTEGER, dimension( N )
!>           En la entrada, IPIV contiene el vector de índices de pivote
!>           tal como lo determinó ZSPTRF.
!>           Si IPIV( K ) = K, no se realizó ningún intercambio.
!>           Si IPIV( K ) <> K pero IPIV( K ) > 0, entonces la fila K se
!>           intercambió con la fila IPIV( K ) y se usó un bloque pivote de
!>           1x1.
!>           Si IPIV( K ) < 0 y UPLO = 'U', entonces la fila K-1 se intercambió
!>           con la fila | IPIV( K ) | y se usó un bloque pivote de 2x2.
!>           Si IPIV( K ) < 0 y UPLO = 'L', entonces la fila K+1 se intercambió
!>           con la fila | IPIV( K ) | y se usó un bloque pivote de 2x2.
!>
!>  B      - Arreglo COMPLEX*16, dimension( LDB, NRHS )
!>           En la entrada, B contiene NRHS vectores de longitud N.
!>           A la salida, B se sobrescribe con el producto A * B.
!>
!>  LDB    - INTEGER
!>           En la entrada, LDB contiene la dimensión principal de B tal como
!>           está declarada en el programa llamador. LDB debe ser al menos
!>           max( 1, N ).
!>           Sin cambios a la salida.
!>
!>  INFO   - INTEGER
!>           INFO es el indicador de error.
!>           A la salida, un valor de 0 indica una salida exitosa.
!>           Un valor negativo, por ejemplo -K, indica que el K-ésimo argumento
!>           tiene un valor ilegal.
!>
