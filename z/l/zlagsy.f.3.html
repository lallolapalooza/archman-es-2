subroutine zlagsy (integer n, integer k, double precision, dimension( * ) d, complex*16, dimension( lda, * ) a, integer lda, integer, dimension( 4 ) iseed, complex*16, dimension( * ) work, integer info)

ZLAGSY

Propósito:

!>
!> ZLAGSY genera una matriz simétrica compleja A, multiplicando
!> previamente y posteriormente una matriz diagonal real D con una matriz
!> unitaria aleatoria: A = U*D*U**T. El semi-ancho de banda puede entonces
!> reducirse a k mediante transformaciones unitarias adicionales.
!>

Parámetros

N 
!>          N es ENTERO
!>          El orden de la matriz A.  N >= 0.
!>

K

!>          K es ENTERO
!>          El número de subdiagonales distintas de cero dentro de la banda
!>          de A.  0 <= K <= N-1.
!>

D

!>          D es array de DOBLE PRECISION, dimensión (N)
!>          Los elementos diagonales de la matriz diagonal D.
!>

A

!>          A es array de COMPLEX*16, dimensión (LDA,N)
!>          La matriz simétrica A generada de n por n (se almacena la matriz
!>          completa).
!>

LDA

!>          LDA es ENTERO
!>          La dimensión principal del array A.  LDA >= N.
!>

ISEED

!>          ISEED es array de ENTEROS, dimensión (4)
!>          En la entrada, la semilla del generador de números aleatorios;
!>          los elementos del array deben estar entre 0 y 4095, e
!>          ISEED(4) debe ser impar.
!>          En la salida, la semilla se actualiza.
!>

WORK

!>          WORK es array de COMPLEX*16, dimensión (2*N)
!>

INFO

!>          INFO es ENTERO
!>          = 0: salida exitosa
!>          < 0: si INFO = -i, el i-ésimo argumento tenía un valor ilegal
!>
