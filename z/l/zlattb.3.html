subroutine zlattb (integer imat, character uplo, character trans, character diag, integer, dimension( 4 ) iseed, integer n, integer kd, complex*16, dimension( ldab, * ) ab, integer ldab, complex*16, dimension( * ) b, complex*16, dimension( * ) work, double precision, dimension( * ) rwork, integer info)

ZLATTB

Propósito:

!>
!> ZLATTB genera una matriz de prueba triangular en almacenamiento
!> bidimensional. IMAT y UPLO especifican de forma única las propiedades de
!> la matriz de prueba, que se devuelve en el array A.
!>

Parámetros

IMAT 
!>          IMAT es un ENTERO
!>          Una clave entera que describe qué matriz generar para esta
!>          ruta.
!>

UPLO

!>          UPLO es CHARACTER*1
!>          Especifica si la matriz A será superior o inferior
!>          triangular.
!>          = 'U': Triangular superior
!>          = 'L': Triangular inferior
!>

TRANS

!>          TRANS es CHARACTER*1
!>          Especifica si se usará la matriz o su transpuesta.
!>          = 'N': Sin transpuesta
!>          = 'T': Transpuesta
!>          = 'C': Transpuesta conjugada (= transpuesta)
!>

DIAG

!>          DIAG es CHARACTER*1
!>          Especifica si la matriz A es unitaria triangular o no.
!>          = 'N': Triangular no unitaria
!>          = 'U': Triangular unitaria
!>

ISEED

!>          ISEED es un array de ENTEROS, dimension (4)
!>          El vector semilla para el generador de números aleatorios
!>          (usado en ZLATMS). Modificado al salir.
!>

N

!>          N es un ENTERO
!>          El orden de la matriz a generar.
!>

KD

!>          KD es un ENTERO
!>          El número de superdiagonales o subdiagonales de la matriz
!>          triangular bandeada A. KD >= 0.
!>

AB

!>          AB es un array COMPLEX*16, dimension (LDAB,N)
!>          La matriz triangular superior o inferior bandeada A, almacenada
!>          en las primeras KD+1 filas de AB. Sea j una columna de A,
!>          1<=j<=n.
!>          Si UPLO = 'U', AB(kd+1+i-j,j) = A(i,j) para max(1,j-kd)<=i<=j.
!>          Si UPLO = 'L', AB(1+i-j,j)    = A(i,j) para j<=i<=min(n,j+kd).
!>

LDAB

!>          LDAB es un ENTERO
!>          La dimensión principal del array AB. LDAB >= KD+1.
!>

B

!>          B es un array COMPLEX*16, dimension (N)
!>

WORK

!>          WORK es un array COMPLEX*16, dimension (2*N)
!>

RWORK

!>          RWORK es un array DOUBLE PRECISION, dimension (N)
!>

INFO

!>          INFO es un ENTERO
!>          = 0: Salida exitosa
!>          < 0: Si INFO = -i, el argumento i tuvo un valor ilegal
!>
