subrutina zlatm5 (integer prtype, integer m, integer n, complex*16, dimension( lda, * ) a, integer lda, complex*16, dimension( ldb, * ) b, integer ldb, complex*16, dimension( ldc, * ) c, integer ldc, complex*16, dimension( ldd, * ) d, integer ldd, complex*16, dimension( lde, * ) e, integer lde, complex*16, dimension( ldf, * ) f, integer ldf, complex*16, dimension( ldr, * ) r, integer ldr, complex*16, dimension( ldl, * ) l, integer ldl, double precision alpha, integer qblcka, integer qblckb)

ZLATM5

Propósito:

!>
!> ZLATM5 genera matrices involucradas en la ecuación de Sylvester Generalizada:
!>
!>     A * R - L * B = C
!>     D * R - L * E = F
!>
!> También satisfacen (la condición de diagonalización)
!>
!>  [ I -L ] ( [ A  -C ], [ D -F ] ) [ I  R ] = ( [ A    ], [ D    ] )
!>  [    I ] ( [     B ]  [    E ] ) [    I ]   ( [    B ]  [    E ] )
!>
!>

Parámetros

PRTYPE 
!>          PRTYPE es INTEGER
!>           a un cierto tipo de matrices a generar
!>          (ver más detalles).
!>

M

!>          M es INTEGER
!>          Especifica el orden de A y D y el número de filas en
!>          C, F, R y L.
!>

N

!>          N es INTEGER
!>          Especifica el orden de B y E y el número de columnas en
!>          C, F, R y L.
!>

A

!>          A es un array COMPLEX*16, dimension (LDA, M).
!>          Al salir, A M-por-M se inicializa según PRTYPE.
!>

LDA

!>          LDA es INTEGER
!>          La dimensión principal de A.
!>

B

!>          B es un array COMPLEX*16, dimension (LDB, N).
!>          Al salir, B N-por-N se inicializa según PRTYPE.
!>

LDB

!>          LDB es INTEGER
!>          La dimensión principal de B.
!>

C

!>          C es un array COMPLEX*16, dimension (LDC, N).
!>          Al salir, C M-por-N se inicializa según PRTYPE.
!>

LDC

!>          LDC es INTEGER
!>          La dimensión principal de C.
!>

D

!>          D es un array COMPLEX*16, dimension (LDD, M).
!>          Al salir, D M-por-M se inicializa según PRTYPE.
!>

LDD

!>          LDD es INTEGER
!>          La dimensión principal de D.
!>

E

!>          E es un array COMPLEX*16, dimension (LDE, N).
!>          Al salir, E N-por-N se inicializa según PRTYPE.
!>

LDE

!>          LDE es INTEGER
!>          La dimensión principal de E.
!>

F

!>          F es un array COMPLEX*16, dimension (LDF, N).
!>          Al salir, F M-por-N se inicializa según PRTYPE.
!>

LDF

!>          LDF es INTEGER
!>          La dimensión principal de F.
!>

R

!>          R es un array COMPLEX*16, dimension (LDR, N).
!>          Al salir, R M-por-N se inicializa según PRTYPE.
!>

LDR

!>          LDR es INTEGER
!>          La dimensión principal de R.
!>

L

!>          L es un array COMPLEX*16, dimension (LDL, N).
!>          Al salir, L M-por-N se inicializa según PRTYPE.
!>

LDL

!>          LDL es INTEGER
!>          La dimensión principal de L.
!>

ALPHA

!>          ALPHA es DOUBLE PRECISION
!>          Parámetro usado en la generación de matrices PRTYPE = 1 y 5.
!>

QBLCKA

!>          QBLCKA es INTEGER
!>          Cuando PRTYPE = 3, especifica la distancia entre bloques de 2x2
!>          en la diagonal de A. De lo contrario, QBLCKA no se referencia.
!>          QBLCKA > 1.
!>

QBLCKB

!>          QBLCKB es INTEGER
!>          Cuando PRTYPE = 3, especifica la distancia entre bloques de 2x2
!>          en la diagonal de B. De lo contrario, QBLCKB no se referencia.
!>          QBLCKB > 1.
!>

Detalles adicionales:

!>
!>  PRTYPE = 1: A y B son bloques de Jordan, D y E son matrices identidad
!>
!>             A : si (i == j) entonces A(i, j) = 1.0
!>                 si (j == i + 1) entonces A(i, j) = -1.0
!>                 else A(i, j) = 0.0,            i, j = 1...M
!>
!>             B : si (i == j) entonces B(i, j) = 1.0 - ALPHA
!>                 si (j == i + 1) entonces B(i, j) = 1.0
!>                 else B(i, j) = 0.0,            i, j = 1...N
!>
!>             D : si (i == j) entonces D(i, j) = 1.0
!>                 else D(i, j) = 0.0,            i, j = 1...M
!>
!>             E : si (i == j) entonces E(i, j) = 1.0
!>                 else E(i, j) = 0.0,            i, j = 1...N
!>
!>             L = R se eligen de [-10...10],
!>                  lo que especifica los lados derechos (C, F).
!>
!>  PRTYPE = 2 o 3: Triangular(es) y/o cuasi-triangular(es).
!>
!>             A : si (i <= j) entonces A(i, j) = [-1...1]
!>                 else A(i, j) = 0.0,             i, j = 1...M
!>
!>                 si (PRTYPE = 3) entonces
!>                    A(k + 1, k + 1) = A(k, k)
!>                    A(k + 1, k) = [-1...1]
!>                    signo(A(k, k + 1) = -(sin(A(k + 1, k))
!>                        k = 1, M - 1, QBLCKA
!>
!>             B : si (i <= j) entonces B(i, j) = [-1...1]
!>                 else B(i, j) = 0.0,            i, j = 1...N
!>
!>                 si (PRTYPE = 3) entonces
!>                    B(k + 1, k + 1) = B(k, k)
!>                    B(k + 1, k) = [-1...1]
!>                    signo(B(k, k + 1) = -(signo(B(k + 1, k))
!>                        k = 1, N - 1, QBLCKB
!>
!>             D : si (i <= j) entonces D(i, j) = [-1...1].
!>                 else D(i, j) = 0.0,            i, j = 1...M
!>
!>
!>             E : si (i <= j) entonces D(i, j) = [-1...1]
!>                 else E(i, j) = 0.0,            i, j = 1...N
!>
!>                 L, R se eligen de [-10...10],
!>                 lo que especifica los lados derechos (C, F).
!>
!>  PRTYPE = 4 Completa
!>             A(i, j) = [-10...10]
!>             D(i, j) = [-1...1]    i,j = 1...M
!>             B(i, j) = [-10...10]
!>             E(i, j) = [-1...1]    i,j = 1...N
!>             R(i, j) = [-10...10]
!>             L(i, j) = [-1...1]    i = 1..M ,j = 1...N
!>
!>             L, R especifican los lados derechos (C, F).
!>
!>  PRTYPE = 5 caso especial eigs comunes y/o cercanos.
!>
