subroutine zlatm4 (integer itype, integer n, integer nz1, integer nz2, logical rsign, double precision amagn, double precision rcond, double precision triang, integer idist, integer, dimension( 4 ) iseed, complex*16, dimension( lda, * ) a, integer lda)

ZLATM4

Propósito:

!>
!> ZLATM4 genera matrices cuadradas básicas, que pueden ser
!> multiplicadas posteriormente por otras para producir matrices de prueba. Está
!> pensada principalmente para probar las rutinas de valores propios generalizados.
!>
!> Primero genera la diagonal y la (posiblemente) subdiagonal,
!> según los valores de ITYPE, NZ1, NZ2, RSIGN, AMAGN y RCOND.
!> Luego rellena el triángulo superior con números aleatorios, si TRIANG es
!> distinto de cero.
!>

Parámetros

ITYPE 
!>          ITYPE es INTEGER
!>          El tipo de matriz en la diagonal y subdiagonal.
!>          Si ITYPE < 0, entonces se genera el tipo abs(ITYPE) y luego
!>             se invierte extremo a extremo (A(I,J) := A'(N-J,N-I).)  Ver también
!>             la descripción de AMAGN y RSIGN.
!>
!>          Tipos especiales:
!>          = 0:  la matriz cero.
!>          = 1:  la identidad.
!>          = 2:  un bloque de Jordan transpuesto.
!>          = 3:  Si N es impar, entonces un bloque de Jordan transpuesto de k+1 x k+1
!>                seguido por un bloque de identidad de k x k, donde k=(N-1)/2.
!>                Si N es par, entonces k=(N-2)/2, y se añade una entrada diagonal cero
!>                al final.
!>
!>          Tipos de diagonal. La diagonal consiste en NZ1 ceros, luego
!>             k=N-NZ1-NZ2 no ceros. La subdiagonal es cero. ITYPE
!>             especifica las entradas diagonales no nulas como sigue:
!>          = 4:  1, ..., k
!>          = 5:  1, RCOND, ..., RCOND
!>          = 6:  1, ..., 1, RCOND
!>          = 7:  1, a, a^2, ..., a^(k-1)=RCOND
!>          = 8:  1, 1-d, 1-2*d, ..., 1-(k-1)*d=RCOND
!>          = 9:  números aleatorios elegidos de (RCOND,1)
!>          = 10: números aleatorios con distribución IDIST (ver ZLARND.)
!>

N

!>          N es INTEGER
!>          El orden de la matriz.
!>

NZ1

!>          NZ1 es INTEGER
!>          Si abs(ITYPE) > 3, entonces las primeras NZ1 entradas diagonales serán
!>          cero.
!>

NZ2

!>          NZ2 es INTEGER
!>          Si abs(ITYPE) > 3, entonces las últimas NZ2 entradas diagonales serán
!>          cero.
!>

RSIGN

!>          RSIGN es LOGICAL
!>          = .TRUE.:  Las entradas de la diagonal y subdiagonal se
!>                     multiplicarán por números aleatorios de magnitud 1.
!>          = .FALSE.: Las entradas de la diagonal y subdiagonal se
!>                     dejarán como están (generalmente reales no negativos.)
!>

AMAGN

!>          AMAGN es DOUBLE PRECISION
!>          Las entradas de la diagonal y subdiagonal se multiplicarán por
!>          AMAGN.
!>

RCOND

!>          RCOND es DOUBLE PRECISION
!>          Si abs(ITYPE) > 4, entonces la entrada diagonal más pequeña será
!>          RCOND. RCOND debe estar entre 0 y 1.
!>

TRIANG

!>          TRIANG es DOUBLE PRECISION
!>          Las entradas sobre la diagonal serán números aleatorios con
!>          magnitud acotada por TRIANG (es decir, números aleatorios multiplicados
!>          por TRIANG.)
!>

IDIST

!>          IDIST es INTEGER
!>          En la entrada, DIST especifica el tipo de distribución que se usará
!>          para generar una matriz aleatoria.
!>          = 1: partes real e imaginaria cada una UNIFORM( 0, 1 )
!>          = 2: partes real e imaginaria cada una UNIFORM( -1, 1 )
!>          = 3: partes real e imaginaria cada una NORMAL( 0, 1 )
!>          = 4: número complejo uniforme en DISCO( 0, 1 )
!>

ISEED

!>          ISEED es un array INTEGER, dimension (4)
!>          En la entrada, ISEED especifica la semilla del generador de números
!>          aleatorios. Los valores de ISEED se cambian en la salida y pueden
!>          ser usados en la siguiente llamada a ZLATM4 para continuar la misma
!>          secuencia de números aleatorios.
!>          Nota: ISEED(4) debe ser impar, para el generador de números aleatorios
!>          utilizado actualmente.
!>

A

!>          A es un array COMPLEX*16, dimension (LDA, N)
!>          Array a calcular.
!>

LDA

!>          LDA es INTEGER
!>          Dimensión líder de A. Debe ser al menos 1 y al menos N.
!>
