subroutine zlattp (integer imat, character uplo, character trans, character diag, integer, dimension( 4 ) iseed, integer n, complex*16, dimension( * ) ap, complex*16, dimension( * ) b, complex*16, dimension( * ) work, double precision, dimension( * ) rwork, integer info)

ZLATTP

Propósito:

!>
!> ZLATTP genera una matriz de prueba triangular en almacenamiento compactado.
!> IMAT y UPLO especifican de forma única las propiedades de la matriz de prueba,
!> que se devuelve en el array AP.
!>

Parámetros

IMAT 
!>          IMAT es INTEGER
!>          Una clave entera que describe qué matriz generar para esta
!>          ruta.
!>

UPLO

!>          UPLO es CHARACTER*1
!>          Especifica si la matriz A será triangular superior o inferior.
!>          = 'U':  Triangular superior
!>          = 'L':  Triangular inferior
!>

TRANS

!>          TRANS es CHARACTER*1
!>          Especifica si se utilizará la matriz o su transpuesta.
!>          = 'N':  Sin transpuesta
!>          = 'T':  Transpuesta
!>          = 'C':  Transpuesta conjugada
!>

DIAG

!>          DIAG es CHARACTER*1
!>          Especifica si la matriz A es unitriangular o no.
!>          = 'N':  Triangular no unitaria
!>          = 'U':  Triangular unitaria
!>

ISEED

!>          ISEED es un array INTEGER, dimension (4)
!>          El vector semilla para el generador de números aleatorios (usado en
!>          ZLATMS). Modificado a la salida.
!>

N

!>          N es INTEGER
!>          El orden de la matriz a generar.
!>

AP

!>          AP es un array COMPLEX*16, dimension (N*(N+1)/2)
!>          La matriz triangular superior o inferior A, compactada por columnas en
!>          un array lineal. La j-ésima columna de A se almacena en el array
!>          AP como sigue:
!>          si UPLO = 'U', AP((j-1)*j/2 + i) = A(i,j) para 1<=i<=j;
!>          si UPLO = 'L',
!>             AP((j-1)*(n-j) + j*(j+1)/2 + i-j) = A(i,j) para j<=i<=n.
!>

B

!>          B es un array COMPLEX*16, dimension (N)
!>          El vector del lado derecho, si IMAT > 10.
!>

WORK

!>          WORK es un array COMPLEX*16, dimension (2*N)
!>

RWORK

!>          RWORK es un array DOUBLE PRECISION, dimension (N)
!>

INFO

!>          INFO es INTEGER
!>          = 0:  salida exitosa
!>          < 0:  si INFO = -i, el i-ésimo argumento tenía un valor ilegal
!>
