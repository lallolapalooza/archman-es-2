subroutine zlsets (integer m, integer p, integer n, complex*16, dimension( lda, * ) a, complex*16, dimension( lda, * ) af, integer lda, complex*16, dimension( ldb, * ) b, complex*16, dimension( ldb, * ) bf, integer ldb, complex*16, dimension( * ) c, complex*16, dimension( * ) cf, complex*16, dimension( * ) d, complex*16, dimension( * ) df, complex*16, dimension( * ) x, complex*16, dimension( lwork ) work, integer lwork, double precision, dimension( * ) rwork, double precision, dimension( 2 ) result)

ZLSETS

Propósito:

!>
!> ZLSETS prueba ZGGLSE - una subrutina para resolver el problema de mínimos cuadrados lineales con restricciones de igualdad (LSE).
!>

Parámetros

M 
!>          M es INTEGER
!>          El número de filas de la matriz A.  M >= 0.
!>

P

!>          P es INTEGER
!>          El número de filas de la matriz B.  P >= 0.
!>

N

!>          N es INTEGER
!>          El número de columnas de las matrices A y B.  N >= 0.
!>

A

!>          A es un array COMPLEX*16, dimensión (LDA,N)
!>          La matriz M por N, A.
!>

AF

!>          AF es un array COMPLEX*16, dimensión (LDA,N)
!>

LDA

!>          LDA es INTEGER
!>          La dimensión principal de los arrays A, AF, Q y R.
!>          LDA >= max(M,N).
!>

B

!>          B es un array COMPLEX*16, dimensión (LDB,N)
!>          La matriz P por N, A.
!>

BF

!>          BF es un array COMPLEX*16, dimensión (LDB,N)
!>

LDB

!>          LDB es INTEGER
!>          La dimensión principal de los arrays B, BF, V y S.
!>          LDB >= max(P,N).
!>

C

!>          C es un array COMPLEX*16, dimensión( M )
!>          el vector C en el problema LSE.
!>

CF

!>          CF es un array COMPLEX*16, dimensión( M )
!>

D

!>          D es un array COMPLEX*16, dimensión( P )
!>          el vector D en el problema LSE.
!>

DF

!>          DF es un array COMPLEX*16, dimensión( P )
!>

X

!>          X es un array COMPLEX*16, dimensión( N )
!>          el vector solución X en el problema LSE.
!>

WORK

!>          WORK es un array COMPLEX*16, dimensión (LWORK)
!>

LWORK

!>          LWORK es INTEGER
!>          La dimensión del array WORK.
!>

RWORK

!>          RWORK es un array DOUBLE PRECISION, dimensión (M)
!>

RESULT

!>          RESULT es un array DOUBLE PRECISION, dimensión (2)
!>          Las relaciones de prueba:
!>            RESULT(1) = norm( A*x - c )/ norm(A)*norm(X)*EPS
!>            RESULT(2) = norm( B*x - d )/ norm(B)*norm(X)*EPS
!>
