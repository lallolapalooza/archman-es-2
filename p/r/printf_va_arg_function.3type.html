DESCRIPCIÓN

Estas funciones sirven para extender y/o modificar el comportamiento de las funciones de la familia printf(3).

register_printf_specifier()

Esta función registra un especificador de conversión personalizado para la familia de funciones printf(3).

spec
El carácter que se usará como especificador de conversión en la cadena de formato.
func
Función de callback que será ejecutada por la familia de funciones printf(3) para formatear los argumentos de entrada en el stream de salida.
stream
Flujo de salida donde se debe imprimir la salida formateada. Este flujo representa la salida de forma transparente, incluso en el caso de funciones que escriben en una cadena.
info
Estructura que contiene información de contexto, incluyendo los modificadores especificados en la cadena de formato. Contiene los mismos contenidos que en arginfo.
args
Array de punteros a los argumentos de la función tipo printf(3).
arginfo
Función de callback que será ejecutada por la familia de funciones printf(3) para saber cuántos argumentos deben ser parseados para el especificador personalizado y también sus tipos.
info
Estructura que contiene información de contexto, incluyendo los modificadores especificados en la cadena de formato. Contiene los mismos contenidos que en func.
n
Número de argumentos restantes por parsear.
argtypes
Este array debe ser configurado para definir el tipo de cada uno de los argumentos que serán parseados. Cada elemento del array representa uno de los argumentos a ser parseados, en el mismo orden en que son pasados a la función tipo printf(3). Cada elemento debe ser configurado a un tipo base (PA_*) de la enumeración anterior, o uno personalizado, y opcionalmente ORed con un modificador de longitud apropiado (PA_FLAG_*).

El tipo se determina usando una de las siguientes constantes:

PA_INT
int.
PA_CHAR
int, casteado a char.
PA_WCHAR
wchar_t.
PA_STRING
const char *, una cadena terminada en '\0'.
PA_WSTRING
const wchar_t *, una cadena ancha terminada en L'\0'.
PA_POINTER
void *.
PA_FLOAT
float.
PA_DOUBLE
double.
PA_LAST
TODO.
size
Para tipos definidos por el usuario, el tamaño del tipo (en bytes) también debe especificarse a través de este array. De lo contrario, déjelo sin usar.

arginfo se llama antes que func, y prepara alguna información necesaria para llamar a func.

VALOR DE RETORNO

register_printf_specifier(), register_printf_modifier(), y register_printf_type() devuelven cero si tienen éxito, o -1 en caso de error.

Callbacks

La función de callback de tipo printf_function debe devolver el número de caracteres escritos, o -1 en caso de error.

La función de callback de tipo printf_arginfo_size_function debe devolver el número de argumentos a ser parseados por este especificador. También pasa información sobre el tipo de esos argumentos al llamador a través de argtypes. En caso de error, debe devolver -1.

ERRORES

EINVAL
El especificador no era un carácter válido.

ESTÁNDARES

GNU.

HISTORIA

register_printf_function(3) es una función antigua similar a register_printf_specifier(), y ahora está obsoleta. Esa función no puede manejar tipos definidos por el usuario.

register_printf_specifier() reemplaza a register_printf_function(3).

EJEMPLOS

El siguiente programa de ejemplo registra los especificadores 'b' y 'B' para imprimir enteros en formato binario, reflejando las reglas para otros especificadores de conversión sin signo como 'x' y 'u'. Esto se puede usar para imprimir en binario antes de C23.

/* Este código es de dominio público */
#include <err.h>
#include <limits.h>
#include <stddef.h>
#include <stdint.h>
#include <stdio.h>
#include <stdlib.h>
#include <string.h>
#include <sys/param.h>
#include <printf.h>
#define GROUP_SEP  '\''
struct Printf_Pad {
    char    ch;
    size_t  len;
};
static int b_printf(FILE *stream, const struct printf_info *info,
                    const void *const args[]);
static int b_arginf_sz(const struct printf_info *info,
                    size_t n, int argtypes[n], int size[n]);
static uintmax_t b_value(const struct printf_info *info,
                    const void *arg);
static size_t b_bin_repr(char bin[UINTMAX_WIDTH],
                    const struct printf_info *info, const void *arg);
static size_t b_bin_len(const struct printf_info *info,
                    ptrdiff_t min_len);
static size_t b_pad_len(const struct printf_info *info,
                    ptrdiff_t bin_len);
static ssize_t b_print_prefix(FILE *stream,
                    const struct printf_info *info);
static ssize_t b_pad_zeros(FILE *stream, const struct printf_info *info,
                    ptrdiff_t min_len);
static ssize_t b_print_number(FILE *stream,
                    const struct printf_info *info,
                    const char bin[UINTMAX_WIDTH],
                    size_t min_len, size_t bin_len);
static char pad_ch(const struct printf_info *info);
static ssize_t pad_spaces(FILE *stream, size_t pad_len);

{

        err(EXIT_FAILURE, "register_printf_specifier('B', ...)");

    printf("%b;\n", 0x5Eu);
    printf("%hB;\n", 0x5Eu);

    printf("%jb;\n", (uintmax_t)0x5E);
    printf("%zb;\n", (size_t)0x5E);
    printf("....----....----....----....----\n");
    printf("%#b;\n", 0x5Eu);
    printf("%#B;\n", 0x5Eu);

static int
b_printf(FILE *stream, const struct printf_info *info,
         const void *const args[])
{
    char               bin[UINTMAX_WIDTH];
    size_t             min_len, bin_len;
    ssize_t            len, tmp;
    struct Printf_Pad  pad = {0};
    len = 0;
    min_len = b_bin_repr(bin, info, args[0]);
    bin_len = b_bin_len(info, min_len);
    pad.ch = pad_ch(info);
    if (pad.ch == ' ')
        pad.len = b_pad_len(info, bin_len);
    /* Relleno con ' ' (alineado a la derecha) */
    if ((pad.ch == ' ') && !info->left) {
        tmp = pad_spaces(stream, pad.len);
        if (tmp == EOF)
            return EOF;
        len += tmp;
    }
    /* Prefijo "0b"/"0B" */
    if (info->alt) {
        tmp = b_print_prefix(stream, info);
        if (tmp == EOF)
            return EOF;
        len += tmp;
    }
    /* Relleno con '0' */
    if (pad.ch == '0') {
        tmp = b_pad_zeros(stream, info, min_len);
        if (tmp == EOF)
            return EOF;
        len += tmp;
    }
    /* Imprimir número (incluyendo ceros iniciales para completar la precisión) */
    tmp = b_print_number(stream, info, bin, min_len, bin_len);
    if (tmp == EOF)
        return EOF;
    len += tmp;
    /* Relleno con ' ' (alineado a la izquierda) */
    if (info->left) {
        tmp = pad_spaces(stream, pad.len);
        if (tmp == EOF)
            return EOF;
        len += tmp;
    }
    return len;
}
static int
b_arginf_sz(const struct printf_info *info, size_t n, int argtypes[n],
            [[maybe_unused]] int size[n])
{
    if (n < 1)
        return -1;
    if (info->is_long_double)
        argtypes[0] = PA_INT | PA_FLAG_LONG_LONG;
    else if (info->is_long)
        argtypes[0] = PA_INT | PA_FLAG_LONG;
    else
        argtypes[0] = PA_INT;
    return 1;
}
static uintmax_t
b_value(const struct printf_info *info, const void *arg)
{
    if (info->is_long_double)
        return *(const unsigned long long *)arg;
    if (info->is_long)
        return *(const unsigned long *)arg;
    /* short y char se promocionan a int */
    return *(const unsigned int *)arg;
}
static size_t
b_bin_repr(char bin[UINTMAX_WIDTH],
           const struct printf_info *info, const void *arg)
{
    size_t     min_len;
    uintmax_t  val;
    val = b_value(info, arg);
    bin[0] = '0';
    for (min_len = 0; val; min_len++) {
        bin[min_len] = '0' + (val % 2);
        val >>= 1;
    }
    return MAX(min_len, 1);
}
static size_t
b_bin_len(const struct printf_info *info, ptrdiff_t min_len)
{
    return MAX(info->prec, min_len);
}
static size_t
b_pad_len(const struct printf_info *info, ptrdiff_t bin_len)
{
    ptrdiff_t  pad_len;
    pad_len = info->width - bin_len;
    if (info->alt)
        pad_len -= 2;
    if (info->group)
        pad_len -= (bin_len - 1) / 4;
    return MAX(pad_len, 0);
}
static ssize_t
b_print_prefix(FILE *stream, const struct printf_info *info)
{
    ssize_t len;
    len = 0;
    if (fputc('0', stream) == EOF)
        return EOF;
    len++;
    if (fputc(info->spec, stream) == EOF)
        return EOF;
    len++;
    return len;
}
static ssize_t
b_pad_zeros(FILE *stream, const struct printf_info *info,
            ptrdiff_t min_len)
{
    ssize_t    len;
    ptrdiff_t  tmp;
    len = 0;
    tmp = info->width - (info->alt * 2);
    if (info->group)
        tmp -= tmp / 5 - !(tmp % 5);
    for (ptrdiff_t i = tmp - 1; i > min_len - 1; i--) {
        if (fputc('0', stream) == EOF)
            return EOF;
        len++;
        if (!info->group || (i % 4))
            continue;
        if (fputc(GROUP_SEP, stream) == EOF)
            return EOF;
        len++;
    }
    return len;
}
static ssize_t
b_print_number(FILE *stream, const struct printf_info *info,
               const char bin[UINTMAX_WIDTH],
               size_t min_len, size_t bin_len)
{
    ssize_t  len;
    len = 0;
    /* Imprimir ceros iniciales para rellenar la precisión */
    for (size_t i = bin_len - 1; i > min_len - 1; i--) {
        if (fputc('0', stream) == EOF)
            return EOF;
        len++;
        if (!info->group || (i % 4))
            continue;
        if (fputc(GROUP_SEP, stream) == EOF)
            return EOF;
        len++;
    }
    /* Imprimir número */
    for (size_t i = min_len - 1; i < min_len; i--) {
        if (fputc(bin[i], stream) == EOF)
            return EOF;
        len++;
        if (!info->group || (i % 4) || !i)
            continue;
        if (fputc(GROUP_SEP, stream) == EOF)
            return EOF;
        len++;
    }
    return len;
}
static char
pad_ch(const struct printf_info *info)
{
    if ((info->prec != -1) || (info->pad == ' ') || info->left)
        return ' ';
    return '0';
}
static ssize_t
pad_spaces(FILE *stream, size_t pad_len)
{
    ssize_t  len;
    len = 0;
    for (size_t i = pad_len - 1; i < pad_len; i--) {
        if (fputc(' ', stream) == EOF)
            return EOF;
        len++;
    }
    return len;
}
