NOMBRE

pacutils-mtree - leer datos mtree para paquetes instalados

SINOPSIS

#include <pacutils/mtree.h>
typedef struct pu_mtree_t {
  char *path;
  char *type;
  uid_t uid;
  gid_t gid;
  mode_t mode;
  off_t size;
  char *md5digest;
  char *sha256digest;
} pu_mtree_t;
typedef struct {
  FILE *stream;
  int eof;
  pu_mtree_t defaults;
} pu_mtree_reader_t;
pu_mtree_reader_t *pu_mtree_reader_open_stream(FILE *stream);
pu_mtree_reader_t *pu_mtree_reader_open_package(alpm_handle_t *h, alpm_pkg_t *p);
pu_mtree_t *pu_mtree_reader_next(pu_mtree_reader_t *reader, pu_mtree_t *dest);
void pu_mtree_reader_free(pu_mtree_reader_t *reader);
void pu_mtree_free(pu_mtree_t *mtree);
/* obsoleto */
alpm_list_t *pu_mtree_load_pkg_mtree(alpm_handle_t *handle, alpm_pkg_t *pkg);

DESCRIPCIÓN

Se proporcionan funciones de lectura de mtree de paquetes para cubrir las lagunas en el lector mtree de libarchive, en particular "md5digest" y "sha256digest" que actualmente no son leídos por libarchive. El soporte es incompleto y solo debe usarse para complementar el lector mtree nativo de libalpm.

pu_mtree_reader_t *pu_mtree_reader_open_stream(FILE *stream);
Abre un flujo de archivo para su análisis.
pu_mtree_reader_t *pu_mtree_reader_open_package(alpm_handle_t *h, alpm_pkg_t *p);
Abre el archivo mtree de un paquete instalado para su análisis. Los resultados no están definidos si "p" no es un paquete instalado localmente.
pu_mtree_t *pu_mtree_reader_next(pu_mtree_reader_t *reader, pu_mtree_t *dest);
Lee y devuelve la siguiente entrada en el archivo mtree. Si "dest" es "NULL" se asignará un nuevo objeto "pu_mtree_t" que debe ser liberado por el llamador. De lo contrario, se llenará con los datos analizados. La memoria asignada internamente se liberará automáticamente según sea necesario.
void pu_mtree_free(pu_mtree_t *mtree);
Libera una estructura "pu_mtree_t".
void pu_mtree_reader_free(pu_mtree_reader_t *reader);
Libera un objeto "pu_mtree_reader_t".
alpm_list_t *pu_mtree_load_pkg_mtree(alpm_handle_t *handle, alpm_pkg_t *pkg);
Devuelve una lista de entradas mtree para "pkg". OBSOLETO: use "pu_mtree_reader_open_package" en su lugar.

EJEMPLOS

Imprimir sumas md5 de archivos para un paquete:
pu_mtree_t *m;
pu_mtree_reader_t *r;
if((r = pu_mtree_reader_open_package(handle, pkg)) == NULL) {
    fprintf(stderr, "error: no se pudo cargar los datos mtree para '%s'\n",
       alpm_pkg_get_name(pkg));
    return;
}
while((m = pu_mtree_reader_next(r, NULL))) {
    const char *md5 = m->md5digest;
    printf("%s: %s\n", m->path,
        md5 && md5[0] != '\0' ? md5 : "(no se proporcionó suma md5)");
    pu_mtree_free(m);
}
if(!reader->eof) {
    fprintf(stderr, "error: no se pudo leer los datos mtree para '%s'\n",
       alpm_pkg_get_name(pkg));
}
pu_mtree_reader_free(r);
