pthread_attr_init(3) Manual de Funciones de Biblioteca pthread_attr_init(3)

NOMBRE

pthread_attr_init, pthread_attr_destroy - inicializar y destruir objeto de atributos de hilo

BIBLIOTECA

Biblioteca de hilos POSIX (libpthread, -lpthread)

SINOPSIS

#include <pthread.h>
int pthread_attr_init(pthread_attr_t *attr);
int pthread_attr_destroy(pthread_attr_t *attr);

DESCRIPCIÓN

La función pthread_attr_init() inicializa el objeto de atributos de hilo apuntado por attr con valores de atributos predeterminados. Después de esta llamada, los atributos individuales del objeto se pueden establecer utilizando varias funciones relacionadas (listadas en VEASE TAMBIÉN), y luego el objeto se puede utilizar en una o más llamadas a pthread_create(3) que crean hilos.

Llamar a pthread_attr_init() en un objeto de atributos de hilo que ya ha sido inicializado produce un comportamiento indefinido.

Cuando un objeto de atributos de hilo ya no se requiere, debe ser destruido utilizando la función pthread_attr_destroy(). Destruir un objeto de atributos de hilo no tiene ningún efecto en los hilos que fueron creados utilizando ese objeto.

Una vez que un objeto de atributos de hilo ha sido destruido, puede ser reinicializado usando pthread_attr_init(). Cualquier otro uso de un objeto de atributos de hilo destruido tiene resultados indefinidos.

VALOR DE RETORNO

En caso de éxito, estas funciones devuelven 0; en caso de error, devuelven un número de error distinto de cero.

ERRORES

POSIX.1 documenta un error ENOMEM para pthread_attr_init(); en Linux estas funciones siempre tienen éxito (pero las aplicaciones portables y preparadas para el futuro, no obstante, deben manejar un posible retorno de error).

ATRIBUTOS

Para obtener una explicación de los términos utilizados en esta sección, consulte attributes(7).

Interfaz Atributo Valor
pthread_attr_init (), pthread_attr_destroy () Seguridad de hilos MT-Safe

ESTANDARES

POSIX.1-2008.

HISTORIA

POSIX.1-2001.

NOTAS

El tipo pthread_attr_t debe tratarse como opaco: cualquier acceso al objeto que no sea a través de funciones pthreads no es portátil y produce resultados indefinidos.

EJEMPLOS

El programa a continuación utiliza opcionalmente pthread_attr_init() y varias funciones relacionadas para inicializar un objeto de atributos de hilo que se utiliza para crear un solo hilo. Una vez creado, el hilo utiliza la función pthread_getattr_np(3) (una extensión GNU no estándar) para recuperar los atributos del hilo, y luego muestra esos atributos.

static void
display_pthread_attr(pthread_attr_t *attr, char *prefix)
{
    int s, i;
    size_t v;
    void *stkaddr;
    struct sched_param sp;
    s = pthread_attr_getdetachstate(attr, &i);
    if (s != 0)
        errc(EXIT_FAILURE, s, "pthread_attr_getdetachstate");
    printf("%sEstado de separación    = %s\n", prefix,
           (i == PTHREAD_CREATE_DETACHED) ? "PTHREAD_CREATE_DETACHED" :
           (i == PTHREAD_CREATE_JOINABLE) ? "PTHREAD_CREATE_JOINABLE" :
           "???");
    s = pthread_attr_getscope(attr, &i);
    if (s != 0)
        errc(EXIT_FAILURE, s, "pthread_attr_getscope");
    printf("%sÁmbito                 = %s\n", prefix,
           (i == PTHREAD_SCOPE_SYSTEM)  ? "PTHREAD_SCOPE_SYSTEM" :
           (i == PTHREAD_SCOPE_PROCESS) ? "PTHREAD_SCOPE_PROCESS" :
           "???");
    s = pthread_attr_getinheritsched(attr, &i);
    if (s != 0)
        errc(EXIT_FAILURE, s, "pthread_attr_getinheritsched");
    printf("%sHeredar planificador   = %s\n", prefix,
           (i == PTHREAD_INHERIT_SCHED)  ? "PTHREAD_INHERIT_SCHED" :
           (i == PTHREAD_EXPLICIT_SCHED) ? "PTHREAD_EXPLICIT_SCHED" :
           "???");
    s = pthread_attr_getschedpolicy(attr, &i);
    if (s != 0)
        errc(EXIT_FAILURE, s, "pthread_attr_getschedpolicy");
    printf("%sPolítica de planificación = %s\n", prefix,
           (i == SCHED_OTHER) ? "SCHED_OTHER" :
           (i == SCHED_FIFO)  ? "SCHED_FIFO" :
           (i == SCHED_RR)    ? "SCHED_RR" :
           "???");
    s = pthread_attr_getschedparam(attr, &sp);
    if (s != 0)
        errc(EXIT_FAILURE, s, "pthread_attr_getschedparam");
    printf("%sPrioridad de planificación = %d\n", prefix, sp.sched_priority);
    s = pthread_attr_getguardsize(attr, &v);
    if (s != 0)
        errc(EXIT_FAILURE, s, "pthread_attr_getguardsize");
    printf("%sTamaño de guardia       = %zu bytes\n", prefix, v);
    s = pthread_attr_getstack(attr, &stkaddr, &v);
    if (s != 0)
        errc(EXIT_FAILURE, s, "pthread_attr_getstack");
    printf("%sDirección de pila       = %p\n", prefix, stkaddr);
    printf("%sTamaño de pila          = %#zx bytes\n", prefix, v);
}
static void *
thread_start(void *arg)
{
    int s;
    pthread_attr_t gattr;
    /* pthread_getattr_np() es una extensión GNU no estándar que
       recupera los atributos del hilo especificado en su
       primer argumento.  */
    s = pthread_getattr_np(pthread_self(), &gattr);
    if (s != 0)
        errc(EXIT_FAILURE, s, "pthread_getattr_np");
    printf("Atributos del hilo:\n");
    display_pthread_attr(&gattr, "\t");
    exit(EXIT_SUCCESS);         /* Termina todos los hilos */
}

EXAMPLES


The following program creates a thread using a thread attributes object.
The thread attributes object is configured to create a detached thread,
and to make the thread's scheduling policy and priority explicit.
The stack size is set to the value specified on the command line.

#include <pthread.h>
#include <stdio.h>
#include <stdlib.h>
#include <string.h>
#include <errno.h>
#include <unistd.h>
#include <err.h>

#define STACK_SIZE_DEFAULT 1024*1024 // 1MB

static void *thread_start(void *arg)
{
    /*
     * The thread does nothing but return.
     */
    pthread_exit(NULL);
}

int main(int argc, char *argv[])
{
    pthread_t thr;
    pthread_attr_t attr;
    int s;
    void *sp;
    size_t stack_size;
    pthread_attr_t *attrp = NULL; /* NULL if not using thread attributes */

    /* Create a thread attributes object, and initialize it with
     * default attribute values. */
    s = pthread_attr_init(&attr);
    if (s != 0)
        errc(EXIT_FAILURE, s, "pthread_attr_init");
    attrp = &attr;

    if (argc > 1) {
        /* If a stack size is provided on the command line, use it. */
        stack_size = strtoul(argv[1], NULL, 0);
        if (stack_size == 0) {
            fprintf(stderr, "Invalid stack size: %s\n", argv[1]);
            exit(EXIT_FAILURE);
        }
    } else {
        stack_size = STACK_SIZE_DEFAULT;
    }

    /* Set the detach state attribute to PTHREAD_CREATE_DETACHED */
    s = pthread_attr_setdetachstate(&attr, PTHREAD_CREATE_DETACHED);
    if (s != 0)
        errc(EXIT_FAILURE, s, "pthread_attr_setdetachstate");

    /* Set the inheritable-sched attribute to PTHREAD_EXPLICIT_SCHED */
    s = pthread_attr_setinheritsched(&attr, PTHREAD_EXPLICIT_SCHED);
    if (s != 0)
        errc(EXIT_FAILURE, s, "pthread_attr_setinheritsched");

    /* Allocate stack memory */
    s = posix_memalign(&sp, sysconf(_SC_PAGESIZE), stack_size);
    if (s != 0)
        errc(EXIT_FAILURE, s, "posix_memalign");
    printf("posix_memalign() allocated at %p\n", sp);

    /* Set the stack attribute */
    s = pthread_attr_setstack(&attr, sp, stack_size);
    if (s != 0)
        errc(EXIT_FAILURE, s, "pthread_attr_setstack");

    /* Create the thread using the attributes object. */
    s = pthread_create(&thr, attrp, &thread_start, NULL);
    if (s != 0)
        errc(EXIT_FAILURE, s, "pthread_create");

    /* Destroy the thread attributes object */
    s = pthread_attr_destroy(attrp);
    if (s != 0)
        errc(EXIT_FAILURE, s, "pthread_attr_destroy");

    /*
     * The thread is detached, so we don't need to join it.
     * We can simply pause the main thread and wait for the
     * program to be terminated by another thread calling exit().
     */
    pause();

    /* Should not reach here */
    return 0;
}
