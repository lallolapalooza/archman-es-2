gevent-websocket - Documentación de gevent-websocket

from geventwebsocket import WebSocketServer, WebSocketError
from flask import Flask, request, render_template
app = Flask(__name__)
@app.route('/')
def index():
    return render_template('index.html')
@app.route('/api')
def api():
    ws = request.environ.get('wsgi.websocket')
    if not ws:
        abort(400, "Se esperaba una solicitud WebSocket")
    while True:
        try:
            message = ws.receive()
            ws.send("Tu mensaje fue: {}".format(message))
        except WebSocketError:
            # Posibilidad de ejecutar código cuando la conexión se cierra
            break
if __name__ == '__main__':
    server = WebSocketServer(("", 8000), app)
    server.serve_forever()
