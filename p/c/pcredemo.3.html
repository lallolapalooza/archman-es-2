/*************************************************
*           PROGRAMA DE DEMOSTRACIÓN PCRE           *
*************************************************/
/* Este es un programa de demostración para ilustrar las formas más sencillas de
llamar a la biblioteca de expresiones regulares PCRE desde un programa C. Ver la
documentación de pcresample para una breve discusión ("man pcresample" si tiene
las páginas del manual de PCRE instaladas).
En entornos similares a Unix, si PCRE está instalado en las bibliotecas del sistema
estándar, debería poder compilar este programa usando este comando:
gcc -Wall pcredemo.c -lpcre -o pcredemo
Si PCRE no está instalado en un lugar estándar, es probable que esté instalado con
soporte para el mecanismo pkg-config. Si tiene pkg-config, puede compilar este
programa usando este comando:
gcc -Wall pcredemo.c `pkg-config --cflags --libs libpcre` -o pcredemo
Si no tiene pkg-config, puede tener que usar esto:
gcc -Wall pcredemo.c -I/usr/local/include -L/usr/local/lib \
  -R/usr/local/lib -lpcre -o pcredemo
Reemplace "/usr/local/include" y "/usr/local/lib" por donde estén instalados los
archivos de inclusión y de bibliotecas de PCRE en su sistema. Solo algunos sistemas
operativos (por ejemplo, Solaris) usan la opción -R.
Compilación en Windows:
Si desea enlazar estáticamente este programa contra un archivo .a no DLL, debe
definir PCRE_STATIC antes de incluir pcre.h, de lo contrario, las funciones
pcre_malloc() y pcre_free() exportadas se declararán como __declspec(dllimport),
con resultados no deseados. Así que en este entorno, descomente la siguiente línea. */
/* #define PCRE_STATIC */
#include <stdio.h>
#include <string.h>
#include <pcre.h>
#define OVECCOUNT 30    /* debe ser un múltiplo de 3 */
int main(int argc, char **argv)
{
pcre *re;
const char *error;
char *pattern;
char *subject;
unsigned char *name_table;
unsigned int option_bits;
int erroffset;
int find_all;
int crlf_is_newline;
int namecount;
int name_entry_size;
int ovector[OVECCOUNT];
int subject_length;
int rc, i;
int utf8;
/**************************************************************************
* Primero, ordenemos la línea de comandos. Solo hay una opción posible en *
* el momento, "-g" para solicitar coincidencias repetidas para encontrar *
* todas las ocurrencias, como la opción /g de Perl. Establecemos la variable *
* find_all a un valor distinto de cero si está presente la opción -g.  *
* Aparte de eso, debe haber exactamente dos argumentos.                    *
**************************************************************************/
find_all = 0;
for (i = 1; i < argc; i++)
  {
  if (strcmp(argv[i], "-g") == 0) find_all = 1;
    else break;
  }
/* Después de las opciones, requerimos exactamente dos argumentos, que son el patrón
y la cadena de sujeto. */
if (argc - i != 2)
  {
  printf("Se requieren dos argumentos: una expresión regular y una cadena de sujeto\n");
  return 1;
  }
pattern = argv[i];
subject = argv[i+1];
subject_length = (int)strlen(subject);
/*************************************************************************
* Ahora vamos a compilar el patrón de expresión regular, y manejar       *
* cualquier error que se detecte.                                          *
*************************************************************************/
re = pcre_compile(
  pattern,              /* el patrón */
  0,                    /* opciones por defecto */
  &error,               /* para el mensaje de error */
  &erroffset,           /* para el desplazamiento del error */
  NULL);                /* usar tablas de caracteres por defecto */
/* La compilación falló: imprimir el mensaje de error y salir */
if (re == NULL)
  {
  printf("La compilación de PCRE falló en el desplazamiento %d: %s\n", erroffset, error);
  return 1;
  }
/*************************************************************************
* Si la compilación tuvo éxito, llamamos a PCRE de nuevo, para realizar   *
* una coincidencia de patrón contra la cadena de sujeto. Esto hace solo UNA*
* coincidencia. Si se necesitan más coincidencias, se harán abajo.         *
*************************************************************************/
rc = pcre_exec(
  re,                   /* el patrón compilado */
  NULL,                 /* sin datos extra - no estudiamos el patrón */
  subject,              /* la cadena de sujeto */
  subject_length,       /* la longitud del sujeto */
  0,                    /* empezar en el desplazamiento 0 en el sujeto */
  0,                    /* opciones por defecto */
  ovector,              /* vector de salida para información de subcadenas */
  OVECCOUNT);           /* número de elementos en el vector de salida */
/* La coincidencia falló: manejar los casos de error */
if (rc < 0)
  {
  switch(rc)
    {
    case PCRE_ERROR_NOMATCH: printf("No hay coincidencia\n"); break;
    /*
    Manejar otros casos especiales si lo desea
    */
    default: printf("Error de coincidencia %d\n", rc); break;
    }
  pcre_free(re);     /* Liberar la memoria usada para el patrón compilado */
  return 1;
  }
/* Coincidencia exitosa */
printf("\nCoincidencia exitosa en el desplazamiento %d\n", ovector[0]);
/*************************************************************************
* Hemos encontrado la primera coincidencia dentro de la cadena de sujeto.  *
* Si el vector de salida no era lo suficientemente grande, decirlo. Luego, *
* mostrar cualquier subcadena que haya sido capturada.                   *
*************************************************************************/
/* El vector de salida no era lo suficientemente grande */
if (rc == 0)
  {
  rc = OVECCOUNT/3;
  printf("ovector solo tiene espacio para %d subcadenas capturadas\n", rc - 1);
  }
/* Mostrar subcadenas almacenadas en el vector de salida por número. Obviamente, en una
aplicación real querría hacer cosas aparte de imprimirlas. */
for (i = 0; i < rc; i++)
  {
  char *substring_start = subject + ovector[2*i];
  int substring_length = ovector[2*i+1] - ovector[2*i];
  printf("%2d: %.*s\n", i, substring_length, substring_start);
  }
/**************************************************************************
* Eso concluye la parte básica de este programa de demostración. Hemos    *
* compilado un patrón y realizado una única coincidencia. El código que   *
* sigue muestra primero cómo acceder a subcadenas con nombre, y luego cómo *
* codificar para coincidencias repetidas en el mismo sujeto.               *
**************************************************************************/
/* Ver si hay alguna subcadena con nombre, y si es así, mostrarla por nombre. Primero
tenemos que extraer el recuento de na
/* Elimina los paréntesis de la expresión regular. */
(void)pcre_fullinfo(
  re,                   /* el patrón compilado */
  NULL,                 /* sin datos adicionales - no hemos estudiado el patrón */
  PCRE_INFO_NAMECOUNT,  /* número de subcadenas nombradas */
  &namecount);          /* dónde poner la respuesta */
if (namecount <= 0) printf("No hay subcadenas nombradas\n"); else
  {
  unsigned char *tabptr;
  printf("Subcadenas nombradas\n");
  /* Antes de poder acceder a las subcadenas, debemos extraer la tabla para
  traducir nombres a números, y el tamaño de cada entrada en la tabla. */
  (void)pcre_fullinfo(
    re,                       /* el patrón compilado */
    NULL,                     /* sin datos adicionales - no hemos estudiado el patrón */
    PCRE_INFO_NAMETABLE,      /* dirección de la tabla */
    &name_table);             /* dónde poner la respuesta */
  (void)pcre_fullinfo(
    re,                       /* el patrón compilado */
    NULL,                     /* sin datos adicionales - no hemos estudiado el patrón */
    PCRE_INFO_NAMEENTRYSIZE,  /* tamaño de cada entrada en la tabla */
    &name_entry_size);        /* dónde poner la respuesta */
  /* Ahora podemos escanear la tabla y, para cada entrada, imprimir el número, el nombre,
  y la subcadena en sí. */
  tabptr = name_table;
  for (i = 0; i < namecount; i++)
    {
    int n = (tabptr[0] << 8) | tabptr[1];
    printf("(%d) %*s: %.*s\n", n, name_entry_size - 3, tabptr + 2,
      ovector[2*n+1] - ovector[2*n], subject + ovector[2*n]);
    tabptr += name_entry_size;
    }
  }
/*************************************************************************
* Si la opción "-g" se dio en la línea de comandos, queremos continuar *
* buscando coincidencias adicionales en la cadena del sujeto, de forma similar *
* a la opción /g en Perl. Esto resulta ser más complicado de lo que *
* podrías pensar debido a la posibilidad de que coincida una cadena vacía. *
* Lo que sucede es lo siguiente: *
* *
* Si la coincidencia anterior NO fue de una cadena vacía, podemos simplemente comenzar *
* la siguiente coincidencia al final de la anterior. *
* *
* Si la coincidencia anterior FUE de una cadena vacía, no podemos hacer eso, ya que *
* llevaría a un bucle infinito. En su lugar, se realiza una llamada especial a pcre_exec() *
* con las banderas PCRE_NOTEMPTY_ATSTART y PCRE_ANCHORED establecidas. *
* La primera de ellas indica a PCRE que una cadena vacía al principio del *
* sujeto no es una coincidencia válida; deben probarse otras posibilidades. La *
* segunda bandera restringe PCRE a un intento de coincidencia en la posición inicial de la cadena. *
* Si esta coincidencia tiene éxito, se ha encontrado una alternativa a la coincidencia de cadena vacía, *
* y podemos imprimirla y continuar el bucle, avanzando la longitud de lo que se encontró. Si esta coincidencia NO *
* tiene éxito, todavía permanecemos en el bucle, avanzando solo un carácter. *
* En modo UTF-8, que se puede establecer mediante (*UTF8) en el patrón, esto puede ser *
* más de un byte. *
* *
* Sin embargo, hay una complicación relacionada con los saltos de línea. Cuando la *
* convención de salto de línea es tal que CRLF es un salto de línea válido, debemos *
* avanzar dos caracteres en lugar de uno. La convención de salto de línea puede *
* establecerse en la expresión regular mediante (*CR), etc.; si no, debemos encontrar la configuración predeterminada. *
*************************************************************************/
if (!find_all)     /* Comprobar si está la opción -g */
  {
  pcre_free(re);   /* Liberar la memoria utilizada para el patrón compilado */
  return 0;        /* Terminar a menos que se haya dado -g */
  }
/* Antes de ejecutar el bucle, comprobar si está en UTF-8 y si CRLF es una secuencia de salto de línea válida
Por el momento, encontrar las opciones con las que se compiló la expresión regular; extraer
el estado UTF-8, y enmascarar todas las opciones excepto las de salto de línea. */
(void)pcre_fullinfo(re, NULL, PCRE_INFO_OPTIONS, &option_bits);
utf8 = option_bits & PCRE_UTF8;
option_bits &= PCRE_NEWLINE_CR|PCRE_NEWLINE_LF|PCRE_NEWLINE_CRLF|
               PCRE_NEWLINE_ANY|PCRE_NEWLINE_ANYCRLF;
/* Si no se establecieron opciones de salto de línea, encontrar la convención de salto de línea predeterminada de la
configuración de compilación. */
if (option_bits == 0)
  {
  int d;
  (void)pcre_config(PCRE_CONFIG_NEWLINE, &d);
  /* Nótese que estos valores son siempre los de ASCII, incluso en
  entornos EBCDIC. CR = 13, NL = 10. */
  option_bits = (d == 13)? PCRE_NEWLINE_CR :
          (d == 10)? PCRE_NEWLINE_LF :
          (d == (13<<8 | 10))? PCRE_NEWLINE_CRLF :
          (d == -2)? PCRE_NEWLINE_ANYCRLF :
          (d == -1)? PCRE_NEWLINE_ANY : 0;
  }
/* Ver si CRLF es una secuencia de salto de línea válida. */
crlf_is_newline =
     option_bits == PCRE_NEWLINE_ANY ||
     option_bits == PCRE_NEWLINE_CRLF ||
     option_bits == PCRE_NEWLINE_ANYCRLF;
/* Bucle para la segunda y subsiguientes coincidencias */
for (;;)
  {
  int options = 0;                 /* Normalmente sin opciones */
  int start_offset = ovector[1];   /* Empezar al final de la coincidencia anterior */
  /* Si la coincidencia anterior fue de una cadena vacía, hemos terminado si estamos
  al final del sujeto. De lo contrario, organizar para ejecutar otra coincidencia en el
  mismo punto para ver si se puede encontrar una coincidencia no vacía. */
  if (ovector[0] == ovector[1])
    {
    if (ovector[0] == subject_length) break;
    options = PCRE_NOTEMPTY_ATSTART | PCRE_ANCHORED;
    }
  /* Ejecutar la siguiente operación de coincidencia */
  rc = pcre_exec(
    re,                   /* el patrón compilado */
    NULL,                 /* sin datos adicionales - no hemos estudiado el patrón */
    subject,              /* la cadena del sujeto */
    subject_length,       /* la longitud del sujeto */
    start_offset,         /* offset de inicio en el sujeto */
    options,              /* opciones */
    ovector,              /* vector de salida para información de subcadena */
    OVECCOUNT);           /* número de elementos en el vector de salida */
  /* Esta vez, un resultado de NOMATCH no es un error. Si el valor en "options"
  es cero, solo significa que hemos encontrado todas las coincidencias posibles, por lo que el bucle termina.
  De lo contrario, significa que no hemos logrado encontrar una coincidencia de cadena no vacía en un
  punto donde hubo una coincidencia de cadena vacía anterior. En este caso, hacemos lo que
  hace Perl: avanzamos la posición de coincidencia en un carácter y continuamos. Hacemos esto
  estableciendo el desplazamiento "fin de la coincidencia anterior", porque eso se recoge al principio del bucle como el punto en el que comenzar de nuevo.
  Hay dos complicaciones: (a) Cuando CRLF es un salto de línea válido, y
  la posición actual está justo antes de él, avanzar un byte adicional. (b)
  De lo contrario, debemos asegurarnos de saltar un carácter UTF-8 completo si estamos en
  modo UTF-8. */
  if (rc == PCRE_ERROR_NOMATCH)
    {
    if (options == 0) break;                    /* Todas las coincidencias encontradas */
    ovector[1] = start_offset + 1;              /* Avanzar un byte */
    if (crlf_is_newline &&                      /* Si CRLF es salto de línea y */
        start_offset < subject_length - 1 &&    /* estamos en CRLF, */
        subject[start_offset] == '\r' &&
        subject[start_offset + 1] == '\n')
      ovector[1] += 1;                          /* Avanzar uno más. */
    else if (utf8)                              /* De lo contrario, asegurar que */
      {                                         /* avanzamos un carácter UTF-8 completo */
      while (ovector[1] < subject_length)       /* . */
        {
        if ((subject[ovector[1]] & 0xc0) != 0x80) break;
        ovector[1] += 1;
        }
      }
    continue;    /* Volver a empezar el bucle */
    }
  /* Otros errores de coincidencia no son recuperables. */
  if (rc < 0)
    {
    printf("Error de coincidencia %d\n", rc);
    pcre_free(re);    /* Liberar la memoria utilizada para el patrón compilado */
    return 1;
    }
  /* Coincidencia exitosa */
  printf("\nCoincidencia exitosa de nuevo en el offset %d\n", ovector[0]);
  /* La coincidencia tuvo éxito, pero el vector de salida no era lo suficientemente grande. */
  if (rc == 0)
    {
    rc = OVECCOUNT/3;
    printf("ovector solo tiene espacio para %d subcadenas capturadas\n", rc - 1);
    }
  /* Ahora imprimir la subcadena capturada si existe. */
  for (i = 1; i < rc; i++)
    {
    printf("  $%d: %.*s\n", i, ovector[2*i+1] - ovector[2*i],
      subject + ovector[2*i]);
    }
  /* Si la coincidencia fue de una cadena vacía, debemos salir aquí para evitar un bucle infinito.
  Esto se maneja de forma especial al principio del bucle. */
  if (ovector[0] == ovector[1]) break;
  }
pcre_free(re); /* Liberar la memoria utilizada para el patrón compilado */
return 0;
}
    }
  /* Igual que antes, muestra las subcadenas almacenadas en el vector de salida por número, y luego
  también cualquier subcadena con nombre. */
  for (i = 0; i < rc; i++)
    {
    char *substring_start = subject + ovector[2*i];
    int substring_length = ovector[2*i+1] - ovector[2*i];
    printf("%2d: %.*s\n", i, substring_length, substring_start);
    }
  if (namecount <= 0) printf("No hay subcadenas con nombre\n"); else
    {
    unsigned char *tabptr = name_table;
    printf("Subcadenas con nombre\n");
    for (i = 0; i < namecount; i++)
      {
      int n = (tabptr[0] << 8) | tabptr[1];
      printf("(%d) %*s: %.*s\n", n, name_entry_size - 3, tabptr + 2,
        ovector[2*n+1] - ovector[2*n], subject + ovector[2*n]);
      tabptr += name_entry_size;
      }
    }
  }      /* Fin del bucle para encontrar la segunda y subsiguientes coincidencias */
printf("\n");
pcre_free(re);       /* Libera la memoria utilizada para el patrón compilado */
return 0;
}
/* Fin de pcredemo.c */
