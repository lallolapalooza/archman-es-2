ts_read(3)

Paquete: tslib

Licencias: LGPL-2.1-only

TS_READ(3) tslib TS_READ(3)

NOMBRE

ts_read, ts_read_raw, ts_read_mt, ts_read_raw_mt - leer muestras táctiles de tslib

SINOPSIS

#include <tslib.h>

int ts_read(struct tsdev *dev, struct ts_sample *samp, int nr);

int ts_read_raw(struct tsdev *dev, struct ts_sample *samp, int nr);

int ts_read_mt(struct tsdev *dev, struct ts_sample_mt **samp, int slots, int nr);

int ts_read_raw_mt(struct tsdev *dev, struct ts_sample_mt **samp, int slots, int nr);

DESCRIPCIÓN

ts_read() lee nr muestras de entrada con los filtros de tslib aplicados. struct ts_sample se define de la siguiente manera:

struct ts_sample {
        int             x;
        int             y;
        unsigned int    pressure;
        struct timeval  tv;
};

ts_read_mt() lee nr * slots muestras de entrada con los filtros de tslib aplicados. struct ts_sample_mt se define de la siguiente manera:

struct ts_sample_mt {
        /* códigos de evento ABS_MT_* más recientes.
         * ver linux/input.h para descripciones */
        int             x;
        int             y;
        unsigned int    pressure;
        int             slot;
        int             tracking_id;
        int             tool_type;
        int             tool_x;
        int             tool_y;
        unsigned int    touch_major;
        unsigned int    width_major;
        unsigned int    touch_minor;
        unsigned int    width_minor;
        int             orientation;
        int             distance;
        int             blob_id;
        struct timeval  tv;
        /* estado BTN_TOUCH */
        short           pen_down;
        /* el bit TSLIB_MT_VALID se establece en valid si esta muestra
         * contiene datos nuevos;
	 * valid se establece a 0 en caso contrario */
        short           valid;
};

El usuario debe proporcionar la cantidad de memoria descrita en nr y slots para contenerlas.

ts_read_raw() y ts_read_raw_mt() hacen lo mismo sin aplicar los filtros de tslib.

VALOR DE RETORNO

Se devuelve el número de muestras leídas realmente. Especialmente cuando se abre en modo no bloqueante, ver ts_setup(), puede ser menor que lo solicitado en la llamada. En caso de fallo, se devuelve un número de error negativo.

EJEMPLO

El siguiente programa lee continuamente muestras de entrada multitáctiles de tslib e imprime los valores de ranura y posición en la salida estándar a medida que se toca la pantalla táctil.

#include <stdio.h>
#include <stdlib.h>
#include <fcntl.h>
#include <tslib.h>
#define READ_SAMPLES 1
#define MAX_SLOTS 5
int main(int argc, char **argv)
{
        struct tsdev *ts;
        struct ts_sample_mt **samp_mt = NULL;
        int i, j;
        int ret;
        ts = ts_setup(NULL, 0);
        if (!ts)
                return -1;
        samp_mt = malloc(READ_SAMPLES * sizeof(struct ts_sample_mt **));
        if (!samp_mt)
                return -1;
        for (i = 0; i < READ_SAMPLES; i++) {
                samp_mt[i] = calloc(MAX_SLOTS, sizeof(struct ts_sample_mt));
                if (!samp_mt[i])
                        return -1;
        }
        while(1) {
                ret = ts_read_mt(ts, samp_mt, MAX_SLOTS, READ_SAMPLES);
                for (i = 0; i < ret; i++) {
                        printf("muestra nr %d0, i);
                        for (j = 0; i < MAX_SLOTS; j++) {
                                if (!(samp_mt[i][j].valid & TSLIB_MT_VALID))
                                        continue;
                                printf("ranura %d: X:%d Y: %d0,
                                       samp_mt[i][j].slot,
                                       samp_mt[i][j].x,
                                       samp_mt[i][j].y);
                        }
                }
        }
}
