tracefs_event_filter_clear(3)

Paquete: libtracefs

Licencias: GPL-2.0-only, LGPL-2.1-only

LIBTRACEFS(3) libtracefs Manual LIBTRACEFS(3)

NOMBRE

tracefs_filter_string_append, tracefs_filter_string_verify, tracefs_event_filter_apply, tracefs_event_filter_clear - Añadir, verificar y aplicar filtros de eventos

SINOPSIS

#include <tracefs.h>
int tracefs_filter_string_append(struct tep_event *event, char **filter,
                                 struct tracefs_filter type, const char *field,
                                 enum tracefs_synth_compare compare, const char *val);
int tracefs_filter_string_verify(struct tep_event *event, const char *filter, char **err);
int tracefs_event_filter_apply(struct tracefs_instance *instance, struct tep_event *event, const char *filter);
int tracefs_event_filter_clear(struct tracefs_instance *instance, struct tep_event *event);

DESCRIPCIÓN

tracefs_filter_string_append() es una forma de crear y verificar filtros de eventos para un evento dado. Verificará que el campo pertenece al evento y que la opción comparar que se utiliza es válida para el tipo de campo, así como val. Para el tipo que no sea de TRACEFS_FILTER_COMPARE, construirá la cadena lógica y también se asegurará de que la sintaxis sea correcta. Por ejemplo, no hay más paréntesis de cierre que de apertura. Un AND (&&) o OR (||) no está mal colocado, etc.

tracefs_synth_append_start_filter() crea un filtro o se añade a él para el evento de inicio. Dependiendo de type, construirá una cadena de tokens para paréntesis o declaraciones lógicas, o puede añadir una comparación de campo a val basándose en compare.

Si type es: TRACEFS_FILTER_COMPARE - Ver abajo TRACEFS_FILTER_AND - Añadir "&&" al filtro TRACEFS_FILTER_OR - Añadir "||" al filtro TRACEFS_FILTER_NOT - Añadir "!" al filtro TRACEFS_FILTER_OPEN_PAREN - Añadir "(" al filtro TRACEFS_FILTER_CLOSE_PAREN - Añadir ")" al filtro

field, compare y val se ignoran a menos que type sea igual a TRACEFS_FILTER_COMPARE, en cuyo caso se usará _compare para lo siguiente:

TRACEFS_COMPARE_EQ - field == val

TRACEFS_COMPARE_NE - field != val

TRACEFS_COMPARE_GT - field > val

TRACEFS_COMPARE_GE - field >= val

TRACEFS_COMPARE_LT - field < val

TRACEFS_COMPARE_LE - field <= val

TRACEFS_COMPARE_RE - field ~ "val" : donde field es una cadena.

TRACEFS_COMPARE_AND - field & val : donde field es un campo de flags.

tracefs_filter_string_verify() analizará filter para asegurarse de que los campos pertenecen al event y que la sintaxis es correcta. Si hay un error en la sintaxis, y err no es NULL, entonces se asignará con un mensaje de error indicando qué salió mal con el filtro. err debe ser liberado con free().

tracefs_event_filter_apply() aplica la cadena filter dada en el event en la instance dada.

tracefs_event_filter_clear() borra todos los filtros del event en la instance dada.

VALOR_DE_RETORNO

tracefs_filter_string_append() devuelve 0 en caso de éxito y -1 en caso de error.

tracefs_filter_string_verify() devuelve 0 en caso de éxito y -1 en caso de error. si hay un error, y errno no es ENOMEM, entonces err se asigna y contendrá una cadena que describe lo que salió mal con filter. err debe ser liberado con free().

tracefs_event_filter_apply() devuelve 0 en caso de éxito y -1 en caso de error.

tracefs_event_filter_clear() devuelve 0 en caso de éxito y -1 en caso de error.

EJEMPLO

#include <stdlib.h>
#include <stdio.h>
#include <ctype.h>
#include <tracefs.h>
static void usage(char **argv)
{
        fprintf(stderr, "usage: %s [system] event filter\n", argv[0]);
        exit(-1);
}
int main (int argc, char **argv)
{
        struct tep_handle *tep;
        struct tep_event *event;
        const char *system = NULL;
        const char *event_name;
        const char *filter;
        char *new_filter = NULL;
        char *err = NULL;
        int i;
        if (argc < 3)
                usage(argv);
        if (argc < 4) {
                event_name = argv[1];
                filter = argv[2];
        } else {
                system = argv[1];
                event_name = argv[2];
                filter = argv[3];
        }
        /* Load all events from the system */
        tep = tracefs_local_events(NULL);
        if (!tep) {
                perror("tep");
                exit(-1);
        }
        event = tep_find_event_by_name(tep, system, event_name);
        if (!event) {
                fprintf(stderr, "Event %s%s%s not found\n",
                        system ? system : "" , system ? " " : "",
                        event_name);
                exit(-1);
        }
        if (tracefs_filter_string_verify(event, filter, &err) < 0) {
                perror("tracecfs_event_verify_filter");
                if (err)
                        fprintf(stderr, "%s", err);
                free(err);
                exit(-1);
        }
        for (i = 0; filter[i]; i++) {
                char buf[strlen(filter)];
                char *field = NULL;
                char *val = NULL;
                enum tracefs_filter type;
                enum tracefs_compare compare = 0;
                int start_i, n;
                int quote;
                bool backslash;
                while (isspace(filter[i]))
                        i++;
                switch(filter[i]) {
                case '(':
                        type = TRACEFS_FILTER_OPEN_PAREN;
                        break;
                case ')':
                        type = TRACEFS_FILTER_CLOSE_PAREN;
                        break;
                case '!':
                        type = TRACEFS_FILTER_NOT;
                        break;
                case '&':
                        type = TRACEFS_FILTER_AND;
                        i++;
                        break;
                case '|':
                        type = TRACEFS_FILTER_OR;
                        i++;
                        break;
                default:
                        type = TRACEFS_FILTER_COMPARE;
                        while (isspace(filter[i]))
                                i++;
                        start_i = i;
                        for (; filter[i]; i++) {
                                switch(filter[i]) {
                                case 'a' ... 'z':
                                case 'A' ... 'Z':
                                case '0' ... '9':
                                case '_':
                                        continue;
                                }
                                break;
                        }
                        n = i - start_i;
                        field = buf;
                        strncpy(field, filter + start_i, n);
                        field[n++] = '\0';
                        val = buf + n;
                        while (isspace(filter[i]))
                                i++;
                        start_i = i;
                        switch(filter[i++]) {
                        case '>':
                                compare = TRACEFS_COMPARE_GT;
                                if (filter[i] == '=') {
                                        i++;
                                        compare = TRACEFS_COMPARE_GE;
                                }
                                break;
                        case '<':
                                compare = TRACEFS_COMPARE_LT;
                                if (filter[i] == '=') {
                                        i++;
                                        compare = TRACEFS_COMPARE_LE;
                                }
                                break;
                        case '=':
                                compare = TRACEFS_COMPARE_EQ;
                                i++;
                                break;
                        case '!':
                                compare = TRACEFS_COMPARE_NE;
                                i++;
                                break;
                        case '~':
                                compare = TRACEFS_COMPARE_RE;
                                break;
                        case '
&':
                                compare = TRACEFS_COMPARE_AND;
                                break;
                        }
                        while (isspace(filter[i]))
                                i++;
                        quote = 0;
                        backslash = false;
                        start_i = i;
                        for (; filter[i]; i++) {
                                if (quote) {
                                        if (backslash)
                                                backslash = false;
                                        else if (filter[i] == '\\')
                                                backslash = true;
                                        else if (filter[i] == quote)
                                                quote = 0;
                                        continue;
                                }
                                switch(filter[i]) {
                                case '"': case '\'':
                                        quote = filter[i];
                                        continue;
                                case 'a' ... 'z':
                                case 'A' ... 'Z':
                                case '0' ... '9':
                                case '_':
                                        continue;
                                }
                                break;
                        }
                        n = i - start_i;
                        strncpy(val, filter + start_i, n);
                        val[n] = '\0';
                        break;
                }
                n = tracefs_filter_string_append(event, &new_filter, type,
                                                field, compare, val);
                if (n < 0) {
                        fprintf(stderr, "Fallo al crear nuevo filtro:\n'%s'\n",
                                new_filter ? new_filter : "(nulo)");
                        exit(-1);
                }
        }
        if (tracefs_event_filter_apply(NULL, event, new_filter))
                fprintf(stderr, "Fallo al aplicar filtro en evento");
        tep_free(tep);
        printf("Creado nuevo filtro: '%s'\n", new_filter);
        free(new_filter);
        exit(0);
}
