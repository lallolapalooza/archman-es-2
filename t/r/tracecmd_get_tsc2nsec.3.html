tracecmd_get_tsc2nsec(3)

Paquete: trace-cmd

Licencias: GPL-2.0-only, LGPL-2.1-only

LIBTRACECMD(3) Manual de libtracefs LIBTRACECMD(3)

NOMBRE

tracecmd_get_first_ts, tracecmd_add_ts_offset, tracecmd_get_tsc2nsec - Manejar marcas de tiempo de un archivo de rastreo.

SINOPSIS

#include <trace-cmd.h>
unsigned long long tracecmd_get_first_ts(struct tracecmd_input *handle);
void tracecmd_add_ts_offset(struct tracecmd_input *handle, long long offset);
int tracecmd_get_tsc2nsec(struct tracecmd_input *handle, int *mult, int pass[*]shift, unsigned long long *offset);

DESCRIPCIÓN

Este conjunto de APIs se puede usar para leer datos de rastreo de un archivo de rastreo abierto con tracecmd_open()(3), tracecmd_open_fd()(3) o tracecmd_open_head()(3).

La función tracecmd_get_first_ts() devuelve la marca de tiempo del primer registro en el handle.

La función tracecmd_add_ts_offset() añade un desplazamiento a cada uno de los registros en el handle que representa un archivo de rastreo. Esto es útil para asociar dos archivos de rastreo diferentes por su desplazamiento (por ejemplo, un archivo de rastreo de un host y un archivo de rastreo de un invitado que no se sincronizaron al crearse).

La función tracecmd_get_tsc2nsec devuelve los valores de cálculo que convierten la marca de tiempo en bruto a nanosegundos. Los parámetros son punteros al almacenamiento para guardar los valores, o NULL para ignorarlos. El multiplicador se guardará en mult, el valor de desplazamiento se guardará en shift, y el valor de compensación se guardará en offset, si los parámetros correspondientes no son NULL.

VALOR DEVUELTO

La función tracecmd_get_first_ts() devuelve la marca de tiempo del primer registro en un archivo de rastreo para el handle dado.

La función tracecmd_get_tsc2nsec() devuelve 0 si el reloj de rastreo soporta los valores de desplazamiento y -1 en caso contrario. Tenga en cuenta que, si el archivo de rastreo tiene la opción TSC2NSEC, los valores devueltos en los parámetros aún pueden ser válidos incluso si la función devuelve -1. El valor de retorno solo indica si los valores se utilizarán en los cálculos del reloj dado.

EJEMPLO

#include <stdlib.h>
#include <trace-cmd.h>
static int print_events(struct tracecmd_input *handle, struct tep_record *record, int cpu, void *data)
{
        static struct trace_seq seq;
        struct tep_handle *tep = tracecmd_get_tep(handle);
        const char *file = tracecmd_get_private(handle);
        if (!seq.buffer)
                trace_seq_init(&seq);
        trace_seq_reset(&seq);
        trace_seq_printf(&seq, "%s: ", file);
        trace_seq_printf(&seq, "%6.1000d [%03d] %s-%d %s: %s\n",
                        TEP_PRINT_TIME, TEP_PRINT_CPU, TEP_PRINT_COMM, TEP_PRINT_PID,
                        TEP_PRINT_NAME, TEP_PRINT_INFO);
        trace_seq_terminate(&seq);
        trace_seq_do_printf(&seq);
        return 0;
}
int main(int argc, char **argv)
{
        struct tracecmd_input **handles = NULL;
        unsigned long long ts, first_ts = 0;
        unsigned long long offset;
        int multi;
        int shift;
        int nr_handles = 0;
        int ret;
        int i;
        if (argc < 2) {
                printf("uso: %s trace.dat [trace.dat ...]\n", argv[0]);
                exit(-1);
        }
        for (i = 1; i < argc; i++) {
                handles = realloc(handles, sizeof(*handles) * (nr_handles + 1));
                if (!handles)
                        exit(-1);
                handles[nr_handles] = tracecmd_open(argv[i], 0);
                if (!handles[nr_handles])
                        exit(-1);
                ret = tracecmd_get_tsc2nsec(handles[nr_handles], &multi, &shift, &offset);
                if (!ret)
                        printf(" %s tiene cálculos tsc2nsec de mult:%d shift:%d offset:%lld\n",
                                argv[i], multi, shift, offset);
                tracecmd_set_private(handles[nr_handles], argv[i]);
                ts = tracecmd_get_first_ts(handles[nr_handles]);
                if (!first_ts || ts < first_ts)
                        first_ts = ts;
                nr_handles++;
        }
        /* Establecer la marca de tiempo para que comience en el primer registro encontrado */
        for (i = 0; i < nr_handles; i++)
                tracecmd_add_ts_offset(handles[i], -first_ts);
        tracecmd_iterate_events_multi(handles, nr_handles, print_events, NULL);
        for (i = 0; i < nr_handles; i++)
                tracecmd_close(handles[i]);
        free(handles);
}
