tracefs_hist_reset(3)

Paquete: libtracefs

Licencias: GPL-2.0-only, LGPL-2.1-only

LIBTRACEFS(3) Manual de libtracefs LIBTRACEFS(3)

NOMBRE

tracefs_hist_start, tracefs_hist_destroy, tracefs_hist_pause, tracefs_hist_continue, tracefs_hist_reset - Pausar, continuar o borrar un histrograma existente

SINOPSIS

#include <tracefs.h>
int tracefs_hist_start(struct tracefs_instance *instance, struct tracefs_hist *hist);
int tracefs_hist_destroy(struct tracefs_instance *instance, struct tracefs_hist *hist);
int tracefs_hist_pause(struct tracefs_instance *instance, struct tracefs_hist *hist);
int tracefs_hist_continue(struct tracefs_instance *instance, struct tracefs_hist *hist);
int tracefs_hist_reset(struct tracefs_instance *instance, struct tracefs_hist *hist);

DESCRIPCIÓN

tracefs_hist_start() se llama para iniciar realmente el histrograma hist. instance es la instancia en la que se iniciará el histrograma, NULL si debe iniciarse en el nivel superior.

tracefs_hist_pause() se llama para pausar el histrograma hist. instance es la instancia en la que se pausará el histrograma, NULL si está en el nivel superior.

tracefs_hist_continue() se llama para continuar un histrograma pausado hist. instance es la instancia para continuar el histrograma, NULL si está en el nivel superior.

tracefs_hist_reset() se llama para borrar / restablecer el histrograma hist. instance es la instancia para borrar el histrograma, NULL si está en el nivel superior.

tracefs_hist_destroy() se llama para eliminar el histrograma donde ya no existirá. instance es la instancia de la que se eliminará el histrograma, NULL si está en el nivel superior.

VALOR DE RETORNO

Todas las funciones devuelven cero en caso de éxito o -1 en caso de error.

EJEMPLO

#include <stdlib.h>
#include <tracefs.h>
enum commands {
        START,
        PAUSE,
        CONT,
        RESET,
        DELETE,
        SHOW,
};
int main (int argc, char **argv, char **env)
{
        struct tracefs_instance *instance;
        struct tracefs_hist *hist;
        struct tep_handle *tep;
        enum commands cmd;
        char *cmd_str;
        int ret;
        if (argc < 2) {
                fprintf(stderr, "uso: %s comando\n", argv[0]);
                exit(-1);
        }
        cmd_str = argv[1];
        if (!strcmp(cmd_str, "start"))
                cmd = START;
        else if (!strcmp(cmd_str, "pause"))
                cmd = PAUSE;
        else if (!strcmp(cmd_str, "cont"))
                cmd = CONT;
        else if (!strcmp(cmd_str, "reset"))
                cmd = RESET;
        else if (!strcmp(cmd_str, "delete"))
                cmd = DELETE;
        else if (!strcmp(cmd_str, "show"))
                cmd = SHOW;
        else {
                fprintf(stderr, "Comando desconocido %s\n", cmd_str);
                exit(-1);
        }
        tep = tracefs_local_events(NULL);
        if (!tep) {
                perror("Leyendo tracefs");
                exit(-1);
        }
        instance = tracefs_instance_create("hist_test");
        if (!instance) {
                fprintf(stderr, "Fallo al crear instancia\n");
                exit(-1);
        }
        hist = tracefs_hist_alloc_2d(tep, "kmem", "kmalloc",
                                     "call_site",TRACEFS_HIST_KEY_SYM,
                                     "bytes_req", 0);
        if (!hist) {
                fprintf(stderr, "Fallo al crear histrograma\n");
                exit(-1);
        }
        ret = tracefs_hist_add_value(hist, "bytes_alloc");
        ret |= tracefs_hist_add_sort_key(hist, "bytes_req");
        ret |= tracefs_hist_add_sort_key(hist, "bytes_alloc");
        ret |= tracefs_hist_sort_key_direction(hist, "bytes_alloc",
                                               TRACEFS_HIST_SORT_DESCENDING);
        if (ret) {
                fprintf(stderr, "Fallo al modificar el histrograma\n");
                exit(-1);
        }
        tracefs_error_clear(instance);
        switch (cmd) {
        case START:
                ret = tracefs_hist_start(instance, hist);
                if (ret) {
                        char *err = tracefs_error_last(instance);
                        if (err)
                                fprintf(stderr, "\n%s\n", err);
                }
                break;
        case PAUSE:
                ret = tracefs_hist_pause(instance, hist);
                break;
        case CONT:
                ret = tracefs_hist_continue(instance, hist);
                break;
        case RESET:
                ret = tracefs_hist_reset(instance, hist);
                break;
        case DELETE:
                ret = tracefs_hist_destroy(instance, hist);
                break;
        case SHOW: {
                char *content;
                content = tracefs_event_file_read(instance, "kmem", "kmalloc",
                                                  "hist", NULL);
                ret = content ? 0 : -1;
                if (content) {
                        printf("%s\n", content);
                        free(content);
                }
                break;
        }
        }
        if (ret)
                fprintf(stderr, "Fallo: comando\n");
        exit(ret);
}

NAME

tracefs_hist_reset - reset all trace ring buffers

SYNOPSIS

#include <tracefs.h>

int tracefs_hist_reset(struct tracefs_session *session);

DESCRIPCIÓN

The tracefs_hist_reset() function resets all trace ring buffers. This function clears all the data stored in the buffers, making them ready for new trace events.

VALOR DE RETORNO

On success, tracefs_hist_reset() returns 0. On failure, a negative error code is returned.

EJEMPLO

#include <stdio.h>
#include <stdlib.h>
#include <tracefs.h>

int main(void) {
    struct tracefs_session *session;
    int ret;

    session = tracefs_session_open(NULL);
    if (!session) {
        perror("tracefs_session_open");
        return EXIT_FAILURE;
    }

    ret = tracefs_hist_reset(session);
    if (ret < 0) {
        fprintf(stderr, "Failed to reset trace buffers: %s\n", tracefs_error(ret));
        tracefs_session_close(session);
        return EXIT_FAILURE;
    }

    printf("Trace buffers reset successfully.\n");

    tracefs_session_close(session);
    return EXIT_SUCCESS;
}
