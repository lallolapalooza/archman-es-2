tracefs_iterate_raw_events(3)

Paquete: libtracefs

NOMBRE

tracefs_iterate_raw_events, tracefs_iterate_stop, tracefs_follow_event, tracefs_follow_missed_events, tracefs_follow_event_clear, tracefs_follow_missed_events_clear, tracefs_iterate_snapshot_events - Iterar sobre eventos en el búfer de anillo

                                          int, void *),
                          void *callback_data);

                          int (*callback)(struct tep_event *,
                                          struct tep_record *,
                                          int, void *),
                          void *callback_data);

                          const char *system, const char *event_name);

DESCRIPCIÓN

Iterador de trazas sobre eventos brutos.

La función tracefs_iterate_raw_events() leerá los búferes de datos brutos de tracefs y llamará a la función callback especificada para cada evento que encuentre. Los eventos se iteran en orden ordenado: el más antiguo primero. Se requiere un manejador tep inicializado (Ver tracefs_local_events(3)). Si instance es NULL, se utilizará el búfer tracefs de nivel superior, de lo contrario se leerá el búfer para la instance correspondiente. Para filtrar solo en un subconjunto de CPUs, cpus y cpu_size pueden configurarse para llamar a callback solo con eventos que ocurrieron en las CPUs especificadas, de lo contrario, si cpus es NULL, la función callback se llamará para todos los eventos y cpu_size se ignorará. La función callback se llamará con los siguientes parámetros: Un puntero a una struct tep_event que corresponde al tipo de evento del registro; El registro que representa el evento; La CPU en la que ocurrió el evento; y un puntero al callback_context especificado por el usuario. Si el callback devuelve un valor distinto de cero, la iteración se detiene.

La función tracefs_iterate_snapshot_events() funciona igual que tracefs_iterate_raw_events() excepto que opera en el búfer de instantánea.

Utilice tracefs_iterate_stop() para forzar la detención de una ejecución de tracefs_iterate_raw_events(). Esto puede llamarse desde un callback que es llamado por el iterador (aunque devolver un valor distinto de cero lo detendrá), o desde otro hilo.

La función tracefs_follow_event() se utiliza con tracefs_iterate_raw_events(), pero en lugar de llamar al callback para cada evento, solo se llama para el system / event_name especificado a la función. El callback es el mismo que para tracefs_iterate_raw_events(), y el callback_context pasado se pasará también al callback. Nota: si devuelve algo distinto de 0, detendrá el bucle antes de que se llame al callback de tracefs_iterate_raw_events().

La función tracefs_follow_missed_events() llamará al callback cuando se detecten eventos omitidos. Establecerá el parámetro record del callback al registro que vino después de los eventos omitidos y event será del tipo de evento que es el record. cpu se establecerá a la CPU que omitió los eventos, y callback_data será el contenido que se pasó a la función.

La función tracefs_follow_event_clear() eliminará los seguidores de instance que coincidan con system y event_name. Si system y event_name son ambos NULL, eliminará todos los seguidores de eventos asociados a instance. Si solo system es NULL, eliminará todos los seguidores que siguen eventos que coinciden con event_name. Si solo event_name es NULL, eliminará todos los seguidores que están adjuntos a eventos que forman parte de un sistema que coincide con system.

La función tracefs_follow_missed_events_clear() eliminará todos los seguidores de eventos omitidos.

VALOR DE RETORNO

La función tracefs_iterate_raw_events() devuelve -1 en caso de error o 0 en caso contrario.

Tanto tracefs_follow_event_clear() como tracefs_follow_missed_events_clear() devuelven 0 en caso de éxito y -1 en caso de error, o si no encontró seguidores que coincidieran y debieran ser eliminados.

EJEMPLO

#include <unistd.h>
#include <tracefs.h>
#include <stdbool.h>
#include <signal.h>
struct my_struct {
        bool            stopped;
};
#define MAX_COUNT 500000
static int counter;
static int callback(struct tep_event *event, struct tep_record *record,
                    int cpu, void *data)
{
        struct my_struct *my_data = data;
        static struct trace_seq seq;
        if (counter++ > MAX_COUNT) {
                my_data->stopped = true;
                return 1;
        }
        if (!seq.buffer)
                trace_seq_init(&seq);
        tep_print_event(event->tep, &seq, record, "%16s-%-5d [%03d] %6.1000d %s: %s\n",
                        TEP_PRINT_COMM, TEP_PRINT_PID, TEP_PRINT_CPU,
                        TEP_PRINT_TIME, TEP_PRINT_NAME, TEP_PRINT_INFO);
        trace_seq_terminate(&seq);
        trace_seq_do_printf(&seq);
        trace_seq_reset(&seq);
        return 0;
}
static int sched_callback(struct tep_event *event, struct tep_record *record,
                          int cpu, void *data)
{
        static struct tep_format_field *prev_pid;
        static struct tep_format_field *next_pid;
        unsigned long long pid;
        int this_pid = *(int *)data;
        if (!prev_pid) {
                prev_pid = tep_find_field(event, "prev_pid");
                next_pid = tep_find_field(event, "next_pid");
                if (!prev_pid || !next_pid) {
                        fprintf(stderr, "No pid fields??\n");
                        return -1;
                }
        }
        tep_read_number_field(prev_pid, record->data, &pid);
        if (pid == this_pid)
                printf("WE ARE LEAVING!\n");
        tep_read_number_field(next_pid, record->data, &pid);
        if (pid == this_pid)
                printf("WE ARE ARRIVING!\n");
        return 0;
}
static int missed_callback(struct tep_event *event, struct tep_record *record,
                           int cpu, void *data)
{
        printf("OOPS! cpu %d dropped ", cpu);
        if (record->missed_events > 0)
                printf("%lld ", record->missed_events);
        printf("events\n");
        return 0;
}

        my_data.stopped = true;
}

        tracefs_event_enable(instance, NULL, NULL);
        sleep(1);
