tracefs_snapshot_free(3)

Paquete: libtracefs

Licencias: GPL-2.0-only, LGPL-2.1-only

LIBTRACEFS(3) Manual de libtracefs LIBTRACEFS(3)

NOMBRE

tracefs_snapshot_snap, tracefs_snapshot_clear, tracefs_snapshot_free - API para crear, borrar y leer instantáneas

SINOPSIS

#include <tracefs.h>
int tracefs_snapshot_snap(struct tracefs_instance *instance);
int tracefs_snapshot_clear(struct tracefs_instance *instance);
int tracefs_snapshot_free(struct tracefs_instance *instance);

DESCRIPCIÓN

El rastreo del kernel de Linux proporciona una funcionalidad de "instantánea". El kernel tiene dos búferes de anillo. Uno al que escribe el sistema de rastreo y otro que es el búfer de "instantánea". Cuando se toma una instantánea, los dos búferes se intercambian, y el búfer de instantánea actual se convierte en el que se está escribiendo, y el búfer que se estaba escribiendo se convierte en la instantánea guardada.

Nota: el búfer de instantánea se asigna la primera vez que se toma, por lo que es mejor tomar una instantánea al principio antes de que se necesite para que se asigne y una instantánea esté lista. Luego, la instantánea ocurrirá inmediatamente.

tracefs_snapshot_snap() asignará (si aún no ha sido asignado) el búfer de instantánea y luego tomará una "instantánea" (intercambiará el búfer principal que se está escribiendo con el búfer de instantánea asignado). Hará esto al búfer de instancia dado o a la instancia superior si instance es NULL.

tracefs_snapshot_clear() borrará el contenido del búfer de instantánea para la instancia dada o la instancia de nivel superior si instance es NULL.

tracefs_snapshot_free() liberará la instantánea asignada para la instancia dada o la instancia de nivel superior si instance es NULL. Es decir, si se realiza otra llamada a tracefs_snapshot_snap() después de esta, deberá asignar el búfer de instantánea nuevamente antes de poder tomar una instantánea. Esta función debe usarse para liberar la memoria del kernel utilizada por el búfer de instantánea cuando ya no se necesite.

VALOR DE RETORNO

tracefs_snapshot_snap(), tracefs_snapshot_clear() y tracefs_snapshot_free() devuelven 0 en caso de éxito y -1 en caso de error.

EJEMPLO

#include <stdlib.h>
#include <stdio.h>
#include <tracefs.h>
static int callback(struct tep_event *event, struct tep_record *record, int cpu, void *data)
{
        static struct trace_seq seq;
        struct tep_handle *tep = event->tep;
        if (!seq.buffer)
                trace_seq_init(&seq);
        trace_seq_reset(&seq);
        tep_print_event(tep, &seq, record, "[%03d] %s-%d %6.1000d\t%s: %s\n",
                                TEP_PRINT_CPU,
                                TEP_PRINT_COMM,
                                TEP_PRINT_PID,
                                TEP_PRINT_TIME,
                                TEP_PRINT_NAME,
                                TEP_PRINT_INFO);
        trace_seq_do_printf(&seq);
        return 0;
}
int main (int argc, char **argv)
{
        struct tracefs_instance *instance;
        struct tep_handle *tep;
        char *line = NULL;
        size_t len = 0;
        int ret;
        instance = tracefs_instance_create("mi_instantanea");
        if (!instance) {
                perror("creando instancia");
                exit(-1);
        }
        tep = tracefs_local_events(NULL);
        if (!tep) {
                perror("leyendo formatos de evento");
                goto out;
        }
        /* Asegurarse de que el búfer de instantánea esté asignado */
        ret = tracefs_snapshot_snap(instance);
        if (ret < 0)
                goto out;
        ret = tracefs_event_enable(instance, "sched", NULL);
        if (ret < 0) {
                perror("habilitando evento");
                goto out;
        }
        for (;;) {
                printf("¡Presiona Enter sin texto para tomar una instantánea!\n");
                printf("Introduce cualquier texto para mostrar la instantánea\n");
                printf("Introduce 'quit' para salir\n");
                getline(&line, &len, stdin);
                ret = tracefs_snapshot_snap(instance);
                if (ret < 0) {
                        perror("tomando instantánea");
                        goto out;
                }
                if (!line)
                        break;
                if (strlen(line) < 2)
                        continue;
                if (strncmp(line, "quit", 4) == 0)
                        break;
                tracefs_iterate_snapshot_events(tep, instance, NULL, 0, callback, NULL);
        }
        free(line);
        tracefs_instance_clear(instance);
 out:
        tracefs_snapshot_free(instance);
        tracefs_event_disable(instance, "sched", NULL);
        tracefs_instance_destroy(instance);
        tracefs_instance_free(instance);
        exit(0);
}

tracefs_snapshot_free(3)

tracefs_snapshot_free - Libera un objeto snapshot de tracefs.

SINOPSIS

#include <tracefs.h>

void tracefs_snapshot_free(struct tracefs_snapshot *snapshot);

DESCRIPCIÓN

La función tracefs_snapshot_free libera la memoria asignada para el objeto struct tracefs_snapshot proporcionado. Este objeto se obtiene típicamente de una llamada a tracefs_snapshot_create().

ARGUMENTOS

  • snapshot: Un puntero al objeto struct tracefs_snapshot que se va a liberar. Si el puntero es NULL, la función no hace nada.

VALOR DE RETORNO

Ninguno.

EJEMPLO

#include <stdio.h>
#include <tracefs.h>

int main() {
    struct tracefs_snapshot *snapshot;

    // Intentar crear un snapshot. En un entorno real, esto podría fallar.
    snapshot = tracefs_snapshot_create();
    if (!snapshot) {
        fprintf(stderr, "Error al crear el snapshot.\n");
        return 1;
    }

    printf("Snapshot creado con éxito.\n");

    // ... hacer algo con el snapshot ...

    // Liberar el snapshot
    tracefs_snapshot_free(snapshot);
    printf("Snapshot liberado.\n");

    return 0;
}
