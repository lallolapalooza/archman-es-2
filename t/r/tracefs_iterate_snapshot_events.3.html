tracefs_iterate_snapshot_events(3)

LIBTRACEFS(3) Manual de libtracefs LIBTRACEFS(3)

NOMBRE

tracefs_iterate_raw_events, tracefs_iterate_stop, tracefs_follow_event, tracefs_follow_missed_events, tracefs_follow_event_clear, tracefs_follow_missed_events_clear, tracefs_iterate_snapshot_events - Iterar sobre eventos en el búfer de anillo

SINOPSIS

#include <tracefs.h>
int tracefs_iterate_raw_events(struct tep_handle *tep, struct tracefs_instance *instance,
                                 cpu_set_t *cpus, int cpu_size,
                                 int (*callback)(struct tep_event *, struct tep_record *, int, void *),
                                 void *callback_context);
void tracefs_iterate_stop(struct tracefs_instance *instance);
int tracefs_follow_event(struct tep_handle *tep, struct tracefs_instance *instance,
                          const char *system, const char *event_name,
                          int (*callback)(struct tep_event *,
                                          struct tep_record *,
                                          int, void *),
                          void *callback_data);
int tracefs_follow_missed_events(struct tracefs_instance *instance,
                          int (*callback)(struct tep_event *,
                                          struct tep_record *,
                                          int, void *),
                          void *callback_data);
int tracefs_follow_event_clear(struct tracefs_instance *instance,
                          const char *system, const char *event_name);
int tracefs_follow_missed_events_clear(struct tracefs_instance *instance);
int tracefs_iterate_snapshot_events(struct tep_handle *tep, struct tracefs_instance *instance,
                                 cpu_set_t *cpus, int cpu_size,
                                 int (*callback)(struct tep_event *, struct tep_record *, int, void *),
                                 void *callback_context);

DESCRIPCIÓN

Iterador de trazas sobre eventos crudos.

La función tracefs_iterate_raw_events() leerá los búferes de datos crudos de tracefs y llamará a la función callback especificada para cada evento que encuentre. Los eventos se iteran en orden ordenado: el más antiguo primero. Se requiere un manejador tep inicializado (ver tracefs_local_events(3)). Si instance es NULL, se usará el búfer tracefs de nivel superior, de lo contrario se leerá el búfer para la instance correspondiente. Para filtrar solo en un subconjunto de CPUs, cpus y cpu_size pueden establecerse para llamar a callback solo con eventos que ocurrieron en las CPUs especificadas, de lo contrario, si cpus es NULL, la función callback será llamada para todos los eventos y cpu_size se ignorará. La función callback será llamada con los siguientes parámetros: un puntero a una struct tep_event que corresponde al tipo de evento del registro; el registro que representa el evento; la CPU en la que ocurrió el evento; y un puntero al callback_context especificado por el usuario. Si el callback devuelve un valor distinto de cero, la iteración se detiene.

La función tracefs_iterate_snapshot_events() funciona igual que tracefs_iterate_raw_events() excepto que funciona en el búfer de instantánea.

Use tracefs_iterate_stop() para forzar la interrupción de una ejecución de tracefs_iterate_raw_events(). Esto se puede llamar desde un callback que es llamado por el iterador (aunque un retorno distinto de cero lo detendrá) o desde otro hilo.

La función tracefs_follow_event() se usa con tracefs_iterate_raw_events() pero en lugar de llamar al callback para cada evento, solo se llama para el system / event_name especificado a la función. El callback es el mismo que para tracefs_iterate_raw_events(), y el callback_context pasado se pasará también al callback. Tenga en cuenta que si devuelve algo distinto de 0, detendrá el bucle antes de que se llame al callback de tracefs_iterate_raw_events().

La función tracefs_follow_missed_events() llamará al callback cuando se detecten eventos perdidos. Establecerá el parámetro record del callback al registro que vino después de los eventos perdidos y event será del tipo de evento que es record. cpu se establecerá a la CPU que perdió los eventos, y callback_data será el contenido que se pasó a la función.

La función tracefs_follow_event_clear() eliminará los seguidores de instance que coincidan con system y event_name. Si system y event_name son ambos NULL, eliminará todos los seguidores de eventos asociados a instance. Si solo system es NULL, eliminará todos los seguidores que siguen eventos que coinciden con event_name. Si solo event_name es NULL, eliminará todos los seguidores que están adjuntos a eventos que forman parte de un sistema que coincide con system.

La función tracefs_follow_missed_events_clear() eliminará todos los seguidores de eventos perdidos.

VALOR DE RETORNO

La función tracefs_iterate_raw_events() devuelve -1 en caso de error o 0 en caso contrario.

Tanto tracefs_follow_event_clear() como tracefs_follow_missed_events_clear() devuelven 0 en caso de éxito y -1 en caso de error, o si no encontró seguidores que coincidieran y debieran ser eliminados.

EJEMPLO

#include <unistd.h>
#include <tracefs.h>
#include <stdbool.h>
#include <signal.h>
struct my_struct {
        bool            stopped;
};
#define MAX_COUNT 500000
static int counter;
static int callback(struct tep_event *event, struct tep_record *record,
                    int cpu, void *data)
{
        struct my_struct *my_data = data;
        static struct trace_seq seq;
        if (counter++ > MAX_COUNT) {
                my_data->stopped = true;
                return 1;
        }
        if (!seq.buffer)
                trace_seq_init(&seq);
        tep_print_event(event->tep, &seq, record, "%16s-%-5d [%03d] %6.1000d %s: %s\n",
                        TEP_PRINT_COMM, TEP_PRINT_PID, TEP_PRINT_CPU,
                        TEP_PRINT_TIME, TEP_PRINT_NAME, TEP_PRINT_INFO);
        trace_seq_terminate(&seq);
        trace_seq_do_printf(&seq);
        trace_seq_reset(&seq);
        return 0;
}
static int sched_callback(struct tep_event *event, struct tep_record *record,
                          int cpu, void *data)
{
        static struct tep_format_field *prev_pid;
        static struct tep_format_field *next_pid;
        unsigned long long pid;
        int this_pid = *(int *)data;
        if (!prev_pid) {
                prev_pid = tep_find_field(event, "prev_pid");
                next_pid = tep_find_field(event, "next_pid");
                if (!prev_pid || !next_pid) {
                        fprintf(stderr, "¿No hay campos de pid?\n");
                        return -1;
                }
        }
        tep_read_number_field(prev_pid, record->data, &pid);
        if (pid == this_pid)
                printf("¡NOS VAMOS!\n");
        tep_read_number_field(next_pid, record->data, &pid);
        if (pid == this_pid)
                printf("¡NOS LLEGAMOS!\n");
        return 0;
}
static int missed_callback(struct tep_event *event, struct tep_record *record,
                           int cpu, void *data)
{
        printf("¡Opa! cpu %d perdió ", cpu);
        if (record->missed_events > 0)
                printf("%lld ", record->missed_events);
        printf("eventos\n");
        return 0;
}
static struct tracefs_instance *instance;
static struct my_struct my_data;
static void sig(int s)
{
        tracefs_iterate_stop(instance);
        my_data.stopped = true;
}
int main (int argc, char **argv, char **env)
{
        struct tep_handle *tep;
        int this_pid = getpid();
        instance = tracefs_instance_create("mi-buffer");
        if (!instance)
                return -1;
        signal(SIGINT, sig);
        tracefs_event_enable(instance, NULL, NULL);
        sleep(1);
        tracefs_event_disable(instance, NULL, NULL);
        tep = tracefs_local_events(NULL);
        tep_load_plugins(tep);
        tracefs_follow_missed_events(instance, missed_callback, NULL);
        tracefs_follow_event(tep, instance, "sched", "sched_switch", sched_callback, &this_pid);
        tracefs_iterate_raw_events(tep, instance, NULL, 0, callback, &my_data);
        /* Nota, la limpieza aquí es para mostrar cómo limpiar todos los seguidores
         * en caso de que tracefs_iterate_raw_events() se llame de nuevo, pero
         * no quiera incluir los
         */
        tracefs_follow_event_clear(instance, NULL, NULL);
        tracefs_follow_missed_events_clear(instance);
        return 0;
}
