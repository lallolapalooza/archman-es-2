tracefs_cpu_buffered_read(3)

Paquete: libtracefs

Licencias: GPL-2.0-only, LGPL-2.1-only

LIBTRACEFS(3) Manual de libtracefs LIBTRACEFS(3)

NOMBRE

tracefs_cpu_read_size, tracefs_cpu_read, tracefs_cpu_buffered_read, tracefs_cpu_write, tracefs_cpu_stop, tracefs_cpu_flush, tracefs_cpu_flush_write, tracefs_cpu_pipe - Lectura de datos de trace_pipe_raw

SINOPSIS

#include <tracefs.h>
int tracefs_cpu_read_size(struct tracefs_cpu *tcpu);
int tracefs_cpu_read(struct tracefs_cpu *tcpu, void *buffer, bool nonblock);
int tracefs_cpu_buffered_read(struct tracefs_cpu *tcpu, void *buffer, bool nonblock);
int tracefs_cpu_write(struct tracefs_cpu *tcpu, int wfd, bool nonblock);
int tracefs_cpu_stop(struct tracefs_cpu *tcpu);
int tracefs_cpu_flush(struct tracefs_cpu *tcpu, void *buffer);
int tracefs_cpu_flush_write(struct tracefs_cpu *tcpu, int wfd);
int tracefs_cpu_pipe(struct tracefs_cpu *tcpu, int wfd, bool nonblock);

DESCRIPCIÓN

Este conjunto de APIs se puede utilizar para leer los datos sin procesar de los archivos trace_pipe_raw en el sistema de archivos tracefs.

tracefs_cpu_read_size() devuelve el tamaño del sub-búfer de trace_pipe_raw. Esto devuelve el tamaño mínimo del búfer que se pasa a las funciones inferiores.

tracefs_cpu_read() lee los archivos trace_pipe_raw asociados con tcpu en buffer. buffer debe ser al menos del tamaño del sub-búfer del búfer circular, que se devuelve por tracefs_cpu_read_size(). Si nonblock está configurado y no hay datos disponibles, devolverá inmediatamente. De lo contrario, dependiendo de cómo se abrió tcpu, bloqueará. Si tcpu se abrió con nonblock configurado, entonces este nonblock no hará ninguna diferencia.

tracefs_cpu_buffered_read() es básicamente lo mismo que tracefs_cpu_read() excepto que utiliza una tubería a través de splice para almacenar en búfer las lecturas. Esto agrupará lecturas, manteniendo la lectura del búfer circular menos intrusiva para el sistema, ya que simplemente leer todo el tiempo puede causar bastante disturbio. Tenga en cuenta que una diferencia entre esto y tracefs_cpu_read() es que solo leerá en páginas de sub-búfer. Si el búfer circular no ha llenado una página, entonces no devolverá nada, incluso con nonblock configurado. Se deben realizar llamadas a tracefs_cpu_flush() para leer el resto del archivo al final del rastreo.

tracefs_cpu_write() enviará datos desde el archivo trace_pipe_raw asociado con tcpu al descriptor de archivo wfd. Si nonblock está configurado, entonces no se bloqueará si no hay nada que escribir. Tenga en cuenta que solo escribirá datos del tamaño del sub-búfer en wfd. Se necesitan llamadas a tracefs_cpu_flush_write() para escribir el resto.

tracefs_cpu_stop() intentará desbloquear una tarea bloqueada en la lectura de tcpu. En núcleos más antiguos, puede que no haga nada para las lecturas de tuberías, ya que los núcleos más antiguos no despiertan a las tareas que esperan en el búfer circular. Devuelve 0 si definitivamente despertó a todos los posibles esperadores, pero devuelve 1 si no está seguro de que funcionó y los esperadores pueden necesitar que se les envíe una señal.

tracefs_cpu_flush() lee el archivo trace_pipe_raw asociado por tcpu y lo coloca en buffer, que debe tener el tamaño del sub-búfer que se recupera con tracefs_cpu_read_size(). Esto debería llamarse al final del rastreo para obtener el resto de los datos. Esta llamada convertirá el descriptor de archivo de trace_pipe_raw al modo no bloqueante.

tracefs_cpu_flush_write() es lo mismo que trace_cpu_flush() excepto que toma un descriptor de archivo wfd para vaciar los datos.

tracefs_cpu_pipe() es similar a tracefs_cpu_write() pero el descriptor de archivo wfd debe ser una tubería. Esta llamada es una optimización de tracefs_cpu_write() que utiliza dos llamadas a splice(2) para conectar el descriptor de archivo trace_pipe_raw con el descriptor de archivo de escritura. splice(2) requiere que uno de los descriptores de archivo pasados sea una tubería. Si la aplicación quiere pasar los datos a una tubería existente, no hay razón para que haya dos llamadas al sistema splice(2) y tracefs_cpu_pipe() puede simplemente usar una llamada a wfd.

VALOR DE RETORNO

tracefs_cpu_open() devuelve un descriptor de struct tracefs_cpu que puede ser utilizado por las otras funciones o NULL en caso de error.

tracefs_cpu_read_size() devuelve el tamaño mínimo de los búferes a utilizar con tracefs_cpu_read(), tracefs_cpu_buffered_read() y tracefs_cpu_flush(). Devuelve negativo en caso de error.

tracefs_cpu_read() devuelve el número de bytes leídos, o negativo en caso de error.

tracefs_cpu_buffered_read() devuelve el número de bytes leídos o negativo en caso de error.

tracefs_cpu_write() devuelve el número de bytes escritos en el archivo o negativo en caso de error.

tracefs_cpu_stop() devuelve cero si se garantiza que los esperadores fueron despertados de esperar la entrada, o devuelve uno si se trata de un núcleo antiguo que no proporciona esa garantía, y puede ser necesario enviar una señal a los esperadores. Devuelve negativo en caso de error.

tracefs_cpu_flush() devuelve el número de bytes leídos o negativo en caso de error.

tracefs_cpu_flush_write() devuelve el número de bytes escritos en el archivo o negativo en caso de error.

EJEMPLO

#define _LARGEFILE64_SOURCE
#include <stdlib.h>
#include <ctype.h>
#include <pthread.h>
#include <unistd.h>
#include <tracefs.h>
struct thread_data {
        struct tracefs_cpu      *tcpu;
        int                     done;
        int                     fd;
};
static void *thread_run(void *arg)
{
        struct thread_data *data = arg;
        struct tracefs_cpu *tcpu = data->tcpu;
        int fd = data->fd;
        int ret;
        while (!data->done) {
                ret = tracefs_cpu_write(tcpu, fd, false);
                printf("wrote %d\n", ret);
        }
        return NULL;
}
int main (int argc, char **argv)
{
        struct tracefs_instance *instance;
        struct thread_data data;
        pthread_t thread;
        char *file;
        int secs = 10;
        int cpu;
        int ret;
        if (argc < 3 || !isdigit(argv[1][0])) {
                printf("usage: %s cpu file_destination [sleep secs]\n\n", argv[0]);
                exit(-1);
        }
        cpu = atoi(argv[1]);
        file = argv[2];
        if (argc > 3)
                secs = atoi(argv[3]);
        instance = tracefs_instance_create("cpu_write");
        if (!instance) {
                perror("create instance");
                exit(-1);
        }
        memset(&data, 0, sizeof(data));
        data.tcpu = tracefs_cpu_open(instance, cpu, 0);
        if (!data.tcpu) {
                perror("Open instance");
                exit(-1);
        }
        data.fd = open(file, O_WRONLY | O_CREAT | O_TRUNC | O_LARGEFILE, 0644);
        if (data.fd < 0) {
                perror(file);
                exit(-1);
        }
        pthread_create(&thread, NULL, thread_run, &data);
        sleep(secs);
        data.done = 1;
        printf("stopping\n");
        ret = tracefs_cpu_stop(data.tcpu);
        printf("joining %d\n", ret);
        pthread_join(thread, NULL);
        tracefs_trace_off(instance);
        do {
                ret = tracefs_cpu_flush_write(data.tcpu, data.fd);
                printf("flushed %d\n", ret);
        } while (ret > 0);
        tracefs_trace_on(instance);
        tracefs_cpu_close(data.tcpu);
        close(data.fd);
        return 0;
}
