LIBTRACEFS(3) Manual de libtracefs LIBTRACEFS(3)

NOMBRE

tracefs_find_cid_pid, tracefs_instance_find_cid_pid, tracefs_time_conversion - funciones auxiliares para manejar el rastreo de invitados

SINOPSIS

#include <tracefs.h>
char *tracefs_find_cid_pid(int cid);
char *tracefs_instance_find_cid_pid(struct tracefs_instance *instance, int cid);
int tracefs_time_conversion(int cpu, int *shift, int *multi, long long *offset);

DESCRIPCIÓN

tracefs_find_cid_pid() utilizará el rastreo para seguir las activaciones (wakeups) de la conexión al cid dado con el fin de encontrar el pid del hilo del invitado que pertenece al cid de vsocket. Luego, leerá el sistema de archivos proc para encontrar el líder del hilo y devolverá el pid del líder del hilo.

tracefs_instance_find_cid_pid() es lo mismo que tracefs_find_cid_pid() pero define la instancia a utilizar para realizar el rastreo. Si es NULL, utilizará el búfer de nivel superior para realizar el rastreo.

tracefs_time_conversion() devolverá los valores utilizados por el kernel para convertir el contador de marcas de tiempo (timestamp) crudo en nanosegundos para la cpu dada. Los punteros para shift, multi y offset pueden ser NULL para ser ignorados, de lo contrario, se establecen con el desplazamiento (shift), el multiplicador y el desplazamiento (offset) respectivamente.

VALOR DE RETORNO

Tanto tracefs_find_cid_pid() como tracefs_instance_find_cid_pid() devolverán el pid del hilo principal del invitado que pertenece al cid, o -1 en caso de error (o si no se encuentra).

EJEMPLO

#include <stdlib.h>
#include <unistd.h>
#include <tracefs.h>
#define MAX_CID         256
static void find_cid(struct tracefs_instance *instance, int cid)
{
        int pid;
        pid = tracefs_instance_find_cid_pid(instance, cid);
        if (pid >= 0)
                printf("%d\t%d\n", cid, pid);
}
static int find_cids(void)
{
        struct tracefs_instance *instance;
        char *name;
        int cid;
        int ret;
        ret = asprintf(&name, "vsock_find-%d\n", getpid());
        if (ret < 0)
                return ret;
        instance = tracefs_instance_create(name);
        free(name);
        if (!instance)
                return -1;
        for (cid = 0; cid < MAX_CID; cid++)
                find_cid(instance, cid);
        tracefs_event_disable(instance, NULL, NULL);
        tracefs_instance_destroy(instance);
        tracefs_instance_free(instance);
        return 0;
}
struct time_info {
        int             shift;
        int             multi;
};
static void show_time_conversion(void)
{
        struct time_info *tinfo;
        int cpus;
        int cpu;
        int ret;
        cpus = sysconf(_SC_NPROCESSORS_CONF);
        tinfo = calloc(cpus, sizeof(*tinfo));
        if (!tinfo)
                exit(-1);
        for (cpu = 0; cpu < cpus; cpu++) {
                ret  = tracefs_time_conversion(cpu,
                                                &tinfo[cpu].shift,
                                                &tinfo[cpu].multi,
                                                NULL);
                if (ret)
                        break;
        }
        if (cpu != cpus) {
                if (!cpu) {
                        perror("tracefs_time_conversion");
                        exit(-1);
                }
                printf("Solo se leyeron %d de %d CPUs", cpu, cpus);
                cpus = cpu + 1;
        }
        /* Comprobar si todos los valores de shift y mult son los mismos */
        for (cpu = 1; cpu < cpus; cpu++) {
                if (tinfo[cpu - 1].shift != tinfo[cpu].shift)
                        break;
                if (tinfo[cpu - 1].multi != tinfo[cpu].multi)
                        break;
        }
        if (cpu == cpus) {
                printf("Todas las CPUs tienen:\n");
                printf(" shift:  %d\n", tinfo[0].shift);
                printf(" multi:  %d\n", tinfo[0].multi);
                printf("\n");
                return;
        }
        for (cpu = 0; cpu < cpus; cpu++) {
                printf("CPU: %d\n", cpu);
                printf(" shift:  %d\n", tinfo[cpu].shift);
                printf(" multi:  %d\n", tinfo[cpu].multi);
                printf("\n");
        }
}
int main(int argc, char *argv[])
{
        show_time_conversion();
        find_cids();
        exit(0);
}

NAME

tracefs_time_conversion - tracefs time conversion functions

SYNOPSIS

#include <tracefs.h>
unsigned long long tracefs_time_to_ns(const char *ts);
char *tracefs_ns_to_time(unsigned long long ns, char *buf, size_t size);

DESCRIPTION

The tracefs filesystem provides timestamps in a variety of formats. These functions provide a simple way to convert between the string format (which is typically microsecond resolution, but can vary) and nanoseconds.

tracefs_time_to_ns() converts the timestamp string {@code ts} to nanoseconds. The string is expected to be a human-readable time string, e.g., "12345.678901". It handles both integer and fractional parts.

tracefs_ns_to_time() converts the nanosecond value {@code ns} to a human-readable time string. The result is stored in {@code buf}, which must have a size of at least {@code size}. The function returns a pointer to the buffer. The resulting string will have microsecond precision.

RETURN VALUE

tracefs_time_to_ns() returns the timestamp in nanoseconds on success. If the conversion fails, it returns 0.

tracefs_ns_to_time() returns a pointer to the buffer {@code buf} on success. If the conversion fails (e.g., due to insufficient buffer size), it returns NULL.

EXAMPLE


                    #include <stdio.h>
                    #include <tracefs.h>

                    int main(void) {
                        const char *ts = "12345.678901";
                        unsigned long long ns;
                        char buf[32];

                        ns = tracefs_time_to_ns(ts);
                        if (ns) {
                            printf("Timestamp %s is %llu ns\n", ts, ns);
                        } else {
                            printf("Failed to convert %s to ns\n", ts);
                        }

                        if (tracefs_ns_to_time(ns, buf, sizeof(buf))) {
                            printf("Nanoseconds %llu is %s\n", ns, buf);
                        } else {
                            printf("Failed to convert %llu ns to time string\n", ns);
                        }

                        return 0;
                    }
