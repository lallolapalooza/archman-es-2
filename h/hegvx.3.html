CHEGVX

Propósito:

!>
!> CHEGVX calcula autovalores seleccionados y, opcionalmente, autovectores
!> de un problema de autovalores generalizado hermítico-definido complejo, de la forma
!> A*x=(lambda)*B*x,  A*Bx=(lambda)*x,  o B*A*x=(lambda)*x. Aquí A y
!> B se suponen hermíticos y B es además positivo definido.
!> Los autovalores y autovectores pueden ser seleccionados especificando un
!> rango de valores o un rango de índices para los autovalores deseados.
!>

Parámetros

ITYPE 
!>          ITYPE es INTEGER
!>          Especifica el tipo de problema a resolver:
!>          = 1:  A*x = (lambda)*B*x
!>          = 2:  A*B*x = (lambda)*x
!>          = 3:  B*A*x = (lambda)*x
!>

JOBZ

!>          JOBZ es CHARACTER*1
!>          = 'N':  Calcula solo autovalores;
!>          = 'V':  Calcula autovalores y autovectores.
!>

RANGE

!>          RANGE es CHARACTER*1
!>          = 'A': se encontrarán todos los autovalores.
!>          = 'V': se encontrarán todos los autovalores en el intervalo semiabierto (VL,VU]
!>                 que se esté buscando.
!>          = 'I': se encontrarán los autovalores del IL-ésimo al IU-ésimo.
!>

UPLO

!>          UPLO es CHARACTER*1
!>          = 'U':  Los triángulos superiores de A y B están almacenados;
!>          = 'L':  Los triángulos inferiores de A y B están almacenados.
!>

N

!>          N es INTEGER
!>          El orden de las matrices A y B.  N >= 0.
!>

A

!>          A es un array COMPLEX, de dimensión (LDA, N)
!>          En la entrada, la matriz hermítica A. Si UPLO = 'U', la
!>          parte superior triangular de N por N de A contiene la
!>          parte superior triangular de la matriz A. Si UPLO = 'L',
!>          la parte inferior triangular de N por N de A contiene
!>          la parte inferior triangular de la matriz A.
!>
!>          En la salida, el triángulo inferior (si UPLO='L') o el triángulo
!>          superior (si UPLO='U') de A, incluyendo la diagonal, se
!>          destruye.
!>

LDA

!>          LDA es INTEGER
!>          La dimensión principal del array A.  LDA >= max(1,N).
!>

B

!>          B es un array COMPLEX, de dimensión (LDB, N)
!>          En la entrada, la matriz hermítica B. Si UPLO = 'U', la
!>          parte superior triangular de N por N de B contiene la
!>          parte superior triangular de la matriz B. Si UPLO = 'L',
!>          la parte inferior triangular de N por N de B contiene
!>          la parte inferior triangular de la matriz B.
!>
!>          En la salida, si INFO <= N, la parte de B que contiene la matriz se

!>

LDB

!>          La dimensión principal del array B.  LDB >= max(1,N).
!>

VL

!>          VL es REAL
!>
!>          Si RANGE='V', el límite inferior del intervalo a
!>          buscar para los autovalores. VL < VU.
!>          No se referencia si RANGE = 'A' o 'I'.
!>

VU

!>          VU es REAL
!>
!>          Si RANGE='V', el límite superior del intervalo a
!>          buscar para los autovalores. VL < VU.
!>          No se referencia si RANGE = 'A' o 'I'.
!>

IL

!>          IL es INTEGER
!>
!>          Si RANGE='I', el índice del
!>          autovalor más pequeño a devolver.
!>          1 <= IL <= IU <= N, si N > 0; IL = 1 e IU = 0 si N = 0.

IU

!>          IU es INTEGER

!>          Si RANGE='I', el índice del
!>          autovalor más grande a devolver.

!>          No se referencia si RANGE = 'A' o 'V'.

!>          ABSTOL es REAL
!>          La tolerancia de error absoluta para los autovalores.
!>          Un autovalor aproximado se acepta como convergido
!>          cuando se determina que yace en un intervalo [a,b]
!>          de anchura menor o igual a

M

subroutine   dsygvx (integer itype, character jobz, character range, character uplo,   integer n, double precision, dimension( lda, * ) a, integer lda, double precision,   dimension( ldb, * ) b, integer ldb, real vl, real vu, integer il, integer   iu, real abstol, integer m, real, dimension( * ) w, double precision,   dimension( ldz, * ) z, integer ldz, double precision, dimension( * ) work,   integer lwork, integer, dimension( * ) iwork, integer, dimension( * ) ifail,   integer info)


DSYGVX


Purpose:




!>
!> DSYGVX computes selected eigenvalues, and optionally, eigenvectors
!> of a real generalized symmetric-definite eigenproblem, of the form
!> A*x=(lambda)*B*x,  A*Bx=(lambda)*x,  or B*A*x=(lambda)*x.  Here A
!> and B are assumed to be symmetric and B is also positive definite.
!> Eigenvalues and eigenvectors can be selected by specifying either a
!> range of values or a range of indices for the desired eigenvalues.
!> 




Parameters


ITYPE

!>          ITYPE is INTEGER
!>          Specifies the problem type to be solved:
!>          = 1:  A*x = (lambda)*B*x
!>          = 2:  A*B*x = (lambda)*x
!>          = 3:  B*A*x = (lambda)*x
!> 


JOBZ


!>          JOBZ is CHARACTER*1
!>          = 'N':  Compute eigenvalues only;
!>          = 'V':  Compute eigenvalues and eigenvectors.
!> 


RANGE


!>          RANGE is CHARACTER*1
!>          = 'A': all eigenvalues will be found.
!>          = 'V': all eigenvalues in the half-open interval (VL,VU]
!>                 will be found.
!>          = 'I': the IL-th through IU-th eigenvalues will be found.
!> 


UPLO


!>          UPLO is CHARACTER*1
!>          = 'U':  Upper triangle of A and B are stored;
!>          = 'L':  Lower triangle of A and B are stored.
!> 


N


!>          N is INTEGER
!>          The order of the matrix pencil (A,B).  N >= 0.
!> 


A


!>          A is DOUBLE PRECISION array, dimension (LDA, N)
!>          On entry, the symmetric matrix A.  If UPLO = 'U', the
!>          leading N-by-N upper triangular part of A contains the
!>          upper triangular part of the matrix A.  If UPLO = 'L',
!>          the leading N-by-N lower triangular part of A contains
!>          the lower triangular part of the matrix A.
!>
!>          On exit, the lower triangle (if UPLO='L') or the upper
!>          triangle (if UPLO='U') of A, including the diagonal, is
!>          destroyed.
!> 


LDA


!>          LDA is INTEGER
!>          The leading dimension of the array A.  LDA >= max(1,N).
!> 


B


!>          B is DOUBLE PRECISION array, dimension (LDB, N)
!>          On entry, the symmetric matrix B.  If UPLO = 'U', the
!>          leading N-by-N upper triangular part of B contains the
!>          upper triangular part of the matrix B.  If UPLO = 'L',
!>          the leading N-by-N lower triangular part of B contains
!>          the lower triangular part of the matrix B.
!>
!>          On exit, if INFO <= N, the part of B containing the matrix is
!>          overwritten by the triangular factor U or L from the Cholesky
!>          factorization B = U**T*U or B = L*L**T.
!> 


LDB


!>          LDB is INTEGER
!>          The leading dimension of the array B.  LDB >= max(1,N).
!> 


VL


!>          VL is REAL
!>          If RANGE='V', the lower bound of the interval to
!>          be searched for eigenvalues. VL < VU.
!>          Not referenced if RANGE = 'A' or 'I'.
!> 


VU


!>          VU is REAL
!>          If RANGE='V', the upper bound of the interval to
!>          be searched for eigenvalues. VL < VU.
!>          Not referenced if RANGE = 'A' or 'I'.
!> 


IL


!>          IL is INTEGER
!>          If RANGE='I', the index of the
!>          smallest eigenvalue to be returned.
!>          1 <= IL <= IU <= N, if N > 0; IL = 1 and IU = 0 if N = 0.
!>          Not referenced if RANGE = 'A' or 'V'.
!> 


IU


!>          IU is INTEGER
!>          If RANGE='I', the index of the
!>          largest eigenvalue to be returned.
!>          1 <= IL <= IU <= N, if N > 0; IL = 1 and IU = 0 if N = 0.
!>          Not referenced if RANGE = 'A' or 'V'.
!> 


ABSTOL


!>          ABSTOL is REAL
!>          The absolute error tolerance for the eigenvalues.
!>          An approximate eigenvalue is accepted as converged
!>          when it is determined to lie in an interval [a,b]
!>          of width less than or equal to
!>
!>          ABSTOL + EPS *   max( |a|,|b| ) ,
!>
!>          where EPS is the machine precision.  If ABSTOL is less than
!>          or equal to zero, then  EPS*|T|  will be used in its place,
!>          where |T| is the 1-norm of the tridiagonal matrix obtained
!>          by reducing C to tridiagonal form, where C is the symmetric
!>          matrix of the standard symmetric problem to which the
!>          generalized problem is transformed.
!>
!>          Eigenvalues will be computed most accurately when ABSTOL is
!>          set to twice the underflow threshold 2*DLAMCH('S'), not zero.
!>          If this routine returns with INFO>0, indicating that some
!>          eigenvectors did not converge, try setting ABSTOL to
!>          2*DLAMCH('S').
!> 


M


!>          M is INTEGER
!>          The total number of eigenvalues found.  0 <= M <= N.
!>          If RANGE = 'A', M = N, and if RANGE = 'I', M = IU-IL+1.
